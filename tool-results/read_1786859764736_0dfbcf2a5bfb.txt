    49→// ─── Error boundary for individual sections ──────────────────────────
    50→// Prevents a malformed section from crashing the entire page.
    51→
    52→interface SectionErrorBoundaryState { hasError: boolean; error: Error | null }
    53→class SectionErrorBoundary extends Component<
    54→  { children: React.ReactNode; sectionType: string },
    55→  SectionErrorBoundaryState
    56→> {
    57→  constructor(props: { children: React.ReactNode; sectionType: string }) {
    58→    super(props);
    59→    this.state = { hasError: false, error: null };
    60→  }
    61→  static getDerivedStateFromError(error: Error) {
    62→    return { hasError: true, error };
    63→  }
    64→  render() {
    65→    if (this.state.hasError) {
    66→      return (
    67→        <div className="flex items-center justify-center py-8 px-6 text-center">
    68→          <div className="max-w-sm">
    69→            <p className="text-sm font-medium" style={{ color: '#888' }}>
    70→              Section render error ({this.props.sectionType})
    71→            </p>
    72→            <p className="mt-1 text-xs" style={{ color: '#aaa' }}>
    73→              {this.state.error?.message || 'Unknown error'}
    74→            </p>
    75→          </div>
    76→        </div>
    77→      );
    78→    }
    79→    return this.props.children;
    80→  }
    81→}

    85→interface SectionWrapperProps {
    86→  section: Section;
    87→  theme: StoreTheme;
    88→  selectedSectionId?: string | null;
    89→  onSelectSection?: (sectionId: string | null) => void;
    90→  children: React.ReactNode;
    91→}
    92→
    93→function SectionWrapper({
    94→  section,
    95→  theme,
    96→  selectedSectionId,
    97→  onSelectSection,
    98→  children,
    99→}: SectionWrapperProps) {
   100→  const isSelected = selectedSectionId === section.id;
   101→  const style = section.style;
   102→
   103→  const sectionBg = style.backgroundColor || theme.colors.background;
   104→  const sectionText = style.textColor || theme.colors.text;
   105→
   106→  const bgImage = style.backgroundImage
   107→    ? `url(${style.backgroundImage})`
   108→    : undefined;
   109→
   110→  const outerClasses = [
   111→    'relative transition-all duration-200 cursor-pointer',
   112→    pyClass(style.paddingY),
   113→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   114→  ].join(' ');
   115→
   116→  return (
   117→    <div
   118→      className={outerClasses}
   119→      style={{
   120→        backgroundColor: sectionBg,
   121→        color: sectionText,
   122→        backgroundImage: bgImage,
   123→        backgroundSize: 'cover',
   124→        backgroundPosition: 'center',
   125→      }}
   126→      onClick={(e) => {
   127→        e.stopPropagation();
   128→        onSelectSection?.(isSelected ? null : section.id);
   129→      }}
   130→      data-section-id={section.id}
   131→      data-section-type={section.type}
   132→    >
   133→      {/* Overlay for background images */}
   134→      {style.backgroundImage && style.overlay && (
   135→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   136→      )}
   137→      <div
   138→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   139→      >
   140→        {/* Selected section label */}
   141→        {isSelected && (
   142→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   143→            {section.type.replace(/-/g, ' ')}
   144→          </div>
   145→        )}
   146→        {children}
   147→      </div>
   148→    </div>
   149→  );
   150→}

   152→// ─── Product Card (shared) ─────────────────────────────────────────────

   154→function ProductCard({
   155→  product,
   156→  theme,
   157→  showAddToCart,
   158→  borderRadius,
   159→  buttonBgOverride,
   160→  buttonTextOverride,
   161→  onViewProduct,
   162→}: {
   163→  product: StoreProduct;
   164→  theme: StoreTheme;
   165→  showAddToCart: boolean;
   166→  borderRadius: string;
   167→  buttonBgOverride?: string;
   168→  buttonTextOverride?: string;
   169→  onViewProduct?: (productId: string) => void;
   170→}) {
   171→  const [hovered, setHovered] = useState(false);
   172→  const [addedFeedback, setAddedFeedback] = useState(false);
   173→  const addToCart = useCartStore((s) => s.addItem);
   174→  const imgColor = stringToColor(product.id, theme);
   175→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   176→
   177→  const handleAddToCart = (e: React.MouseEvent) => {
   178→    e.stopPropagation();
   179→    addToCart(product);
   180→    setAddedFeedback(true);
   181→    setTimeout(() => setAddedFeedback(false), 1200);
   182→  };
   183→
   184→  return (
   185→    <div
   186→      className={`${borderRadius} ${onViewProduct ? 'cursor-pointer' : ''} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   187→      style={{
   188→        borderColor: theme.colors.border,
   189→      }}
   190→      onMouseEnter={() => setHovered(true)}
   191→      onMouseLeave={() => setHovered(false)}
   192→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   193→      role={onViewProduct ? 'button' : undefined}
   194→      tabIndex={onViewProduct ? 0 : undefined}
   195→    >
   196→      {/* Product image with real image + fallback */}
   197→      <div
   198→        className="relative aspect-square w-full overflow-hidden"
   199→        style={{ backgroundColor: imgColor }}
   200→      >
   201→        <StoreImage
   202→          src={product.images[0] || ''}
   203→          alt={product.name}
   204→          fallbackColor={imgColor}
   205→          className="absolute inset-0 h-full w-full object-cover transition-transform duration-300 group-hover:scale-105"
   206→          iconSize="lg"
   207→        />
   208→        {/* Out of stock overlay */}
   209→        {!product.inStock && (
   210→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   211→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   212→              Sold Out
   213→            </span>
   214→          </div>
   215→        )}
   216→        {/* Discount badge */}
   217→        {hasDiscount && product.compareAtPrice && (
   218→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   219→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   220→          </div>
   221→        )}
   222→      </div>
   223→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   224→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   225→        {product.category && (
   226→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   227→            {product.category}
   228→          </p>
   229→        )}
   230→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   231→          {product.name}
   232→        </h3>
   233→        <div className="mt-1.5 flex items-center gap-2">
   234→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   235→            {formatPrice(product.price)}
   236→          </span>
   237→          {hasDiscount && product.compareAtPrice && (
   238→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   239→              {formatPrice(product.compareAtPrice)}
   240→            </span>
   241→          )}
   242→        </div>
   243→        {showAddToCart && product.inStock && (
   244→          <button
   245→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 cursor-pointer ${
   246→              hovered
   247→                ? 'opacity-100 translate-y-0'
   248→                : 'opacity-0 translate-y-1 pointer-events-none'
   249→            }`}
   250→            style={{
   251→              backgroundColor: addedFeedback ? '#16a34a' : (buttonBgOverride || theme.colors.primary),
   252→              color: addedFeedback ? '#ffffff' : (buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary)),
   253→            }}
   254→            onClick={handleAddToCart}
   255→          >
   256→            {addedFeedback ? 'Added!' : 'Add to Cart'}
   257→          </button>
   258→        )}
   259→      </div>
   260→    </div>
   261→  );
   262→}

   359→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   360→  const content = section.content as unknown as HeroContent;
   361→  const style = section.style;
   362→  const isSelected = selectedSectionId === section.id;
   363→
   364→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   365→  const alignMap = {
   366→    left: 'items-start text-left',
   367→    center: 'items-center text-center',
   368→    right: 'items-end text-right',
   369→  };
   370→
   371→  // Only apply theme gradient as default background when no custom color or image is set.
   372→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   373→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   374→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   375→    : undefined;
   376→
   377→  const outerClasses = [
   378→    'relative flex transition-all duration-200 cursor-pointer',
   379→    heightMap[content.height] || heightMap.md,
   380→    alignMap[content.alignment],
   381→    pyClass(style.paddingY),
   382→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   383→  ].join(' ');
   384→
   385→  return (
   386→    <div
   387→      className={outerClasses}
   388→      style={{
   389→        backgroundColor: style.backgroundColor || theme.colors.primary,
   390→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   391→        backgroundSize: 'cover',
   392→        backgroundPosition: 'center',
   393→        color: style.textColor || '#ffffff',
   394→      }}
   395→      onClick={(e) => {
   396→        e.stopPropagation();
   397→        onSelectSection?.(isSelected ? null : section.id);
   398→      }}
   399→      data-section-id={section.id}
   400→      data-section-type={section.type}
   401→    >
   402→      {/* Overlay */}
   403→      {style.backgroundImage && style.overlay && (
   404→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   405→      )}
   406→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   407→      {!style.backgroundImage && !style.backgroundColor && (
   408→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   409→      )}
   410→      {!style.backgroundImage && !style.backgroundColor && (
   411→        <>
   412→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   413→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />
   414→        </>
   415→      )}
   416→
   417→      <div className={`relative z-10 mx-auto w-full ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}>
   418→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   419→          {content.headline}
   420→        </h1>
   421→        {content.subheadline && (
   422→          <p className="mt-4 max-w-xl text-base opacity-80 sm:text-lg md:text-xl">
   423→            {content.subheadline}
   424→          </p>
   425→        )}
   426→        {content.ctaText && (
   427→          <button className="mt-8 inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-transform hover:scale-105 sm:text-base"
   428→            style={{
   429→              backgroundColor: section.style.buttonBackgroundColor || '#ffffff',
   430→              color: section.style.buttonTextColor || contrastTextColor(section.style.buttonBackgroundColor || '#ffffff'),
   431→            }}>
   432→            {content.ctaText}
   433→            <ArrowRight className="h-4 w-4" />

   443→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   444→  const content = section.content as unknown as FeaturedProductsContent;
   445→  const borderRadius = borderRadiusClass(theme.borderRadius);
   446→  const featuredProducts = useMemo(
   447→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   448→    [content.productIds, products],
   449→  );
   450→
   451→  return (
   452→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   453→      {content.headline && (
   454→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   455→          {content.headline}
   456→        </h2>
   457→      )}
   458→      {content.subtitle && (
   459→        <p className="mb-8 text-sm opacity-65">
   460→          {content.subtitle}
   461→        </p>
   462→      )}
   463→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   464→        {featuredProducts.map((product) => (
   465→          <ProductCard
   466→            key={product.id}
   467→            product={product}
   468→            theme={theme}
   469→            showAddToCart={content.showAddToCart && !forceHideAddToCart}
   470→            borderRadius={borderRadius}
   471→            buttonBgOverride={section.style.buttonBackgroundColor}
   472→            buttonTextOverride={section.style.buttonTextColor}
   473→            onViewProduct={onViewProduct}
   474→          />
   475→        ))}
   476→      </div>
   477→    </SectionWrapper>
   478→  );

   483→export function ProductGridSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   484→  const content = section.content as unknown as ProductGridContent;
   485→  const borderRadius = borderRadiusClass(theme.borderRadius);
   486→
   487→  const filteredProducts = useMemo(() => {
   488→    let prods = products;
   489→    if (content.filterByCategory) {
   490→      prods = prods.filter((p) => p.category === content.filterByCategory);
   491→    }
   492→    return prods;
   493→  }, [products, content.filterByCategory]);
   494→
   495→  return (
   496→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   497→      {content.headline && (
   498→        <h2 className="mb-8 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   499→          {content.headline}
   500→        </h2>
   501→      )}
   502→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   503→        {filteredProducts.map((product) => (
   504→          <ProductCard
   505→            key={product.id}
   506→            product={product}
   507→            theme={theme}
   508→            showAddToCart={content.showAddToCart && !forceHideAddToCart}
   509→            borderRadius={borderRadius}
   510→            buttonBgOverride={section.style.buttonBackgroundColor}
   511→            buttonTextOverride={section.style.buttonTextColor}
   512→            onViewProduct={onViewProduct}
   513→          />
   514→        ))}
   515→      </div>
   516→      {filteredProducts.length === 0 && (
   517→        <div className="py-16 text-center opacity-65">
   518→          <p className="text-sm">No products found.</p>
   519→        </div>
   520→      )}
   521→    </SectionWrapper>
   522→  );
   523→}

  1032→// ─── Main Section Router ────────────────────────────────────────────────
  1033→
  1034→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1035→  const { section } = props;
  1036→
  1037→  if (!section.visible) return null;
  1038→
  1039→  let element: React.ReactNode = null;
  1040→
  1041→  switch (section.type) {
  1042→    case 'header':
  1043→      element = <HeaderSection {...props} />; break;
  1044→    case 'hero':
  1045→      element = <HeroSection {...props} />; break;
  1046→    case 'featured-products':
  1047→      element = <FeaturedProductsSection {...props} />; break;
  1048→    case 'product-grid':
  1049→      element = <ProductGridSection {...props} />; break;
  1050→    case 'text-banner':
  1051→      element = <TextBannerSection {...props} />; break;
  1052→    case 'image-gallery':
  1053→      element = <ImageGallerySection {...props} />; break;
  1054→    case 'testimonials':
  1055→      element = <TestimonialsSection {...props} />; break;
  1056→    case 'newsletter':
  1057→      element = <NewsletterSection {...props} />; break;
  1058→    case 'faq':
  1059→      element = <FAQSection {...props} />; break;
  1060→    case 'cta':
  1061→      element = <CTASection {...props} />; break;
  1062→    case 'categories':
  1063→      element = <CategoriesSection {...props} />; break;
  1064→    case 'footer':
  1065→      element = <FooterSection {...props} />; break;
  1066→    case 'rich-text':
  1067→      element = <RichTextSection {...props} />; break;
  1068→    case 'spacer':
  1069→      element = <SpacerSection {...props} />; break;
  1070→    case 'divider':
  1071→      element = <DividerSection {...props} />; break;
  1072→    default:
  1073→      return null;
