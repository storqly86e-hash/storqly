    95→function LandingPage() {
    96→  const { data: session } = useSession()
    97→  const [promptText, setPromptText] = useState('')
    98→  const [mkOpen, setMkOpen] = useState(false)
    99→  const [authOpen, setAuthOpen] = useState(false)
   100→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   101→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   102→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   103→  const abortRef = useRef<AbortController | null>(null)
   104→
   105→  const {
   106→    isGenerating,
   107→    setIsGenerating,
   108→    setStore,
   109→    setStoreWithFallback,
   110→  } = useStoreEditor()
   111→
   112→  // Local UI state (not in Zustand — this is view-level)
   113→  const [generationStatus, setGenerationStatus] = useState('')
   114→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   115→  const [error, setError] = useState<string | null>(null)
   116→
   117→  const clearTimers = useCallback(() => {
   118→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   119→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }
   120→  }, [])
   121→
   122→  const handleCancel = useCallback(() => {
   123→    abortRef.current?.abort()
   124→    clearTimers()
   125→    setIsGenerating(false)
   126→    setGenerationStatus('')
   127→    setElapsedSeconds(0)
   128→    setError(null)
   129→  }, [clearTimers, setIsGenerating])
   130→
   131→  const handleGenerate = useCallback(async () => {
   132→    const trimmed = promptText.trim()
   133→    if (!trimmed) {
   134→      setError('Please describe your store to get started.')
   135→      textareaRef.current?.focus()
   136→      return
   137→    }
   138→
   139→    // Auth gate — block generation for logged-out users (show modal instead)
   140→    if (!session?.user?.id) {
   141→      setAuthOpen(true)
   142→      return
   143→    }
   144→
   145→    // Reset state
   146→    setError(null)
   147→    setIsGenerating(true)
   148→    setGenerationStatus('Starting generation...')
   149→    setElapsedSeconds(0)
   150→
   151→    // Create abort controller for this request
   152→    const controller = new AbortController()
   153→    abortRef.current = controller
   154→
   155→    // Elapsed time counter
   156→    elapsedTimerRef.current = setInterval(() => {
   157→      setElapsedSeconds((prev) => prev + 1)
   158→    }, 1000)
   159→
   160→    // Fallback progress cycling (used only if SSE progress events stop)
   161→    let fallbackIdx = 0
   162→    let receivedSSEProgress = false
   163→    progressTimerRef.current = setInterval(() => {
   164→      if (!receivedSSEProgress) {
   165→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   166→        setGenerationStatus(progressMessages[fallbackIdx])
   167→      }
   168→    }, 3000)
   169→
   170→    try {
   171→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   172→
   173→      // ── Fetch with auto-retry for transient gateway errors (502/503/504) ──
   174→      // Caddy returns 502 when Turbopack is briefly unavailable during recompilation.
   175→      // One retry with a 2s pause is enough to let the server finish recompiling.
   176→      const RETRYABLE_STATUSES = [502, 503, 504]
   177→      let res: Response | undefined
   178→      let retryCount = 0
   179→      const maxRetries = 1
   180→
   181→      while (retryCount <= maxRetries) {
   182→        try {
   183→          res = await fetch('/api/store/generate', {
   184→            method: 'POST',
   185→            headers: { 'Content-Type': 'application/json' },
   186→            body: JSON.stringify({ prompt: trimmed }),
   187→            signal: controller.signal,
   188→          })
   189→
   190→          // If the status is retryable, wait and loop again
   191→          if (!res.ok && RETRYABLE_STATUSES.includes(res.status) && retryCount < maxRetries) {
   192→            retryCount++
   193→            console.warn(`[Storqly] Gateway error ${res.status}, retrying (${retryCount}/${maxRetries}) in 2s...`)
   194→            setGenerationStatus('Connection issue — retrying...')
   195→            await new Promise(r => setTimeout(r, 2000))
   196→            continue
   197→          }
   198→          break // Success or non-retryable error — exit loop
   199→        } catch (fetchErr: unknown) {
   200→          // Network-level error (DNS failure, connection refused, etc.)
   201→          if (fetchErr instanceof TypeError && retryCount < maxRetries) {
   202→            retryCount++
   203→            console.warn(`[Storqly] Network error, retrying (${retryCount}/${maxRetries}) in 2s...`)
   204→            setGenerationStatus('Connection issue — retrying...')
   205→            await new Promise(r => setTimeout(r, 2000))
   206→            continue
   207→          }
   208→          throw fetchErr // Re-throw if out of retries or not a network error
   209→        }
   210→      }
   211→
   212→      if (!res!.ok) {
   213→        let errorMsg = `Server error (${res!.status})`
   214→        try {
   215→          const errorData = await res!.json()
   216→          errorMsg = errorData.error || errorMsg
   217→        } catch { /* ignore parse error */ }
   218→        throw new Error(errorMsg)
   219→      }
   220→
   221→      // ── Consume SSE stream ──
   222→      const reader = res.body?.getReader()
   223→      if (!reader) throw new Error('No response stream received.')
   224→
   225→      const decoder = new TextDecoder()
   226→      let buffer = ''
   227→      let resolved = false
   228→
   229→      while (true) {
   230→        const { done, value } = await reader.read()
   231→        if (done) break
   232→
   233→        buffer += decoder.decode(value, { stream: true })
   234→
   235→        // Parse SSE events from buffer
   236→        // SSE format: "event: <name>\ndata: <json>\n\n" or ": comment\n\n"
   237→        const lines = buffer.split('\n')
   238→        // Keep incomplete last line in buffer
   239→        buffer = lines.pop() || ''
   240→
   241→        let currentEvent = ''
   242→        for (const line of lines) {
   243→          if (line.startsWith(':')) {
   244→            // SSE comment (heartbeat) — skip
   245→            continue
   246→          }
   247→          if (line.startsWith('event: ')) {
   248→            currentEvent = line.slice(7).trim()
   249→          } else if (line.startsWith('data: ')) {
   250→            const dataStr = line.slice(6)
   251→
   252→            // Handle SSE events
   253→            if (currentEvent === 'progress') {
   254→              receivedSSEProgress = true
   255→              try {
   256→                const data = JSON.parse(dataStr)
   257→                setGenerationStatus(data.message || 'Processing...')
   258→              } catch { /* ignore */ }
   259→            } else if (currentEvent === 'result') {
   260→              try {
   261→                const data = JSON.parse(dataStr)
   262→                if (!data.store) {
   263→                  throw new Error('The AI response was missing store data.')
   264→                }
   265→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   266→                resolved = true
   267→                clearTimers()
   268→
   269→                if (data._isFallback) {
   270→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   271→                } else {
   272→                  setStore(data.store)
   273→                }
   274→              } catch (parseErr) {
   275→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   276→                console.error('[Storqly] Failed to parse result event:', msg)
   277→                throw new Error(msg)
   278→              }
   279→            } else if (currentEvent === 'error') {
   280→              try {
   281→                const data = JSON.parse(dataStr)
   282→                throw new Error(data.message || 'Generation failed')
   283→              } catch (e) {
   284→                if (e instanceof Error && e.message !== 'Generation failed') throw e
   285→                throw new Error('Generation failed')
   286→              }
   287→            }
   288→            // Reset event after processing its data
   289→            currentEvent = ''
   290→          }
   291→        }
   292→      }
   293→
   294→      if (!resolved) {
   295→        throw new Error('Stream ended without a result. The server may have disconnected.')
   296→      }
   297→    } catch (err: unknown) {
   298→      clearTimers()
   299→
   300→      let message = 'Something went wrong. Please try again.'
   301→      if (err instanceof Error) {
   302→        if (err.name === 'AbortError') {
   303→          message = 'Generation was cancelled.'
   304→        } else {
   305→          console.error('[Storqly] Generation error:', err)
   306→          message = err.message
   307→        }
   308→      }
   309→
   310→      setError(message)
   311→      setIsGenerating(false)
   312→      setGenerationStatus('')
   313→      setElapsedSeconds(0)
   314→      toast.error('Store generation failed', { description: message })
   315→    }
   316→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])
   317→
   318→  // Cleanup on unmount
   319→  useEffect(() => {
   320→    return () => {
   321→      clearTimers()
   322→      abortRef.current?.abort()
   323→    }
   324→  }, [clearTimers])
   325→
   326→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   327→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   328→      e.preventDefault()
   329→      handleGenerate()
   330→    }
   331→  }
   332→
   333→  const formatElapsed = (seconds: number) => {
   334→    if (seconds < 60) return `${seconds}s`
   335→    const m = Math.floor(seconds / 60)
   336→    const s = seconds % 60
   337→    return `${m}:${s.toString().padStart(2, '0')}`
   338→  }

   343→      <header className="fixed top-0 left-0 right-0 z-50 border-b border-white/[0.06] bg-[#09090b]/80 backdrop-blur-xl">
   344→        <nav className="mx-auto flex h-16 max-w-6xl items-center justify-between px-5 sm:px-8">
   345→          <span className="text-2xl font-bold tracking-tight bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   346→            Storqly
   347→          </span>
   348→          <div className="hidden sm:flex items-center gap-4">
   349→            <span className="text-sm text-zinc-500">
   350→              AI-Powered Store Builder
   351→            </span>
   352→            <AuthButton onSignIn={() => setAuthOpen(true)} />
   353→          </div>
   354→          <div className="flex sm:hidden">
   355→            <AuthButton onSignIn={() => setAuthOpen(true)} />
   356→          </div>
   357→        </nav>
   358→      </header>
   359→
   360→      <AuthModal open={authOpen} onOpenChange={setAuthOpen} />
   361→
   362→      {/* ── Hero Section ── */}
   363→      <section className="relative flex flex-1 flex-col items-center justify-center px-5 pt-32 pb-20 sm:pt-40 sm:pb-28">
   364→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   365→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />
   366→        </div>
   367→
   368→        <div className="relative z-10 mx-auto w-full max-w-3xl text-center">
   369→          <motion.h1
   370→            custom={0}
   371→            variants={fadeUp}
   372→            initial="hidden"
   373→            animate="visible"
   374→            className="text-4xl font-extrabold leading-[1.1] tracking-tight sm:text-5xl md:text-6xl lg:text-7xl"
   375→          >
   376→            Build Your Store{' '}
   377→            <span className="bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   378→              in Seconds
   379→            </span>
   380→          </motion.h1>
   381→
   382→          <motion.p
   383→            custom={1}
   384→            variants={fadeUp}
   385→            initial="hidden"
   386→            animate="visible"
   387→            className="mt-6 text-lg text-zinc-400 sm:text-xl"
   388→          >
   389→            Describe your store. AI builds it. You customize and publish.
   390→          </motion.p>
   391→
   392→          <motion.div

   395→            initial="hidden"

   400→              <Textarea
   401→                ref={textareaRef}
   402→                value={promptText}
   403→                onChange={(e) => { setPromptText(e.target.value); setError(null) }}
   404→                onKeyDown={handleKeyDown}
   405→                disabled={isGenerating}

   702→function EditorToolbar({
   703→  onToggleLeft,
   704→  onToggleRight,
   705→  showLeft,
   706→  showRight,
   707→}: {
   708→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   709→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   710→  showLeft: boolean
   711→  showRight: boolean
   712→}) {
   713→  const store = useStoreEditor((s) => s.store)
   714→  const reset = useStoreEditor((s) => s.reset)
   715→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   716→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   717→  const isPublished = useStoreEditor((s) => s.isPublished)
   718→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   719→  const [isSaving, setIsSaving] = useState(false)
   720→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
   721→  const [copied, setCopied] = useState(false)
   722→  const [authOpen, setAuthOpen] = useState(false)
   723→
   724→  const handleBack = () => {
   725→    reset()
   726→  }
   727→
   728→  const handleSave = async () => {
   729→    if (!store || isSaving) return
   730→    setIsSaving(true)
   731→    try {
   732→      const res = await fetch('/api/store/save', {
   733→        method: 'POST',
   734→        headers: { 'Content-Type': 'application/json' },
   735→        body: JSON.stringify({ store }),
   736→      })
   737→      if (!res.ok) throw new Error('Save failed')
   738→      const data = await res.json()
   739→      toast.success(`Draft saved — ID: ${data.id}`, {
   740→        description: `Slug: ${data.slug}`,
   741→        duration: 5000,
   742→      })
   743→    } catch {
   744→      toast.error('Failed to save store')
   745→    } finally {
   746→      setIsSaving(false)
   747→    }
   748→  }
   749→
   750→  const handlePublish = async () => {
   751→    if (!store || isPublishing) return
   752→    setIsPublishing(true)
   753→    try {
   754→      const res = await fetch('/api/store/publish', {
   755→        method: 'POST',
   756→        headers: { 'Content-Type': 'application/json' },
   757→        body: JSON.stringify({ store }),
   758→      })
   759→      if (!res.ok) throw new Error('Publish failed')
   760→      const data = await res.json()
   761→      const slug = data.slug
   762→      setIsPublished(true)
   763→      // Build a real viewable URL using the current origin
   764→      const baseUrl = window.location.origin
   765→      const viewUrl = `${baseUrl}/?store=${slug}`
   766→      setPublishedUrl(viewUrl)
   767→      toast.success('Store published successfully!')
   768→    } catch {
   769→      toast.error('Failed to publish store')
   770→    } finally {
   771→      setIsPublishing(false)
   772→    }
   773→  }
   774→
   775→  const handleCopyUrl = useCallback(async () => {
   776→    if (!publishedUrl) return
   777→    try {
   778→      await navigator.clipboard.writeText(publishedUrl)
   779→      setCopied(true)
   780→      toast.success('URL copied to clipboard!')
   781→      setTimeout(() => setCopied(false), 2000)
   782→    } catch {
   783→      // Fallback for older browsers
   784→      const textarea = document.createElement('textarea')
   785→      textarea.value = publishedUrl
   786→      document.body.appendChild(textarea)
   787→      textarea.select()
   788→      document.execCommand('copy')
   789→      document.body.removeChild(textarea)
   790→      setCopied(true)
   791→      toast.success('URL copied to clipboard!')
   792→      setTimeout(() => setCopied(false), 2000)
   793→    }
   794→  }, [publishedUrl])
   795→
   796→  return (
   797→    <>
   798→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
   799→        <div className="flex items-center gap-1">
   800→          <Button
   801→            variant="ghost"
   802→            size="icon"
   803→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   804→            onClick={handleBack}
   805→            aria-label="Back to home"
   806→          >
   807→            <ArrowLeft className="h-4 w-4" />
   808→          </Button>
   809→          <div className="h-5 w-px bg-zinc-800" />
   810→          <Button
   811→            variant={showLeft ? 'secondary' : 'ghost'}
   812→            size="sm"
   813→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   814→            onClick={() => onToggleLeft((v) => !v)}
   815→          >
   816→            <Layers className="h-3.5 w-3.5" />
   817→            <span className="hidden sm:inline">Sections</span>
   818→          </Button>
   819→          <Button
   820→            variant={showRight ? 'secondary' : 'ghost'}
   821→            size="sm"
   822→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   823→            onClick={() => onToggleRight((v) => !v)}
   824→          >
   825→            <MessageSquare className="h-3.5 w-3.5" />
   826→            <span className="hidden sm:inline">Chat</span>
   827→          </Button>
   828→        </div>
   829→
   830→        <div className="flex items-center gap-1.5">
   831→          {publishedUrl && (
   832→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
   833→              <Check className="h-3 w-3 text-emerald-400" />
   834→              <span className="max-w-[180px] truncate text-xs font-medium text-emerald-300">
   835→                {publishedUrl}
   836→              </span>
   837→              <button
   838→                onClick={handleCopyUrl}
   839→                className="ml-1 rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
   840→                aria-label="Copy URL"
   841→              >
   842→                {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
   843→              </button>
   844→              <a
   845→                href={publishedUrl}
   846→                target="_blank"
   847→                rel="noopener noreferrer"
   848→                className="rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
   849→                aria-label="Open published store"
   850→              >
   851→                <ExternalLink className="h-3 w-3" />
   852→              </a>
   853→            </div>
   854→          )}

   858→            {store?.name || 'Untitled Store'}
   859→          </span>
   860→          <Button
   861→            variant="ghost"
   862→            size="sm"
   863→            className="h-8 gap-1.5 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   864→            onClick={handleSave}
   865→            disabled={isSaving}
   866→          >
   867→            <Save className={`h-3.5 w-3.5 ${isSaving ? 'animate-pulse' : ''}`} />
   868→            <span className="hidden sm:inline">{isSaving ? 'Saving...' : 'Save'}</span>
   869→          </Button>
   870→          <Button
   871→            size="sm"
   872→            className="h-8 gap-1.5 bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] text-white shadow-none hover:opacity-90"

   888→      {/* Publish Success Dialog */}
   889→      <AnimatePresence>
   890→        {publishedUrl && !isPublishing && (
   891→          <motion.div
   892→            initial={{ opacity: 0 }}
   893→            animate={{ opacity: 1 }}
   894→            exit={{ opacity: 0 }}
   895→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 backdrop-blur-sm"
   896→            onClick={() => setPublishedUrl(null)}
   897→          >
   898→            <motion.div
   899→              initial={{ scale: 0.95, opacity: 0 }}
   900→              animate={{ scale: 1, opacity: 1 }}
   901→              exit={{ scale: 0.95, opacity: 0 }}

   965→// ─── Published Store Viewer (read-only, shown via ?store=slug) ────────

   967→function PublishedStoreViewer({ slug }: { slug: string }) {
   968→  const [store, setStore] = useState<Store | null>(null)
   969→  const [loading, setLoading] = useState(true)
   970→  const [error, setError] = useState<string | null>(null)
   971→
   972→  useEffect(() => {
   973→    let cancelled = false
   974→    async function load() {
   975→      try {
   976→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
   977→        if (!res.ok) {
   978→          const data = await res.json().catch(() => ({}))
   979→          throw new Error(data.error || `Store not found (${res.status})`)
   980→        }
   981→        const data = await res.json()
   982→        if (!cancelled) setStore(data.store)
   983→      } catch (err: unknown) {
   984→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
   985→      } finally {
   986→        if (!cancelled) setLoading(false)
   987→      }
   988→    }
   989→    load()
   990→    return () => { cancelled = true }
   991→  }, [slug])
   992→
   993→  if (loading) {
   994→    return (
   995→      <div className="flex min-h-screen items-center justify-center bg-white">
   996→        <div className="text-center">
   997→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
   998→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
   999→        </div>
  1000→      </div>
  1001→    )
  1002→  }
  1003→
  1004→  if (error || !store) {
  1005→    return (
  1006→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
  1007→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
  1008→          <AlertCircle className="h-8 w-8 text-red-400" />
  1009→        </div>
  1010→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
  1011→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
  1012→          {error || 'This store could not be found or has not been published.'}
  1013→        </p>
  1014→        <Button
  1015→          variant="outline"
  1016→          onClick={() => window.location.href = '/'}
  1017→          className="gap-2"
  1018→        >
  1019→          <ArrowLeft className="h-4 w-4" />
  1020→          Back to Storqly
  1021→        </Button>
  1022→      </div>
  1023→    )
  1024→  }
  1025→
  1026→  return (
  1027→    <>
  1028→      <StoreRenderer store={store} />
  1029→      {/* Minimal footer for published stores */}
  1030→      <div className="fixed bottom-3 right-3 z-40">
  1031→        <a
  1032→          href="/"
  1033→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
  1034→        >
  1035→          <Sparkles className="h-3 w-3" />
  1036→          Built with Storqly
  1037→        </a>
  1038→      </div>
  1039→    </>
  1040→  )
  1041→}
