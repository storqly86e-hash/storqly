    36→// ─── Shared Helpers ─────────────────────────────────────────────────────
    37→
    38→/** Get padding-y class from style config */
    39→function pyClass(paddingY?: SectionStyle['paddingY']) {
    40→  switch (paddingY) {
    41→    case 'sm': return 'py-6';
    42→    case 'md': return 'py-10';
    43→    case 'lg': return 'py-16';
    44→    case 'xl': return 'py-24';
    45→    default: return 'py-12';
    46→  }
    47→}
    48→
    49→/** Get padding-x class from style config */
    50→function pxClass(paddingX?: SectionStyle['paddingX']) {
    51→  switch (paddingX) {
    52→    case 'sm': return 'px-4';
    53→    case 'md': return 'px-6';
    54→    case 'lg': return 'px-10';
    55→    default: return 'px-6';
    56→  }
    57→}
    58→
    59→/** Get max-width class from style config */
    60→function maxWidthClass(maxWidth?: SectionStyle['maxWidth']) {
    61→  switch (maxWidth) {
    62→    case 'sm': return 'max-w-2xl';
    63→    case 'md': return 'max-w-4xl';
    64→    case 'lg': return 'max-w-6xl';
    65→    case 'xl': return 'max-w-7xl';
    66→    case 'full': return 'max-w-full';
    67→    default: return 'max-w-6xl';
    68→  }
    69→}
    70→
    71→/** Get border-radius class from theme */
    72→function borderRadiusClass(radius?: StoreTheme['borderRadius']) {
    73→  switch (radius) {
    74→    case 'none': return 'rounded-none';
    75→    case 'sm': return 'rounded-sm';
    76→    case 'md': return 'rounded-md';
    77→    case 'lg': return 'rounded-lg';
    78→    case 'xl': return 'rounded-xl';
    79→    default: return 'rounded-md';
    80→  }
    81→}
    82→
    83→/** Generate a consistent color from a string for image placeholders */
    84→function stringToColor(str: string, theme: StoreTheme): string {
    85→  let hash = 0;
    86→  for (let i = 0; i < str.length; i++) {
    87→    hash = str.charCodeAt(i) + ((hash << 5) - hash);
    88→  }
    89→  const hue = Math.abs(hash) % 360;
    90→  return `hsl(${hue}, 35%, 88%)`;
    91→}
    92→
    93→/** Format price */
    94→function formatPrice(price: number): string {
    95→  return new Intl.NumberFormat('en-US', {
    96→    style: 'currency',
    97→    currency: 'USD',
    98→  }).format(price);
    99→}
   100→
   101→/** Get initials from a name */
   102→function getInitials(name: string): string {
   103→  return name
   104→    .split(' ')
   105→    .map((n) => n[0])
   106→    .join('')
   107→    .toUpperCase()
   108→    .slice(0, 2);
   109→}

   111→// ─── Section Wrapper ────────────────────────────────────────────────────

   113→interface SectionWrapperProps {
   114→  section: Section;
   115→  theme: StoreTheme;
   116→  selectedSectionId?: string | null;
   117→  onSelectSection?: (sectionId: string | null) => void;
   118→  children: React.ReactNode;
   119→}
   120→
   121→function SectionWrapper({
   122→  section,
   123→  theme,
   124→  selectedSectionId,
   125→  onSelectSection,
   126→  children,
   127→}: SectionWrapperProps) {
   128→  const isSelected = selectedSectionId === section.id;
   129→  const style = section.style;
   130→
   131→  const sectionBg = style.backgroundColor || theme.colors.background;
   132→  const sectionText = style.textColor || theme.colors.text;
   133→
   134→  const bgImage = style.backgroundImage
   135→    ? `url(${style.backgroundImage})`
   136→    : undefined;
   137→
   138→  const outerClasses = [
   139→    'relative transition-all duration-200 cursor-pointer',
   140→    pyClass(style.paddingY),
   141→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   142→  ].join(' ');
   143→
   144→  return (
   145→    <div
   146→      className={outerClasses}
   147→      style={{
   148→        backgroundColor: sectionBg,
   149→        color: sectionText,
   150→        backgroundImage: bgImage,
   151→        backgroundSize: 'cover',
   152→        backgroundPosition: 'center',
   153→      }}
   154→      onClick={(e) => {
   155→        e.stopPropagation();
   156→        onSelectSection?.(isSelected ? null : section.id);
   157→      }}
   158→      data-section-id={section.id}
   159→      data-section-type={section.type}
   160→    >
   161→      {/* Overlay for background images */}
   162→      {style.backgroundImage && style.overlay && (
   163→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   164→      )}
   165→      <div
   166→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   167→      >
   168→        {/* Selected section label */}
   169→        {isSelected && (
   170→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   171→            {section.type.replace(/-/g, ' ')}
   172→          </div>
   173→        )}
   174→        {children}
   175→      </div>
   176→    </div>
   177→  );
   178→}

   180→// ─── Product Card (shared) ─────────────────────────────────────────────
   181→
   182→function ProductCard({
   183→  product,
   184→  theme,
   185→  showAddToCart,
   186→  borderRadius,
   187→}: {
   188→  product: StoreProduct;
   189→  theme: StoreTheme;
   190→  showAddToCart: boolean;
   191→  borderRadius: string;
   192→}) {
   193→  const [hovered, setHovered] = useState(false);
   194→  const imgColor = stringToColor(product.id, theme);
   195→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   196→
   197→  return (
   198→    <div
   199→      className={`${borderRadius} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   200→      style={{
   201→        borderColor: theme.colors.border,
   202→      }}
   203→      onMouseEnter={() => setHovered(true)}
   204→      onMouseLeave={() => setHovered(false)}
   205→    >
   206→ {/* Image placeholder */}
   207→      <div
   208→        className="relative aspect-square w-full overflow-hidden"
   209→        style={{ backgroundColor: imgColor }}
   210→      >
   211→        <div className="absolute inset-0 flex items-center justify-center">
   212→          <div
   213→            className="h-16 w-16 rounded-full opacity-30"
   214→            style={{ backgroundColor: theme.colors.primary }}
   215→          />
   216→        </div>
   217→        {/* Out of stock overlay */}
   218→        {!product.inStock && (
   219→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   220→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   221→              Sold Out
   222→            </span>
   223→          </div>
   224→        )}
   225→        {/* Discount badge */}
   226→        {hasDiscount && product.compareAtPrice && (
   227→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   228→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   229→          </div>
   230→        )}
   231→      </div>
   232→      {/* Info */}
   233→      <div className="p-3 sm:p-4">
   234→        {product.category && (
   235→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   236→            {product.category}
   237→          </p>
   238→        )}
   239→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   240→          {product.name}
   241→        </h3>
   242→        <div className="mt-1.5 flex items-center gap-2">
   243→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   244→            {formatPrice(product.price)}
   245→          </span>
   246→          {hasDiscount && product.compareAtPrice && (
   247→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   248→              {formatPrice(product.compareAtPrice)}
   249→            </span>
   250→          )}
   251→        </div>
   252→        {showAddToCart && product.inStock && (
   253→          <button
   254→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 ${
   255→              hovered
   256→                ? 'opacity-100 translate-y-0'
   257→                : 'opacity-0 translate-y-1'
   258→            }`}
   259→            style={{
   260→              backgroundColor: theme.colors.primary,
   261→              color: '#ffffff',
   262→            }}
   263→          >
   264→            Add to Cart
   265→          </button>
   266→        )}
   267→      </div>
   268→    </div>
   269→  );
   270→}

   272→// ─── Column grid class helper ───────────────────────────────────────────
   273→
   274→function gridCols(columns: 2 | 3 | 4) {
   275→  switch (columns) {
   276→    case 2: return 'grid-cols-1 sm:grid-cols-2';
   277→    case 3: return 'grid-cols-1 sm:grid-cols-2 lg:grid-cols-3';
   278→    case 4: return 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-4';
   279→    default: return 'grid-cols-1 sm:grid-cols-2 lg:grid-cols-3';
   280→  }
   281→}

   377→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   378→  const content = section.content as unknown as HeroContent;
   379→  const style = section.style;
   380→  const isSelected = selectedSectionId === section.id;
   381→
   382→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };

   384→    left: 'items-start text-left',
   385→    center: 'items-center text-center',
   386→    right: 'items-end text-right',

   388→
   389→  const bgGradient = !style.backgroundImage
   390→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`

   392→
   393→  const outerClasses = [
   394→    'relative flex transition-all duration-200 cursor-pointer',

   396→    alignMap[content.alignment],

   398→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',

   401→  return (
   402→    <div
   403→      className={outerClasses}
   404→      style={{
   405→        backgroundColor: style.backgroundColor || theme.colors.primary,
   406→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   407→        backgroundSize: 'cover',
   408→        backgroundPosition: 'center',
   409→      }}
   410→      onClick={(e) => {
   411→        e.stopPropagation();
   412→        onSelectSection?.(isSelected ? null : section.id);
   413→      }}
   414→      data-section-id={section.id}
   415→      data-section-type={section.type}

   417→      {/* Overlay */}
   418→      {style.backgroundImage && style.overlay && (

   451→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   452→  const content = section.content as unknown as FeaturedProductsContent;
   453→  const borderRadius = borderRadiusClass(theme.borderRadius);
   454→  const featuredProducts = useMemo(
   455→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   456→    [content.productIds, products],
   457→  );
   458→
   459→  return (
   460→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   461→      {content.headline && (
   462→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={{ color: theme.colors.text }}>
   463→          {content.headline}
   464→        </h2>
   465→      )}
   466→      {content.subtitle && (
   467→        <p className="mb-8 text-sm" style={{ color: theme.colors.textMuted }}>
   468→          {content.subtitle}
   469→        </p>
   470→      )}
   471→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   472→        {featuredProducts.map((product) => (
   473→          <ProductCard
   474→            key={product.id}
   475→            product={product}
   476→            theme={theme}
   477→            showAddToCart={content.showAddToCart}
   478→            borderRadius={borderRadius}
   479→          />
   480→        ))}
   481→      </div>
   482→    </SectionWrapper>
   483→  );
   484→}

   486→// ─── 4. Product Grid ───────────────────────────────────────────────────
   487→
   488→export function ProductGridSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   489→  const content = section.content as unknown as ProductGridContent;
   490→  const borderRadius = borderRadiusClass(theme.borderRadius);
   491→
   492→  const filteredProducts = useMemo(() => {
   493→    let prods = products;
   494→    if (content.filterByCategory) {
   495→      prods = prods.filter((p) => p.category === content.filterByCategory);
   496→    }
   497→    return prods;
   498→  }, [products, content.filterByCategory]);
   499→
   500→  return (
   501→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   502→      {content.headline && (
   503→        <h2 className="mb-8 text-2xl font-bold sm:text-3xl" style={{ color: theme.colors.text }}>
   504→          {content.headline}
   505→        </h2>
   506→      )}
   507→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   508→        {filteredProducts.map((product) => (
   509→          <ProductCard
   510→            key={product.id}
   511→            product={product}
   512→            theme={theme}
   513→            showAddToCart={content.showAddToCart}
   514→            borderRadius={borderRadius}
   515→          />
   516→        ))}
   517→      </div>
   518→      {filteredProducts.length === 0 && (
   519→        <div className="py-16 text-center" style={{ color: theme.colors.textMuted }}>
   520→          <p className="text-sm">No products found.</p>
   521→        </div>
   522→      )}
   523→    </SectionWrapper>
   524→  );
   525→}

   529→export function TextBannerSection({ section, theme, selectedSectionId, onSelectSection }: SectionRendererProps) {

   561→export function ImageGallerySection({ section, theme, selectedSectionId, onSelectSection }: SectionRendererProps) {
   562→  const content = section.content as unknown as ImageGalleryContent;
   563→  const borderRadius = borderRadiusClass(theme.borderRadius);
   564→  const gapMap = { sm: 'gap-2', md: 'gap-4', lg: 'gap-6' };

   568→      <div className={`grid ${gridCols(content.columns)} ${gapMap[content.gap] || gapMap.md}`}>
   569→        {content.images.map((img, i) => (

  1019→// ─── Main Section Router ────────────────────────────────────────────────
  1020→
  1021→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1022→  const { section } = props;
  1023→
  1024→  if (!section.visible) return null;
  1025→
  1026→  switch (section.type) {
  1027→    case 'header':
  1028→      return <HeaderSection {...props} />;
  1029→    case 'hero':
  1030→      return <HeroSection {...props} />;
  1031→    case 'featured-products':
  1032→      return <FeaturedProductsSection {...props} />;
  1033→    case 'product-grid':
  1034→      return <ProductGridSection {...props} />;
  1035→    case 'text-banner':
  1036→      return <TextBannerSection {...props} />;
  1037→    case 'image-gallery':
  1038→      return <ImageGallerySection {...props} />;
  1039→    case 'testimonials':
  1040→      return <TestimonialsSection {...props} />;
  1041→    case 'newsletter':
  1042→      return <NewsletterSection {...props} />;
  1043→    case 'faq':
  1044→      return <FAQSection {...props} />;
  1045→    case 'cta':
  1046→      return <CTASection {...props} />;
  1047→    case 'categories':
  1048→      return <CategoriesSection {...props} />;
  1049→    case 'footer':
  1050→      return <FooterSection {...props} />;
  1051→    case 'rich-text':
  1052→      return <RichTextSection {...props} />;
  1053→    case 'spacer':
  1054→      return <SpacerSection {...props} />;
  1055→    case 'divider':
  1056→      return <DividerSection {...props} />;
  1057→    default:
  1058→      return null;
  1059→  }
  1060→}
