   363→    // If the server is hung or the proxy drops the connection, the user gets a

   380→    let receivedSSEProgress = false
   381→    progressTimerRef.current = setInterval(() => {
   382→      if (!receivedSSEProgress) {
   383→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   384→        setGenerationStatus(progressMessages[fallbackIdx])
   385→      }
   386→    }, 3000)

   408→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)

   410→      // ── Fetch with auto-retry for transient gateway errors (502/503/504) ──
   411→      const RETRYABLE_STATUSES = [502, 503, 504]
   412→      let res: Response | undefined
   413→      let retryCount = 0
   414→      const maxRetries = 2
   415→
   416→      while (retryCount <= maxRetries) {
   417→        try {
   418→          res = await fetch('/api/store/generate', {
   419→            method: 'POST',
   420→            headers: { 'Content-Type': 'application/json' },
   421→            body: JSON.stringify({ prompt: trimmed }),
   422→            signal: controller.signal,
   423→          })
   424→
   425→          // If the status is retryable, wait and loop again
   426→          if (!res.ok && RETRYABLE_STATUSES.includes(res.status) && retryCount < maxRetries) {
   427→            retryCount++
   428→            console.warn(`[Storqly] Gateway error ${res.status}, retrying (${retryCount}/${maxRetries}) in 3s...`)
   429→            setGenerationStatus('Connection issue — retrying...')
   430→            await new Promise(r => setTimeout(r, 3000))
   431→            continue

   433→          break // Success or non-retryable error — exit loop
   434→        } catch (fetchErr: unknown) {
   435→          // Network-level error (DNS failure, connection refused, etc.)
   436→          if (fetchErr instanceof TypeError && retryCount < maxRetries) {
   437→            retryCount++
   438→            console.warn(`[Storqly] Network error, retrying (${retryCount}/${maxRetries}) in 3s...`)
   439→            setGenerationStatus('Connection issue — retrying...')
   440→            await new Promise(r => setTimeout(r, 3000))
   441→            continue
   442→          }
   443→          // Out of retries or not a network error — handle gracefully
   444→          const msg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr)
   445→          if (controller.signal.aborted) {
   446→            finishWithError('Generation was cancelled.')
   447→          } else {
   448→            console.warn('[Storqly] Network error after retries:', msg)
   449→            finishWithError('Could not connect to the server. Please check your connection and try again.')
   450→          }
   451→          return
   452→        }
   453→      }
   454→
   455→      if (!res!.ok) {
   456→        let errorMsg = `Server error (${res!.status})`
   457→        try {
   458→          const errorData = await res!.json()
   459→          errorMsg = errorData.error || errorMsg
   460→        } catch { /* ignore parse error */ }
   461→        console.warn('[Storqly] HTTP error:', errorMsg)
   462→        finishWithError(errorMsg)
   463→        return
   464→      }
   465→
   466→      // ── Consume SSE stream ──
   467→      const reader = res.body?.getReader()
   468→      if (!reader) {
   469→        finishWithError('No response stream received from server.')
   470→        return
   471→      }
   472→
   473→      const decoder = new TextDecoder()
   474→      let buffer = ''
   475→      let resolved = false
   476→
   477→      while (true) {
   478→        const { done, value } = await reader.read()
   479→        if (done) break
   480→
   481→        buffer += decoder.decode(value, { stream: true })
   482→
   483→        // Parse SSE events from buffer
   484→        const lines = buffer.split('\n')
   485→        buffer = lines.pop() || ''
   486→
   487→        let currentEvent = ''
   488→        for (const line of lines) {
   489→          if (line.startsWith(':')) continue
   490→          if (line.startsWith('event: ')) {
   491→            currentEvent = line.slice(7).trim()
   492→          } else if (line.startsWith('data: ')) {
   493→            const dataStr = line.slice(6)
   494→
   495→            if (currentEvent === 'progress') {
   496→              receivedSSEProgress = true
   497→              try {
   498→                const data = JSON.parse(dataStr)
   499→                setGenerationStatus(data.message || 'Processing...')
   500→              } catch { /* ignore */ }
   501→            } else if (currentEvent === 'result') {
   502→              try {
   503→                const data = JSON.parse(dataStr)
   504→                if (!data.store) {
   505→                  console.warn('[Storqly] Result event missing store data')
   506→                  finishWithError('The AI response was missing store data.')
   507→                  return
   508→                }
   509→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   510→                resolved = true
   511→
   512→                if (data._isFallback) {
   513→                  // Differentiate "all providers down" from other fallback reasons
   514→                  const reason = data._fallbackReason || 'AI generation failed'
   515→                  const isProviderFailure = reason.includes('All') || reason.includes('429') || reason.includes('403') || reason.includes('404') || reason.includes('providers failed')
   516→                  toast.warning(
   517→                    isProviderFailure
   518→                      ? 'All AI providers are currently unavailable. You are viewing a starter template. Try again later.'
   519→                      : 'AI service returned invalid data — showing starter template. Try again.',
   520→                    { duration: 8000 }
   521→                  )
   522→                  setStoreWithFallback(data.store, true, reason)
   523→                } else {
   524→                  setStore(data.store)
   525→                  // Trigger lazy background image enrichment (non-blocking)
   526→                  triggerBackgroundImageEnrichment(data.store)
   527→                }
   528→
   529→                // Soft cap toast
   530→                if (data._productCapHit) {
   531→                  toast.info(
   532→                    `Generated ${data._generatedCount} products — for larger catalogs, you can add more via the chat editor.`,
   533→                    { duration: 8000 }
   534→                  )
   535→                }
   536→
   537→                finishOk()
   538→                return
   539→              } catch (parseErr) {
   540→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   541→                console.warn('[Storqly] Failed to parse result event:', msg)
   542→                finishWithError(msg)
   543→                return
   544→              }
   545→            } else if (currentEvent === 'error') {
   546→              try {
   547→                const data = JSON.parse(dataStr)
   548→                const serverMsg = data.message || 'Generation failed'
   549→                console.warn('[Storqly] Server error event:', serverMsg)
   550→                finishWithError(serverMsg)
   551→                return
   552→              } catch {
   553→                finishWithError('Generation failed')
   554→                return
   555→              }
   556→            }
   557→            currentEvent = ''
   558→          }
   559→        }
   560→      }
   561→
   562→      if (!resolved) {
   563→        console.warn('[Storqly] Stream ended without a result event')
   564→        finishWithError('Stream ended without a result. The server may have disconnected.')
   565→      }
   566→    } catch (err: unknown) {
   567→      // Last-resort safety net — should rarely be reached now
   568→      clearTimeout(timeoutId)
   569→      clearTimers()
   570→      let message = 'Something went wrong. Please try again.'
   571→      if (err instanceof Error) {
   572→        if (err.name === 'AbortError') {
   573→          message = timedOut
   574→            ? 'Generation timed out after 2 minutes. The server may be overloaded — please try again.'
   575→            : 'Generation was cancelled.'
   576→          console.warn(`[Storqly] Request aborted: ${message}`)
   577→        } else {
   578→          console.warn('[Storqly] Unexpected error (caught by safety net):', err.message)
   579→          message = err.message
   580→        }
   581→      }
   582→      setError(message)
   583→      setIsGenerating(false)
   584→      setGenerationStatus('')
   585→      setElapsedSeconds(0)
   586→      toast.error('Store generation failed', { description: message })
   587→    }
   588→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])

   632→      <AuthModal open={authOpen} onOpenChange={setAuthOpen} />

   634→      {/* ── Hero Section ── */}

   636→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   637→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />
   638→        </div>

  1133→function EditorToolbar({
  1134→  onToggleLeft,
  1135→  onToggleRight,
  1136→  showLeft,
  1137→  showRight,
  1138→}: {
  1139→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
  1140→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
  1141→  showLeft: boolean
  1142→  showRight: boolean
  1143→}) {
  1144→  const store = useStoreEditor((s) => s.store)
  1145→  const reset = useStoreEditor((s) => s.reset)
  1146→  const isPublishing = useStoreEditor((s) => s.isPublishing)
  1147→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
  1148→  const isPublished = useStoreEditor((s) => s.isPublished)
  1149→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
  1150→  const [isSaving, setIsSaving] = useState(false)
  1151→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
  1152→  const [copied, setCopied] = useState(false)
  1153→  const [authOpen, setAuthOpen] = useState(false)
  1154→
  1155→  const handleBack = () => {
  1156→    reset()
  1157→  }
  1158→
  1159→  const handleSave = async () => {
  1160→    if (!store || isSaving) return
  1161→    setIsSaving(true)
  1162→    try {
  1163→      const res = await fetch('/api/store/save', {
  1164→        method: 'POST',
  1165→        headers: { 'Content-Type': 'application/json' },
  1166→        body: JSON.stringify({ store }),
  1167→      })
  1168→      if (!res.ok) throw new Error(`Save failed (${res.status})`)
  1169→      const data = await res.json()
  1170→      toast.success(`Draft saved — ID: ${data.id}`, {
  1171→        description: `Slug: ${data.slug}`,
  1172→        duration: 5000,
  1173→      })
  1174→    } catch (err: unknown) {
  1175→      const msg = err instanceof Error ? err.message : ''
  1176→      if (msg.includes('Failed to fetch') || msg.includes('NetworkError') || msg.includes('fetch')) {
  1177→        toast.error('Connection lost while saving', {
  1178→          description: 'Your work is still in the editor. Please check your connection and try again.',
  1179→          duration: 6000,
  1180→        })
  1181→      } else {
  1182→        toast.error('Failed to save store', {
  1183→          description: msg || 'An unexpected error occurred. Please try again.',
  1184→          duration: 5000,
  1185→        })
  1186→      }
  1187→    } finally {
  1188→      setIsSaving(false)
  1189→    }
  1190→  }
  1191→
  1192→  const handlePublish = async () => {
  1193→    if (!store || isPublishing) return
  1194→    setIsPublishing(true)
  1195→    try {
  1196→      const res = await fetch('/api/store/publish', {
  1197→        method: 'POST',
  1198→        headers: { 'Content-Type': 'application/json' },
  1199→        body: JSON.stringify({ store }),
  1200→      })
  1201→      if (!res.ok) {
  1202→        // Try to get server error details
  1203→        let serverMsg = ''
  1204→        try {
  1205→          const errorData = await res.json()
  1206→          serverMsg = errorData.error || ''
  1207→        } catch { /* ignore */ }
  1208→
  1209→        if (res.status === 401) {
  1210→          throw new Error('AUTH_REQUIRED')
  1211→        } else if (res.status === 502 || res.status === 503 || res.status === 504) {
  1212→          throw new Error('GATEWAY_ERROR')
  1213→        } else {
  1214→          throw new Error(serverMsg || `Server error (${res.status})`)
  1215→        }
  1216→      }
  1217→      const data = await res.json()
  1218→      const slug = data.slug
  1219→      setIsPublished(true)
  1220→      // Build a real viewable URL using the current origin
  1221→      const baseUrl = window.location.origin
  1222→      const viewUrl = `${baseUrl}/?store=${slug}`
  1223→      setPublishedUrl(viewUrl)
  1224→      toast.success('Store published successfully!')
  1225→    } catch (err: unknown) {
  1226→      const msg = err instanceof Error ? err.message : ''
  1227→
  1228→      if (msg === 'AUTH_REQUIRED') {
  1229→        toast.error('Please sign in to publish', {
  1230→          description: 'You need to be signed in before publishing your store.',
  1231→          duration: 6000,
  1232→        })
  1233→        setAuthOpen(true)
  1234→      } else if (msg === 'GATEWAY_ERROR') {
  1235→        toast.error('Server temporarily unavailable', {
  1236→          description: 'The server seems to be restarting. Please wait a moment and try again.',
  1237→          duration: 6000,
  1238→        })
  1239→      } else if (msg.includes('Failed to fetch') || msg.includes('NetworkError') || msg.includes('fetch') || msg.includes('Load failed')) {
  1240→        toast.error('Connection lost while publishing', {
  1241→          description: 'Your store data is safe. Please check your connection and try again.',
  1242→          duration: 6000,
  1243→        })
  1244→      } else {
  1245→        toast.error('Failed to publish store', {
  1246→          description: msg || 'An unexpected error occurred. Please try again.',
  1247→          duration: 5000,
  1248→        })
  1249→      }
  1250→    } finally {
  1251→      setIsPublishing(false)
  1252→    }
  1253→  }
  1254→
  1255→  const handleCopyUrl = useCallback(async () => {
  1256→    if (!publishedUrl) return
  1257→    try {
  1258→      await navigator.clipboard.writeText(publishedUrl)
  1259→      setCopied(true)
  1260→      toast.success('URL copied to clipboard!')
  1261→      setTimeout(() => setCopied(false), 2000)
  1262→    } catch {
  1263→      // Fallback for older browsers
  1264→      const textarea = document.createElement('textarea')
  1265→      textarea.value = publishedUrl
  1266→      document.body.appendChild(textarea)
  1267→      textarea.select()

  1447→function PublishedStoreViewer({ slug }: { slug: string }) {
  1448→  const [store, setStore] = useState<Store | null>(null)
  1449→  const [loading, setLoading] = useState(true)
  1450→  const [error, setError] = useState<string | null>(null)
  1451→
  1452→  useEffect(() => {
  1453→    let cancelled = false
  1454→    async function load() {
  1455→      try {
  1456→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
  1457→        if (!res.ok) {
  1458→          const data = await res.json().catch(() => ({}))
  1459→          throw new Error(data.error || `Store not found (${res.status})`)
  1460→        }
  1461→        const data = await res.json()
  1462→        if (!cancelled) setStore(data.store)
  1463→      } catch (err: unknown) {
  1464→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
  1465→      } finally {
  1466→        if (!cancelled) setLoading(false)
  1467→      }
  1468→    }
  1469→    load()
  1470→    return () => { cancelled = true }
  1471→  }, [slug])
  1472→
  1473→  if (loading) {
  1474→    return (
  1475→      <div className="flex min-h-screen items-center justify-center bg-white">
  1476→        <div className="text-center">
  1477→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
  1478→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
  1479→        </div>
  1480→      </div>
  1481→    )
  1482→  }
  1483→
  1484→  if (error || !store) {
  1485→    return (
  1486→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
  1487→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
  1488→          <AlertCircle className="h-8 w-8 text-red-400" />
  1489→        </div>
  1490→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
  1491→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
  1492→          {error || 'This store could not be found or has not been published.'}
  1493→        </p>
  1494→        <Button
  1495→          variant="outline"
  1496→          onClick={() => window.location.href = '/'}
  1497→          className="gap-2"
  1498→        >
  1499→          <ArrowLeft className="h-4 w-4" />
  1500→          Back to Storqly
  1501→        </Button>
  1502→      </div>
  1503→    )
  1504→  }
  1505→
  1506→  return (
  1507→    <>
  1508→      <StoreRenderer store={store} />
  1509→      <Toaster />
  1510→      {/* Minimal footer for published stores */}
  1511→      <div className="fixed bottom-3 right-3 z-40">
  1512→        <a
  1513→          href="/"
  1514→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
  1515→        >
  1516→          <Sparkles className="h-3 w-3" />
  1517→          Built with Storqly
  1518→        </a>
  1519→      </div>
  1520→    </>
  1521→  )
  1522→}
