   256→   256→function SortableSectionItem({
   257→   257→  section,
   258→   258→  isSelected,
   259→   259→  onSelect,
   260→   260→  onToggleVisibility,
   261→   261→  onDelete,
   262→   262→}: SortableSectionItemProps) {
   263→   263→  const {
   264→   264→    attributes,
   265→   265→    listeners,
   266→   266→    setNodeRef,
   267→   267→    transform,
   268→   268→    transition,

   270→   270→  } = useSortable({ id: section.id });
   271→   271→
   272→   272→  const style = {
   273→   273→    transform: CSS.Transform.toString(transform),
   274→   274→    transition,
   275→   275→  };
   276→   276→
   277→   277→  const Icon = SECTION_TYPE_ICONS[section.type];
   278→   278→  const label = SECTION_TYPE_LABELS[section.type];
   279→   279→
   280→   280→  return (
   281→   281→    <div
   282→   282→      ref={setNodeRef}
   283→   283→      style={style}
   284→   284→      className={[
   285→   285→        'group flex items-center gap-2 rounded-lg px-3 py-2 text-sm cursor-pointer transition-colors',
   286→   286→        isDragging
   287→   287→          ? 'z-50 bg-zinc-700/80 shadow-xl ring-1 ring-white/10'
   288→   288→          : isSelected

   290→   290→            : 'text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200',
   291→   291→      ].join(' ')}
   292→   292→      onClick={onSelect}
   293→   293→    >
   294→   294→      {/* Drag handle */}
   295→   295→      <button
   296→   296→        className="flex-shrink-0 cursor-grab text-zinc-600 hover:text-zinc-400 active:cursor-grabbing focus:outline-none"

   298→   298→        {...listeners}

   303→   303→      </button>

   305→   305→      {/* Icon */}

   313→   313→        className="flex-shrink-0 rounded p-1 text-zinc-500 hover:text-zinc-300 hover:bg-zinc-700/50 focus:outline-none"
   314→   314→        onClick={(e) => {
   315→   315→          e.stopPropagation();
   316→   316→          onToggleVisibility();

   320→   320→        {section.visible ? (

   329→   329→        className="flex-shrink-0 rounded p-1 text-zinc-600 opacity-0 group-hover:opacity-100 hover:text-red-400 hover:bg-red-400/10 transition-opacity focus:outline-none"

   396→   396→function ContentFieldRenderer({
   397→   397→  keyName,
   398→   398→  value,
   399→   399→  onChange,
   400→   400→  depth = 0,
   401→   401→}: {
   402→   402→  keyName: string;
   403→   403→  value: unknown;
   404→   404→  onChange: (key: string, value: unknown) => void;
   405→   405→  depth?: number;
   406→   406→}) {
   407→   407→  const label = keyName
   408→   408→    .replace(/([A-Z])/g, ' $1')
   409→   409→    .replace(/^./, (s) => s.toUpperCase())
   410→   410→    .trim();
   411→   411→
   412→   412→  if (value === null || value === undefined) {
   413→   413→    return null;
   414→   414→  }
   415→   415→
   416→   416→  if (typeof value === 'string') {
   417→   417→    if (keyName === 'html' && value.includes('<')) {
   418→   418→      return (
   419→   419→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   420→   420→          <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   421→   421→          <Textarea
   422→   422→            value={value}
   423→   423→            onChange={(e) => onChange(keyName, e.target.value)}
   424→   424→            rows={4}
   425→   425→            className="bg-zinc-800 border-zinc-700 text-zinc-100 text-xs font-mono resize-none"
   426→   426→          />
   427→   427→        </div>
   428→   428→      );
   429→   429→    }
   430→   430→    return (
   431→   431→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   432→   432→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   433→   433→        <Input
   434→   434→          value={value}
   435→   435→          onChange={(e) => onChange(keyName, e.target.value)}
   436→   436→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   437→   437→        />
   438→   438→      </div>
   439→   439→    );
   440→   440→  }
   441→   441→
   442→   442→  if (typeof value === 'number') {
   443→   443→    return (
   444→   444→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   445→   445→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   446→   446→        <Input
   447→   447→          type="number"
   448→   448→          value={value}
   449→   449→          onChange={(e) => onChange(keyName, Number(e.target.value))}
   450→   450→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   451→   451→        />
   452→   452→      </div>
   453→   453→    );
   454→   454→  }
   455→   455→
   456→   456→  if (typeof value === 'boolean') {
   457→   457→    return (
   458→   458→      <div
   459→   459→        key={keyName}
   460→   460→        className={`flex items-center justify-between ${depth > 0 ? 'ml-4' : ''}`}
   461→   461→      >
   462→   462→        <Label className="text-xs text-zinc-400">{label}</Label>
   463→   463→        <Switch
   464→   464→          checked={value}
   465→   465→          onCheckedChange={(checked) => onChange(keyName, checked)}
   466→   466→        />
   467→   467→      </div>
   468→   468→    );
   469→   469→  }
   470→   470→
   471→   471→  if (Array.isArray(value)) {
   472→   472→    return (
   473→   473→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   474→   474→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   475→   475→        <div className="rounded-lg bg-zinc-800/60 border border-zinc-700/50 px-3 py-2 text-xs text-zinc-500">
   476→   476→          {value.length} item{value.length !== 1 ? 's' : ''}
   477→   477→        </div>
   478→   478→      </div>
   479→   479→    );
   480→   480→  }
   481→   481→
   482→   482→  if (typeof value === 'object') {
   483→   483→    const obj = value as Record<string, unknown>;
   484→   484→    const keys = Object.keys(obj);
   485→   485→
   486→   486→    const allPrimitive = keys.every(
   487→   487→      (k) =>
   488→   488→        typeof obj[k] === 'string' ||
   489→   489→        typeof obj[k] === 'number' ||
   490→   490→        typeof obj[k] === 'boolean' ||
   491→   491→        obj[k] === null ||
   492→   492→        obj[k] === undefined
   493→   493→    );
   494→   494→
   495→   495→    if (allPrimitive && keys.length > 0) {
   496→   496→      return (
   497→   497→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   498→   498→          <p className="text-xs font-medium text-zinc-300 mb-2 mt-1">{label}</p>
   499→   499→          <div className="space-y-3 border-l border-zinc-700/50 pl-3 ml-1">
   500→   500→            {keys.map((k) => (
   501→   501→              <ContentFieldRenderer
   502→   502→                key={k}
   503→   503→                keyName={k}
   504→   504→                value={obj[k]}
   505→   505→                onChange={(subKey, subVal) => {
   506→   506→                  onChange(keyName, { ...obj, [subKey]: subVal });
   507→   507→                }}
   508→   508→                depth={depth + 1}
   509→   509→              />
   510→   510→            ))}
   511→   511→          </div>
   512→   512→        </div>
   513→   513→      );
   514→   514→    }
   515→   515→
   516→   516→    return (
   517→   517→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   518→   518→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   519→   519→        <Textarea
   520→   520→          value={JSON.stringify(value, null, 2)}
   521→   521→          readOnly
   522→   522→          rows={3}

   916→   916→    setSelectedSectionId,
   917→   917→    moveSection,
   918→   918→    addSection,
   919→   919→    removeSection,
   920→   920→  } = useStoreEditor();
   921→   921→
   922→   922→  // Get the current page (homepage for now)
   923→   923→  const currentPage = useMemo(
   924→   924→    () => store?.pages.find((p) => p.isHomepage) ?? store?.pages[0],
   925→   925→    [store?.pages]
   926→   926→  );
   927→   927→
   928→   928→  const sections = currentPage?.sections ?? [];
   929→   929→  const pageId = currentPage?.id ?? '';
   930→   930→
   931→   931→  const selectedSection = useMemo(
   932→   932→    () => sections.find((s) => s.id === selectedSectionId) ?? null,
   933→   933→    [sections, selectedSectionId]
   934→   934→  );
   935→   935→
   936→   936→  // Drag end handler
   937→   937→  const handleDragEnd = useCallback(
   938→   938→    (event: DragEndEvent) => {
   939→   939→      const { active, over } = event;
   940→   940→      if (!over || active.id === over.id || !pageId) return;
   941→   941→
   942→   942→      const oldIndex = sections.findIndex((s) => s.id === active.id);
   943→   943→      const newIndex = sections.findIndex((s) => s.id === over.id);
   944→   944→
   945→   945→      if (oldIndex !== -1 && newIndex !== -1 && oldIndex !== newIndex) {
   946→   946→        moveSection(pageId, oldIndex, newIndex);
   947→   947→      }
   948→   948→    },
   949→   949→    [sections, pageId, moveSection]
   950→   950→  );
   951→   951→
   952→   952→  // Add section handler
   953→   953→  const handleAddSection = useCallback(
   954→   954→    (type: SectionType) => {
   955→   955→      if (!pageId) return;
   956→   956→      const newSection = createDefaultSection(type);
   957→   957→      addSection(pageId, newSection);
   958→   958→      setSelectedSectionId(newSection.id);
   959→   959→      toast.success(`Added ${SECTION_TYPE_LABELS[type]} section`);
   960→   960→    },
   961→   961→    [pageId, addSection, setSelectedSectionId]
   962→   962→  );
   963→   963→
   964→   964→  // Toggle visibility for section list items
   965→   965→  const handleToggleVisibility = useCallback(
   966→   966→    (sectionId: string) => {
   967→   967→      if (!store) return;
   968→   968→      const updatedStore = {
   969→   969→        ...store,
   970→   970→        updatedAt: new Date().toISOString(),
   971→   971→        pages: store.pages.map((page) => ({
   972→   972→          ...page,
   973→   973→          sections: page.sections.map((s) =>
   974→   974→            s.id === sectionId ? { ...s, visible: !s.visible } : s
   975→   975→          ),
   976→   976→        })),
   977→   977→      };
   978→   978→      setStore(updatedStore);
   979→   979→    },
   980→   980→    [store, setStore]
   981→   981→  );
   982→   982→
   983→   983→  const handleDeleteSection = useCallback(
   984→   984→    (sectionId: string) => {
   985→   985→      if (!pageId) return;
   986→   986→      const section = sections.find((s) => s.id === sectionId);
   987→   987→      removeSection(pageId, sectionId);
   988→   988→      if (selectedSectionId === sectionId) {
   989→   989→        setSelectedSectionId(null);
   990→   990→      }
   991→   991→      if (section) {
   992→   992→        toast.success(`Deleted ${SECTION_TYPE_LABELS[section.type]} section`);
   993→   993→      }
   994→   994→    },
   995→   995→    [pageId, sections, selectedSectionId, removeSection, setSelectedSectionId]
   996→   996→  );
   997→   997→
   998→   998→  if (!store || !currentPage) {
   999→   999→    return (
  1000→  1000→      <div className="flex h-full items-center justify-center bg-zinc-950">
  1001→  1001→        <p className="text-sm text-zinc-600">No store loaded</p>
  1002→  1002→      </div>
  1003→  1003→    );
  1004→  1004→  }
  1005→  1005→
  1006→  1006→  return (
  1007→  1007→    <div className="flex h-full overflow-hidden bg-zinc-950">
