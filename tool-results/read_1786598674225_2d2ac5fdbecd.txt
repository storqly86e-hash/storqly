    49→// ─── Section Wrapper ────────────────────────────────────────────────────
    50→
    51→interface SectionWrapperProps {
    52→  section: Section;
    53→  theme: StoreTheme;
    54→  selectedSectionId?: string | null;
    55→  onSelectSection?: (sectionId: string | null) => void;
    56→  children: React.ReactNode;
    57→}
    58→
    59→function SectionWrapper({
    60→  section,
    61→  theme,
    62→  selectedSectionId,
    63→  onSelectSection,
    64→  children,
    65→}: SectionWrapperProps) {
    66→  const isSelected = selectedSectionId === section.id;
    67→  const style = section.style;
    68→
    69→  const sectionBg = style.backgroundColor || theme.colors.background;
    70→  const sectionText = style.textColor || theme.colors.text;
    71→
    72→  const bgImage = style.backgroundImage
    73→    ? `url(${style.backgroundImage})`
    74→    : undefined;
    75→
    76→  const outerClasses = [
    77→    'relative transition-all duration-200 cursor-pointer',
    78→    pyClass(style.paddingY),
    79→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
    80→  ].join(' ');
    81→
    82→  return (
    83→    <div
    84→      className={outerClasses}
    85→      style={{
    86→        backgroundColor: sectionBg,
    87→        color: sectionText,
    88→        backgroundImage: bgImage,
    89→        backgroundSize: 'cover',
    90→        backgroundPosition: 'center',
    91→      }}
    92→      onClick={(e) => {
    93→        e.stopPropagation();
    94→        onSelectSection?.(isSelected ? null : section.id);
    95→      }}
    96→      data-section-id={section.id}
    97→      data-section-type={section.type}
    98→    >
    99→      {/* Overlay for background images */}
   100→      {style.backgroundImage && style.overlay && (
   101→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   102→      )}
   103→      <div
   104→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   105→      >
   106→        {/* Selected section label */}
   107→        {isSelected && (
   108→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   109→            {section.type.replace(/-/g, ' ')}
   110→          </div>
   111→        )}
   112→        {children}
   113→      </div>
   114→    </div>
   115→  );
   116→}

   118→// ─── Product Card (shared) ─────────────────────────────────────────────
   119→
   120→function ProductCard({
   121→  product,
   122→  theme,
   123→  showAddToCart,
   124→  borderRadius,
   125→  buttonBgOverride,
   126→  buttonTextOverride,
   127→  onViewProduct,
   128→}: {
   129→  product: StoreProduct;
   130→  theme: StoreTheme;
   131→  showAddToCart: boolean;
   132→  borderRadius: string;
   133→  buttonBgOverride?: string;
   134→  buttonTextOverride?: string;
   135→  onViewProduct?: (productId: string) => void;
   136→}) {
   137→  const [hovered, setHovered] = useState(false);
   138→  const addToCart = useCartStore((s) => s.addItem);
   139→  const imgColor = stringToColor(product.id, theme);
   140→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   141→
   142→  return (
   143→    <div
   144→      className={`${borderRadius} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   145→      style={{
   146→        borderColor: theme.colors.border,
   147→      }}
   148→      onMouseEnter={() => setHovered(true)}
   149→      onMouseLeave={() => setHovered(false)}
   150→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   151→      role={onViewProduct ? 'button' : undefined}
   152→      tabIndex={onViewProduct ? 0 : undefined}
   153→    >
   154→ {/* Image placeholder */}
   155→      <div
   156→        className="relative aspect-square w-full overflow-hidden"
   157→        style={{ backgroundColor: imgColor }}
   158→      >
   159→        <div className="absolute inset-0 flex items-center justify-center">
   160→          <div
   161→            className="h-16 w-16 rounded-full opacity-30"
   162→            style={{ backgroundColor: theme.colors.primary }}
   163→          />
   164→        </div>
   165→        {/* Out of stock overlay */}
   166→        {!product.inStock && (
   167→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   168→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   169→              Sold Out
   170→            </span>
   171→          </div>
   172→        )}
   173→        {/* Discount badge */}
   174→        {hasDiscount && product.compareAtPrice && (
   175→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   176→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   177→          </div>
   178→        )}
   179→      </div>
   180→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   181→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   182→        {product.category && (
   183→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   184→            {product.category}
   185→          </p>
   186→        )}
   187→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   188→          {product.name}
   189→        </h3>
   190→        <div className="mt-1.5 flex items-center gap-2">
   191→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   192→            {formatPrice(product.price)}
   193→          </span>
   194→          {hasDiscount && product.compareAtPrice && (
   195→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   196→              {formatPrice(product.compareAtPrice)}
   197→            </span>
   198→          )}
   199→        </div>
   200→        {showAddToCart && product.inStock && (
   201→          <button
   202→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 ${
   203→              hovered
   204→                ? 'opacity-100 translate-y-0'
   205→                : 'opacity-0 translate-y-1'
   206→            }`}
   207→            style={{
   208→              backgroundColor: buttonBgOverride || theme.colors.primary,
   209→              color: buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary),
   210→            }}
   211→            onClick={(e) => {
   212→              e.stopPropagation();
   213→              addToCart(product);
   214→            }}
   215→          >
   216→            Add to Cart
   217→          </button>
   218→        )}
   219→      </div>
   220→    </div>
   221→  );
   222→}

   319→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   320→  const content = section.content as unknown as HeroContent;
   321→  const style = section.style;
   322→  const isSelected = selectedSectionId === section.id;
   323→
   324→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   325→  const alignMap = {
   326→    left: 'items-start text-left',
   327→    center: 'items-center text-center',
   328→    right: 'items-end text-right',
   329→  };
   330→
   331→  // Only apply theme gradient as default background when no custom color or image is set.
   332→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   333→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   334→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   335→    : undefined;
   336→
   337→  const outerClasses = [
   338→    'relative flex transition-all duration-200 cursor-pointer',
   339→    heightMap[content.height] || heightMap.md,
   340→    alignMap[content.alignment],
   341→    pyClass(style.paddingY),
   342→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   343→  ].join(' ');
   344→
   345→  return (
   346→    <div
   347→      className={outerClasses}
   348→      style={{
   349→        backgroundColor: style.backgroundColor || theme.colors.primary,
   350→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   351→        backgroundSize: 'cover',
   352→        backgroundPosition: 'center',
   353→        color: style.textColor || '#ffffff',
   354→      }}
   355→      onClick={(e) => {
   356→        e.stopPropagation();
   357→        onSelectSection?.(isSelected ? null : section.id);
   358→      }}
   359→      data-section-id={section.id}
   360→      data-section-type={section.type}
   361→    >
   362→      {/* Overlay */}
   363→      {style.backgroundImage && style.overlay && (
   364→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   365→      )}
   366→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   367→      {!style.backgroundImage && !style.backgroundColor && (
   368→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   369→      )}
   370→      {!style.backgroundImage && !style.backgroundColor && (
   371→        <>
   372→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   373→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />
   374→        </>
   375→      )}
   376→
   377→      <div className={`relative z-10 mx-auto w-full ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}>
   378→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   379→          {content.headline}
   380→        </h1>
   381→        {content.subheadline && (
   382→          <p className="mt-4 max-w-xl text-base opacity-80 sm:text-lg md:text-xl">
   383→            {content.subheadline}
   384→          </p>
   385→        )}
   386→        {content.ctaText && (
   387→          <button className="mt-8 inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-transform hover:scale-105 sm:text-base"
   388→            style={{
   389→              backgroundColor: section.style.buttonBackgroundColor || '#ffffff',
   390→              color: section.style.buttonTextColor || contrastTextColor(section.style.buttonBackgroundColor || '#ffffff'),
   391→            }}>
   392→            {content.ctaText}
   393→            <ArrowRight className="h-4 w-4" />
   394→          </button>
   395→        )}
   396→      </div>
   397→    </div>
   398→  );
   399→}

   403→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct }: SectionRendererProps) {
   404→  const content = section.content as unknown as FeaturedProductsContent;
   405→  const borderRadius = borderRadiusClass(theme.borderRadius);
   406→  const featuredProducts = useMemo(
   407→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   408→    [content.productIds, products],
   409→  );
   410→
   411→  return (
   412→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   413→      {content.headline && (
   414→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   415→          {content.headline}
   416→        </h2>
   417→      )}
   418→      {content.subtitle && (
   419→        <p className="mb-8 text-sm opacity-65">
   420→          {content.subtitle}
   421→        </p>
   422→      )}
   423→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   424→        {featuredProducts.map((product) => (
   425→          <ProductCard
   426→            key={product.id}
   427→            product={product}
   428→            theme={theme}
   429→            showAddToCart={content.showAddToCart}
   430→            borderRadius={borderRadius}
   431→            buttonBgOverride={section.style.buttonBackgroundColor}
   432→            buttonTextOverride={section.style.buttonTextColor}
   433→            onViewProduct={onViewProduct}
   434→          />
   435→        ))}
   436→      </div>
   437→    </SectionWrapper>
   438→  );
   439→}

   441→// ─── 4. Product Grid ───────────────────────────────────────────────────
   442→
   443→export function ProductGridSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct }: SectionRendererProps) {
   444→  const content = section.content as unknown as ProductGridContent;
   445→  const borderRadius = borderRadiusClass(theme.borderRadius);
   446→
   447→  const filteredProducts = useMemo(() => {
   448→    let prods = products;
   449→    if (content.filterByCategory) {
   450→      prods = prods.filter((p) => p.category === content.filterByCategory);
   451→    }
   452→    return prods;
   453→  }, [products, content.filterByCategory]);
   454→
   455→  return (
   456→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   457→      {content.headline && (
   458→        <h2 className="mb-8 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   459→          {content.headline}
   460→        </h2>
   461→      )}
   462→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   463→        {filteredProducts.map((product) => (
   464→          <ProductCard
   465→            key={product.id}
   466→            product={product}
   467→            theme={theme}
   468→            showAddToCart={content.showAddToCart}
   469→            borderRadius={borderRadius}
   470→            buttonBgOverride={section.style.buttonBackgroundColor}
   471→            buttonTextOverride={section.style.buttonTextColor}
   472→            onViewProduct={onViewProduct}
   473→          />
   474→        ))}
   475→      </div>
   476→      {filteredProducts.length === 0 && (
   477→        <div className="py-16 text-center opacity-65">
   478→          <p className="text-sm">No products found.</p>
   479→        </div>
   480→      )}
   481→    </SectionWrapper>
   482→  );
   483→}

   487→export function TextBannerSection({ section, theme, selectedSectionId, onSelectSection }: SectionRendererProps) {
   488→  const content = section.content as unknown as TextBannerContent;
   489→  const alignMap = {
   490→    left: 'text-left',
   491→    center: 'text-center',
   492→    right: 'text-right',
   493→  };
   494→  const sizeMap = {
   495→    sm: { headline: 'text-xl sm:text-2xl', body: 'text-sm' },
   496→    md: { headline: 'text-2xl sm:text-3xl md:text-4xl', body: 'text-base' },
   497→    lg: { headline: 'text-3xl sm:text-4xl md:text-5xl lg:text-6xl', body: 'text-lg' },
   498→  };
   499→  const sizes = sizeMap[content.size] || sizeMap.md;
   500→
   501→  return (
   502→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   503→      <div className={`${alignMap[content.alignment]}`}>
   504→        <h2 className={`font-bold leading-tight tracking-tight ${sizes.headline}`} style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   505→          {content.headline}
   506→        </h2>
   507→        {content.body && (
   508→          <p className={`mt-3 max-w-2xl ${sizes.body} opacity-65`}>
   509→            {content.body}
   510→          </p>
   511→        )}
   512→      </div>
   513→    </SectionWrapper>
   514→  );
   515→}

   974→// ─── Section Renderer Props (shared) ────────────────────────────────────

   976→interface SectionRendererProps {
   977→  section: Section;
   978→  theme: StoreTheme;
   979→  selectedSectionId?: string | null;
   980→  onSelectSection?: (sectionId: string | null) => void;
   981→  products: StoreProduct[];

   986→// ─── Main Section Router ────────────────────────────────────────────────
   987→
   988→export function renderSection(props: SectionRendererProps): React.ReactNode {
   989→  const { section } = props;
   990→
   991→  if (!section.visible) return null;
   992→
   993→  switch (section.type) {
   994→    case 'header':
   995→      return <HeaderSection {...props} />;
   996→    case 'hero':
   997→      return <HeroSection {...props} />;
   998→    case 'featured-products':
   999→      return <FeaturedProductsSection {...props} />;
  1000→    case 'product-grid':
  1001→      return <ProductGridSection {...props} />;
  1002→    case 'text-banner':
  1003→      return <TextBannerSection {...props} />;
  1004→    case 'image-gallery':
  1005→      return <ImageGallerySection {...props} />;
  1006→    case 'testimonials':
  1007→      return <TestimonialsSection {...props} />;
  1008→    case 'newsletter':
  1009→      return <NewsletterSection {...props} />;
  1010→    case 'faq':
  1011→      return <FAQSection {...props} />;
  1012→    case 'cta':
  1013→      return <CTASection {...props} />;
  1014→    case 'categories':
  1015→      return <CategoriesSection {...props} />;
  1016→    case 'footer':
  1017→      return <FooterSection {...props} />;
  1018→    case 'rich-text':
  1019→      return <RichTextSection {...props} />;
  1020→    case 'spacer':
  1021→      return <SpacerSection {...props} />;
  1022→    case 'divider':
  1023→      return <DividerSection {...props} />;
  1024→    default:
  1025→      return null;
  1026→  }
  1027→}
