    49→// ─── Shared Image Component with onError Fallback ──────────────
    50→// Renders <img> with automatic fallback to a colored placeholder.
    51→// Used everywhere images appear: ProductCard, CollectionPage, ProductDetail, Cart, Gallery, Categories.
    52→
    53→function StoreImage({
    54→  src,
    55→  alt,
    56→  fallbackColor,
    57→  className = '',
    58→  iconSize = 'md',
    59→}: {
    60→  src: string;
    61→  alt: string;
    62→  fallbackColor: string;
    63→  className?: string;
    64→  /** Size of the placeholder icon: sm=6, md=10, lg=16 (in rem units /4) */
    65→  iconSize?: 'sm' | 'md' | 'lg';
    66→}) {
    67→  const [failed, setFailed] = useState(false);
    68→  const handleError = useCallback(() => setFailed(true), []);
    69→
    70→  const iconPx = iconSize === 'sm' ? 'h-6 w-6' : iconSize === 'lg' ? 'h-16 w-16' : 'h-10 w-10';
    71→
    72→  if (!src || failed) {
    73→    return (
    74→      <div
    75→        className={`flex items-center justify-center ${className}`}
    76→        style={{ backgroundColor: fallbackColor }}
    77→        aria-label={alt}
    78→        role='img'
    79→      >
    80→        <svg className={`${iconPx} opacity-20`} fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={1.5}>
    81→          <path strokeLinecap="round" strokeLinejoin="round" d="M2.25 15.75l5.159-5.159a2.25 2.25 0 013.182 0l5.159 5.159m-1.5-1.5l1.409-1.409a2.25 2.25 0 013.182 0l2.909 2.909M3.75 21h16.5A2.25 2.25 0 0022.5 18.75V5.25A2.25 2.25 0 0020.25 3H3.75A2.25 2.25 0 001.5 5.25v13.5A2.25 2.25 0 003.75 21z" />
    82→        </svg>
    83→      </div>
    84→    );
    85→  }
    86→
    87→  return (
    88→    <img
    89→      src={src}
    90→      alt={alt}
    91→      className={className}
    92→      onError={handleError}
    93→      loading="lazy"
    94→    />
    95→  );
    96→}

    99→// ─── Section Wrapper ────────────────────────────────────────────────────
   100→
   101→interface SectionWrapperProps {
   102→  section: Section;
   103→  theme: StoreTheme;
   104→  selectedSectionId?: string | null;
   105→  onSelectSection?: (sectionId: string | null) => void;
   106→  children: React.ReactNode;
   107→}
   108→
   109→function SectionWrapper({
   110→  section,
   111→  theme,
   112→  selectedSectionId,
   113→  onSelectSection,
   114→  children,
   115→}: SectionWrapperProps) {
   116→  const isSelected = selectedSectionId === section.id;
   117→  const style = section.style;
   118→
   119→  const sectionBg = style.backgroundColor || theme.colors.background;
   120→  const sectionText = style.textColor || theme.colors.text;
   121→
   122→  const bgImage = style.backgroundImage
   123→    ? `url(${style.backgroundImage})`
   124→    : undefined;
   125→
   126→  const outerClasses = [
   127→    'relative transition-all duration-200 cursor-pointer',
   128→    pyClass(style.paddingY),
   129→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   130→  ].join(' ');
   131→
   132→  return (
   133→    <div
   134→      className={outerClasses}
   135→      style={{
   136→        backgroundColor: sectionBg,
   137→        color: sectionText,
   138→        backgroundImage: bgImage,
   139→        backgroundSize: 'cover',
   140→        backgroundPosition: 'center',
   141→      }}
   142→      onClick={(e) => {
   143→        e.stopPropagation();
   144→        onSelectSection?.(isSelected ? null : section.id);
   145→      }}
   146→      data-section-id={section.id}
   147→      data-section-type={section.type}
   148→    >
   149→      {/* Overlay for background images */}
   150→      {style.backgroundImage && style.overlay && (
   151→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   152→      )}
   153→      <div
   154→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   155→      >
   156→        {/* Selected section label */}
   157→        {isSelected && (
   158→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   159→            {section.type.replace(/-/g, ' ')}
   160→          </div>
   161→        )}
   162→        {children}
   163→      </div>
   164→    </div>
   165→  );
   166→}

   170→function ProductCard({
   171→  product,
   172→  theme,
   173→  showAddToCart,
   174→  borderRadius,
   175→  buttonBgOverride,
   176→  buttonTextOverride,
   177→  onViewProduct,
   178→}: {
   179→  product: StoreProduct;
   180→  theme: StoreTheme;
   181→  showAddToCart: boolean;
   182→  borderRadius: string;
   183→  buttonBgOverride?: string;
   184→  buttonTextOverride?: string;
   185→  onViewProduct?: (productId: string) => void;
   186→}) {
   187→  const [hovered, setHovered] = useState(false);
   188→  const [addedFeedback, setAddedFeedback] = useState(false);
   189→  const addToCart = useCartStore((s) => s.addItem);
   190→  const imgColor = stringToColor(product.id, theme);
   191→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   192→
   193→  const handleAddToCart = (e: React.MouseEvent) => {
   194→    e.stopPropagation();
   195→    addToCart(product);
   196→    setAddedFeedback(true);
   197→    setTimeout(() => setAddedFeedback(false), 1200);
   198→  };
   199→
   200→  return (
   201→    <div
   202→      className={`${borderRadius} ${onViewProduct ? 'cursor-pointer' : ''} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   203→      style={{
   204→        borderColor: theme.colors.border,
   205→      }}
   206→      onMouseEnter={() => setHovered(true)}
   207→      onMouseLeave={() => setHovered(false)}
   208→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   209→      role={onViewProduct ? 'button' : undefined}
   210→      tabIndex={onViewProduct ? 0 : undefined}
   211→    >
   212→ {/* Product image with real image + fallback */
   213→      <div
   214→        className="relative aspect-square w-full overflow-hidden"
   215→        style={{ backgroundColor: imgColor }}
   216→      >
   217→        <StoreImage
   218→          src={product.images[0] || ''}
   219→          alt={product.name}
   220→          fallbackColor={imgColor}
   221→          className="absolute inset-0 h-full w-full object-cover transition-transform duration-300 group-hover:scale-105"
   222→          iconSize="lg"
   223→        />
   224→        {/* Out of stock overlay */}
   225→        {!product.inStock && (
   226→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   227→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   228→              Sold Out
   229→            </span>
   230→          </div>
   231→        )}
   232→        {/* Discount badge */}
   233→        {hasDiscount && product.compareAtPrice && (
   234→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   235→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   236→          </div>
   237→        )}
   238→      </div>
   239→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   240→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   241→        {product.category && (
   242→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   243→            {product.category}
   244→          </p>
   245→        )}
   246→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   247→          {product.name}
   248→        </h3>
   249→        <div className="mt-1.5 flex items-center gap-2">
   250→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   251→            {formatPrice(product.price)}
   252→          </span>
   253→          {hasDiscount && product.compareAtPrice && (
   254→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   255→              {formatPrice(product.compareAtPrice)}
   256→            </span>
   257→          )}
   258→        </div>
   259→        {showAddToCart && product.inStock && (
   260→          <button
   261→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 cursor-pointer ${
   262→              hovered
   263→                ? 'opacity-100 translate-y-0'
   264→                : 'opacity-0 translate-y-1 pointer-events-none'
   265→            }`}
   266→            style={{
   267→              backgroundColor: addedFeedback ? '#16a34a' : (buttonBgOverride || theme.colors.primary),
   268→              color: addedFeedback ? '#ffffff' : (buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary)),
   269→            }}
   270→            onClick={handleAddToCart}
   271→          >
   272→            {addedFeedback ? 'Added!' : 'Add to Cart'}
   273→          </button>
   274→        )}
   275→      </div>
   276→    </div>
   277→  );
   278→}

   375→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   376→  const content = section.content as unknown as HeroContent;
   377→  const style = section.style;
   378→  const isSelected = selectedSectionId === section.id;
   379→
   380→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   381→  const alignMap = {
   382→    left: 'items-start text-left',
   383→    center: 'items-center text-center',
   384→    right: 'items-end text-right',
   385→  };
   386→
   387→  // Only apply theme gradient as default background when no custom color or image is set.
   388→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   389→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   390→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   391→    : undefined;
   392→
   393→  const outerClasses = [
   394→    'relative flex transition-all duration-200 cursor-pointer',
   395→    heightMap[content.height] || heightMap.md,
   396→    alignMap[content.alignment],
   397→    pyClass(style.paddingY),
   398→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   399→  ].join(' ');
   400→
   401→  return (
   402→    <div
   403→      className={outerClasses}
   404→      style={{
   405→        backgroundColor: style.backgroundColor || theme.colors.primary,
   406→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   407→        backgroundSize: 'cover',
   408→        backgroundPosition: 'center',
   409→        color: style.textColor || '#ffffff',
   410→      }}
   411→      onClick={(e) => {
   412→        e.stopPropagation();
   413→        onSelectSection?.(isSelected ? null : section.id);
   414→      }}
   415→      data-section-id={section.id}
   416→      data-section-type={section.type}
   417→    >
   418→      {/* Overlay */}
   419→      {style.backgroundImage && style.overlay && (
   420→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   421→      )}
   422→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   423→      {!style.backgroundImage && !style.backgroundColor && (
   424→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   425→      )}
   426→      {!style.backgroundImage && !style.backgroundColor && (
   427→        <>
   428→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   429→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />
   430→        </>
   431→      )}
   432→
   433→      <div className={`relative z-10 mx-auto w-full ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}>
   434→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   435→          {content.headline}
   436→        </h1>

   438→          <p className="mt-4 max-w-xl text-base opacity-80 sm:text-lg md:text-xl">
   439→            {content.subheadline}

   443→          <button className="mt-8 inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-transform hover:scale-105 sm:text-base"
   444→            style={{
   445→              backgroundColor: section.style.buttonBackgroundColor || '#ffffff',

   475→        <p className="mb-8 text-sm opacity-65">
   476→          {content.subtitle}
   477→        </p>
   478→      )}
   479→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>

   482→            key={product.id}

   499→export function ProductGridSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   500→  const content = section.content as unknown as ProductGridContent;
   501→  const borderRadius = borderRadiusClass(theme.borderRadius);
   502→
   503→  const filteredProducts = useMemo(() => {
   504→    let prods = products;
   505→    if (content.filterByCategory) {
   506→      prods = prods.filter((p) => p.category === content.filterByCategory);
   507→    }
   508→    return prods;
   509→  }, [products, content.filterByCategory]);
   510→
   511→  return (
   512→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   513→      {content.headline && (
   514→        <h2 className="mb-8 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   515→          {content.headline}
   516→        </h2>
   517→      )}
   518→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   519→        {filteredProducts.map((product) => (
   520→          <ProductCard
   521→            key={product.id}
   522→            product={product}
   523→            theme={theme}
   524→            showAddToCart={content.showAddToCart && !forceHideAddToCart}
   525→            borderRadius={borderRadius}
   526→            buttonBgOverride={section.style.buttonBackgroundColor}
   527→            buttonTextOverride={section.style.buttonTextColor}
   528→            onViewProduct={onViewProduct}
   529→          />
   530→        ))}
   531→      </div>
   532→      {filteredProducts.length === 0 && (
   533→        <div className="py-16 text-center opacity-65">
   534→          <p className="text-sm">No products found.</p>
   535→        </div>
   536→      )}
   537→    </SectionWrapper>
   538→  );
   539→}

   543→export function TextBannerSection({ section, theme, selectedSectionId, onSelectSection }: SectionRendererProps) {
   544→  const content = section.content as unknown as TextBannerContent;
   545→  const alignMap = {
   546→    left: 'text-left',
   547→    center: 'text-center',
   548→    right: 'text-right',
   549→  };
   550→  const sizeMap = {
   551→    sm: { headline: 'text-xl sm:text-2xl', body: 'text-sm' },
   552→    md: { headline: 'text-2xl sm:text-3xl md:text-4xl', body: 'text-base' },
   553→    lg: { headline: 'text-3xl sm:text-4xl md:text-5xl lg:text-6xl', body: 'text-lg' },
   554→  };
   555→  const sizes = sizeMap[content.size] || sizeMap.md;

  1045→// ─── Main Section Router ────────────────────────────────────────────────
  1046→
  1047→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1048→  const { section } = props;
  1049→
  1050→  if (!section.visible) return null;
  1051→
  1052→  switch (section.type) {
  1053→    case 'header':
  1054→      return <HeaderSection {...props} />;
  1055→    case 'hero':
  1056→      return <HeroSection {...props} />;
  1057→    case 'featured-products':
  1058→      return <FeaturedProductsSection {...props} />;
  1059→    case 'product-grid':
  1060→      return <ProductGridSection {...props} />;
  1061→    case 'text-banner':
  1062→      return <TextBannerSection {...props} />;
  1063→    case 'image-gallery':
  1064→      return <ImageGallerySection {...props} />;
  1065→    case 'testimonials':
  1066→      return <TestimonialsSection {...props} />;
  1067→    case 'newsletter':
  1068→      return <NewsletterSection {...props} />;
  1069→    case 'faq':
  1070→      return <FAQSection {...props} />;
  1071→    case 'cta':
  1072→      return <CTASection {...props} />;
  1073→    case 'categories':
