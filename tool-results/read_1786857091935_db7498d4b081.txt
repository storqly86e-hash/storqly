   259→function SortableSectionItem({
   260→  section,
   261→  isSelected,
   262→  onSelect,
   263→  onToggleVisibility,
   264→  onDelete,
   265→}: SortableSectionItemProps) {
   266→  const {
   267→    attributes,
   268→    listeners,
   269→    setNodeRef,
   270→    transform,
   271→    transition,
   272→    isDragging,
   273→  } = useSortable({ id: section.id });
   274→
   275→  const style = {
   276→    transform: CSS.Transform.toString(transform),
   277→    transition,
   278→  };
   279→
   280→  const Icon = SECTION_TYPE_ICONS[section.type];
   281→  const label = SECTION_TYPE_LABELS[section.type];
   282→
   283→  return (
   284→    <div
   285→      ref={setNodeRef}
   286→      style={style}
   287→      className={[
   288→        'group flex items-center gap-2 rounded-lg px-3 py-2 text-sm cursor-pointer transition-colors',
   289→        isDragging
   290→          ? 'z-50 bg-zinc-700/80 shadow-xl ring-1 ring-white/10'
   291→          : isSelected
   292→            ? 'bg-amber-500/15 text-amber-200 ring-1 ring-amber-500/40'
   293→            : 'text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200',
   294→      ].join(' ')}
   295→      onClick={onSelect}
   296→    >
   297→      {/* Drag handle */}
   298→      <button
   299→        className="flex-shrink-0 cursor-grab text-zinc-600 hover:text-zinc-400 active:cursor-grabbing focus:outline-none"
   300→        {...attributes}
   301→        {...listeners}
   302→        onClick={(e) => e.stopPropagation()}

   305→        <GripVertical className="h-4 w-4" />

   307→

   309→      <Icon className="h-4 w-4 flex-shrink-0 opacity-60" />
   310→
   311→      {/* Label */}
   312→      <span className="flex-1 truncate font-medium">{label}</span>

   397→// ── Content Field Renderer ───────────────────────────────────────────
   398→
   399→function ContentFieldRenderer({
   400→  keyName,
   401→  value,
   402→  onChange,
   403→  depth = 0,
   404→}: {
   405→  keyName: string;
   406→  value: unknown;
   407→  onChange: (key: string, value: unknown) => void;
   408→  depth?: number;
   409→}) {
   410→  const label = keyName
   411→    .replace(/([A-Z])/g, ' $1')
   412→    .replace(/^./, (s) => s.toUpperCase())
   413→    .trim();
   414→
   415→  if (value === null || value === undefined) {
   416→    return null;
   417→  }
   418→
   419→  if (typeof value === 'string') {
   420→    if (keyName === 'html' && value.includes('<')) {
   421→      return (
   422→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   423→          <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   424→          <Textarea
   425→            value={value}
   426→            onChange={(e) => onChange(keyName, e.target.value)}
   427→            rows={4}
   428→            className="bg-zinc-800 border-zinc-700 text-zinc-100 text-xs font-mono resize-none"
   429→          />
   430→        </div>
   431→      );
   432→    }
   433→    return (
   434→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   435→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   436→        <Input
   437→          value={value}
   438→          onChange={(e) => onChange(keyName, e.target.value)}
   439→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   440→        />
   441→      </div>
   442→    );
   443→  }
   444→
   445→  if (typeof value === 'number') {
   446→    return (
   447→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   448→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   449→        <Input
   450→          type="number"
   451→          value={value}
   452→          onChange={(e) => onChange(keyName, Number(e.target.value))}
   453→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   454→        />
   455→      </div>
   456→    );
   457→  }
   458→
   459→  if (typeof value === 'boolean') {
   460→    return (
   461→      <div
   462→        key={keyName}
   463→        className={`flex items-center justify-between ${depth > 0 ? 'ml-4' : ''}`}
   464→      >
   465→        <Label className="text-xs text-zinc-400">{label}</Label>
   466→        <Switch
   467→          checked={value}
   468→          onCheckedChange={(checked) => onChange(keyName, checked)}
   469→        />
   470→      </div>
   471→    );
   472→  }
   473→
   474→  if (Array.isArray(value)) {
   475→    return (
   476→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   477→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   478→        <div className="rounded-lg bg-zinc-800/60 border border-zinc-700/50 px-3 py-2 text-xs text-zinc-500">
   479→          {value.length} item{value.length !== 1 ? 's' : ''}
   480→        </div>
   481→      </div>
   482→    );
   483→  }
   484→
   485→  if (typeof value === 'object') {
   486→    const obj = value as Record<string, unknown>;
   487→    const keys = Object.keys(obj);
   488→
   489→    const allPrimitive = keys.every(
   490→      (k) =>
   491→        typeof obj[k] === 'string' ||
   492→        typeof obj[k] === 'number' ||
   493→        typeof obj[k] === 'boolean' ||
   494→        obj[k] === null ||
   495→        obj[k] === undefined
   496→    );
   497→
   498→    if (allPrimitive && keys.length > 0) {
   499→      return (
   500→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   501→          <p className="text-xs font-medium text-zinc-300 mb-2 mt-1">{label}</p>
   502→          <div className="space-y-3 border-l border-zinc-700/50 pl-3 ml-1">
   503→            {keys.map((k) => (
   504→              <ContentFieldRenderer
   505→                key={k}
   506→                keyName={k}
   507→                value={obj[k]}
   508→                onChange={(subKey, subVal) => {
   509→                  onChange(keyName, { ...obj, [subKey]: subVal });
   510→                }}
   511→                depth={depth + 1}
   512→              />
   513→            ))}
   514→          </div>
   515→        </div>
   516→      );
   517→    }
   518→
   519→    return (
   520→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   521→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   522→        <Textarea
   523→          value={JSON.stringify(value, null, 2)}
   524→          readOnly
   525→          rows={3}
   526→          className="bg-zinc-800/60 border-zinc-700/50 text-zinc-500 text-xs font-mono resize-none"
   527→        />
   528→      </div>
   529→    );
   530→  }
   531→
   532→  return null;
   533→}

   542→function PropertiesPanel({ section, pageId }: PropertiesPanelProps) {
   543→  const {
   544→    updateSectionContent,
   545→    updateSectionStyle,
   546→    removeSection,
   547→    setSelectedSectionId,
   548→    store,
   549→    setStore,
   550→  } = useStoreEditor();
   551→
   552→  const [localStyle, setLocalStyle] = useState<SectionStyle>(section.style);
   553→
   554→  // Sync style from store when section changes
   555→  if (section.style !== localStyle) {
   556→    setLocalStyle(section.style);
   557→  }
   558→
   559→  const handleContentChange = useCallback(
   560→    (key: string, value: unknown) => {
   561→      updateSectionContent(pageId, section.id, { [key]: value });
   562→    },
   563→    [pageId, section.id, updateSectionContent]
   564→  );
   565→
   566→  const handleStyleChange = useCallback(
   567→    (patch: Partial<SectionStyle>) => {
   568→      const updated = { ...localStyle, ...patch };
   569→      setLocalStyle(updated);
   570→      updateSectionStyle(pageId, section.id, patch);
   571→    },
   572→    [pageId, section.id, localStyle, updateSectionStyle]
   573→  );
   574→
   575→  const handleDelete = useCallback(() => {
   576→    removeSection(pageId, section.id);
   577→    setSelectedSectionId(null);
   578→    toast.success('Section deleted');
   579→  }, [pageId, section.id, removeSection, setSelectedSectionId]);
   580→
   581→  const handleVisibilityToggle = useCallback(() => {
   582→    if (!store) return;
   583→    const updatedStore = {
   584→      ...store,
   585→      updatedAt: new Date().toISOString(),
   586→      pages: store.pages.map((page) => ({
   587→        ...page,
   588→        sections: page.sections.map((s) =>
   589→          s.id === section.id ? { ...s, visible: !s.visible } : s
   590→        ),
   591→      })),
   592→    };
   593→    setStore(updatedStore);
   594→  }, [store, section.id, setStore]);
   595→
   596→  const Icon = SECTION_TYPE_ICONS[section.type];
   597→  const typeLabel = SECTION_TYPE_LABELS[section.type];
   598→
   599→  const renderSelect = (
   600→    lbl: string,
   601→    value: string | undefined,
   602→    options: { label: string; value: string }[],
   603→    onChange: (val: string) => void
   604→  ) => (
   605→    <div>
   606→      <Label className="text-xs text-zinc-400 mb-1.5 block">{lbl}</Label>
   607→      <Select value={value ?? ''} onValueChange={onChange}>
   608→        <SelectTrigger className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm">
   609→          <SelectValue />
   610→        </SelectTrigger>
   611→        <SelectContent className="bg-zinc-900 border-zinc-700">
   612→          {options.map((opt) => (
   613→            <SelectItem
   614→              key={opt.value}
   615→              value={opt.value}
   616→              className="text-zinc-300 focus:bg-zinc-800 focus:text-zinc-100"
   617→            >
   618→              {opt.label}
   619→            </SelectItem>
   620→          ))}
   621→        </SelectContent>
   622→      </Select>
   623→    </div>
   624→  );
   625→
   626→  return (
   627→    <div className="flex h-full flex-col min-h-0">
   628→      {/* Header */}
   629→      <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-3">
   630→        <div className="flex items-center gap-2">
   631→          <Settings2 className="h-4 w-4 text-zinc-500" />
   632→          <span className="text-sm font-semibold text-zinc-200">Properties</span>
   633→        </div>
   634→        <button
   635→          className="rounded-md p-1 text-zinc-500 hover:text-zinc-300 hover:bg-zinc-800 focus:outline-none"
   636→          onClick={() => setSelectedSectionId(null)}
   637→          aria-label="Close properties"
   638→        >
   639→          <X className="h-4 w-4" />
   640→        </button>

   933→
   934→export function VisualEditor() {

   940→    moveSection,
   941→    addSection,
   942→    removeSection,

   945→    addCustomPage,
   946→    removeCustomPage,
   947→    renameCustomPage,
   948→  } = useStoreEditor();
   949→
   950→  // Pages available for tab switching (real pages only, no dynamic product pages)
   951→  // Includes custom pages (section-editable like Home)
   952→  const editorPages = useMemo(
   953→    () => (store?.pages.filter((p) => !p.type || p.type === 'home' || p.type === 'custom' || p.type === 'collection' || p.type === 'cart' || p.type === 'checkout') ?? []),
   954→    [store?.pages]
   955→  );
   956→
   957→  // Custom pages only (for delete/rename)
   958→  const customPages = useMemo(
   959→    () => (store?.pages.filter((p) => p.type === 'custom') ?? []),
   960→    [store?.pages]
   961→  );
   962→
   963→  // State for inline rename input
   964→  const [renamingPageId, setRenamingPageId] = useState<string | null>(null);
   965→  const [renameValue, setRenameValue] = useState('');
   966→
   967→  // State for add-page dialog
   968→  const [showAddPageDialog, setShowAddPageDialog] = useState(false);
   969→  const [newPageName, setNewPageName] = useState('');
   970→
   971→  // Handle adding a new custom page
   972→  const handleAddPage = useCallback(() => {
   973→    const name = newPageName.trim();
   974→    if (!name) return;
   975→    const newId = addCustomPage(name);
   976→    if (newId) {
   977→      toast.success(`Added "${name}" page`);
   978→      setNewPageName('');
   979→      setShowAddPageDialog(false);
   980→    }
   981→  }, [newPageName, addCustomPage]);
   982→
   983→  // Handle deleting a custom page
   984→  const handleDeletePage = useCallback((pageId: string, pageName: string) => {
   985→    removeCustomPage(pageId);
   986→    toast.success(`Deleted "${pageName}" page`);
   987→  }, [removeCustomPage]);
   988→
   989→  // Handle starting rename
   990→  const handleStartRename = useCallback((pageId: string, currentName: string) => {
   991→    setRenamingPageId(pageId);
   992→    setRenameValue(currentName);
   993→  }, []);
   994→
   995→  // Handle committing rename
   996→  const handleCommitRename = useCallback(() => {
   997→    if (!renamingPageId || !renameValue.trim()) {
   998→      setRenamingPageId(null);
   999→      return;
  1000→    }
  1001→    renameCustomPage(renamingPageId, renameValue.trim());
  1002→    toast.success(`Renamed to "${renameValue.trim()}"`);
  1003→    setRenamingPageId(null);
  1004→  }, [renamingPageId, renameValue, renameCustomPage]);
  1005→
  1006→  // Current page for section editing (only home-type pages have editable sections)
  1007→  const currentPage = useMemo(
  1008→    () => store?.pages.find((p) => p.id === editorCurrentPageId) ?? store?.pages.find((p) => p.isHomepage) ?? store?.pages[0],
  1009→    [store?.pages, editorCurrentPageId]
  1010→  );
  1011→
  1012→  // Whether the currently selected page is a template (non-editable via visual editor)
  1013→  // Custom pages are section-editable (like home), only collection/cart/checkout/product are templates
  1014→  const isTemplatePage = currentPage && currentPage.type && currentPage.type !== 'home' && currentPage.type !== 'custom';
  1015→

  1017→  const pageId = currentPage?.id ?? '';
  1018→
  1019→  const selectedSection = useMemo(
  1020→    () => sections.find((s) => s.id === selectedSectionId) ?? null,
  1021→    [sections, selectedSectionId]

  1026→    (event: DragEndEvent) => {

  1028→      if (!over || active.id === over.id || !pageId) return;
  1029→
  1030→      const oldIndex = sections.findIndex((s) => s.id === active.id);
  1031→      const newIndex = sections.findIndex((s) => s.id === over.id);
  1032→
  1033→      if (oldIndex !== -1 && newIndex !== -1 && oldIndex !== newIndex) {
  1034→        moveSection(pageId, oldIndex, newIndex);
  1035→      }
  1036→    },
  1037→    [sections, pageId, moveSection]
  1038→  );
  1039→
  1040→  // Add section handler
  1041→  const handleAddSection = useCallback(
  1042→    (type: SectionType) => {
  1043→      if (!pageId) return;
  1044→      const newSection = createDefaultSection(type);

  1062→            s.id === sectionId ? { ...s, visible: !s.visible } : s

  1064→        })),
  1065→      };
  1066→      setStore(updatedStore);
  1067→    },

  1081→      }

  1093→    },
  1094→    [setEditorCurrentPageId, renamingPageId]
  1095→  );
  1096→
  1097→  if (!store || !currentPage) {
  1098→    return (
  1099→      <div className="flex h-full items-center justify-center bg-zinc-950">
  1100→        <p className="text-sm text-zinc-600">No store loaded</p>

  1102→    );
  1103→  }
  1104→

  1110→          'flex flex-col h-full overflow-hidden border-r border-zinc-800',

  1114→        {/* Page Tabs + Add Page button */}

  1121→
