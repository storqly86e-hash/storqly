    64→// ─── Background Image Enrichment (lazy, non-blocking) ─────
    65→// Triggered after store generation completes. Fetches real product
    66→// images sequentially without blocking the user or exhausting rate limits.
    67→
    68→function triggerBackgroundImageEnrichment(store: Store) {
    69→  if (!store.products || store.products.length === 0) return
    70→
    71→  // Only enrich products that still have AI-hallucinated Unsplash URLs
    72→  // (Real enriched URLs from the image search service will have different patterns)
    73→  const needsEnrichment = store.products.filter(p =>
    74→    p.images.length > 0 && p.images[0].includes('unsplash.com/photo-')
    75→  )
    76→
    77→  if (needsEnrichment.length === 0) return
    78→
    79→  console.log(`[Storqly] Background image enrichment: ${needsEnrichment.length}/${store.products.length} products need enrichment`)
    80→
    81→  fetch('/api/store/enrich-images', {
    82→    method: 'POST',
    83→    headers: { 'Content-Type': 'application/json' },
    84→    body: JSON.stringify({
    85→      products: needsEnrichment.map(p => ({
    86→        id: p.id,
    87→        name: p.name,
    88→        images: [...p.images],
    89→        category: p.category,
    90→        description: p.description,
    91→      })),
    92→      storeName: store.name,
    93→    }),
    94→  })
    95→    .then(res => res.json())
    96→    .then(data => {
    97→      if (data.enriched > 0) {
    98→        console.log(`[Storqly] Background enrichment complete: ${data.enriched} enriched, ${data.kept} kept, ${data.failed} failed in ${data.latencyMs}ms`)
    99→        // Update product images in the store via the Zustand store
   100→        // The enrich-images endpoint mutated the products array in-place,
   101→        // but we need to update the Zustand store to trigger re-renders.
   102→        // Dynamic import to avoid circular dependency
   103→        import('@/lib/store').then(({ useStoreEditor }) => {
   104→          const currentStore = useStoreEditor.getState().store
   105→          if (currentStore) {
   106→            const updatedProducts = currentStore.products.map(prod => {
   107→              const enriched = data.products?.find((ep: { id: string; images: string[] }) => ep.id === prod.id)
   108→              if (enriched && enriched.images[0] !== prod.images[0]) {
   109→                return { ...prod, images: enriched.images }
   110→              }
   111→              return prod
   112→            })
   113→            useStoreEditor.getState().setStore({ ...currentStore, products: updatedProducts })
   114→          }
   115→        })
   116→      }
   117→    })
   118→    .catch(err => {
   119→      // Non-fatal — products still show AI placeholder images
   120→      console.warn('[Storqly] Background enrichment failed (non-fatal):', err)
   121→    })
   122→}

   169→function LandingPage() {

   172→  const [mkOpen, setMkOpen] = useState(false)

   176→  const [editingSlug, setEditingSlug] = useState<string | null>(null)
   177→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   178→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   179→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   180→  const abortRef = useRef<AbortController | null>(null)
   181→
   182→  const {
   183→    isGenerating,
   184→    setIsGenerating,
   185→    setStore,
   186→    setStoreWithFallback,
   187→  } = useStoreEditor()
   188→
   189→  // Local UI state (not in Zustand — this is view-level)

   191→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   192→  const [error, setError] = useState<string | null>(null)
   193→
   194→  const clearTimers = useCallback(() => {
   195→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   196→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }
   197→  }, [])
   198→

   204→    }

   206→    setStoresLoading(true)

   209→      .then((data) => { if (!cancelled) setMyStores(data.stores ?? []) })
   210→      .catch(() => { if (!cancelled) setMyStores([]) })
   211→      .finally(() => { if (!cancelled) setStoresLoading(false) })
   212→    return () => { cancelled = true }
   213→  }, [session?.user?.id])

   229→

   231→  const formatTimeAgo = useCallback((iso: string) => {
   232→    const diff = Date.now() - new Date(iso).getTime()
   233→    const mins = Math.floor(diff / 60000)
   234→    if (mins < 1) return 'just now'
   235→    if (mins < 60) return `${mins}m ago`
   236→    const hours = Math.floor(mins / 60)
   237→    if (hours < 24) return `${hours}h ago`
   238→    const days = Math.floor(hours / 24)
   239→    if (days < 30) return `${days}d ago`
   240→    return new Date(iso).toLocaleDateString()
   241→  }, [])
   242→
   243→  const handleCancel = useCallback(() => {
   244→    abortRef.current?.abort()
   245→    clearTimers()
   246→    setIsGenerating(false)
   247→    setGenerationStatus('')
   248→    setElapsedSeconds(0)
   249→    setError(null)
   250→  }, [clearTimers, setIsGenerating])
   251→
   252→  const handleGenerate = useCallback(async () => {
   253→    const trimmed = promptText.trim()
   254→    if (!trimmed) {
   255→      setError('Please describe your store to get started.')
   256→      textareaRef.current?.focus()
   257→      return
   258→    }
   259→
   260→    // Auth gate — block generation for logged-out users (show modal instead)
   261→    if (!session?.user?.id) {
   262→      setAuthOpen(true)
   263→      return
   264→    }
   265→
   266→    // Reset state
   267→    setError(null)
   268→    setIsGenerating(true)
   269→    setGenerationStatus('Starting generation...')
   270→    setElapsedSeconds(0)
   271→
   272→    // Create abort controller for this request
   273→    const controller = new AbortController()
   274→    abortRef.current = controller
   275→
   276→    // Hard timeout: 120 seconds — no request should ever take longer than this.
   277→    // If the server is hung or the proxy drops the connection, the user gets a
   278→    // clear message instead of waiting indefinitely.
   279→    const HARD_TIMEOUT_MS = 120_000
   280→    let timedOut = false
   281→    const timeoutId = setTimeout(() => {
   282→      timedOut = true
   283→      controller.abort()
   284→      console.warn(`[Storqly] Hard timeout (${HARD_TIMEOUT_MS / 1000}s) reached — aborting generation`)
   285→    }, HARD_TIMEOUT_MS)
   286→
   287→    // Elapsed time counter
   288→    elapsedTimerRef.current = setInterval(() => {
   289→      setElapsedSeconds((prev) => prev + 1)
   290→    }, 1000)
   291→
   292→    // Fallback progress cycling (used only if SSE progress events stop)
   293→    let fallbackIdx = 0
   294→    let receivedSSEProgress = false
   295→    progressTimerRef.current = setInterval(() => {
   296→      if (!receivedSSEProgress) {
   297→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   298→        setGenerationStatus(progressMessages[fallbackIdx])
   299→      }
   300→    }, 3000)
   301→
   302→    // ── Helper: clean up on any error path (no throw, no crash overlay) ──
   303→    const finishWithError = (message: string) => {
   304→      clearTimeout(timeoutId)
   305→      clearTimers()
   306→      setError(message)
   307→      setIsGenerating(false)
   308→      setGenerationStatus('')
   309→      setElapsedSeconds(0)
   310→      toast.error('Store generation failed', { description: message })
   311→    }
   312→
   313→    const finishOk = () => {
   314→      clearTimeout(timeoutId)
   315→      clearTimers()
   316→      setIsGenerating(false)
   317→      setGenerationStatus('')
   318→      setElapsedSeconds(0)
   319→    }
   320→
   321→    try {
   322→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   323→
   324→      // ── Fetch with auto-retry for transient gateway errors (502/503/504) ──
   325→      const RETRYABLE_STATUSES = [502, 503, 504]
   326→      let res: Response | undefined
   327→      let retryCount = 0
   328→      const maxRetries = 2
   329→
   330→      while (retryCount <= maxRetries) {
   331→        try {
   332→          res = await fetch('/api/store/generate', {
   333→            method: 'POST',
   334→            headers: { 'Content-Type': 'application/json' },
   335→            body: JSON.stringify({ prompt: trimmed }),
   336→            signal: controller.signal,
   337→          })
   338→
   339→          // If the status is retryable, wait and loop again
   340→          if (!res.ok && RETRYABLE_STATUSES.includes(res.status) && retryCount < maxRetries) {
   341→            retryCount++
   342→            console.warn(`[Storqly] Gateway error ${res.status}, retrying (${retryCount}/${maxRetries}) in 3s...`)
   343→            setGenerationStatus('Connection issue — retrying...')
   344→            await new Promise(r => setTimeout(r, 3000))
   345→            continue
   346→          }
   347→          break // Success or non-retryable error — exit loop
   348→        } catch (fetchErr: unknown) {
   349→          // Network-level error (DNS failure, connection refused, etc.)
   350→          if (fetchErr instanceof TypeError && retryCount < maxRetries) {
   351→            retryCount++
   352→            console.warn(`[Storqly] Network error, retrying (${retryCount}/${maxRetries}) in 3s...`)
   353→            setGenerationStatus('Connection issue — retrying...')
   354→            await new Promise(r => setTimeout(r, 3000))
   355→            continue
   356→          }
   357→          // Out of retries or not a network error — handle gracefully
   358→          const msg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr)
   359→          if (controller.signal.aborted) {
   360→            finishWithError('Generation was cancelled.')
   361→          } else {
   362→            console.warn('[Storqly] Network error after retries:', msg)
   363→            finishWithError('Could not connect to the server. Please check your connection and try again.')
   364→          }
   365→          return
   366→        }
   367→      }
   368→
   369→      if (!res!.ok) {
   370→        let errorMsg = `Server error (${res!.status})`
   371→        try {
   372→          const errorData = await res!.json()
   373→          errorMsg = errorData.error || errorMsg
   374→        } catch { /* ignore parse error */ }
   375→        console.warn('[Storqly] HTTP error:', errorMsg)
   376→        finishWithError(errorMsg)
   377→        return
   378→      }
   379→
   380→      // ── Consume SSE stream ──
   381→      const reader = res.body?.getReader()
   382→      if (!reader) {
   383→        finishWithError('No response stream received from server.')
   384→        return
   385→      }
   386→
   387→      const decoder = new TextDecoder()
   388→      let buffer = ''
   389→      let resolved = false
   390→
   391→      while (true) {
   392→        const { done, value } = await reader.read()
   393→        if (done) break
   394→
   395→        buffer += decoder.decode(value, { stream: true })
   396→
   397→        // Parse SSE events from buffer
   398→        const lines = buffer.split('\n')
   399→        buffer = lines.pop() || ''
   400→
   401→        let currentEvent = ''
   402→        for (const line of lines) {
   403→          if (line.startsWith(':')) continue
   404→          if (line.startsWith('event: ')) {
   405→            currentEvent = line.slice(7).trim()
   406→          } else if (line.startsWith('data: ')) {
   407→            const dataStr = line.slice(6)
   408→
   409→            if (currentEvent === 'progress') {
   410→              receivedSSEProgress = true
   411→              try {
   412→                const data = JSON.parse(dataStr)
   413→                setGenerationStatus(data.message || 'Processing...')
   414→              } catch { /* ignore */ }
   415→            } else if (currentEvent === 'result') {
   416→              try {
   417→                const data = JSON.parse(dataStr)
   418→                if (!data.store) {
   419→                  console.warn('[Storqly] Result event missing store data')
   420→                  finishWithError('The AI response was missing store data.')
   421→                  return
   422→                }
   423→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   424→                resolved = true
   425→
   426→                if (data._isFallback) {
   427→                  toast.warning('AI service unavailable — showing starter template. Try again in a moment.', { duration: 6000 })
   428→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   429→                } else {
   430→                  setStore(data.store)
   431→                  // Trigger lazy background image enrichment (non-blocking)
   432→                  triggerBackgroundImageEnrichment(data.store)
   433→                }
   434→
   435→                // Soft cap toast
   436→                if (data._productCapHit) {
   437→                  toast.info(
   438→                    `Generated ${data._generatedCount} products — for larger catalogs, you can add more via the chat editor.`,
   439→                    { duration: 8000 }
   440→                  )
   441→                }
   442→
   443→                finishOk()
   444→                return
   445→              } catch (parseErr) {
   446→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   447→                console.warn('[Storqly] Failed to parse result event:', msg)
   448→                finishWithError(msg)
   449→                return
   450→              }
   451→            } else if (currentEvent === 'error') {
   452→              try {
   453→                const data = JSON.parse(dataStr)
   454→                const serverMsg = data.message || 'Generation failed'
   455→                console.warn('[Storqly] Server error event:', serverMsg)
   456→                finishWithError(serverMsg)
   457→                return
   458→              } catch {
   459→                finishWithError('Generation failed')
   460→                return
   461→              }
   462→            }
   463→            currentEvent = ''
   464→          }
   465→        }
   466→      }
   467→
   468→      if (!resolved) {
   469→        console.warn('[Storqly] Stream ended without a result event')
   470→        finishWithError('Stream ended without a result. The server may have disconnected.')
   471→      }
   472→    } catch (err: unknown) {
   473→      // Last-resort safety net — should rarely be reached now
   474→      clearTimeout(timeoutId)
   475→      clearTimers()
   476→      let message = 'Something went wrong. Please try again.'
   477→      if (err instanceof Error) {
   478→        if (err.name === 'AbortError') {
   479→          message = timedOut
   480→            ? 'Generation timed out after 2 minutes. The server may be overloaded — please try again.'
   481→            : 'Generation was cancelled.'
   482→          console.warn(`[Storqly] Request aborted: ${message}`)
   483→        } else {
   484→          console.warn('[Storqly] Unexpected error (caught by safety net):', err.message)
   485→          message = err.message
   486→        }
   487→      }
   488→      setError(message)
   489→      setIsGenerating(false)
   490→      setGenerationStatus('')
   491→      setElapsedSeconds(0)
   492→      toast.error('Store generation failed', { description: message })
   493→    }
   494→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])
   495→
   496→  // Cleanup on unmount
   497→  useEffect(() => {
   498→    return () => {
   499→      clearTimers()
   500→      abortRef.current?.abort()
   501→    }
   502→  }, [clearTimers])
   503→
   504→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   505→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   506→      e.preventDefault()
   507→      handleGenerate()
   508→    }
   509→  }

   515→    return `${m}:${s.toString().padStart(2, '0')}`
   516→  }

   524→            Storqly

   543→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />

  1005→function EditorToolbar({
  1006→  onToggleLeft,
  1007→  onToggleRight,
  1008→  showLeft,
  1009→  showRight,
  1010→}: {
  1011→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
  1012→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
  1013→  showLeft: boolean
  1014→  showRight: boolean
  1015→}) {
  1016→  const store = useStoreEditor((s) => s.store)
  1017→  const reset = useStoreEditor((s) => s.reset)
  1018→  const isPublishing = useStoreEditor((s) => s.isPublishing)
  1019→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
  1020→  const isPublished = useStoreEditor((s) => s.isPublished)
  1021→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
  1022→  const [isSaving, setIsSaving] = useState(false)
  1023→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
  1024→  const [copied, setCopied] = useState(false)
  1025→  const [authOpen, setAuthOpen] = useState(false)
  1026→
  1027→  const handleBack = () => {
  1028→    reset()
  1029→  }
  1030→
  1031→  const handleSave = async () => {
  1032→    if (!store || isSaving) return
  1033→    setIsSaving(true)
  1034→    try {
  1035→      const res = await fetch('/api/store/save', {
  1036→        method: 'POST',
  1037→        headers: { 'Content-Type': 'application/json' },
  1038→        body: JSON.stringify({ store }),
  1039→      })
  1040→      if (!res.ok) throw new Error('Save failed')
  1041→      const data = await res.json()
  1042→      toast.success(`Draft saved — ID: ${data.id}`, {
  1043→        description: `Slug: ${data.slug}`,
  1044→        duration: 5000,
  1045→      })
  1046→    } catch {
  1047→      toast.error('Failed to save store')
  1048→    } finally {
  1049→      setIsSaving(false)
  1050→    }
  1051→  }
  1052→
  1053→  const handlePublish = async () => {
  1054→    if (!store || isPublishing) return
  1055→    setIsPublishing(true)
  1056→    try {
  1057→      const res = await fetch('/api/store/publish', {
  1058→        method: 'POST',
  1059→        headers: { 'Content-Type': 'application/json' },
  1060→        body: JSON.stringify({ store }),
  1061→      })
  1062→      if (!res.ok) throw new Error('Publish failed')
  1063→      const data = await res.json()
  1064→      const slug = data.slug
  1065→      setIsPublished(true)
  1066→      // Build a real viewable URL using the current origin
  1067→      const baseUrl = window.location.origin
  1068→      const viewUrl = `${baseUrl}/?store=${slug}`
  1069→      setPublishedUrl(viewUrl)
  1070→      toast.success('Store published successfully!')
  1071→    } catch {
  1072→      toast.error('Failed to publish store')
  1073→    } finally {
  1074→      setIsPublishing(false)
  1075→    }
  1076→  }
  1077→
  1078→  const handleCopyUrl = useCallback(async () => {
  1079→    if (!publishedUrl) return
  1080→    try {
  1081→      await navigator.clipboard.writeText(publishedUrl)
  1082→      setCopied(true)
  1083→      toast.success('URL copied to clipboard!')

  1268→// ─── Published Store Viewer (read-only, shown via ?store=slug) ────────
  1269→
  1270→function PublishedStoreViewer({ slug }: { slug: string }) {
  1271→  const [store, setStore] = useState<Store | null>(null)
  1272→  const [loading, setLoading] = useState(true)
  1273→  const [error, setError] = useState<string | null>(null)
  1274→
  1275→  useEffect(() => {
  1276→    let cancelled = false
  1277→    async function load() {
  1278→      try {
  1279→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
  1280→        if (!res.ok) {
  1281→          const data = await res.json().catch(() => ({}))
  1282→          throw new Error(data.error || `Store not found (${res.status})`)
  1283→        }
  1284→        const data = await res.json()
  1285→        if (!cancelled) setStore(data.store)
  1286→      } catch (err: unknown) {
  1287→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
  1288→      } finally {
  1289→        if (!cancelled) setLoading(false)
  1290→      }
  1291→    }
  1292→    load()
  1293→    return () => { cancelled = true }
  1294→  }, [slug])
  1295→
  1296→  if (loading) {
  1297→    return (
  1298→      <div className="flex min-h-screen items-center justify-center bg-white">
  1299→        <div className="text-center">
  1300→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
  1301→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
  1302→        </div>
  1303→      </div>
  1304→    )
  1305→  }
  1306→
  1307→  if (error || !store) {
  1308→    return (
  1309→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
  1310→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
  1311→          <AlertCircle className="h-8 w-8 text-red-400" />
  1312→        </div>
  1313→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
  1314→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
  1315→          {error || 'This store could not be found or has not been published.'}
  1316→        </p>
  1317→        <Button
  1318→          variant="outline"
  1319→          onClick={() => window.location.href = '/'}
  1320→          className="gap-2"
  1321→        >
  1322→          <ArrowLeft className="h-4 w-4" />
  1323→          Back to Storqly
  1324→        </Button>
  1325→      </div>
  1326→    )
  1327→  }
  1328→
  1329→  return (
  1330→    <>
  1331→      <StoreRenderer store={store} />
  1332→      <Toaster />
  1333→      {/* Minimal footer for published stores */}
  1334→      <div className="fixed bottom-3 right-3 z-40">
  1335→        <a
  1336→          href="/"
  1337→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
  1338→        >
  1339→          <Sparkles className="h-3 w-3" />
  1340→          Built with Storqly
  1341→        </a>
  1342→      </div>
  1343→    </>
  1344→  )
  1345→}
