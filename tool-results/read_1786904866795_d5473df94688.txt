    50→// ─── Error boundary for individual sections ──────────────────────────
    51→// Prevents a malformed section from crashing the entire page.
    52→
    53→interface SectionErrorBoundaryState { hasError: boolean; error: Error | null }
    54→class SectionErrorBoundary extends Component<
    55→  { children: React.ReactNode; sectionType: string },
    56→  SectionErrorBoundaryState
    57→> {
    58→  constructor(props: { children: React.ReactNode; sectionType: string }) {
    59→    super(props);
    60→    this.state = { hasError: false, error: null };
    61→  }
    62→  static getDerivedStateFromError(error: Error) {
    63→    return { hasError: true, error };
    64→  }
    65→  render() {
    66→    if (this.state.hasError) {
    67→      return (
    68→        <div className="flex items-center justify-center py-8 px-6 text-center">
    69→          <div className="max-w-sm">
    70→            <p className="text-sm font-medium" style={{ color: '#888' }}>
    71→              Section render error ({this.props.sectionType})
    72→            </p>
    73→            <p className="mt-1 text-xs" style={{ color: '#aaa' }}>
    74→              {this.state.error?.message || 'Unknown error'}
    75→            </p>
    76→          </div>
    77→        </div>
    78→      );
    79→    }
    80→    return this.props.children;
    81→  }
    82→}

    86→interface SectionWrapperProps {
    87→  section: Section;
    88→  theme: StoreTheme;
    89→  selectedSectionId?: string | null;
    90→  onSelectSection?: (sectionId: string | null) => void;

    94→function SectionWrapper({
    95→  section,
    96→  theme,
    97→  selectedSectionId,
    98→  onSelectSection,
    99→  children,
   100→}: SectionWrapperProps) {
   101→  const isSelected = selectedSectionId === section.id;
   102→  const style = section.style;
   103→
   104→  const sectionBg = style.backgroundColor || theme.colors.background;
   105→  const sectionText = style.textColor || theme.colors.text;
   106→
   107→  const bgImage = style.backgroundImage
   108→    ? `url(${style.backgroundImage})`
   109→    : undefined;
   110→
   111→  const outerClasses = [
   112→    'relative transition-all duration-200 cursor-pointer',
   113→    pyClass(style.paddingY),
   114→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   115→  ].join(' ');
   116→
   117→  return (
   118→    <div
   119→      className={outerClasses}
   120→      style={{
   121→        backgroundColor: sectionBg,
   122→        color: sectionText,
   123→        backgroundImage: bgImage,
   124→        backgroundSize: 'cover',
   125→        backgroundPosition: 'center',
   126→      }}
   127→      onClick={(e) => {
   128→        e.stopPropagation();
   129→        onSelectSection?.(isSelected ? null : section.id);
   130→      }}
   131→      data-section-id={section.id}
   132→      data-section-type={section.type}
   133→    >
   134→      {/* Overlay for background images */}
   135→      {style.backgroundImage && style.overlay && (
   136→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   137→      )}
   138→      <div
   139→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   140→      >
   141→        {/* Selected section label */}
   142→        {isSelected && (
   143→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   144→            {section.type.replace(/-/g, ' ')}
   145→          </div>
   146→        )}
   147→        {children}
   148→      </div>
   149→    </div>
   150→  );
   151→}

   155→function ProductCard({
   156→  product,
   157→  theme,
   158→  showAddToCart,
   159→  borderRadius,
   160→  buttonBgOverride,
   161→  buttonTextOverride,
   162→  onViewProduct,
   163→}: {
   164→  product: StoreProduct;
   165→  theme: StoreTheme;
   166→  showAddToCart: boolean;
   167→  borderRadius: string;
   168→  buttonBgOverride?: string;
   169→  buttonTextOverride?: string;
   170→  onViewProduct?: (productId: string) => void;
   171→}) {
   172→  const [hovered, setHovered] = useState(false);
   173→  const [addedFeedback, setAddedFeedback] = useState(false);
   174→  const addToCart = useCartStore((s) => s.addItem);
   175→  const imgColor = stringToColor(product.id, theme);
   176→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   177→
   178→  const handleAddToCart = (e: React.MouseEvent) => {
   179→    e.stopPropagation();
   180→    addToCart(product);
   181→    setAddedFeedback(true);
   182→    setTimeout(() => setAddedFeedback(false), 1200);
   183→  };
   184→
   185→  return (
   186→    <div
   187→      className={`${borderRadius} ${onViewProduct ? 'cursor-pointer' : ''} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   188→      style={{
   189→        borderColor: theme.colors.border,
   190→      }}
   191→      onMouseEnter={() => setHovered(true)}
   192→      onMouseLeave={() => setHovered(false)}
   193→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   194→      role={onViewProduct ? 'button' : undefined}
   195→      tabIndex={onViewProduct ? 0 : undefined}
   196→    >
   197→      {/* Product image with real image + fallback */}
   198→      <div
   199→        className="relative aspect-square w-full overflow-hidden"
   200→        style={{ backgroundColor: imgColor }}
   201→      >
   202→        <StoreImage
   203→          src={product.images[0] || ''}
   204→          alt={product.name}
   205→          fallbackColor={imgColor}
   206→          className="absolute inset-0 h-full w-full object-cover transition-transform duration-300 group-hover:scale-105"
   207→          iconSize="lg"
   208→        />
   209→        {/* Out of stock overlay */}
   210→        {!product.inStock && (
   211→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   212→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   213→              Sold Out
   214→            </span>
   215→          </div>
   216→        )}
   217→        {/* Discount badge */}
   218→        {hasDiscount && product.compareAtPrice && (
   219→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   220→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   221→          </div>
   222→        )}
   223→      </div>
   224→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   225→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   226→        {product.category && (
   227→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   228→            {product.category}
   229→          </p>
   230→        )}
   231→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   232→          {product.name}
   233→        </h3>
   234→        <div className="mt-1.5 flex items-center gap-2">
   235→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   236→            {formatPrice(product.price)}
   237→          </span>
   238→          {hasDiscount && product.compareAtPrice && (
   239→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   240→              {formatPrice(product.compareAtPrice)}
   241→            </span>
   242→          )}
   243→        </div>
   244→        {showAddToCart && product.inStock && (
   245→          <button
   246→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 cursor-pointer ${
   247→              hovered
   248→                ? 'opacity-100 translate-y-0'
   249→                : 'opacity-0 translate-y-1 pointer-events-none'
   250→            }`}
   251→            style={{
   252→              backgroundColor: addedFeedback ? '#16a34a' : (buttonBgOverride || theme.colors.primary),
   253→              color: addedFeedback ? '#ffffff' : (buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary)),
   254→            }}
   255→            onClick={handleAddToCart}
   256→          >
   257→            {addedFeedback ? 'Added!' : 'Add to Cart'}
   258→          </button>
   259→        )}
   260→      </div>
   261→    </div>
   262→  );
   263→}

   361→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   362→  const content = section.content as unknown as HeroContent;
   363→  const style = section.style;
   364→  const isSelected = selectedSectionId === section.id;
   365→
   366→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   367→  const alignMap = {
   368→    left: 'items-start text-left',
   369→    center: 'items-center text-center',
   370→    right: 'items-end text-right',
   371→  };
   372→
   373→  // Phase 3A: layout mode
   374→  const layout = content.layout || 'centered';
   375→  const isSplit = layout !== 'centered';
   376→  const imageOnRight = layout === 'split-left'; // text left, image right
   377→  const imageOnLeft = layout === 'split-right'; // image left, text right
   378→
   379→  // Resolve hero image: explicit content.heroImage, or first featured product image
   380→  const resolvedHeroImage = content.heroImage
   381→    || (isSplit ? products.find(p => p.featured)?.images[0] || products[0]?.images[0] : undefined);
   382→
   383→  // Background: gradient default, or custom color/image
   384→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   385→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   386→    : undefined;
   387→
   388→  const outerClasses = [
   389→    'relative flex transition-all duration-200 cursor-pointer overflow-hidden',
   390→    heightMap[content.height] || heightMap.md,
   391→    isSplit ? 'items-center' : alignMap[content.alignment],
   392→    pyClass(style.paddingY),
   393→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   394→  ].join(' ');
   395→
   396→  // CTA click handler: stop propagation to prevent section selection.
   397→  // Navigation to specific pages is handled by the parent via ctaLink.
   398→  const handleCtaClick = (e: React.MouseEvent) => {
   399→    e.stopPropagation();
   400→  };
   401→
   402→  const textColor = style.textColor || '#ffffff';
   403→  const buttonBg = section.style.buttonBackgroundColor || '#ffffff';
   404→  const buttonFg = section.style.buttonTextColor || contrastTextColor(buttonBg);
   405→  const headlineColor = section.style.headlineColor || undefined;
   406→
   407→  return (
   408→    <div
   409→      className={outerClasses}
   410→      style={{
   411→        backgroundColor: style.backgroundColor || theme.colors.primary,
   412→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   413→        backgroundSize: 'cover',
   414→        backgroundPosition: 'center',
   415→        color: textColor,
   416→      }}
   417→      onClick={(e) => {
   418→        e.stopPropagation();
   419→        onSelectSection?.(isSelected ? null : section.id);
   420→      }}
   421→      data-section-id={section.id}
   422→      data-section-type={section.type}
   423→    >
   424→      {/* Image background overlay */}
   425→      {style.backgroundImage && style.overlay && (
   426→        <div className="absolute inset-0 pointer-events-none" style={{ background: 'linear-gradient(to bottom, rgba(0,0,0,0.15) 0%, rgba(0,0,0,0.55) 100%)' }} />
   427→      )}
   428→      {/* Subtle gradient overlay for depth */}
   429→      {bgGradient && (
   430→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   431→      )}
   432→      {/* Decorative blur orbs — always present for visual richness */}
   433→      <div className="absolute right-[10%] top-[10%] h-40 w-40 rounded-full bg-white/10 blur-3xl pointer-events-none" />
   434→      <div className="absolute bottom-[10%] left-[10%] h-32 w-32 rounded-full bg-white/5 blur-2xl pointer-events-none" />
   435→      {/* Extra decorative orb for split layouts */}
   436→      {isSplit && (
   437→        <div className="absolute left-[30%] top-[20%] h-24 w-24 rounded-full bg-white/5 blur-2xl pointer-events-none" />
   438→      )}
   439→
   440→      {/* Content area */}
   441→      <div className={`relative z-10 mx-auto w-full ${isSplit ? 'max-w-7xl' : maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)} ${isSplit ? 'grid grid-cols-1 gap-8 lg:grid-cols-2 lg:gap-12 items-center' : ''}`}>
   442→        {/* Text column */}
   443→        <div className={isSplit ? (imageOnLeft ? 'lg:order-2' : 'lg:order-1') : ''}>
   444→          {/* Badge */}
   445→          {content.badge && (
   446→            <span
   447→              className="mb-4 inline-block rounded-full border border-white/25 px-3 py-1 text-xs font-semibold uppercase tracking-widest backdrop-blur-sm"

   452→          )}

   454→          <h1
   455→            className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl"
   456→            style={{ textShadow: style.backgroundImage ? '0 2px 12px rgba(0,0,0,0.3)' : undefined, ...(headlineColor ? { color: headlineColor } : {}) }}
   457→          >
   458→            {content.headline}
   459→          </h1>
   460→
   461→          {content.subheadline && (
   462→            <p
   463→              className={`mt-4 text-base opacity-80 sm:text-lg md:text-xl ${isSplit ? 'max-w-xl' : 'max-w-2xl'} mx-${isSplit ? '0' : 'auto'}`}
   464→              style={{ textShadow: style.backgroundImage ? '0 1px 8px rgba(0,0,0,0.2)' : undefined }}
   465→            >
   466→              {content.subheadline}
   467→            </p>
   468→          )}
   469→
   470→          {/* CTA buttons */}
   471→          {(content.ctaText || content.secondaryCtaText) && (
   472→            <div className={`mt-8 flex flex-wrap gap-3 ${!isSplit && content.alignment === 'center' ? 'justify-center' : ''} ${!isSplit && content.alignment === 'right' ? 'justify-end' : ''}`}>
   473→              {content.ctaText && (
   474→                <button
   475→                  className="inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-all hover:scale-105 hover:shadow-xl sm:text-base"
   476→                  style={{ backgroundColor: buttonBg, color: buttonFg }}
   477→                  onClick={handleCtaClick}
   478→                >
   479→                  {content.ctaText}
   480→                  <ArrowRight className="h-4 w-4" />
   481→                </button>
   482→              )}
   483→              {content.secondaryCtaText && (
   484→                <button
   485→                  className="inline-flex items-center gap-2 rounded-lg border border-white/30 px-6 py-3 text-sm font-semibold backdrop-blur-sm transition-all hover:bg-white/10 hover:scale-105 sm:text-base"
   486→                  style={{ color: textColor, borderColor: textColor + '40' }}
   487→                  onClick={handleCtaClick}
   488→                >
   489→                  {content.secondaryCtaText}
   490→                </button>
   491→              )}
   492→            </div>
   493→          )}
   494→        </div>
   495→
   496→        {/* Image column (split layouts only) */}
   497→        {isSplit && resolvedHeroImage && (
   498→          <div className={`relative ${imageOnLeft ? 'lg:order-1' : 'lg:order-2'} hidden lg:block`}>
   499→            <div className="relative">
   500→              {/* Decorative background shape behind image */}
   501→              <div
   502→                className="absolute -inset-4 rounded-2xl opacity-20 blur-xl"
   503→                style={{ backgroundColor: textColor }}
   504→              />
   505→              <StoreImage
   506→                src={resolvedHeroImage}
   507→                alt={content.headline}
   508→                className="relative rounded-2xl shadow-2xl object-cover w-full aspect-[4/3]"
   509→              />
   510→            </div>
   511→          </div>
   512→        )}
   513→      </div>
   514→    </div>
   515→  );
   516→}

   520→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   521→  const content = section.content as unknown as FeaturedProductsContent;
   522→  const borderRadius = borderRadiusClass(theme.borderRadius);
   523→  const featuredProducts = useMemo(
   524→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   525→    [content.productIds, products],
   526→  );
   527→
   528→  return (
   529→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   530→      {content.headline && (
   531→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   532→          {content.headline}
   533→        </h2>
   534→      )}
   535→      {content.subtitle && (
   536→        <p className="mb-8 text-sm opacity-65">
   537→          {content.subtitle}
   538→        </p>
   539→      )}
   540→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   541→        {featuredProducts.map((product) => (

  1243→// ─── Main Section Router ────────────────────────────────────────────────
  1244→
  1245→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1246→  const { section } = props;
  1247→
  1248→  if (!section.visible) return null;
  1249→
  1250→  let element: React.ReactNode = null;
  1251→
  1252→  switch (section.type) {
  1253→    case 'header':
  1254→      element = <HeaderSection {...props} />; break;
  1255→    case 'hero':
  1256→      element = <HeroSection {...props} />; break;
  1257→    case 'featured-products':
  1258→      element = <FeaturedProductsSection {...props} />; break;
  1259→    case 'product-grid':
  1260→      element = <ProductGridSection {...props} />; break;
  1261→    case 'text-banner':
  1262→      element = <TextBannerSection {...props} />; break;
  1263→    case 'image-gallery':
  1264→      element = <ImageGallerySection {...props} />; break;
  1265→    case 'testimonials':
  1266→      element = <TestimonialsSection {...props} />; break;
  1267→    case 'newsletter':
  1268→      element = <NewsletterSection {...props} />; break;
  1269→    case 'faq':
  1270→      element = <FAQSection {...props} />; break;
  1271→    case 'cta':
  1272→      element = <CTASection {...props} />; break;
  1273→    case 'categories':
  1274→      element = <CategoriesSection {...props} />; break;
  1275→    case 'footer':
  1276→      element = <FooterSection {...props} />; break;
  1277→    case 'rich-text':
  1278→      element = <RichTextSection {...props} />; break;
  1279→    case 'spacer':
  1280→      element = <SpacerSection {...props} />; break;
  1281→    case 'divider':
  1282→      element = <DividerSection {...props} />; break;
  1283→    default:
  1284→      // Unknown section type — render a visible placeholder instead of invisible null
  1285→      return (
  1286→        <SectionErrorBoundary sectionType={section.type}>
  1287→          <div className="py-8 text-center text-sm text-gray-400">
  1288→            <p className="font-medium">Unsupported section type: {section.type}</p>
  1289→            <p className="mt-1 text-xs text-gray-300">This section can be edited or removed.</p>
  1290→          </div>
  1291→        </SectionErrorBoundary>
  1292→      );
  1293→  }
  1294→
  1295→  // Wrap each section in its own error boundary so a malformed
  1296→  // section (e.g. AI-generated content with wrong shape) can never
  1297→  // crash the entire page.
  1298→  return (
  1299→    <SectionErrorBoundary sectionType={section.type}>
  1300→      {element}
  1301→    </SectionErrorBoundary>
  1302→  );
  1303→}
