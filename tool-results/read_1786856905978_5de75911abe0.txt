   259→   259→function SortableSectionItem({
   260→   260→  section,
   261→   261→  isSelected,
   262→   262→  onSelect,
   263→   263→  onToggleVisibility,
   264→   264→  onDelete,
   265→   265→}: SortableSectionItemProps) {
   266→   266→  const {
   267→   267→    attributes,
   268→   268→    listeners,
   269→   269→    setNodeRef,
   270→   270→    transform,
   271→   271→    transition,
   272→   272→    isDragging,
   273→   273→  } = useSortable({ id: section.id });
   274→   274→
   275→   275→  const style = {
   276→   276→    transform: CSS.Transform.toString(transform),
   277→   277→    transition,
   278→   278→  };
   279→   279→
   280→   280→  const Icon = SECTION_TYPE_ICONS[section.type];
   281→   281→  const label = SECTION_TYPE_LABELS[section.type];
   282→   282→
   283→   283→  return (
   284→   284→    <div
   285→   285→      ref={setNodeRef}
   286→   286→      style={style}
   287→   287→      className={[
   288→   288→        'group flex items-center gap-2 rounded-lg px-3 py-2 text-sm cursor-pointer transition-colors',
   289→   289→        isDragging
   290→   290→          ? 'z-50 bg-zinc-700/80 shadow-xl ring-1 ring-white/10'
   291→   291→          : isSelected
   292→   292→            ? 'bg-amber-500/15 text-amber-200 ring-1 ring-amber-500/40'
   293→   293→            : 'text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200',
   294→   294→      ].join(' ')}
   295→   295→      onClick={onSelect}
   296→   296→    >
   297→   297→      {/* Drag handle */}
   298→   298→      <button
   299→   299→        className="flex-shrink-0 cursor-grab text-zinc-600 hover:text-zinc-400 active:cursor-grabbing focus:outline-none"
   300→   300→        {...attributes}
   301→   301→        {...listeners}
   302→   302→        onClick={(e) => e.stopPropagation()}
   303→   303→        aria-label="Drag to reorder"
   304→   304→      >
   305→   305→        <GripVertical className="h-4 w-4" />
   306→   306→      </button>
   307→   307→
   308→   308→      {/* Icon */}
   309→   309→      <Icon className="h-4 w-4 flex-shrink-0 opacity-60" />
   310→   310→
   311→   311→      {/* Label */}

   315→   315→      <button
   316→   316→        className="flex-shrink-0 rounded p-1 text-zinc-500 hover:text-zinc-300 hover:bg-zinc-700/50 focus:outline-none"
   317→   317→        onClick={(e) => {
   318→   318→          e.stopPropagation();
   319→   319→          onToggleVisibility();
   320→   320→        }}
   321→   321→        aria-label={section.visible ? 'Hide section' : 'Show section'}
   322→   322→      >
   323→   323→        {section.visible ? (
   324→   324→          <Eye className="h-3.5 w-3.5" />

   328→   328→      </button>
   329→   329→
   330→   330→      {/* Delete */}

   332→   332→        className="flex-shrink-0 rounded p-1 text-zinc-600 opacity-0 group-hover:opacity-100 hover:text-red-400 hover:bg-red-400/10 transition-opacity focus:outline-none"

   397→   397→// ── Content Field Renderer ───────────────────────────────────────────
   398→   398→
   399→   399→function ContentFieldRenderer({
   400→   400→  keyName,
   401→   401→  value,
   402→   402→  onChange,
   403→   403→  depth = 0,
   404→   404→}: {
   405→   405→  keyName: string;
   406→   406→  value: unknown;
   407→   407→  onChange: (key: string, value: unknown) => void;
   408→   408→  depth?: number;
   409→   409→}) {
   410→   410→  const label = keyName
   411→   411→    .replace(/([A-Z])/g, ' $1')
   412→   412→    .replace(/^./, (s) => s.toUpperCase())
   413→   413→    .trim();
   414→   414→
   415→   415→  if (value === null || value === undefined) {
   416→   416→    return null;
   417→   417→  }
   418→   418→
   419→   419→  if (typeof value === 'string') {
   420→   420→    if (keyName === 'html' && value.includes('<')) {
   421→   421→      return (
   422→   422→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   423→   423→          <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   424→   424→          <Textarea
   425→   425→            value={value}
   426→   426→            onChange={(e) => onChange(keyName, e.target.value)}
   427→   427→            rows={4}
   428→   428→            className="bg-zinc-800 border-zinc-700 text-zinc-100 text-xs font-mono resize-none"
   429→   429→          />
   430→   430→        </div>
   431→   431→      );
   432→   432→    }
   433→   433→    return (
   434→   434→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   435→   435→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   436→   436→        <Input
   437→   437→          value={value}
   438→   438→          onChange={(e) => onChange(keyName, e.target.value)}
   439→   439→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   440→   440→        />
   441→   441→      </div>
   442→   442→    );
   443→   443→  }
   444→   444→
   445→   445→  if (typeof value === 'number') {
   446→   446→    return (
   447→   447→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   448→   448→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   449→   449→        <Input
   450→   450→          type="number"
   451→   451→          value={value}
   452→   452→          onChange={(e) => onChange(keyName, Number(e.target.value))}
   453→   453→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   454→   454→        />
   455→   455→      </div>
   456→   456→    );
   457→   457→  }
   458→   458→
   459→   459→  if (typeof value === 'boolean') {
   460→   460→    return (
   461→   461→      <div
   462→   462→        key={keyName}
   463→   463→        className={`flex items-center justify-between ${depth > 0 ? 'ml-4' : ''}`}
   464→   464→      >
   465→   465→        <Label className="text-xs text-zinc-400">{label}</Label>
   466→   466→        <Switch
   467→   467→          checked={value}
   468→   468→          onCheckedChange={(checked) => onChange(keyName, checked)}
   469→   469→        />
   470→   470→      </div>
   471→   471→    );
   472→   472→  }
   473→   473→
   474→   474→  if (Array.isArray(value)) {
   475→   475→    return (
   476→   476→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   477→   477→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   478→   478→        <div className="rounded-lg bg-zinc-800/60 border border-zinc-700/50 px-3 py-2 text-xs text-zinc-500">
   479→   479→          {value.length} item{value.length !== 1 ? 's' : ''}
   480→   480→        </div>
   481→   481→      </div>
   482→   482→    );
   483→   483→  }
   484→   484→
   485→   485→  if (typeof value === 'object') {
   486→   486→    const obj = value as Record<string, unknown>;
   487→   487→    const keys = Object.keys(obj);
   488→   488→
   489→   489→    const allPrimitive = keys.every(
   490→   490→      (k) =>
   491→   491→        typeof obj[k] === 'string' ||
   492→   492→        typeof obj[k] === 'number' ||
   493→   493→        typeof obj[k] === 'boolean' ||
   494→   494→        obj[k] === null ||
   495→   495→        obj[k] === undefined
   496→   496→    );
   497→   497→
   498→   498→    if (allPrimitive && keys.length > 0) {
   499→   499→      return (
   500→   500→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   501→   501→          <p className="text-xs font-medium text-zinc-300 mb-2 mt-1">{label}</p>
   502→   502→          <div className="space-y-3 border-l border-zinc-700/50 pl-3 ml-1">
   503→   503→            {keys.map((k) => (
   504→   504→              <ContentFieldRenderer
   505→   505→                key={k}
   506→   506→                keyName={k}
   507→   507→                value={obj[k]}
   508→   508→                onChange={(subKey, subVal) => {
   509→   509→                  onChange(keyName, { ...obj, [subKey]: subVal });
   510→   510→                }}
   511→   511→                depth={depth + 1}
   512→   512→              />
   513→   513→            ))}
   514→   514→          </div>
   515→   515→        </div>
   516→   516→      );
   517→   517→    }
   518→   518→
   519→   519→    return (
   520→   520→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   521→   521→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   522→   522→        <Textarea
   523→   523→          value={JSON.stringify(value, null, 2)}
   524→   524→          readOnly

  1061→  1061→          sections: page.sections.map((s) =>
  1062→  1062→            s.id === sectionId ? { ...s, visible: !s.visible } : s
  1063→  1063→          ),
  1064→  1064→        })),
  1065→  1065→      };
  1066→  1066→      setStore(updatedStore);
  1067→  1067→    },
  1068→  1068→    [store, setStore]

  1070→  1070→

  1074→  1074→      const section = sections.find((s) => s.id === sectionId);
  1075→  1075→      removeSection(pageId, sectionId);
  1076→  1076→      if (selectedSectionId === sectionId) {
  1077→  1077→        setSelectedSectionId(null);
  1078→  1078→      }
  1079→  1079→      if (section) {
  1080→  1080→        toast.success(`Deleted ${SECTION_TYPE_LABELS[section.type]} section`);
  1081→  1081→      }
  1082→  1082→    },
  1083→  1083→    [pageId, sections, selectedSectionId, removeSection, setSelectedSectionId]
  1084→  1084→  );
  1085→  1085→
  1086→  1086→  // Handle page tab click (before early return to satisfy rules-of-hooks)
  1087→  1087→  const handlePageTabClick = useCallback(
  1088→  1088→    (pageId: string) => {
  1089→  1089→      if (renamingPageId && renamingPageId !== pageId) {
  1090→  1090→        setRenamingPageId(null); // Cancel rename if clicking another tab
  1091→  1091→      }
  1092→  1092→      setEditorCurrentPageId(pageId);
  1093→  1093→    },
  1094→  1094→    [setEditorCurrentPageId, renamingPageId]

  1096→  1096→
  1097→  1097→  if (!store || !currentPage) {
  1098→  1098→    return (
  1099→  1099→      <div className="flex h-full items-center justify-center bg-zinc-950">
  1100→  1100→        <p className="text-sm text-zinc-600">No store loaded</p>
  1101→  1101→      </div>
  1102→  1102→    );
  1103→  1103→  }
  1104→  1104→
  1105→  1105→  return (
  1106→  1106→    <div className="flex h-full overflow-hidden bg-zinc-950">
  1107→  1107→      {/* Section List Panel */}
  1108→  1108→      <div
  1109→  1109→        className={[
  1110→  1110→          'flex flex-col border-r border-zinc-800 min-h-0',
  1111→  1111→          selectedSection && !isTemplatePage ? 'w-64 lg:w-72' : 'w-full',
  1112→  1112→        ].join(' ')}
  1113→  1113→      >
  1114→  1114→        {/* Page Tabs + Add Page button */}
  1115→  1115→        <div className="flex items-center gap-0.5 overflow-x-auto border-b border-zinc-800 px-2 py-2" style={{scrollbarWidth: 'none'}}>
  1116→  1116→          {editorPages.map((page) => {

  1118→  1118→            const isCustom = page.type === 'custom';
  1119→  1119→            const Icon = PAGE_TYPE_ICONS[page.type || 'home'] || Layout;
  1120→  1120→            const isRenaming = renamingPageId === page.id;
  1121→  1121→
  1122→  1122→            return (
  1123→  1123→              <div key={page.id} className="flex shrink-0 items-center gap-0.5">
  1124→  1124→                <button
  1125→  1125→                  onClick={() => handlePageTabClick(page.id)}
  1126→  1126→                  className={`flex items-center gap-1.5 rounded-lg px-2.5 py-1.5 text-xs font-medium transition-colors ${
  1127→  1127→                    isActive
  1128→  1128→                      ? 'bg-zinc-800 text-zinc-100'

  1130→  1130→                  }`}
  1131→  1131→                >
  1132→  1132→                  <Icon className="h-3 w-3" />

  1174→  1174→                        <Trash2 className="h-3 w-3" />
  1175→  1175→                        Delete Page
  1176→  1176→                      </button>
  1177→  1177→                    </PopoverContent>
  1178→  1178→                  </Popover>
  1179→  1179→                )}
  1180→  1180→              </div>
  1181→  1181→            );
  1182→  1182→          })}
  1183→  1183→
  1184→  1184→          {/* Add Page button */}
  1185→  1185→          <Popover open={showAddPageDialog} onOpenChange={setShowAddPageDialog}>
  1186→  1186→            <PopoverTrigger asChild>
  1187→  1187→              <button className="flex shrink-0 items-center gap-1 rounded-lg px-2 py-1.5 text-xs text-zinc-600 hover:bg-zinc-800/50 hover:text-zinc-400 transition-colors">
  1188→  1188→                <Plus className="h-3 w-3" />
  1189→  1189→                <span className="whitespace-nowrap">Page</span>
  1190→  1190→              </button>
  1191→  1191→            </PopoverTrigger>
  1192→  1192→            <PopoverContent className="w-56 p-3 bg-zinc-900 border-zinc-800" align="start">

  1194→  1194→              <div className="flex gap-2">
  1195→  1195→                <Input
  1196→  1196→                  autoFocus
  1197→  1197→                  value={newPageName}
  1198→  1198→                  onChange={(e) => setNewPageName(e.target.value)}
  1199→  1199→                  onKeyDown={(e) => {

  1214→  1214→        {/* Panel header — only for section-editable pages */}

  1237→  1237→              <DndContext
  1238→  1238→                collisionDetection={closestCenter}

  1266→  1266→            </div>
  1267→  1267→          </ScrollArea>
  1268→  1268→        ) : (
  1269→  1269→          <div className="flex flex-1 min-h-0 flex-col items-center justify-center px-6 text-center">
  1270→  1270→            <div className="mb-4 flex h-12 w-12 items-center justify-center rounded-xl bg-zinc-800/60">
  1271→  1271→              {(() => {
  1272→  1272→                const TIcon = PAGE_TYPE_ICONS[currentPage.type || 'home'] || Layout;
  1273→  1273→                return <TIcon className="h-6 w-6 text-zinc-500" />;
  1274→  1274→              })()}
  1275→  1275→            </div>
  1276→  1276→            <p className="text-sm font-medium text-zinc-300">{PAGE_TYPE_LABELS[currentPage.type || 'home'] || 'Template'} Page</p>

  1282→  1282→
  1283→  1283→        {/* Add section button — only for section-editable pages */}
  1284→  1284→        {!isTemplatePage && (

  1289→  1289→      </div>
  1290→  1290→
  1291→  1291→      {/* Properties Panel — only for section-editable pages */}
  1292→  1292→      {selectedSection && !isTemplatePage && (

  1299→  1299→}
