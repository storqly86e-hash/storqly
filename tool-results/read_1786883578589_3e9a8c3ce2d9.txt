   888→          <span className="text-lg font-bold">
   889→            {content.storeName}
   890→          </span>
   891→          {content.tagline && (
   892→            <p className="mt-1 text-sm opacity-65">
   893→              {content.tagline}
   894→            </p>
   895→          )}
   896→        </div>
   897→
   898→        {/* Link columns */}
   899→        <div className="grid grid-cols-2 gap-8 sm:grid-cols-3 lg:grid-cols-4">
   900→          {content.columns.map((col) => (
   901→            <div key={col.title}>
   902→              <h4 className="mb-3 text-xs font-semibold uppercase tracking-wider">
   903→                {col.title}
   904→              </h4>
   905→              <ul className="space-y-2">
   906→                {col.links.map((link) => (
   907→                  <li key={link.label}>
   908→                    <span className="cursor-pointer text-sm transition-colors hover:opacity-70 opacity-65">
   909→                      {link.label}
   910→                    </span>
   911→                  </li>
   912→                ))}
   913→              </ul>
   914→            </div>
   915→          ))}
   916→        </div>
   917→
   918→        {/* Bottom row */}
   919→        <div className="flex flex-col items-center justify-between gap-3 border-t pt-6 sm:flex-row" style={{ borderColor: theme.colors.border }}>
   920→          {content.copyrightText && (
   921→            <p className="text-xs opacity-65">
   922→              {content.copyrightText}
   923→            </p>
   924→          )}
   925→          {/* Social icons placeholder */}
   926→          {content.socialLinks && content.socialLinks.length > 0 && (
   927→            <div className="flex gap-3">
   928→              {content.socialLinks.map((social) => (
   929→                <span
   930→                  key={social.platform}
   931→                  className="flex h-8 w-8 items-center justify-center rounded-full transition-colors hover:opacity-70"
   932→                  style={{ backgroundColor: `${theme.colors.primary}15`, color: theme.colors.primary }}
   933→                  title={social.platform}
   934→                >
   935→                  <svg className="h-4 w-4" fill="currentColor" viewBox="0 0 24 24">
   936→                    <circle cx="12" cy="12" r="10" fill="none" stroke="currentColor" strokeWidth="1.5" />
   937→                  </svg>
   938→                </span>
   939→              ))}
   940→            </div>
   941→          )}
   942→        </div>
   943→      </div>
   944→    </SectionWrapper>
   945→  );
   946→}

   965→export function SpacerSection({ section, theme, selectedSectionId, onSelectSection }: SectionRendererProps) {
   966→  const content = section.content as unknown as SpacerContent;
   967→  const heightMap = { sm: 'h-6', md: 'h-12', lg: 'h-20', xl: 'h-32' };
   968→  const isSelected = selectedSectionId === section.id;
   969→
   970→  return (
   971→    <div
   972→      className={`relative transition-all duration-200 ${heightMap[content.height] || heightMap.md} ${
   973→        isSelected
   974→          ? 'ring-2 ring-[#a855f7] ring-offset-2'
   975→          : 'cursor-pointer hover:ring-1 hover:ring-[#a855f7]/40'
   976→      }`}
   977→      onClick={(e) => {
   978→        e.stopPropagation();
   979→        onSelectSection?.(isSelected ? null : section.id);
   980→      }}
   981→      data-section-id={section.id}
   982→      data-section-type={section.type}
   983→    >
   984→      {/* Dashed guide when selected */}
   985→      {isSelected && (
   986→        <div className="absolute inset-x-0 top-1/2 -translate-y-1/2 border-t-2 border-dashed border-[#a855f7]/50" />
   987→      )}
   988→    </div>
   989→  );
   990→}

  1020→interface SectionRendererProps {
  1021→  section: Section;
  1022→  theme: StoreTheme;
  1023→  selectedSectionId?: string | null;
  1024→  onSelectSection?: (sectionId: string | null) => void;
  1025→  products: StoreProduct[];
  1026→  onViewProduct?: (productId: string) => void;
  1027→  onNavigate?: (pageId: string) => void;
  1028→  /** When true, force-hide the Add to Cart button on product cards (e.g. home page) */
  1029→  forceHideAddToCart?: boolean;
  1030→}
  1031→
  1032→// ─── Main Section Router ────────────────────────────────────────────────
  1033→
  1034→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1035→  const { section } = props;
  1036→
  1037→  if (!section.visible) return null;
  1038→
  1039→  let element: React.ReactNode = null;
  1040→
  1041→  switch (section.type) {
  1042→    case 'header':
  1043→      element = <HeaderSection {...props} />; break;
  1044→    case 'hero':
  1045→      element = <HeroSection {...props} />; break;
  1046→    case 'featured-products':
  1047→      element = <FeaturedProductsSection {...props} />; break;
  1048→    case 'product-grid':
  1049→      element = <ProductGridSection {...props} />; break;
  1050→    case 'text-banner':
  1051→      element = <TextBannerSection {...props} />; break;
  1052→    case 'image-gallery':
  1053→      element = <ImageGallerySection {...props} />; break;
  1054→    case 'testimonials':
  1055→      element = <TestimonialsSection {...props} />; break;
  1056→    case 'newsletter':
  1057→      element = <NewsletterSection {...props} />; break;
  1058→    case 'faq':
  1059→      element = <FAQSection {...props} />; break;
  1060→    case 'cta':
  1061→      element = <CTASection {...props} />; break;
  1062→    case 'categories':
  1063→      element = <CategoriesSection {...props} />; break;
  1064→    case 'footer':
  1065→      element = <FooterSection {...props} />; break;
  1066→    case 'rich-text':
  1067→      element = <RichTextSection {...props} />; break;
  1068→    case 'spacer':
  1069→      element = <SpacerSection {...props} />; break;
  1070→    case 'divider':
  1071→      element = <DividerSection {...props} />; break;
  1072→    default:
  1073→      // Unknown section type — render a visible placeholder instead of invisible null
  1074→      return (
  1075→        <SectionErrorBoundary sectionType={section.type}>
  1076→          <div className="py-8 text-center text-sm text-gray-400">
  1077→            <p className="font-medium">Unsupported section type: {section.type}</p>
  1078→            <p className="mt-1 text-xs text-gray-300">This section can be edited or removed.</p>
  1079→          </div>
  1080→        </SectionErrorBoundary>
  1081→      );
  1082→  }
  1083→
  1084→  // Wrap each section in its own error boundary so a malformed
  1085→  // section (e.g. AI-generated content with wrong shape) can never
  1086→  // crash the entire page.
  1087→  return (
  1088→    <SectionErrorBoundary sectionType={section.type}>
  1089→      {element}
  1090→    </SectionErrorBoundary>
  1091→  );
  1092→}
