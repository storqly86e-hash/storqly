     1→     1→// ========================================
     2→     2→// Storqly Schema Normalization Layer
     3→     3→// ========================================
     4→     4→// Deterministic, testable function that takes ANY valid JSON object
     5→     5→// (guaranteed by json_object mode) and reshapes it into a valid Store.
     6→     6→//
     7→     7→// This is NOT JSON repair (that fixes broken syntax).
     8→     8→// This IS schema normalization (coercing valid JSON to match the Store type).
     9→     9→//
    10→    10→// Design principles:
    11→    11→// - Never throws — always returns a valid Store
    12→    12→// - Preserve AI output when valid — only fill defaults for missing/wrong fields
    13→    13→// - Be lenient with types — coerce strings to numbers, etc.
    14→    14→// - Log every normalization for debugging
    15→    15→
    16→    16→import type { Store, StoreTheme, StoreProduct, StorePage, Section, SectionStyle, SectionType, PageType } from './store-schema';
    17→    17→import { defaultTheme } from './store-schema';

    40→    40→// ─── Logging ──────────────────────────────────────────────────────
    41→    41→
    42→    42→type NormLog = { field: string; action: 'missing' | 'coerced' | 'invalid' | 'defaulted'; from?: unknown; to?: unknown };
    43→    43→
    44→    44→function createLogger() {
    45→    45→  const logs: NormLog[] = [];
    46→    46→  return {
    47→    47→    log(entry: NormLog) { logs.push(entry); },
    48→    48→    getLogs() { return logs; },
    49→    49→    summary(): string {
    50→    50→      const counts = { missing: 0, coerced: 0, invalid: 0, defaulted: 0 };
    51→    51→      for (const l of logs) counts[l.action]++;
    52→    52→      return `Normalization: ${logs.length} fixes (${counts.missing} missing, ${counts.coerced} coerced, ${counts.invalid} invalid, ${counts.defaulted} defaulted)`;
    53→    53→    },
    54→    54→  };
    55→    55→}

    57→    57→// ─── Primitive helpers ────────────────────────────────────────────
    58→    58→
    59→    59→function uuid(): string {
    60→    60→  return crypto.randomUUID();
    61→    61→}
    62→    62→
    63→    63→function str(v: unknown, fallback: string): string {
    64→    64→  if (typeof v === 'string' && v.length > 0) return v;
    65→    65→  if (v === null || v === undefined) return fallback;
    66→    66→  return String(v).slice(0, 500) || fallback;
    67→    67→}
    68→    68→
    69→    69→function num(v: unknown, fallback: number, min?: number, max?: number): number {
    70→    70→  if (typeof v === 'number' && isFinite(v)) {
    71→    71→    let n = v;
    72→    72→    if (min !== undefined) n = Math.max(min, n);
    73→    73→    if (max !== undefined) n = Math.min(max, n);
    74→    74→    return n;
    75→    75→  }
    76→    76→  // Coerce string numbers
    77→    77→  if (typeof v === 'string') {
    78→    78→    const n = parseFloat(v.replace(/[^\d.\-]/g, ''));
    79→    79→    if (!isNaN(n) && isFinite(n)) {
    80→    80→      if (min !== undefined) return Math.max(min, n);
    81→    81→      if (max !== undefined) return Math.min(max, n);
    82→    82→      return n;
    83→    83→    }
    84→    84→  }
    85→    85→  return fallback;
    86→    86→}
    87→    87→
    88→    88→function bool(v: unknown, fallback: boolean): boolean {
    89→    89→  if (typeof v === 'boolean') return v;
    90→    90→  if (typeof v === 'string') return v === 'true' || v === '1';
    91→    91→  if (typeof v === 'number') return v !== 0;
    92→    92→  return fallback;
    93→    93→}
    94→    94→
    95→    95→function arr<T>(v: unknown, fallback: T[]): T[] {
    96→    96→  if (Array.isArray(v)) return v;
    97→    97→  return fallback;
    98→    98→}
    99→    99→
   100→   100→function obj(v: unknown, fallback: Record<string, unknown>): Record<string, unknown> {
   101→   101→  if (v !== null && typeof v === 'object' && !Array.isArray(v)) return v as Record<string, unknown>;
   102→   102→  return fallback;
   103→   103→}
   104→   104→
   105→   105→function enumVal(v: unknown, valid: Set<string>, fallback: string): string {
   106→   106→  if (typeof v === 'string' && valid.has(v)) return v;
   107→   107→  // Case-insensitive fallback
   108→   108→  if (typeof v === 'string') {
   109→   109→    const lower = v.toLowerCase();
   110→   110→    for (const validVal of valid) {
   111→   111→      if (validVal === lower) return validVal;
   112→   112→    }
   113→   113→  }
   114→   114→  return fallback;
   115→   115→}
   116→   116→
   117→   117→function oneLine(s: string): string {
   118→   118→  return s.replace(/[\r\n]+/g, ' ').replace(/\s{2,}/g, ' ').trim();
   119→   119→}
   120→   120→
   121→   121→function slugify(s: string): string {
   122→   122→  return s.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '').substring(0, 60) || 'store';
   123→   123→}
   124→   124→
   125→   125→function isUUID(s: string): boolean {
   126→   126→  return /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(s);
   127→   127→}

   129→   129→// ─── Theme normalization ──────────────────────────────────────────
   130→   130→
   131→   131→function normalizeTheme(raw: unknown, log: ReturnType<typeof createLogger>): StoreTheme {
   132→   132→  const t = obj(raw, {} as Record<string, unknown>);
   133→   133→  const theme: StoreTheme = { ...defaultTheme };
   134→   134→
   135→   135→  // Colors
   136→   136→  const colors = obj(t.colors, {} as Record<string, unknown>);
   137→   137→  const colorKeys = ['primary', 'secondary', 'accent', 'background', 'surface', 'text', 'textMuted', 'border'] as const;
   138→   138→  for (const key of colorKeys) {
   139→   139→    const v = colors[key];
   140→   140→    if (typeof v === 'string' && v.startsWith('#') && v.length >= 4) {
   141→   141→      theme.colors[key] = v;
   142→   142→    } else if (v !== undefined) {
   143→   143→      log.log({ field: `theme.colors.${key}`, action: 'invalid', from: v, to: defaultTheme.colors[key] });
   144→   144→    }
   145→   145→  }
   146→   146→
   147→   147→  // Fonts
   148→   148→  const fonts = obj(t.fonts, {} as Record<string, unknown>);
   149→   149→  theme.fonts.heading = str(fonts.heading, 'Inter');
   150→   150→  theme.fonts.body = str(fonts.body, 'Inter');
   151→   151→
   152→   152→  // Spacing
   153→   153→  theme.spacing = enumVal(t.spacing, VALID_SPACING, 'normal') as StoreTheme['spacing'];
   154→   154→  if (t.spacing && theme.spacing === 'normal' && !VALID_SPACING.has(t.spacing as string)) {
   155→   155→    log.log({ field: 'theme.spacing', action: 'invalid', from: t.spacing, to: 'normal' });
   156→   156→  }
   157→   157→
   158→   158→  // Border radius
   159→   159→  theme.borderRadius = enumVal(t.borderRadius, VALID_BORDER_RADIUS, 'md') as StoreTheme['borderRadius'];
   160→   160→
   161→   161→  return theme;
   162→   162→}

   166→   166→function normalizeSectionStyle(raw: unknown, log: ReturnType<typeof createLogger>): SectionStyle {
   167→   167→  const s = obj(raw, {} as Record<string, unknown>);

   173→   173→  if (typeof s.overlay === 'boolean') style.overlay = s.overlay;

   179→   179→
   180→   180→  return style;
   181→   181→}

   351→   351→function normalizeSection(raw: unknown, log: ReturnType<typeof createLogger>): Section {
   352→   352→  const s = obj(raw, {} as Record<string, unknown>);
   353→   353→
   354→   354→  // Validate and normalize type
   355→   355→  let type = str(s.type, 'text-banner').toLowerCase().trim() as SectionType;
   356→   356→  if (!VALID_SECTION_TYPES.has(type)) {
   357→   357→    log.log({ field: 'section.type', action: 'invalid', from: s.type, to: 'text-banner' });
   358→   358→    type = 'text-banner';
   359→   359→  }
   360→   360→
   361→   361→  return {
   362→   362→    id: isUUID(str(s.id, '')) ? str(s.id, '') : uuid(),
   363→   363→    type,
   364→   364→    content: normalizeSectionContent(type, s.content, log),
   365→   365→    style: normalizeSectionStyle(s.style, log),
   366→   366→    visible: bool(s.visible, true),
   367→   367→  };
   368→   368→}

   408→   408→/** Infer page type from slug/name if not explicitly set. */
   409→   409→function normalizePageType(
   410→   410→  rawType: unknown,
   411→   411→  slug: string,
   412→   412→  name: string,
   413→   413→  isHomepage: boolean,
   414→   414→  log: ReturnType<typeof createLogger>
   415→   415→): PageType {
   416→   416→  // Explicit type takes priority
   417→   417→  if (typeof rawType === 'string' && VALID_PAGE_TYPES.has(rawType)) {
   418→   418→    return rawType as PageType;
   419→   419→  }
   420→   420→  // Homepage is always 'home'
   421→   421→  if (isHomepage) return 'home';
   422→   422→  // Infer from slug
   423→   423→  const slugLower = slug.toLowerCase();
   424→   424→  if (slugLower === 'shop' || slugLower === 'products' || slugLower === 'collection') return 'collection';
   425→   425→  if (slugLower === 'cart') return 'cart';
   426→   426→  if (slugLower === 'checkout') return 'checkout';
   427→   427→  // Infer from name
   428→   428→  const nameLower = name.toLowerCase();
   429→   429→  if (nameLower.includes('shop') || nameLower.includes('collection') || nameLower.includes('catalog')) return 'collection';
   430→   430→  if (nameLower.includes('cart')) return 'cart';
   431→   431→  if (nameLower.includes('checkout')) return 'checkout';
   432→   432→  // Default
   433→   433→  if (rawType !== undefined) {
   434→   434→    log.log({ field: 'page.type', action: 'coerced', from: String(rawType), to: 'home' });
   435→   435→  }
   436→   436→  return 'home';
   437→   437→}

   439→   439→function normalizePage(raw: unknown, log: ReturnType<typeof createLogger>): StorePage {

   459→   459→/** Hard-cap enforcement: truncate sections and products to safe output limits.
   460→   460→ *  This is the safety net if the AI ignores the system prompt caps.
   461→   461→ *  Layers of defense:
   462→   462→ *    1. sanitizePrompt() in route.ts strips count requests (prevents AI from trying)
   463→   463→ *    2. System prompt ABSOLUTE CAPS language (strong AI guidance)
   464→   464→ *    3. THIS function: hard truncation (final safety net)
   465→   465→ */
   466→   466→function enforceOutputCaps(store: Store, log: ReturnType<typeof createLogger>): void {
   467→   467→  const MAX_PRODUCTS = 3;
   468→   468→  const MAX_SECTIONS = 4;
   469→   469→
   470→   470→  // ── Cap products ──
   471→   471→  if (store.products.length > MAX_PRODUCTS) {
   472→   472→    const before = store.products.length;
   473→   473→    store.products = store.products.slice(0, MAX_PRODUCTS);
   474→   474→    // Ensure at least 1 featured product after truncation
   475→   475→    if (!store.products.some((p) => p.featured) && store.products.length > 0) {
   476→   476→      store.products[0].featured = true;
   477→   477→    }
   478→   478→    log.log({
   479→   479→      field: 'products',
   480→   480→      action: 'coerced',
   481→   481→      from: `${before} products`,
   482→   482→      to: `${MAX_PRODUCTS} products (capped)`,
   483→   483→    });
   484→   484→  }
   485→   485→
   486→   486→  // ── Cap sections per page (only for 'home' type — template pages have 0 sections by design) ──
   487→   487→  for (const page of store.pages) {
   488→   488→    if (page.type && page.type !== 'home') continue; // Skip non-home pages
   489→   489→    if (page.sections.length > MAX_SECTIONS) {
   490→   490→      const before = page.sections.length;
   491→   491→      page.sections = page.sections.slice(0, MAX_SECTIONS);
   492→   492→      log.log({
   493→   493→        field: `pages[${page.name}].sections`,
   494→   494→        action: 'coerced',
   495→   495→        from: `${before} sections`,
   496→   496→        to: `${MAX_SECTIONS} sections (capped)`,
   497→   497→      });
   498→   498→    }
   499→   499→  }
   500→   500→}

   502→   502→/** Fix productIds in featured-products sections to reference actual products */
   503→   503→function fixProductReferences(store: Store, log: ReturnType<typeof createLogger>): void {
   504→   504→  const validProductIds = new Set(store.products.map((p) => p.id));
   505→   505→  const allProductIds = store.products.map((p) => p.id);
   506→   506→
   507→   507→  for (const page of store.pages) {
   508→   508→    for (const section of page.sections) {
   509→   509→      if (section.type === 'featured-products' && Array.isArray(section.content.productIds)) {
   510→   510→        const before = section.content.productIds.length;
   511→   511→        // Keep only IDs that reference actual products
   512→   512→        const validIds = (section.content.productIds as string[]).filter((id) => validProductIds.has(id));
   513→   513→
   514→   514→        // Fill missing slots with other product IDs so all products are shown
   515→   515→        const usedIds = new Set(validIds);
   516→   516→        for (const pid of allProductIds) {
   517→   517→          if (validIds.length >= store.products.length) break;
   518→   518→          if (!usedIds.has(pid)) {
   519→   519→            validIds.push(pid);
   520→   520→            usedIds.add(pid);
   521→   521→          }
   522→   522→        }
   523→   523→
   524→   524→        if (validIds.length !== before) {
   525→   525→          const invalidCount = before - (validIds.length - (store.products.length - before > 0 ? store.products.length - before : 0));
   526→   526→          log.log({ field: 'featured-products.productIds', action: 'coerced', from: `${before} IDs (some invalid)`, to: `${validIds.length} valid IDs` });
   527→   527→        }
   528→   528→        section.content.productIds = validIds;
   529→   529→      }
   530→   530→    }
   531→   531→  }
   532→   532→}

   636→   636→export interface NormalizeResult {
   637→   637→  store: Store;
   638→   638→  log: string[];        // Human-readable log lines
   639→   639→  summary: string;      // One-line summary
   640→   640→  isFallback: boolean;  // true if the input was too broken to normalize
   641→   641→  normalizationCount: number; // Number of fixes applied
   642→   642→}
   643→   643→
   644→   644→/**
   645→   645→ * Normalize any valid JSON object into a conformant Store.
   646→   646→ *
   647→   647→ * This function NEVER throws. If the input is too malformed to recover,
   648→   648→ * it returns null and the caller should use a hardcoded fallback store.
   649→   649→ *
   650→   650→ * @param raw - Parsed JSON object (JSON.parse already succeeded)
   651→   651→ * @param prompt - Original user prompt (for fallback name extraction)
   652→   652→ */
   653→   653→export function normalizeStore(raw: unknown, prompt?: string): NormalizeResult | null {
   654→   654→  const log = createLogger();
   655→   655→
   656→   656→  // ── Guard: must be a non-null object ──
   657→   657→  if (raw === null || typeof raw !== 'object' || Array.isArray(raw)) {
   658→   658→    console.warn('[normalizeStore] Input is not an object:', typeof raw);
   659→   659→    return null;
   660→   660→  }
   661→   661→
   662→   662→  const r = raw as Record<string, unknown>;
   663→   663→
   664→   664→  // ── Extract store name (used as fallback for missing name) ──
   665→   665→  const storeName = oneLine(str(r.name, prompt ? extractStoreNameFromPrompt(prompt) : 'My Store'));
   666→   666→
   667→   667→  if (!r.name || typeof r.name !== 'string') {
   668→   668→    log.log({ field: 'name', action: 'missing', to: storeName });
   669→   669→  }
   670→   670→
   671→   671→  // ── Top-level fields ──
   672→   672→  const now = new Date().toISOString();
   673→   673→  const store: Store = {
   674→   674→    id: isUUID(str(r.id, '')) ? str(r.id, '') : uuid(),
   675→   675→    name: storeName.substring(0, 100),
   676→   676→    slug: slugify(storeName),
   677→   677→    description: r.description !== undefined ? oneLine(str(r.description, '')) : undefined,
   678→   678→    theme: r.theme ? normalizeTheme(r.theme, log) : defaultTheme,
   679→   679→    pages: Array.isArray(r.pages)
   680→   680→      ? r.pages.map((p: unknown) => normalizePage(p, log))
   681→   681→      : [createDefaultPage(storeName)],
   682→   682→    products: Array.isArray(r.products)
   683→   683→      ? (r.products as unknown[]).map((p: unknown) => normalizeProduct(p, log))
   684→   684→      : [],
   685→   685→    published: bool(r.published, false),
   686→   686→    publishedAt: r.publishedAt !== undefined ? str(r.publishedAt, '') : undefined,
   687→   687→    createdAt: str(r.createdAt, now),
   688→   688→    updatedAt: str(r.updatedAt, now),
   689→   689→  };
   690→   690→
   691→   691→  // Log missing arrays
   692→   692→  if (!Array.isArray(r.pages)) log.log({ field: 'pages', action: 'missing' });
   693→   693→  if (!Array.isArray(r.products)) log.log({ field: 'products', action: 'missing' });
   694→   694→  if (!r.theme) log.log({ field: 'theme', action: 'missing' });
   695→   695→
   696→   696→  // ── Ensure at least one page exists ──
   697→   697→  if (store.pages.length === 0) {
   698→   698→    store.pages.push(createDefaultPage(storeName));
   699→   699→    log.log({ field: 'pages', action: 'defaulted' });
   700→   700→  }
   701→   701→
   702→   702→  // ── Ensure at least one product exists (renderer expects products array) ──
   703→   703→  if (store.products.length === 0) {
   704→   704→    store.products.push(createDefaultProduct());
   705→   705→    log.log({ field: 'products', action: 'defaulted' });
   706→   706→  }
   707→   707→
   708→   708→  // ── Cross-reference fixes ──
   709→   709→  // IMPORTANT: enforceOutputCaps MUST run before fixProductReferences
   710→   710→  // because product references depend on the final product list
   711→   711→  enforceOutputCaps(store, log);
   712→   712→  padProducts(store, log);
   713→   713→  fixProductReferences(store, log);
   714→   714→  stripJunkPages(store, log);
   715→   715→  ensureHomepage(store, log);
   716→   716→  ensureFeaturedProducts(store, log);
   717→   717→  ensureTemplatePages(store, log);
   718→   718→
   719→   719→  // ── Build result ──
   720→   720→  const logEntries = log.getLogs();
   721→   721→  const logLines = logEntries.map(
   722→   722→    (l) => `[${l.action}] ${l.field}${l.from !== undefined ? ` (was: ${JSON.stringify(l.from).substring(0, 60)})` : ''}${l.to !== undefined ? ` → ${JSON.stringify(l.to).substring(0, 60)}` : ''}`
   723→   723→  );
   724→   724→
   725→   725→  return {
   726→   726→    store,
   727→   727→    log: logLines,
   728→   728→    summary: log.summary(),
   729→   729→    isFallback: false,
   730→   730→    normalizationCount: logEntries.length,
   731→   731→  };
   732→   732→}

   774→   774→/** Extract a short store name from a user prompt (lightweight version for normalization) */
   775→   775→function extractStoreNameFromPrompt(prompt: string): string {
   776→   776→  const text = prompt.trim();
   777→   777→
   778→   778→  // Brand name after "called / named / known as"
   779→   779→  const calledMatch = text.match(/(?:called|named|known\s+as)\s+([\w&'\-]+(?:\s+[\w&'\-]+){0,2})/i);
   780→   780→  if (calledMatch?.[1]) {
   781→   781→    const name = calledMatch[1].replace(/\s+(selling|with|that|for|using|featuring)\s*$/i, '').trim();
   782→   782→    if (name.length >= 2 && name.length <= 40) return name;
   783→   783→  }
   784→   784→
   785→   785→  // Quoted brand name
   786→   786→  const quotedMatch = text.match(/["']([^"']{2,40})["']/);
   787→   787→  if (quotedMatch?.[1]) return quotedMatch[1].trim();
   788→   788→
   789→   789→  // Strip common prefixes
   790→   790→  const stripped = text
   791→   791→    .replace(/^(build|create|make|design|set\s+up)\s+(a|an|the|my)\s+/i, '')
   792→   792→    .replace(/\b(online|e-commerce|ecommerce)\s+(store|shop|boutique)\b/gi, '')
   793→   793→    .replace(/\b(store|shop|boutique|website|site|brand)\b/gi, '')
   794→   794→    .replace(/\b(selling|with|that|for|using|featuring)\b.*/i, '')
   795→   795→    .trim();
   796→   796→
   797→   797→  // Find title-case run
   798→   798→  const words = stripped.split(/\s+/).filter(w => w.length >= 2);
   799→   799→  const isTitle = (w: string) => w[0] === w[0].toUpperCase() && w[0] !== w[0].toLowerCase();
   800→   800→  let bestRun: string[] = [];
   801→   801→  let currentRun: string[] = [];
   802→   802→  for (const w of words) {
   803→   803→    if (isTitle(w)) {
   804→   804→      currentRun.push(w.replace(/[^a-zA-Z0-9&'-]/g, ''));
   805→   805→    } else {
   806→   806→      if (currentRun.length > bestRun.length) bestRun = currentRun;
   807→   807→      currentRun = [];
   808→   808→    }
   809→   809→  }
   810→   810→  if (currentRun.length > bestRun.length) bestRun = currentRun;
   811→   811→
   812→   812→  if (bestRun.length >= 1) {
   813→   813→    const candidate = bestRun.slice(0, 3).join(' ');
   814→   814→    if (candidate.length >= 2 && candidate.length <= 40) return candidate;
   815→   815→  }
   816→   816→
   817→   817→  return 'My Store';
   818→   818→}
