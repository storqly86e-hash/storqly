    38→    38→/** Get padding-y class from style config */
    39→    39→function pyClass(paddingY?: SectionStyle['paddingY']) {
    40→    40→  switch (paddingY) {
    41→    41→    case 'sm': return 'py-6';
    42→    42→    case 'md': return 'py-10';
    43→    43→    case 'lg': return 'py-16';
    44→    44→    case 'xl': return 'py-24';
    45→    45→    default: return 'py-12';
    46→    46→  }
    47→    47→}
    48→    48→
    49→    49→/** Get padding-x class from style config */
    50→    50→function pxClass(paddingX?: SectionStyle['paddingX']) {
    51→    51→  switch (paddingX) {
    52→    52→    case 'sm': return 'px-4';
    53→    53→    case 'md': return 'px-6';
    54→    54→    case 'lg': return 'px-10';
    55→    55→    default: return 'px-6';
    56→    56→  }
    57→    57→}
    58→    58→
    59→    59→/** Get max-width class from style config */
    60→    60→function maxWidthClass(maxWidth?: SectionStyle['maxWidth']) {
    61→    61→  switch (maxWidth) {
    62→    62→    case 'sm': return 'max-w-2xl';
    63→    63→    case 'md': return 'max-w-4xl';
    64→    64→    case 'lg': return 'max-w-6xl';
    65→    65→    case 'xl': return 'max-w-7xl';
    66→    66→    case 'full': return 'max-w-full';
    67→    67→    default: return 'max-w-6xl';
    68→    68→  }
    69→    69→}
    70→    70→
    71→    71→/** Get border-radius class from theme */
    72→    72→function borderRadiusClass(radius?: StoreTheme['borderRadius']) {
    73→    73→  switch (radius) {
    74→    74→    case 'none': return 'rounded-none';
    75→    75→    case 'sm': return 'rounded-sm';
    76→    76→    case 'md': return 'rounded-md';
    77→    77→    case 'lg': return 'rounded-lg';
    78→    78→    case 'xl': return 'rounded-xl';
    79→    79→    default: return 'rounded-md';
    80→    80→  }
    81→    81→}

    83→    83→/** Generate a consistent color from a string for image placeholders */
    84→    84→function stringToColor(str: string, theme: StoreTheme): string {
    85→    85→  let hash = 0;
    86→    86→  for (let i = 0; i < str.length; i++) {
    87→    87→    hash = str.charCodeAt(i) + ((hash << 5) - hash);
    88→    88→  }
    89→    89→  const hue = Math.abs(hash) % 360;
    90→    90→  return `hsl(${hue}, 35%, 88%)`;
    91→    91→}

    93→    93→/** Format price */
    94→    94→function formatPrice(price: number): string {
    95→    95→  return new Intl.NumberFormat('en-US', {
    96→    96→    style: 'currency',
    97→    97→    currency: 'USD',
    98→    98→  }).format(price);
    99→    99→}
   100→   100→
   101→   101→/** Get initials from a name */
   102→   102→function getInitials(name: string): string {
   103→   103→  return name
   104→   104→    .split(' ')
   105→   105→    .map((n) => n[0])
   106→   106→    .join('')
   107→   107→    .toUpperCase()
   108→   108→    .slice(0, 2);
   109→   109→}

   113→   113→interface SectionWrapperProps {
   114→   114→  section: Section;
   115→   115→  theme: StoreTheme;
   116→   116→  selectedSectionId?: string | null;
   117→   117→  onSelectSection?: (sectionId: string | null) => void;

   120→   120→
   121→   121→function SectionWrapper({
   122→   122→  section,
   123→   123→  theme,
   124→   124→  selectedSectionId,
   125→   125→  onSelectSection,
   126→   126→  children,
   127→   127→}: SectionWrapperProps) {
   128→   128→  const isSelected = selectedSectionId === section.id;
   129→   129→  const style = section.style;
   130→   130→
   131→   131→  const sectionBg = style.backgroundColor || theme.colors.background;
   132→   132→  const sectionText = style.textColor || theme.colors.text;
   133→   133→
   134→   134→  const bgImage = style.backgroundImage
   135→   135→    ? `url(${style.backgroundImage})`
   136→   136→    : undefined;
   137→   137→
   138→   138→  const outerClasses = [
   139→   139→    'relative transition-all duration-200 cursor-pointer',
   140→   140→    pyClass(style.paddingY),
   141→   141→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   142→   142→  ].join(' ');
   143→   143→
   144→   144→  return (
   145→   145→    <div
   146→   146→      className={outerClasses}
   147→   147→      style={{
   148→   148→        backgroundColor: sectionBg,
   149→   149→        color: sectionText,
   150→   150→        backgroundImage: bgImage,
   151→   151→        backgroundSize: 'cover',
   152→   152→        backgroundPosition: 'center',
   153→   153→      }}
   154→   154→      onClick={(e) => {
   155→   155→        e.stopPropagation();
   156→   156→        onSelectSection?.(isSelected ? null : section.id);
   157→   157→      }}
   158→   158→      data-section-id={section.id}
   159→   159→      data-section-type={section.type}
   160→   160→    >
   161→   161→      {/* Overlay for background images */}
   162→   162→      {style.backgroundImage && style.overlay && (
   163→   163→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   164→   164→      )}
   165→   165→      <div
   166→   166→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   167→   167→      >
   168→   168→        {/* Selected section label */}
   169→   169→        {isSelected && (
   170→   170→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   171→   171→            {section.type.replace(/-/g, ' ')}
   172→   172→          </div>
   173→   173→        )}
   174→   174→        {children}
   175→   175→      </div>
   176→   176→    </div>
   177→   177→  );
   178→   178→}

   182→   182→function ProductCard({
   183→   183→  product,

   187→   187→}: {

   264→   264→            Add to Cart
   265→   265→          </button>
   266→   266→        )}
   267→   267→      </div>
   268→   268→    </div>
   269→   269→  );
   270→   270→}

   377→   377→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {

   381→   381→

   383→   383→  const alignMap = {
   384→   384→    left: 'items-start text-left',

   386→   386→    right: 'items-end text-right',

   388→   388→
   389→   389→  // Only apply theme gradient as default background when no custom color or image is set.

   391→   391→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   392→   392→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   393→   393→    : undefined;

   395→   395→  const outerClasses = [

   400→   400→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',

   402→   402→
   403→   403→  return (

   406→   406→      style={{
   407→   407→        backgroundColor: style.backgroundColor || theme.colors.primary,

   416→   416→      data-section-id={section.id}
