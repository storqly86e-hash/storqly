    38→/** Get padding-y class from style config */
    39→function pyClass(paddingY?: SectionStyle['paddingY']) {
    40→  switch (paddingY) {
    41→    case 'sm': return 'py-6';
    42→    case 'md': return 'py-10';
    43→    case 'lg': return 'py-16';
    44→    case 'xl': return 'py-24';
    45→    default: return 'py-12';
    46→  }
    47→}
    48→
    49→/** Get padding-x class from style config */
    50→function pxClass(paddingX?: SectionStyle['paddingX']) {
    51→  switch (paddingX) {
    52→    case 'sm': return 'px-4';
    53→    case 'md': return 'px-6';
    54→    case 'lg': return 'px-10';
    55→    default: return 'px-6';
    56→  }
    57→}
    58→
    59→/** Get max-width class from style config */
    60→function maxWidthClass(maxWidth?: SectionStyle['maxWidth']) {
    61→  switch (maxWidth) {
    62→    case 'sm': return 'max-w-2xl';
    63→    case 'md': return 'max-w-4xl';
    64→    case 'lg': return 'max-w-6xl';
    65→    case 'xl': return 'max-w-7xl';
    66→    case 'full': return 'max-w-full';
    67→    default: return 'max-w-6xl';
    68→  }
    69→}
    70→
    71→/** Get border-radius class from theme */
    72→function borderRadiusClass(radius?: StoreTheme['borderRadius']) {
    73→  switch (radius) {
    74→    case 'none': return 'rounded-none';
    75→    case 'sm': return 'rounded-sm';
    76→    case 'md': return 'rounded-md';
    77→    case 'lg': return 'rounded-lg';
    78→    case 'xl': return 'rounded-xl';
    79→    default: return 'rounded-md';
    80→  }
    81→}
    82→
    83→/** Generate a consistent color from a string for image placeholders */
    84→function stringToColor(str: string, theme: StoreTheme): string {
    85→  let hash = 0;
    86→  for (let i = 0; i < str.length; i++) {
    87→    hash = str.charCodeAt(i) + ((hash << 5) - hash);
    88→  }
    89→  const hue = Math.abs(hash) % 360;
    90→  return `hsl(${hue}, 35%, 88%)`;
    91→}
    92→
    93→/** Get a contrasting text color for a given background color */
    94→function contrastTextColor(bgColor: string): string {
    95→  const hex = bgColor.replace('#', '');
    96→  if (hex.length !== 6) return '#111827';
    97→  const r = parseInt(hex.substring(0, 2), 16);
    98→  const g = parseInt(hex.substring(2, 4), 16);
    99→  const b = parseInt(hex.substring(4, 6), 16);
   100→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   101→  return luminance > 0.55 ? '#111827' : '#ffffff';
   102→}
   103→
   104→/** Format price */
   105→function formatPrice(price: number): string {
   106→  return new Intl.NumberFormat('en-US', {
   107→    style: 'currency',
   108→    currency: 'USD',
   109→  }).format(price);
   110→}
   111→
   112→/** Get initials from a name */
   113→function getInitials(name: string): string {
   114→  return name
   115→    .split(' ')
   116→    .map((n) => n[0])
   117→    .join('')
   118→    .toUpperCase()
   119→    .slice(0, 2);
   120→}

   122→// ─── Section Wrapper ────────────────────────────────────────────────────

   124→interface SectionWrapperProps {
   125→  section: Section;
   126→  theme: StoreTheme;
   127→  selectedSectionId?: string | null;
   128→  onSelectSection?: (sectionId: string | null) => void;
   129→  children: React.ReactNode;
   130→}
   131→
   132→function SectionWrapper({
   133→  section,
   134→  theme,
   135→  selectedSectionId,
   136→  onSelectSection,
   137→  children,
   138→}: SectionWrapperProps) {
   139→  const isSelected = selectedSectionId === section.id;
   140→  const style = section.style;
   141→
   142→  const sectionBg = style.backgroundColor || theme.colors.background;
   143→  const sectionText = style.textColor || theme.colors.text;
   144→
   145→  const bgImage = style.backgroundImage
   146→    ? `url(${style.backgroundImage})`
   147→    : undefined;
   148→
   149→  const outerClasses = [
   150→    'relative transition-all duration-200 cursor-pointer',
   151→    pyClass(style.paddingY),
   152→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   153→  ].join(' ');
   154→
   155→  return (
   156→    <div
   157→      className={outerClasses}
   158→      style={{
   159→        backgroundColor: sectionBg,
   160→        color: sectionText,
   161→        backgroundImage: bgImage,
   162→        backgroundSize: 'cover',
   163→        backgroundPosition: 'center',
   164→      }}
   165→      onClick={(e) => {
   166→        e.stopPropagation();
   167→        onSelectSection?.(isSelected ? null : section.id);
   168→      }}
   169→      data-section-id={section.id}
   170→      data-section-type={section.type}
   171→    >
   172→      {/* Overlay for background images */}
   173→      {style.backgroundImage && style.overlay && (
   174→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   175→      )}
   176→      <div
   177→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   178→      >
   179→        {/* Selected section label */}
   180→        {isSelected && (
   181→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   182→            {section.type.replace(/-/g, ' ')}
   183→          </div>
   184→        )}
   185→        {children}
   186→      </div>
   187→    </div>
   188→  );
   189→}

   191→// ─── Product Card (shared) ─────────────────────────────────────────────
   192→
   193→function ProductCard({
   194→  product,
   195→  theme,
   196→  showAddToCart,
   197→  borderRadius,
   198→}: {
   199→  product: StoreProduct;
   200→  theme: StoreTheme;
   201→  showAddToCart: boolean;
   202→  borderRadius: string;
   203→}) {
   204→  const [hovered, setHovered] = useState(false);
   205→  const imgColor = stringToColor(product.id, theme);
   206→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   207→
   208→  return (
   209→    <div
   210→      className={`${borderRadius} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   211→      style={{
   212→        borderColor: theme.colors.border,
   213→      }}
   214→      onMouseEnter={() => setHovered(true)}
   215→      onMouseLeave={() => setHovered(false)}
   216→    >
   217→ {/* Image placeholder */}
   218→      <div
   219→        className="relative aspect-square w-full overflow-hidden"
   220→        style={{ backgroundColor: imgColor }}
   221→      >
   222→        <div className="absolute inset-0 flex items-center justify-center">
   223→          <div
   224→            className="h-16 w-16 rounded-full opacity-30"
   225→            style={{ backgroundColor: theme.colors.primary }}
   226→          />
   227→        </div>
   228→        {/* Out of stock overlay */}
   229→        {!product.inStock && (
   230→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   231→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   232→              Sold Out
   233→            </span>
   234→          </div>
   235→        )}
   236→        {/* Discount badge */}
   237→        {hasDiscount && product.compareAtPrice && (
   238→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   239→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   240→          </div>
   241→        )}
   242→      </div>
   243→      {/* Info */}
   244→      <div className="p-3 sm:p-4">
   245→        {product.category && (
   246→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   247→            {product.category}
   248→          </p>
   249→        )}
   250→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   251→          {product.name}
   252→        </h3>
   253→        <div className="mt-1.5 flex items-center gap-2">
   254→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   255→            {formatPrice(product.price)}
   256→          </span>
   257→          {hasDiscount && product.compareAtPrice && (
   258→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   259→              {formatPrice(product.compareAtPrice)}
   260→            </span>
   261→          )}
   262→        </div>
   263→        {showAddToCart && product.inStock && (
   264→          <button
   265→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 ${
   266→              hovered
   267→                ? 'opacity-100 translate-y-0'
   268→                : 'opacity-0 translate-y-1'
   269→            }`}
   270→            style={{
   271→              backgroundColor: theme.colors.primary,
   272→              color: contrastTextColor(theme.colors.primary),
   273→            }}
   274→          >
   275→            Add to Cart
   276→          </button>
   277→        )}
   278→      </div>
   279→    </div>
   280→  );
   281→}

   283→// ─── Column grid class helper ───────────────────────────────────────────
   284→
   285→function gridCols(columns: 2 | 3 | 4) {
   286→  switch (columns) {
   287→    case 2: return 'grid-cols-1 sm:grid-cols-2';
   288→    case 3: return 'grid-cols-1 sm:grid-cols-2 lg:grid-cols-3';
   289→    case 4: return 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-4';
   290→    default: return 'grid-cols-1 sm:grid-cols-2 lg:grid-cols-3';
   291→  }
   292→}

   388→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   389→  const content = section.content as unknown as HeroContent;
   390→  const style = section.style;
   391→  const isSelected = selectedSectionId === section.id;
   392→
   393→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   394→  const alignMap = {
   395→    left: 'items-start text-left',
   396→    center: 'items-center text-center',
   397→    right: 'items-end text-right',
   398→  };
   399→
   400→  // Only apply theme gradient as default background when no custom color or image is set.
   401→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   402→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   403→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   404→    : undefined;
   405→
   406→  const outerClasses = [
   407→    'relative flex transition-all duration-200 cursor-pointer',
   408→    heightMap[content.height] || heightMap.md,
   409→    alignMap[content.alignment],
   410→    pyClass(style.paddingY),
   411→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   412→  ].join(' ');
   413→
   414→  return (
   415→    <div
   416→      className={outerClasses}
   417→      style={{
   418→        backgroundColor: style.backgroundColor || theme.colors.primary,
   419→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   420→        backgroundSize: 'cover',
   421→        backgroundPosition: 'center',
   422→        color: style.textColor || '#ffffff',
   423→      }}
   424→      onClick={(e) => {
   425→        e.stopPropagation();
   426→        onSelectSection?.(isSelected ? null : section.id);
   427→      }}
   428→      data-section-id={section.id}
   429→      data-section-type={section.type}
   430→    >
   431→      {/* Overlay */}
   432→      {style.backgroundImage && style.overlay && (
   433→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   434→      )}
   435→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   436→      {!style.backgroundImage && !style.backgroundColor && (
   437→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   438→      )}
   439→      {!style.backgroundImage && !style.backgroundColor && (
   440→        <>
   441→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   442→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />
   443→        </>
   444→      )}
   445→
   446→      <div className={`relative z-10 mx-auto w-full ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}>
   447→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl">
   448→          {content.headline}
   449→        </h1>
   450→        {content.subheadline && (
   451→          <p className="mt-4 max-w-xl text-base opacity-80 sm:text-lg md:text-xl">
   452→            {content.subheadline}
   453→          </p>
   454→        )}
   455→        {content.ctaText && (
   456→          <button className="mt-8 inline-flex items-center gap-2 rounded-lg bg-white px-6 py-3 text-sm font-semibold shadow-lg transition-transform hover:scale-105 sm:text-base" style={{ color: contrastTextColor('#ffffff') }}>
   457→            {content.ctaText}
   458→            <ArrowRight className="h-4 w-4" />
   459→          </button>
   460→        )}
   461→      </div>
   462→    </div>
   463→  );
   464→}

   468→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   469→  const content = section.content as unknown as FeaturedProductsContent;
   470→  const borderRadius = borderRadiusClass(theme.borderRadius);
   471→  const featuredProducts = useMemo(
   472→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   473→    [content.productIds, products],
   474→  );
   475→
   476→  return (
   477→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   478→      {content.headline && (
   479→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl">
   480→          {content.headline}
   481→        </h2>
   482→      )}
   483→      {content.subtitle && (
   484→        <p className="mb-8 text-sm opacity-65">
   485→          {content.subtitle}
   486→        </p>
   487→      )}
   488→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   489→        {featuredProducts.map((product) => (
   490→          <ProductCard
   491→            key={product.id}
   492→            product={product}
   493→            theme={theme}
   494→            showAddToCart={content.showAddToCart}
   495→            borderRadius={borderRadius}
   496→          />
   497→        ))}
   498→      </div>
   499→    </SectionWrapper>
   500→  );
   501→}

   505→export function ProductGridSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   506→  const content = section.content as unknown as ProductGridContent;
   507→  const borderRadius = borderRadiusClass(theme.borderRadius);
   508→
   509→  const filteredProducts = useMemo(() => {
   510→    let prods = products;
   511→    if (content.filterByCategory) {
   512→      prods = prods.filter((p) => p.category === content.filterByCategory);
   513→    }
   514→    return prods;
   515→  }, [products, content.filterByCategory]);
   516→
   517→  return (
   518→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   519→      {content.headline && (
   520→        <h2 className="mb-8 text-2xl font-bold sm:text-3xl">
   521→          {content.headline}
   522→        </h2>
   523→      )}
   524→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   525→        {filteredProducts.map((product) => (
   526→          <ProductCard
   527→            key={product.id}
   528→            product={product}
   529→            theme={theme}
   530→            showAddToCart={content.showAddToCart}
   531→            borderRadius={borderRadius}
   532→          />
   533→        ))}
   534→      </div>
   535→      {filteredProducts.length === 0 && (
   536→        <div className="py-16 text-center opacity-65">
   537→          <p className="text-sm">No products found.</p>
   538→        </div>
   539→      )}
   540→    </SectionWrapper>
   541→  );
   542→}

   546→export function TextBannerSection({ section, theme, selectedSectionId, onSelectSection }: SectionRendererProps) {
   547→  const content = section.content as unknown as TextBannerContent;
   548→  const alignMap = {
   549→    left: 'text-left',
   550→    center: 'text-center',
   551→    right: 'text-right',
   552→  };
   553→  const sizeMap = {
   554→    sm: { headline: 'text-xl sm:text-2xl', body: 'text-sm' },
   555→    md: { headline: 'text-2xl sm:text-3xl md:text-4xl', body: 'text-base' },
   556→    lg: { headline: 'text-3xl sm:text-4xl md:text-5xl lg:text-6xl', body: 'text-lg' },
   557→  };
   558→  const sizes = sizeMap[content.size] || sizeMap.md;

   560→  return (
   561→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   562→      <div className={`${alignMap[content.alignment]}`}>
   563→        <h2 className={`font-bold leading-tight tracking-tight ${sizes.headline}`}>
   564→          {content.headline}
   565→        </h2>
   566→        {content.body && (
   567→          <p className={`mt-3 max-w-2xl ${sizes.body} opacity-65`}>
   568→            {content.body}
   569→          </p>
   570→        )}
   571→      </div>
   572→    </SectionWrapper>
   573→  );

  1033→// ─── Main Section Router ────────────────────────────────────────────────
  1034→
  1035→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1036→  const { section } = props;
  1037→
  1038→  if (!section.visible) return null;
  1039→
  1040→  switch (section.type) {
  1041→    case 'header':
  1042→      return <HeaderSection {...props} />;
  1043→    case 'hero':
  1044→      return <HeroSection {...props} />;
  1045→    case 'featured-products':
  1046→      return <FeaturedProductsSection {...props} />;
  1047→    case 'product-grid':
  1048→      return <ProductGridSection {...props} />;
  1049→    case 'text-banner':
  1050→      return <TextBannerSection {...props} />;
  1051→    case 'image-gallery':
  1052→      return <ImageGallerySection {...props} />;
  1053→    case 'testimonials':
  1054→      return <TestimonialsSection {...props} />;
  1055→    case 'newsletter':
  1056→      return <NewsletterSection {...props} />;
  1057→    case 'faq':
  1058→      return <FAQSection {...props} />;
  1059→    case 'cta':
  1060→      return <CTASection {...props} />;
  1061→    case 'categories':
  1062→      return <CategoriesSection {...props} />;
  1063→    case 'footer':
  1064→      return <FooterSection {...props} />;
  1065→    case 'rich-text':
  1066→      return <RichTextSection {...props} />;
  1067→    case 'spacer':
  1068→      return <SpacerSection {...props} />;
  1069→    case 'divider':
  1070→      return <DividerSection {...props} />;
  1071→    default:
  1072→      return null;
  1073→  }
  1074→}
