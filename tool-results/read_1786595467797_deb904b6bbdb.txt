   246→function ProductCard({
   247→  product,
   248→  theme,
   249→  showAddToCart,
   250→  borderRadius,
   251→  buttonBgOverride,
   252→  buttonTextOverride,
   253→}: {
   254→  product: StoreProduct;
   255→  theme: StoreTheme;
   256→  showAddToCart: boolean;
   257→  borderRadius: string;
   258→  buttonBgOverride?: string;
   259→  buttonTextOverride?: string;
   260→}) {
   261→  const [hovered, setHovered] = useState(false);
   262→  const imgColor = stringToColor(product.id, theme);
   263→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   264→
   265→  return (
   266→    <div
   267→      className={`${borderRadius} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   268→      style={{
   269→        borderColor: theme.colors.border,
   270→      }}
   271→      onMouseEnter={() => setHovered(true)}
   272→      onMouseLeave={() => setHovered(false)}
   273→    >
   274→ {/* Image placeholder */}
   275→      <div
   276→        className="relative aspect-square w-full overflow-hidden"
   277→        style={{ backgroundColor: imgColor }}
   278→      >
   279→        <div className="absolute inset-0 flex items-center justify-center">
   280→          <div
   281→            className="h-16 w-16 rounded-full opacity-30"
   282→            style={{ backgroundColor: theme.colors.primary }}
   283→          />
   284→        </div>
   285→        {/* Out of stock overlay */}
   286→        {!product.inStock && (
   287→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   288→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   289→              Sold Out
   290→            </span>
   291→          </div>
   292→        )}
   293→        {/* Discount badge */}
   294→        {hasDiscount && product.compareAtPrice && (
   295→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   296→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   297→          </div>
   298→        )}
   299→      </div>
   300→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   301→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   302→        {product.category && (
   303→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   304→            {product.category}
   305→          </p>
   306→        )}
   307→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   308→          {product.name}
   309→        </h3>
   310→        <div className="mt-1.5 flex items-center gap-2">
   311→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   312→            {formatPrice(product.price)}
   313→          </span>
   314→          {hasDiscount && product.compareAtPrice && (
   315→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   316→              {formatPrice(product.compareAtPrice)}
   317→            </span>
   318→          )}
   319→        </div>
   320→        {showAddToCart && product.inStock && (
   321→          <button
   322→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 ${
   323→              hovered
   324→                ? 'opacity-100 translate-y-0'
   325→                : 'opacity-0 translate-y-1'
   326→            }`}
   327→            style={{
   328→              backgroundColor: buttonBgOverride || theme.colors.primary,
   329→              color: buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary),
   330→            }}
   331→          >
   332→            Add to Cart
   333→          </button>
   334→        )}
   335→      </div>
   336→    </div>
   337→  );
   338→}

   445→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   446→  const content = section.content as unknown as HeroContent;
   447→  const style = section.style;
   448→  const isSelected = selectedSectionId === section.id;
   449→
   450→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   451→  const alignMap = {
   452→    left: 'items-start text-left',
   453→    center: 'items-center text-center',
   454→    right: 'items-end text-right',
   455→  };
   456→
   457→  // Only apply theme gradient as default background when no custom color or image is set.
   458→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   459→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   460→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   461→    : undefined;
   462→
   463→  const outerClasses = [
   464→    'relative flex transition-all duration-200 cursor-pointer',
   465→    heightMap[content.height] || heightMap.md,
   466→    alignMap[content.alignment],
   467→    pyClass(style.paddingY),
   468→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   469→  ].join(' ');
   470→
   471→  return (
   472→    <div
   473→      className={outerClasses}
   474→      style={{
   475→        backgroundColor: style.backgroundColor || theme.colors.primary,
   476→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   477→        backgroundSize: 'cover',
   478→        backgroundPosition: 'center',
   479→        color: style.textColor || '#ffffff',
   480→      }}
   481→      onClick={(e) => {
   482→        e.stopPropagation();
   483→        onSelectSection?.(isSelected ? null : section.id);
   484→      }}
   485→      data-section-id={section.id}
   486→      data-section-type={section.type}
   487→    >
   488→      {/* Overlay */}
   489→      {style.backgroundImage && style.overlay && (
   490→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   491→      )}
   492→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   493→      {!style.backgroundImage && !style.backgroundColor && (
   494→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   495→      )}
   496→      {!style.backgroundImage && !style.backgroundColor && (
   497→        <>
   498→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   499→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />

   504→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>

  1108→// ─── Main Section Router ────────────────────────────────────────────────
  1109→
  1110→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1111→  const { section } = props;
  1112→
  1113→  if (!section.visible) return null;
  1114→
  1115→  switch (section.type) {
  1116→    case 'header':
  1117→      return <HeaderSection {...props} />;
  1118→    case 'hero':
  1119→      return <HeroSection {...props} />;
  1120→    case 'featured-products':
  1121→      return <FeaturedProductsSection {...props} />;
  1122→    case 'product-grid':
  1123→      return <ProductGridSection {...props} />;
  1124→    case 'text-banner':
  1125→      return <TextBannerSection {...props} />;
  1126→    case 'image-gallery':
  1127→      return <ImageGallerySection {...props} />;
  1128→    case 'testimonials':
  1129→      return <TestimonialsSection {...props} />;
  1130→    case 'newsletter':
  1131→      return <NewsletterSection {...props} />;
  1132→    case 'faq':
  1133→      return <FAQSection {...props} />;
  1134→    case 'cta':
  1135→      return <CTASection {...props} />;
  1136→    case 'categories':
  1137→      return <CategoriesSection {...props} />;
  1138→    case 'footer':
  1139→      return <FooterSection {...props} />;
  1140→    case 'rich-text':
  1141→      return <RichTextSection {...props} />;
  1142→    case 'spacer':
  1143→      return <SpacerSection {...props} />;
  1144→    case 'divider':
  1145→      return <DividerSection {...props} />;
  1146→    default:
  1147→      return null;
  1148→  }
  1149→}
