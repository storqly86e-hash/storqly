   103→// ─── Background Image Enrichment (lazy, non-blocking) ─────
   104→// Triggered after store generation completes. Fetches real product
   105→// images sequentially without blocking the user or exhausting rate limits.
   106→
   107→function triggerBackgroundImageEnrichment(store: Store) {
   108→  if (!store.products || store.products.length === 0) return
   109→
   110→  // Only enrich products that still have AI-hallucinated Unsplash URLs
   111→  // (Real enriched URLs from the image search service will have different patterns)
   112→  const needsEnrichment = store.products.filter(p =>
   113→    p.images.length > 0 && p.images[0].includes('unsplash.com/photo-')
   114→  )
   115→
   116→  // Scan homepage sections for backgroundImage URLs that need enrichment
   117→  const homepage = store.pages.find(p => p.isHomepage)
   118→  const sectionBackgrounds: { sectionId: string; query: string; currentUrl: string }[] = []
   119→  if (homepage) {
   120→    for (const section of homepage.sections) {
   121→      const bgUrl = section.style?.backgroundImage
   122→      if (bgUrl && bgUrl.includes('unsplash.com/photo-')) {
   123→        // Build a query from the section type and store name
   124→        const query = `${store.name} ${section.type} lifestyle setting photo`
   125→        sectionBackgrounds.push({ sectionId: section.id, query, currentUrl: bgUrl })
   126→      }
   127→    }
   128→  }
   129→
   130→  // Skip if nothing needs enrichment
   131→  if (needsEnrichment.length === 0 && sectionBackgrounds.length === 0) return
   132→
   133→  console.log(`[Storqly] Background image enrichment: ${needsEnrichment.length}/${store.products.length} products, ${sectionBackgrounds.length} section backgrounds`)
   134→
   135→  fetch('/api/store/enrich-images', {
   136→    method: 'POST',
   137→    headers: { 'Content-Type': 'application/json' },
   138→    body: JSON.stringify({
   139→      products: needsEnrichment.map(p => ({
   140→        id: p.id,
   141→        name: p.name,
   142→        images: [...p.images],
   143→        category: p.category,
   144→        description: p.description,
   145→      })),
   146→      storeName: store.name,
   147→      sectionBackgrounds: sectionBackgrounds.length > 0 ? sectionBackgrounds : undefined,
   148→    }),
   149→  })
   150→    .then(res => res.json())
   151→    .then(data => {
   152→      // Update product images in the store via the Zustand store
   153→      const hasProductEnrichment = data.enriched > 0
   154→      const hasSectionEnrichment = data.sectionBackgrounds && data.sectionBackgrounds.length > 0
   155→
   156→      if (hasProductEnrichment || hasSectionEnrichment) {
   157→        console.log(`[Storqly] Background enrichment complete: ${data.enriched} enriched, ${data.kept} kept, ${data.failed} failed in ${data.latencyMs}ms`)
   158→        // Dynamic import to avoid circular dependency
   159→        import('@/lib/store').then(({ useStoreEditor }) => {
   160→          const currentStore = useStoreEditor.getState().store
   161→          if (currentStore) {
   162→            // Update product images
   163→            let updatedProducts = currentStore.products
   164→            if (hasProductEnrichment) {
   165→              updatedProducts = currentStore.products.map(prod => {
   166→                const enriched = data.products?.find((ep: { id: string; images: string[] }) => ep.id === prod.id)
   167→                if (enriched && enriched.images[0] !== prod.images[0]) {
   168→                  return { ...prod, images: enriched.images }
   169→                }
   170→                return prod
   171→              })
   172→            }
   173→
   174→            // Update section background images
   175→            let updatedPages = currentStore.pages
   176→            if (hasSectionEnrichment) {
   177→              const bgMap = new Map(data.sectionBackgrounds.map((sb: { sectionId: string; url: string }) => [sb.sectionId, sb.url]))
   178→              updatedPages = currentStore.pages.map(page => ({
   179→                ...page,
   180→                sections: page.sections.map(section => {
   181→                  const newBg = bgMap.get(section.id)
   182→                  if (newBg && section.style?.backgroundImage !== newBg) {
   183→                    return { ...section, style: { ...section.style, backgroundImage: newBg } }
   184→                  }
   185→                  return section
   186→                }),
   187→              }))
   188→            }
   189→
   190→            useStoreEditor.getState().setStore({ ...currentStore, products: updatedProducts, pages: updatedPages })
   191→          }
   192→        })
   193→      }
   194→    })
   195→    .catch(err => {
   196→      // Non-fatal — products still show AI placeholder images
   197→      console.warn('[Storqly] Background enrichment failed (non-fatal):', err)
   198→    })
   199→}

   255→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)

   258→

   260→    isGenerating,

   361→
   362→    // Hard timeout: 120 seconds — no request should ever take longer than this.
   363→    // If the server is hung or the proxy drops the connection, the user gets a
   364→    // clear message instead of waiting indefinitely.

   366→    let timedOut = false

   372→
   373→    // Elapsed time counter
   374→    elapsedTimerRef.current = setInterval(() => {
   375→      setElapsedSeconds((prev) => prev + 1)
   376→    }, 1000)
   377→
   378→    // Fallback progress cycling (used only if SSE progress events stop)
   379→    let fallbackIdx = 0
   380→    let receivedSSEProgress = false
   381→    progressTimerRef.current = setInterval(() => {
   382→      if (!receivedSSEProgress) {
   383→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   384→        setGenerationStatus(progressMessages[fallbackIdx])
   385→      }
   386→    }, 3000)
   387→
   388→    // ── Helper: clean up on any error path (no throw, no crash overlay) ──
   389→    const finishWithError = (message: string) => {
   390→      clearTimeout(timeoutId)
   391→      clearTimers()
   392→      setError(message)
   393→      setIsGenerating(false)
   394→      setGenerationStatus('')
   395→      setElapsedSeconds(0)
   396→      toast.error('Store generation failed', { description: message })
   397→    }
   398→
   399→    const finishOk = () => {
   400→      clearTimeout(timeoutId)
   401→      clearTimers()
   402→      setIsGenerating(false)
   403→      setGenerationStatus('')
   404→      setElapsedSeconds(0)
   405→    }
   406→
   407→    try {
   408→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   409→
   410→      // ── Fetch with auto-retry for transient gateway errors (502/503/504) ──
   411→      const RETRYABLE_STATUSES = [502, 503, 504]
   412→      let res: Response | undefined
   413→      let retryCount = 0
   414→      const maxRetries = 2
   415→
   416→      while (retryCount <= maxRetries) {
   417→        try {
   418→          res = await fetch('/api/store/generate', {
   419→            method: 'POST',
   420→            headers: { 'Content-Type': 'application/json' },
   421→            body: JSON.stringify({ prompt: trimmed }),
   422→            signal: controller.signal,
   423→          })
   424→
   425→          // If the status is retryable, wait and loop again
   426→          if (!res.ok && RETRYABLE_STATUSES.includes(res.status) && retryCount < maxRetries) {
   427→            retryCount++
   428→            console.warn(`[Storqly] Gateway error ${res.status}, retrying (${retryCount}/${maxRetries}) in 3s...`)
   429→            setGenerationStatus('Connection issue — retrying...')
   430→            await new Promise(r => setTimeout(r, 3000))
   431→            continue
   432→          }
   433→          break // Success or non-retryable error — exit loop
   434→        } catch (fetchErr: unknown) {
   435→          // Network-level error (DNS failure, connection refused, etc.)
   436→          if (fetchErr instanceof TypeError && retryCount < maxRetries) {
   437→            retryCount++
   438→            console.warn(`[Storqly] Network error, retrying (${retryCount}/${maxRetries}) in 3s...`)
   439→            setGenerationStatus('Connection issue — retrying...')
   440→            await new Promise(r => setTimeout(r, 3000))
   441→            continue
   442→          }
   443→          // Out of retries or not a network error — handle gracefully
   444→          const msg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr)
   445→          if (controller.signal.aborted) {
   446→            finishWithError('Generation was cancelled.')
   447→          } else {
   448→            console.warn('[Storqly] Network error after retries:', msg)
   449→            finishWithError('Could not connect to the server. Please check your connection and try again.')
   450→          }
   451→          return
   452→        }
   453→      }
   454→
   455→      if (!res!.ok) {
   456→        let errorMsg = `Server error (${res!.status})`
   457→        try {
   458→          const errorData = await res!.json()
   459→          errorMsg = errorData.error || errorMsg
   460→        } catch { /* ignore parse error */ }
   461→        console.warn('[Storqly] HTTP error:', errorMsg)
   462→        finishWithError(errorMsg)
   463→        return
   464→      }
   465→
   466→      // ── Consume SSE stream ──
   467→      const reader = res.body?.getReader()
   468→      if (!reader) {
   469→        finishWithError('No response stream received from server.')
   470→        return
   471→      }
   472→
   473→      const decoder = new TextDecoder()
   474→      let buffer = ''
   475→      let resolved = false
   476→
   477→      while (true) {
   478→        const { done, value } = await reader.read()
   479→        if (done) break
   480→
   481→        buffer += decoder.decode(value, { stream: true })
   482→
   483→        // Parse SSE events from buffer
   484→        const lines = buffer.split('\n')
   485→        buffer = lines.pop() || ''
   486→
   487→        let currentEvent = ''
   488→        for (const line of lines) {
   489→          if (line.startsWith(':')) continue
   490→          if (line.startsWith('event: ')) {
   491→            currentEvent = line.slice(7).trim()
   492→          } else if (line.startsWith('data: ')) {
   493→            const dataStr = line.slice(6)
   494→
   495→            if (currentEvent === 'progress') {
   496→              receivedSSEProgress = true
   497→              try {
   498→                const data = JSON.parse(dataStr)
   499→                setGenerationStatus(data.message || 'Processing...')
   500→              } catch { /* ignore */ }
   501→            } else if (currentEvent === 'result') {
   502→              try {
   503→                const data = JSON.parse(dataStr)
   504→                if (!data.store) {
   505→                  console.warn('[Storqly] Result event missing store data')
   506→                  finishWithError('The AI response was missing store data.')
   507→                  return
   508→                }
   509→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   510→                resolved = true
   511→
   512→                if (data._isFallback) {
   513→                  // Differentiate "all providers down" from other fallback reasons
   514→                  const reason = data._fallbackReason || 'AI generation failed'
   515→                  const isProviderFailure = reason.includes('All') || reason.includes('429') || reason.includes('403') || reason.includes('404') || reason.includes('providers failed')
   516→                  toast.warning(
   517→                    isProviderFailure
   518→                      ? 'All AI providers are currently unavailable. You are viewing a starter template. Try again later.'
   519→                      : 'AI service returned invalid data — showing starter template. Try again.',
   520→                    { duration: 8000 }
   521→                  )
   522→                  setStoreWithFallback(data.store, true, reason)
   523→                } else {
   524→                  setStore(data.store)
   525→                  // Trigger lazy background image enrichment (non-blocking)
   526→                  triggerBackgroundImageEnrichment(data.store)
   527→                }
   528→
   529→                // Soft cap toast
   530→                if (data._productCapHit) {
   531→                  toast.info(
   532→                    `Generated ${data._generatedCount} products — for larger catalogs, you can add more via the chat editor.`,
   533→                    { duration: 8000 }
   534→                  )
   535→                }
   536→
   537→                finishOk()
   538→                return
   539→              } catch (parseErr) {
   540→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   541→                console.warn('[Storqly] Failed to parse result event:', msg)
   542→                finishWithError(msg)
   543→                return
   544→              }
   545→            } else if (currentEvent === 'error') {
   546→              try {
   547→                const data = JSON.parse(dataStr)
   548→                const serverMsg = data.message || 'Generation failed'
   549→                console.warn('[Storqly] Server error event:', serverMsg)
   550→                finishWithError(serverMsg)
   551→                return
   552→              } catch {
   553→                finishWithError('Generation failed')
   554→                return
   555→              }
   556→            }
   557→            currentEvent = ''
   558→          }
   559→        }
   560→      }
   561→
   562→      if (!resolved) {
   563→        console.warn('[Storqly] Stream ended without a result event')
   564→        finishWithError('Stream ended without a result. The server may have disconnected.')

  1155→function EditorToolbar({
  1156→  onToggleLeft,
  1157→  onToggleRight,
  1158→  showLeft,
  1159→  showRight,
  1160→}: {
  1161→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
  1162→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
  1163→  showLeft: boolean
  1164→  showRight: boolean
  1165→}) {
  1166→  const store = useStoreEditor((s) => s.store)
  1167→  const reset = useStoreEditor((s) => s.reset)
  1168→  const isPublishing = useStoreEditor((s) => s.isPublishing)
  1169→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
  1170→  const isPublished = useStoreEditor((s) => s.isPublished)
  1171→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
  1172→  const [isSaving, setIsSaving] = useState(false)
  1173→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
  1174→  const [copied, setCopied] = useState(false)
  1175→  const [authOpen, setAuthOpen] = useState(false)
  1176→
  1177→  const handleBack = () => {
  1178→    reset()
  1179→  }
  1180→
  1181→  const handleSave = async () => {
  1182→    if (!store || isSaving) return
  1183→    setIsSaving(true)
  1184→    try {
  1185→      const res = await fetch('/api/store/save', {
  1186→        method: 'POST',
  1187→        headers: { 'Content-Type': 'application/json' },
  1188→        body: JSON.stringify({ store }),
  1189→      })
  1190→      if (!res.ok) throw new Error(`Save failed (${res.status})`)
  1191→      const data = await res.json()
  1192→      toast.success(`Draft saved — ID: ${data.id}`, {
  1193→        description: `Slug: ${data.slug}`,
  1194→        duration: 5000,
  1195→      })
  1196→    } catch (err: unknown) {
  1197→      const msg = err instanceof Error ? err.message : ''
  1198→      if (msg.includes('Failed to fetch') || msg.includes('NetworkError') || msg.includes('fetch')) {
  1199→        toast.error('Connection lost while saving', {
  1200→          description: 'Your work is still in the editor. Please check your connection and try again.',
  1201→          duration: 6000,
  1202→        })
  1203→      } else {
  1204→        toast.error('Failed to save store', {
  1205→          description: msg || 'An unexpected error occurred. Please try again.',
  1206→          duration: 5000,
  1207→        })
  1208→      }
  1209→    } finally {
  1210→      setIsSaving(false)
  1211→    }
  1212→  }
  1213→
  1214→  const handlePublish = async () => {
  1215→    if (!store || isPublishing) return
  1216→    setIsPublishing(true)
  1217→    try {
  1218→      const res = await fetch('/api/store/publish', {
  1219→        method: 'POST',
  1220→        headers: { 'Content-Type': 'application/json' },
  1221→        body: JSON.stringify({ store }),
  1222→      })
  1223→      if (!res.ok) {
  1224→        // Try to get server error details
  1225→        let serverMsg = ''
  1226→        try {
  1227→          const errorData = await res.json()
  1228→          serverMsg = errorData.error || ''
  1229→        } catch { /* ignore */ }
  1230→
  1231→        if (res.status === 401) {
  1232→          throw new Error('AUTH_REQUIRED')
  1233→        } else if (res.status === 502 || res.status === 503 || res.status === 504) {
  1234→          throw new Error('GATEWAY_ERROR')
  1235→        } else {
  1236→          throw new Error(serverMsg || `Server error (${res.status})`)
  1237→        }
  1238→      }
  1239→      const data = await res.json()
  1240→      const slug = data.slug
  1241→      setIsPublished(true)
  1242→      // Build a real viewable URL using the current origin
  1243→      const baseUrl = window.location.origin
  1244→      const viewUrl = `${baseUrl}/?store=${slug}`
  1245→      setPublishedUrl(viewUrl)
  1246→      toast.success('Store published successfully!')
  1247→    } catch (err: unknown) {
  1248→      const msg = err instanceof Error ? err.message : ''
  1249→
  1250→      if (msg === 'AUTH_REQUIRED') {
  1251→        toast.error('Please sign in to publish', {
  1252→          description: 'You need to be signed in before publishing your store.',
  1253→          duration: 6000,
  1254→        })
  1255→        setAuthOpen(true)
  1256→      } else if (msg === 'GATEWAY_ERROR') {
  1257→        toast.error('Server temporarily unavailable', {
  1258→          description: 'The server seems to be restarting. Please wait a moment and try again.',
  1259→          duration: 6000,
  1260→        })
  1261→      } else if (msg.includes('Failed to fetch') || msg.includes('NetworkError') || msg.includes('fetch') || msg.includes('Load failed')) {
  1262→        toast.error('Connection lost while publishing', {
  1263→          description: 'Your store data is safe. Please check your connection and try again.',
  1264→          duration: 6000,
  1265→        })
  1266→      } else {
  1267→        toast.error('Failed to publish store', {
  1268→          description: msg || 'An unexpected error occurred. Please try again.',
  1269→          duration: 5000,
  1270→        })
  1271→      }
  1272→    } finally {
  1273→      setIsPublishing(false)
  1274→    }
  1275→  }
  1276→
  1277→  const handleCopyUrl = useCallback(async () => {
  1278→    if (!publishedUrl) return
  1279→    try {
  1280→      await navigator.clipboard.writeText(publishedUrl)
  1281→      setCopied(true)
  1282→      toast.success('URL copied to clipboard!')
  1283→      setTimeout(() => setCopied(false), 2000)
  1284→    } catch {
  1285→      // Fallback for older browsers
  1286→      const textarea = document.createElement('textarea')
  1287→      textarea.value = publishedUrl
  1288→      document.body.appendChild(textarea)
  1289→      textarea.select()
  1290→      document.execCommand('copy')
  1291→      document.body.removeChild(textarea)
  1292→      setCopied(true)
  1293→      toast.success('URL copied to clipboard!')
  1294→      setTimeout(() => setCopied(false), 2000)
  1295→    }
  1296→  }, [publishedUrl])
  1297→
  1298→  return (
  1299→    <>
  1300→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
  1301→        <div className="flex items-center gap-1">
  1302→          <Button
  1303→            variant="ghost"
  1304→            size="icon"
  1305→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
  1306→            onClick={handleBack}
  1307→            aria-label="Back to home"
  1308→          >
  1309→            <ArrowLeft className="h-4 w-4" />
  1310→          </Button>
  1311→          <div className="h-5 w-px bg-zinc-800" />
  1312→          <Button
  1313→            variant={showLeft ? 'secondary' : 'ghost'}
  1314→            size="sm"
  1315→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
  1316→            onClick={() => onToggleLeft((v) => !v)}
  1317→          >
  1318→            <Layers className="h-3.5 w-3.5" />
  1319→            <span className="hidden sm:inline">Sections</span>
  1320→          </Button>
  1321→          <Button
  1322→            variant={showRight ? 'secondary' : 'ghost'}
  1323→            size="sm"
  1324→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
  1325→            onClick={() => onToggleRight((v) => !v)}
  1326→          >
  1327→            <MessageSquare className="h-3.5 w-3.5" />
  1328→            <span className="hidden sm:inline">Chat</span>
  1329→          </Button>
  1330→        </div>
  1331→
  1332→        <div className="flex items-center gap-1.5">
  1333→          {publishedUrl && (
  1334→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
  1335→              <Check className="h-3 w-3 text-emerald-400" />
  1336→              <span className="max-w-[180px] truncate text-xs font-medium text-emerald-300">
  1337→                {publishedUrl}
  1338→              </span>
  1339→              <button
  1340→                onClick={handleCopyUrl}
  1341→                className="ml-1 rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"

  1347→                href={publishedUrl}
  1348→                target="_blank"
  1349→                rel="noopener noreferrer"
  1350→                className="rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
  1351→                aria-label="Open published store"
  1352→              >
  1353→                <ExternalLink className="h-3 w-3" />
