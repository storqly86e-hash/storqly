   256→function SortableSectionItem({
   257→  section,
   258→  isSelected,
   259→  onSelect,
   260→  onToggleVisibility,
   261→  onDelete,
   262→}: SortableSectionItemProps) {
   263→  const {
   264→    attributes,
   265→    listeners,
   266→    setNodeRef,
   267→    transform,
   268→    transition,
   269→    isDragging,
   270→  } = useSortable({ id: section.id });
   271→
   272→  const style = {
   273→    transform: CSS.Transform.toString(transform),
   274→    transition,
   275→  };
   276→
   277→  const Icon = SECTION_TYPE_ICONS[section.type];
   278→  const label = SECTION_TYPE_LABELS[section.type];
   279→
   280→  return (
   281→    <div
   282→      ref={setNodeRef}
   283→      style={style}
   284→      className={[
   285→        'group flex items-center gap-2 rounded-lg px-3 py-2 text-sm cursor-pointer transition-colors',
   286→        isDragging
   287→          ? 'z-50 bg-zinc-700/80 shadow-xl ring-1 ring-white/10'
   288→          : isSelected
   289→            ? 'bg-amber-500/15 text-amber-200 ring-1 ring-amber-500/40'
   290→            : 'text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200',
   291→      ].join(' ')}
   292→      onClick={onSelect}
   293→    >
   294→      {/* Drag handle */}
   295→      <button
   296→        className="flex-shrink-0 cursor-grab text-zinc-600 hover:text-zinc-400 active:cursor-grabbing focus:outline-none"
   297→        {...attributes}
   298→        {...listeners}
   299→        onClick={(e) => e.stopPropagation()}
   300→        aria-label="Drag to reorder"
   301→      >
   302→        <GripVertical className="h-4 w-4" />
   303→      </button>
   304→
   305→      {/* Icon */}
   306→      <Icon className="h-4 w-4 flex-shrink-0 opacity-60" />
   307→
   308→      {/* Label */}
   309→      <span className="flex-1 truncate font-medium">{label}</span>
   310→
   311→      {/* Visibility toggle */}
   312→      <button
   313→        className="flex-shrink-0 rounded p-1 text-zinc-500 hover:text-zinc-300 hover:bg-zinc-700/50 focus:outline-none"
   314→        onClick={(e) => {
   315→          e.stopPropagation();
   316→          onToggleVisibility();
   317→        }}

   326→
   327→      {/* Delete */}
   328→      <button
   329→        className="flex-shrink-0 rounded p-1 text-zinc-600 opacity-0 group-hover:opacity-100 hover:text-red-400 hover:bg-red-400/10 transition-opacity focus:outline-none"
   330→        onClick={(e) => {
   331→          e.stopPropagation();

   394→// ── Content Field Renderer ───────────────────────────────────────────
   395→
   396→function ContentFieldRenderer({
   397→  keyName,
   398→  value,
   399→  onChange,
   400→  depth = 0,
   401→}: {
   402→  keyName: string;
   403→  value: unknown;
   404→  onChange: (key: string, value: unknown) => void;
   405→  depth?: number;
   406→}) {
   407→  const label = keyName
   408→    .replace(/([A-Z])/g, ' $1')
   409→    .replace(/^./, (s) => s.toUpperCase())
   410→    .trim();
   411→
   412→  if (value === null || value === undefined) {
   413→    return null;
   414→  }
   415→
   416→  if (typeof value === 'string') {
   417→    if (keyName === 'html' && value.includes('<')) {
   418→      return (
   419→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   420→          <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   421→          <Textarea
   422→            value={value}
   423→            onChange={(e) => onChange(keyName, e.target.value)}
   424→            rows={4}
   425→            className="bg-zinc-800 border-zinc-700 text-zinc-100 text-xs font-mono resize-none"
   426→          />
   427→        </div>
   428→      );
   429→    }
   430→    return (
   431→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   432→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   433→        <Input
   434→          value={value}
   435→          onChange={(e) => onChange(keyName, e.target.value)}
   436→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   437→        />
   438→      </div>
   439→    );
   440→  }
   441→
   442→  if (typeof value === 'number') {
   443→    return (
   444→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   445→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   446→        <Input
   447→          type="number"
   448→          value={value}
   449→          onChange={(e) => onChange(keyName, Number(e.target.value))}
   450→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   451→        />
   452→      </div>
   453→    );
   454→  }
   455→
   456→  if (typeof value === 'boolean') {
   457→    return (
   458→      <div
   459→        key={keyName}
   460→        className={`flex items-center justify-between ${depth > 0 ? 'ml-4' : ''}`}
   461→      >
   462→        <Label className="text-xs text-zinc-400">{label}</Label>
   463→        <Switch
   464→          checked={value}
   465→          onCheckedChange={(checked) => onChange(keyName, checked)}
   466→        />
   467→      </div>
   468→    );
   469→  }
   470→
   471→  if (Array.isArray(value)) {
   472→    return (
   473→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   474→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   475→        <div className="rounded-lg bg-zinc-800/60 border border-zinc-700/50 px-3 py-2 text-xs text-zinc-500">
   476→          {value.length} item{value.length !== 1 ? 's' : ''}
   477→        </div>
   478→      </div>
   479→    );
   480→  }
   481→
   482→  if (typeof value === 'object') {
   483→    const obj = value as Record<string, unknown>;
   484→    const keys = Object.keys(obj);
   485→
   486→    const allPrimitive = keys.every(
   487→      (k) =>
   488→        typeof obj[k] === 'string' ||
   489→        typeof obj[k] === 'number' ||
   490→        typeof obj[k] === 'boolean' ||
   491→        obj[k] === null ||
   492→        obj[k] === undefined
   493→    );
   494→
   495→    if (allPrimitive && keys.length > 0) {
   496→      return (
   497→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   498→          <p className="text-xs font-medium text-zinc-300 mb-2 mt-1">{label}</p>
   499→          <div className="space-y-3 border-l border-zinc-700/50 pl-3 ml-1">
   500→            {keys.map((k) => (
   501→              <ContentFieldRenderer
   502→                key={k}
   503→                keyName={k}
   504→                value={obj[k]}
   505→                onChange={(subKey, subVal) => {
   506→                  onChange(keyName, { ...obj, [subKey]: subVal });
   507→                }}
   508→                depth={depth + 1}
   509→              />
   510→            ))}
   511→          </div>
   512→        </div>
   513→      );
   514→    }
   515→
   516→    return (
   517→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   518→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   519→        <Textarea
   520→          value={JSON.stringify(value, null, 2)}
   521→          readOnly
   522→          rows={3}
   523→          className="bg-zinc-800/60 border-zinc-700/50 text-zinc-500 text-xs font-mono resize-none"
   524→        />
   525→      </div>
   526→    );
   527→  }
   528→
   529→  return null;
   530→}

   539→function PropertiesPanel({ section, pageId }: PropertiesPanelProps) {
   540→  const {
   541→    updateSectionContent,
   542→    updateSectionStyle,
   543→    removeSection,
   544→    setSelectedSectionId,
   545→    store,
   546→    setStore,
   547→  } = useStoreEditor();
   548→
   549→  const [localStyle, setLocalStyle] = useState<SectionStyle>(section.style);
   550→
   551→  // Sync style from store when section changes
   552→  if (section.style !== localStyle) {
   553→    setLocalStyle(section.style);
   554→  }
   555→
   556→  const handleContentChange = useCallback(
   557→    (key: string, value: unknown) => {
   558→      updateSectionContent(pageId, section.id, { [key]: value });
   559→    },
   560→    [pageId, section.id, updateSectionContent]
   561→  );
   562→
   563→  const handleStyleChange = useCallback(
   564→    (patch: Partial<SectionStyle>) => {
   565→      const updated = { ...localStyle, ...patch };
   566→      setLocalStyle(updated);
   567→      updateSectionStyle(pageId, section.id, patch);
   568→    },
   569→    [pageId, section.id, localStyle, updateSectionStyle]
   570→  );
   571→
   572→  const handleDelete = useCallback(() => {
   573→    removeSection(pageId, section.id);
   574→    setSelectedSectionId(null);
   575→    toast.success('Section deleted');
   576→  }, [pageId, section.id, removeSection, setSelectedSectionId]);
   577→
   578→  const handleVisibilityToggle = useCallback(() => {
   579→    if (!store) return;
   580→    const updatedStore = {
   581→      ...store,
   582→      updatedAt: new Date().toISOString(),
   583→      pages: store.pages.map((page) => ({
   584→        ...page,
   585→        sections: page.sections.map((s) =>
   586→          s.id === section.id ? { ...s, visible: !s.visible } : s
   587→        ),
   588→      })),
   589→    };
   590→    setStore(updatedStore);
   591→  }, [store, section.id, setStore]);
   592→
   593→  const Icon = SECTION_TYPE_ICONS[section.type];
   594→  const typeLabel = SECTION_TYPE_LABELS[section.type];
   595→
   596→  const renderSelect = (
   597→    lbl: string,
   598→    value: string | undefined,
   599→    options: { label: string; value: string }[],
   600→    onChange: (val: string) => void
   601→  ) => (
   602→    <div>
   603→      <Label className="text-xs text-zinc-400 mb-1.5 block">{lbl}</Label>
   604→      <Select value={value ?? ''} onValueChange={onChange}>
   605→        <SelectTrigger className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm">
   606→          <SelectValue />
   607→        </SelectTrigger>
   608→        <SelectContent className="bg-zinc-900 border-zinc-700">
   609→          {options.map((opt) => (
   610→            <SelectItem
   611→              key={opt.value}
   612→              value={opt.value}
   613→              className="text-zinc-300 focus:bg-zinc-800 focus:text-zinc-100"
   614→            >
   615→              {opt.label}
   616→            </SelectItem>
   617→          ))}
   618→        </SelectContent>
   619→      </Select>
   620→    </div>
   621→  );
   622→
   623→  return (
   624→    <div className="flex h-full flex-col min-h-0">
   625→      {/* Header */}
   626→      <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-3">
   627→        <div className="flex items-center gap-2">
   628→          <Settings2 className="h-4 w-4 text-zinc-500" />
   629→          <span className="text-sm font-semibold text-zinc-200">Properties</span>
   630→        </div>
   631→        <button
   632→          className="rounded-md p-1 text-zinc-500 hover:text-zinc-300 hover:bg-zinc-800 focus:outline-none"
   633→          onClick={() => setSelectedSectionId(null)}
   634→          aria-label="Close properties"
   635→        >
   636→          <X className="h-4 w-4" />
   637→        </button>
   638→      </div>

   911→// ── Page type labels & icons for editor tabs ────────────────────────
   912→
   913→const PAGE_TYPE_ICONS: Record<string, LucideIcon> = {
   914→  home: Layout,
   915→  collection: Grid3X3,
   916→  product: Star,
   917→  cart: PanelBottom,
   918→  checkout: MousePointerClick,
   919→};
   920→
   921→const PAGE_TYPE_LABELS: Record<string, string> = {
   922→  home: 'Home',
   923→  collection: 'Shop',
   924→  product: 'Product',
   925→  cart: 'Cart',
   926→  checkout: 'Checkout',
   927→};
   928→
   929→export function VisualEditor() {
   930→  const {
   931→    store,
   932→    setStore,
   933→    selectedSectionId,
   934→    setSelectedSectionId,
   935→    moveSection,
   936→    addSection,
   937→    removeSection,
   938→    editorCurrentPageId,
   939→    setEditorCurrentPageId,
   940→  } = useStoreEditor();
   941→
   942→  // Pages available for tab switching (real pages only, no dynamic product pages)
   943→  const editorPages = useMemo(
   944→    () => (store?.pages.filter((p) => !p.type || p.type === 'home' || p.type === 'collection' || p.type === 'cart' || p.type === 'checkout') ?? []),
   945→    [store?.pages]
   946→  );
   947→
   948→  // Current page for section editing (only home-type pages have editable sections)
   949→  const currentPage = useMemo(
   950→    () => store?.pages.find((p) => p.id === editorCurrentPageId) ?? store?.pages.find((p) => p.isHomepage) ?? store?.pages[0],
   951→    [store?.pages, editorCurrentPageId]
   952→  );
   953→
   954→  // Whether the currently selected page is a template (non-editable via visual editor)
   955→  const isTemplatePage = currentPage && currentPage.type && currentPage.type !== 'home';
   956→
   957→  const sections = currentPage?.sections ?? [];
   958→  const pageId = currentPage?.id ?? '';
   959→
   960→  const selectedSection = useMemo(
   961→    () => sections.find((s) => s.id === selectedSectionId) ?? null,
   962→    [sections, selectedSectionId]
   963→  );
   964→
   965→  // Drag end handler
   966→  const handleDragEnd = useCallback(
   967→    (event: DragEndEvent) => {
   968→      const { active, over } = event;
   969→      if (!over || active.id === over.id || !pageId) return;
   970→
   971→      const oldIndex = sections.findIndex((s) => s.id === active.id);
   972→      const newIndex = sections.findIndex((s) => s.id === over.id);
   973→
   974→      if (oldIndex !== -1 && newIndex !== -1 && oldIndex !== newIndex) {
   975→        moveSection(pageId, oldIndex, newIndex);
   976→      }
   977→    },
   978→    [sections, pageId, moveSection]
   979→  );
   980→
   981→  // Add section handler
   982→  const handleAddSection = useCallback(
   983→    (type: SectionType) => {
   984→      if (!pageId) return;
   985→      const newSection = createDefaultSection(type);
   986→      addSection(pageId, newSection);
   987→      setSelectedSectionId(newSection.id);
   988→      toast.success(`Added ${SECTION_TYPE_LABELS[type]} section`);
   989→    },
   990→    [pageId, addSection, setSelectedSectionId]
   991→  );
   992→
   993→  // Toggle visibility for section list items
   994→  const handleToggleVisibility = useCallback(
   995→    (sectionId: string) => {
   996→      if (!store) return;
   997→      const updatedStore = {
   998→        ...store,
   999→        updatedAt: new Date().toISOString(),
  1000→        pages: store.pages.map((page) => ({
  1001→          ...page,
  1002→          sections: page.sections.map((s) =>
  1003→            s.id === sectionId ? { ...s, visible: !s.visible } : s
  1004→          ),
  1005→        })),
  1006→      };
  1007→      setStore(updatedStore);
  1008→    },
  1009→    [store, setStore]
  1010→  );
  1011→
  1012→  const handleDeleteSection = useCallback(
  1013→    (sectionId: string) => {
  1014→      if (!pageId) return;
  1015→      const section = sections.find((s) => s.id === sectionId);
  1016→      removeSection(pageId, sectionId);
  1017→      if (selectedSectionId === sectionId) {
  1018→        setSelectedSectionId(null);
  1019→      }
  1020→      if (section) {
  1021→        toast.success(`Deleted ${SECTION_TYPE_LABELS[section.type]} section`);
  1022→      }
  1023→    },
  1024→    [pageId, sections, selectedSectionId, removeSection, setSelectedSectionId]
  1025→  );
  1026→
  1027→  // Handle page tab click (before early return to satisfy rules-of-hooks)
  1028→  const handlePageTabClick = useCallback(
  1029→    (pageId: string) => {
  1030→      setEditorCurrentPageId(pageId);
  1031→    },
  1032→    [setEditorCurrentPageId]
  1033→  );
  1034→
  1035→  if (!store || !currentPage) {
  1036→    return (
  1037→      <div className="flex h-full items-center justify-center bg-zinc-950">
  1038→        <p className="text-sm text-zinc-600">No store loaded</p>
  1039→      </div>
  1040→    );
  1041→  }
  1042→
  1043→  return (
  1044→    <div className="flex h-full overflow-hidden bg-zinc-950">
  1045→      {/* Section List Panel */}
  1046→      <div
  1047→        className={[
  1048→          'flex flex-col border-r border-zinc-800 min-h-0',
  1049→          selectedSection && !isTemplatePage ? 'w-64 lg:w-72' : 'w-full',
  1050→        ].join(' ')}
  1051→      >
  1052→        {/* Page Tabs */}
  1053→        {editorPages.length > 1 && (
  1054→          <div className="flex items-center gap-0.5 overflow-x-auto border-b border-zinc-800 px-2 py-2" style={{scrollbarWidth: 'none'}}>
  1055→            {editorPages.map((page) => {
  1056→              const isActive = page.id === editorCurrentPageId;
  1057→              const Icon = PAGE_TYPE_ICONS[page.type || 'home'] || Layout;
  1058→              return (
  1059→                <button
  1060→                  key={page.id}
  1061→                  onClick={() => handlePageTabClick(page.id)}
  1062→                  className={`flex shrink-0 items-center gap-1.5 rounded-lg px-2.5 py-1.5 text-xs font-medium transition-colors ${
  1063→                    isActive
  1064→                      ? 'bg-zinc-800 text-zinc-100'
  1065→                      : 'text-zinc-500 hover:bg-zinc-800/50 hover:text-zinc-300'
  1066→                  }`}
  1067→                >
  1068→                  <Icon className="h-3 w-3" />
  1069→                  <span className="whitespace-nowrap">{page.name}</span>
  1070→                </button>
  1071→              );
  1072→            })}
  1073→          </div>
  1074→        )}
  1075→
  1076→        {/* Panel header — only for section-editable pages */}
  1077→        {!isTemplatePage && (
  1078→          <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-3">
  1079→            <h3 className="text-sm font-semibold text-zinc-200">Sections</h3>
  1080→            <span className="text-xs text-zinc-600 tabular-nums">
  1081→              {sections.length}
  1082→            </span>
  1083→          </div>
  1084→        )}
  1085→
  1086→        {/* Section list with DnD — only for section-editable pages */}
  1087→        {!isTemplatePage ? (
  1088→          <ScrollArea className="flex-1 min-h-0">
  1089→            <div className="p-2">
  1090→              <DndContext
  1091→                collisionDetection={closestCenter}
  1092→                onDragEnd={handleDragEnd}
  1093→              >
  1094→                <SortableContext
  1095→                  items={sections.map((s) => s.id)}
  1096→                  strategy={verticalListSortingStrategy}
  1097→                >
  1098→                  <div className="space-y-0.5">
  1099→                    {sections.map((section) => (
  1100→                      <SortableSectionItem
  1101→                        key={section.id}
  1102→                        section={section}
  1103→                        isSelected={section.id === selectedSectionId}
  1104→                        onSelect={() =>
  1105→                          setSelectedSectionId(
  1106→                            section.id === selectedSectionId ? null : section.id
  1107→                          )
  1108→                        }
  1109→                        onToggleVisibility={() =>
  1110→                          handleToggleVisibility(section.id)
  1111→                        }
  1112→                        onDelete={() => handleDeleteSection(section.id)}
  1113→                      />
  1114→                    ))}
  1115→                  </div>
  1116→                </SortableContext>
  1117→              </DndContext>
  1118→            </div>
  1119→          </ScrollArea>
  1120→        ) : (
  1121→          <div className="flex flex-1 min-h-0 flex-col items-center justify-center px-6 text-center">
  1122→            <div className="mb-4 flex h-12 w-12 items-center justify-center rounded-xl bg-zinc-800/60">
  1123→              {(() => {
  1124→                const TIcon = PAGE_TYPE_ICONS[currentPage.type || 'home'] || Layout;
  1125→                return <TIcon className="h-6 w-6 text-zinc-500" />;
  1126→              })()}
  1127→            </div>
  1128→            <p className="text-sm font-medium text-zinc-300">{PAGE_TYPE_LABELS[currentPage.type || 'home'] || 'Template'} Page</p>
  1129→            <p className="mt-1.5 max-w-[200px] text-xs leading-relaxed text-zinc-600">
  1130→              Switch to Home to edit sections.
  1131→            </p>
  1132→          </div>
  1133→        )}
  1134→
  1135→        {/* Add section button — only for section-editable pages */}
  1136→        {!isTemplatePage && (
  1137→          <div className="border-t border-zinc-800 p-2">
  1138→            <AddSectionPopover onAdd={handleAddSection} />
  1139→          </div>
  1140→        )}
  1141→      </div>
  1142→
  1143→      {/* Properties Panel — only for section-editable pages */}
