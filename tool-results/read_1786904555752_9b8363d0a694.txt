   103→// ─── Background Image Enrichment (lazy, non-blocking) ─────
   104→// Triggered after store generation completes. Fetches real product
   105→// images sequentially without blocking the user or exhausting rate limits.
   106→
   107→function triggerBackgroundImageEnrichment(store: Store) {
   108→  if (!store.products || store.products.length === 0) return
   109→
   110→  // Only enrich products that still have AI-hallucinated Unsplash URLs
   111→  // (Real enriched URLs from the image search service will have different patterns)
   112→  const needsEnrichment = store.products.filter(p =>
   113→    p.images.length > 0 && p.images[0].includes('unsplash.com/photo-')
   114→  )
   115→
   116→  if (needsEnrichment.length === 0) return
   117→
   118→  console.log(`[Storqly] Background image enrichment: ${needsEnrichment.length}/${store.products.length} products need enrichment`)
   119→
   120→  fetch('/api/store/enrich-images', {
   121→    method: 'POST',
   122→    headers: { 'Content-Type': 'application/json' },
   123→    body: JSON.stringify({
   124→      products: needsEnrichment.map(p => ({
   125→        id: p.id,
   126→        name: p.name,
   127→        images: [...p.images],
   128→        category: p.category,
   129→        description: p.description,
   130→      })),
   131→      storeName: store.name,
   132→    }),
   133→  })
   134→    .then(res => res.json())
   135→    .then(data => {
   136→      if (data.enriched > 0) {
   137→        console.log(`[Storqly] Background enrichment complete: ${data.enriched} enriched, ${data.kept} kept, ${data.failed} failed in ${data.latencyMs}ms`)
   138→        // Update product images in the store via the Zustand store
   139→        // The enrich-images endpoint mutated the products array in-place,
   140→        // but we need to update the Zustand store to trigger re-renders.
   141→        // Dynamic import to avoid circular dependency
   142→        import('@/lib/store').then(({ useStoreEditor }) => {
   143→          const currentStore = useStoreEditor.getState().store
   144→          if (currentStore) {
   145→            const updatedProducts = currentStore.products.map(prod => {
   146→              const enriched = data.products?.find((ep: { id: string; images: string[] }) => ep.id === prod.id)
   147→              if (enriched && enriched.images[0] !== prod.images[0]) {
   148→                return { ...prod, images: enriched.images }
   149→              }
   150→              return prod
   151→            })
   152→            useStoreEditor.getState().setStore({ ...currentStore, products: updatedProducts })
   153→          }
   154→        })
   155→      }
   156→    })
   157→    .catch(err => {
   158→      // Non-fatal — products still show AI placeholder images
   159→      console.warn('[Storqly] Background enrichment failed (non-fatal):', err)
   160→    })
   161→}

   354→      setError(message)

   356→      setGenerationStatus('')

   362→      clearTimeout(timeoutId)
   363→      clearTimers()
   364→      setIsGenerating(false)
   365→      setGenerationStatus('')
   366→      setElapsedSeconds(0)
   367→    }
   368→
   369→    try {
   370→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   371→
   372→      // ── Fetch with auto-retry for transient gateway errors (502/503/504) ──
   373→      const RETRYABLE_STATUSES = [502, 503, 504]
   374→      let res: Response | undefined
   375→      let retryCount = 0
   376→      const maxRetries = 2
   377→
   378→      while (retryCount <= maxRetries) {
   379→        try {
   380→          res = await fetch('/api/store/generate', {
   381→            method: 'POST',
   382→            headers: { 'Content-Type': 'application/json' },
   383→            body: JSON.stringify({ prompt: trimmed }),
   384→            signal: controller.signal,
   385→          })
   386→
   387→          // If the status is retryable, wait and loop again
   388→          if (!res.ok && RETRYABLE_STATUSES.includes(res.status) && retryCount < maxRetries) {
   389→            retryCount++
   390→            console.warn(`[Storqly] Gateway error ${res.status}, retrying (${retryCount}/${maxRetries}) in 3s...`)
   391→            setGenerationStatus('Connection issue — retrying...')
   392→            await new Promise(r => setTimeout(r, 3000))
   393→            continue
   394→          }
   395→          break // Success or non-retryable error — exit loop
   396→        } catch (fetchErr: unknown) {
   397→          // Network-level error (DNS failure, connection refused, etc.)
   398→          if (fetchErr instanceof TypeError && retryCount < maxRetries) {
   399→            retryCount++
   400→            console.warn(`[Storqly] Network error, retrying (${retryCount}/${maxRetries}) in 3s...`)
   401→            setGenerationStatus('Connection issue — retrying...')
   402→            await new Promise(r => setTimeout(r, 3000))
   403→            continue
   404→          }
   405→          // Out of retries or not a network error — handle gracefully
   406→          const msg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr)
   407→          if (controller.signal.aborted) {
   408→            finishWithError('Generation was cancelled.')
   409→          } else {
   410→            console.warn('[Storqly] Network error after retries:', msg)
   411→            finishWithError('Could not connect to the server. Please check your connection and try again.')
   412→          }
   413→          return
   414→        }
   415→      }
   416→
   417→      if (!res!.ok) {
   418→        let errorMsg = `Server error (${res!.status})`
   419→        try {
   420→          const errorData = await res!.json()
   421→          errorMsg = errorData.error || errorMsg
   422→        } catch { /* ignore parse error */ }
   423→        console.warn('[Storqly] HTTP error:', errorMsg)
   424→        finishWithError(errorMsg)
   425→        return
   426→      }
   427→
   428→      // ── Consume SSE stream ──
   429→      const reader = res.body?.getReader()
   430→      if (!reader) {
   431→        finishWithError('No response stream received from server.')
   432→        return
   433→      }
   434→
   435→      const decoder = new TextDecoder()
   436→      let buffer = ''
   437→      let resolved = false
   438→
   439→      while (true) {
   440→        const { done, value } = await reader.read()
   441→        if (done) break
   442→
   443→        buffer += decoder.decode(value, { stream: true })
   444→
   445→        // Parse SSE events from buffer
   446→        const lines = buffer.split('\n')
   447→        buffer = lines.pop() || ''
   448→
   449→        let currentEvent = ''
   450→        for (const line of lines) {
   451→          if (line.startsWith(':')) continue
   452→          if (line.startsWith('event: ')) {
   453→            currentEvent = line.slice(7).trim()
   454→          } else if (line.startsWith('data: ')) {
   455→            const dataStr = line.slice(6)
   456→
   457→            if (currentEvent === 'progress') {
   458→              receivedSSEProgress = true
   459→              try {
   460→                const data = JSON.parse(dataStr)
   461→                setGenerationStatus(data.message || 'Processing...')
   462→              } catch { /* ignore */ }
   463→            } else if (currentEvent === 'result') {
   464→              try {
   465→                const data = JSON.parse(dataStr)
   466→                if (!data.store) {
   467→                  console.warn('[Storqly] Result event missing store data')
   468→                  finishWithError('The AI response was missing store data.')
   469→                  return
   470→                }
   471→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   472→                resolved = true
   473→
   474→                if (data._isFallback) {
   475→                  // Differentiate "all providers down" from other fallback reasons
   476→                  const reason = data._fallbackReason || 'AI generation failed'
   477→                  const isProviderFailure = reason.includes('All') || reason.includes('429') || reason.includes('403') || reason.includes('404') || reason.includes('providers failed')
   478→                  toast.warning(
   479→                    isProviderFailure
   480→                      ? 'All AI providers are currently unavailable. You are viewing a starter template. Try again later.'
   481→                      : 'AI service returned invalid data — showing starter template. Try again.',
   482→                    { duration: 8000 }
   483→                  )
   484→                  setStoreWithFallback(data.store, true, reason)
   485→                } else {
   486→                  setStore(data.store)
   487→                  // Trigger lazy background image enrichment (non-blocking)
   488→                  triggerBackgroundImageEnrichment(data.store)
   489→                }
   490→
   491→                // Soft cap toast
   492→                if (data._productCapHit) {
   493→                  toast.info(
   494→                    `Generated ${data._generatedCount} products — for larger catalogs, you can add more via the chat editor.`,
   495→                    { duration: 8000 }
   496→                  )
   497→                }
   498→
   499→                finishOk()
   500→                return
   501→              } catch (parseErr) {
   502→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   503→                console.warn('[Storqly] Failed to parse result event:', msg)
   504→                finishWithError(msg)
   505→                return
   506→              }
   507→            } else if (currentEvent === 'error') {
   508→              try {
   509→                const data = JSON.parse(dataStr)
   510→                const serverMsg = data.message || 'Generation failed'
   511→                console.warn('[Storqly] Server error event:', serverMsg)
   512→                finishWithError(serverMsg)
   513→                return
   514→              } catch {
   515→                finishWithError('Generation failed')
   516→                return
   517→              }
   518→            }
   519→            currentEvent = ''
   520→          }
   521→        }
   522→      }
   523→
   524→      if (!resolved) {
   525→        console.warn('[Storqly] Stream ended without a result event')
   526→        finishWithError('Stream ended without a result. The server may have disconnected.')
   527→      }
   528→    } catch (err: unknown) {
   529→      // Last-resort safety net — should rarely be reached now
   530→      clearTimeout(timeoutId)
   531→      clearTimers()
   532→      let message = 'Something went wrong. Please try again.'
   533→      if (err instanceof Error) {
   534→        if (err.name === 'AbortError') {
   535→          message = timedOut
   536→            ? 'Generation timed out after 2 minutes. The server may be overloaded — please try again.'
   537→            : 'Generation was cancelled.'
   538→          console.warn(`[Storqly] Request aborted: ${message}`)
   539→        } else {
   540→          console.warn('[Storqly] Unexpected error (caught by safety net):', err.message)
   541→          message = err.message
   542→        }
   543→      }
   544→      setError(message)
   545→      setIsGenerating(false)
   546→      setGenerationStatus('')
   547→      setElapsedSeconds(0)
   548→      toast.error('Store generation failed', { description: message })
   549→    }
   550→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])
   551→
   552→  // Cleanup on unmount
   553→  useEffect(() => {
   554→    return () => {
   555→      clearTimers()
   556→      abortRef.current?.abort()
   557→    }
   558→  }, [clearTimers])
   559→
   560→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   561→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   562→      e.preventDefault()
   563→      handleGenerate()
   564→    }
   565→  }
   566→
   567→  const formatElapsed = (seconds: number) => {
   568→    if (seconds < 60) return `${seconds}s`

  1035→// ─── Fallback Banner ──────────────────────────────────────────────
  1036→// Shows when: (1) AI returned a fallback template, or
  1037→// (2) the store ended up in a broken/incomplete state (e.g. server
  1038→//     crashed mid-generation leaving only placeholder sections).
  1039→
  1040→function FallbackBanner() {
  1041→  const reset = useStoreEditor((s) => s.reset)
  1042→  const isFallbackStore = useStoreEditor((s) => s.isFallbackStore)
  1043→  const fallbackReason = useStoreEditor((s) => s.fallbackReason)
  1044→  const [dismissed, setDismissed] = useState(false)
  1045→
  1046→  if (dismissed || !isFallbackStore) return null
  1047→
  1048→  const isIncomplete = fallbackReason.includes('interrupted') || fallbackReason.includes('incomplete')
  1049→  const isProviderDown = fallbackReason.includes('All') || fallbackReason.includes('429') || fallbackReason.includes('403') || fallbackReason.includes('404') || fallbackReason.includes('providers failed')
  1050→
  1051→  // Determine message based on failure type

  1418→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)

  1420→          const data = await res.json().catch(() => ({}))
  1421→          throw new Error(data.error || `Store not found (${res.status})`)

  1424→        if (!cancelled) setStore(data.store)
  1425→      } catch (err: unknown) {
  1426→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
  1427→      } finally {
  1428→        if (!cancelled) setLoading(false)
  1429→      }
