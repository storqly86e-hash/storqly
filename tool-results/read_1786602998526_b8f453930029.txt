    49→// ─── Section Wrapper ────────────────────────────────────────────────────
    50→
    51→interface SectionWrapperProps {
    52→  section: Section;
    53→  theme: StoreTheme;
    54→  selectedSectionId?: string | null;
    55→  onSelectSection?: (sectionId: string | null) => void;
    56→  children: React.ReactNode;
    57→}
    58→
    59→function SectionWrapper({
    60→  section,
    61→  theme,
    62→  selectedSectionId,
    63→  onSelectSection,
    64→  children,
    65→}: SectionWrapperProps) {
    66→  const isSelected = selectedSectionId === section.id;
    67→  const style = section.style;
    68→
    69→  const sectionBg = style.backgroundColor || theme.colors.background;
    70→  const sectionText = style.textColor || theme.colors.text;
    71→
    72→  const bgImage = style.backgroundImage
    73→    ? `url(${style.backgroundImage})`
    74→    : undefined;
    75→
    76→  const outerClasses = [
    77→    'relative transition-all duration-200 cursor-pointer',
    78→    pyClass(style.paddingY),
    79→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
    80→  ].join(' ');
    81→
    82→  return (
    83→    <div
    84→      className={outerClasses}
    85→      style={{
    86→        backgroundColor: sectionBg,
    87→        color: sectionText,
    88→        backgroundImage: bgImage,
    89→        backgroundSize: 'cover',
    90→        backgroundPosition: 'center',
    91→      }}
    92→      onClick={(e) => {
    93→        e.stopPropagation();
    94→        onSelectSection?.(isSelected ? null : section.id);
    95→      }}
    96→      data-section-id={section.id}
    97→      data-section-type={section.type}
    98→    >
    99→      {/* Overlay for background images */}
   100→      {style.backgroundImage && style.overlay && (
   101→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   102→      )}
   103→      <div
   104→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   105→      >
   106→        {/* Selected section label */}
   107→        {isSelected && (
   108→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   109→            {section.type.replace(/-/g, ' ')}
   110→          </div>
   111→        )}
   112→        {children}
   113→      </div>
   114→    </div>
   115→  );
   116→}

   118→// ─── Product Card (shared) ─────────────────────────────────────────────

   120→function ProductCard({
   121→  product,
   122→  theme,
   123→  showAddToCart,
   124→  borderRadius,
   125→  buttonBgOverride,
   126→  buttonTextOverride,
   127→  onViewProduct,
   128→}: {
   129→  product: StoreProduct;
   130→  theme: StoreTheme;
   131→  showAddToCart: boolean;
   132→  borderRadius: string;
   133→  buttonBgOverride?: string;
   134→  buttonTextOverride?: string;
   135→  onViewProduct?: (productId: string) => void;
   136→}) {
   137→  const [hovered, setHovered] = useState(false);
   138→  const [addedFeedback, setAddedFeedback] = useState(false);
   139→  const addToCart = useCartStore((s) => s.addItem);
   140→  const imgColor = stringToColor(product.id, theme);
   141→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   142→
   143→  const handleAddToCart = (e: React.MouseEvent) => {
   144→    e.stopPropagation();
   145→    addToCart(product);
   146→    setAddedFeedback(true);
   147→    setTimeout(() => setAddedFeedback(false), 1200);
   148→  };
   149→
   150→  return (
   151→    <div
   152→      className={`${borderRadius} ${onViewProduct ? 'cursor-pointer' : ''} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   153→      style={{
   154→        borderColor: theme.colors.border,
   155→      }}
   156→      onMouseEnter={() => setHovered(true)}
   157→      onMouseLeave={() => setHovered(false)}
   158→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   159→      role={onViewProduct ? 'button' : undefined}
   160→      tabIndex={onViewProduct ? 0 : undefined}
   161→    >
   162→ {/* Image placeholder */}
   163→      <div
   164→        className="relative aspect-square w-full overflow-hidden"
   165→        style={{ backgroundColor: imgColor }}
   166→      >
   167→        <div className="absolute inset-0 flex items-center justify-center">
   168→          <div
   169→            className="h-16 w-16 rounded-full opacity-30"
   170→            style={{ backgroundColor: theme.colors.primary }}
   171→          />
   172→        </div>
   173→        {/* Out of stock overlay */}
   174→        {!product.inStock && (
   175→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   176→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   177→              Sold Out
   178→            </span>
   179→          </div>
   180→        )}
   181→        {/* Discount badge */}
   182→        {hasDiscount && product.compareAtPrice && (
   183→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   184→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   185→          </div>
   186→        )}
   187→      </div>
   188→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   189→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   190→        {product.category && (
   191→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   192→            {product.category}
   193→          </p>
   194→        )}
   195→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   196→          {product.name}
   197→        </h3>
   198→        <div className="mt-1.5 flex items-center gap-2">
   199→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   200→            {formatPrice(product.price)}
   201→          </span>
   202→          {hasDiscount && product.compareAtPrice && (
   203→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   204→              {formatPrice(product.compareAtPrice)}
   205→            </span>
   206→          )}
   207→        </div>
   208→        {showAddToCart && product.inStock && (
   209→          <button
   210→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 cursor-pointer ${
   211→              hovered
   212→                ? 'opacity-100 translate-y-0'
   213→                : 'opacity-0 translate-y-1 pointer-events-none'
   214→            }`}
   215→            style={{
   216→              backgroundColor: addedFeedback ? '#16a34a' : (buttonBgOverride || theme.colors.primary),
   217→              color: addedFeedback ? '#ffffff' : (buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary)),
   218→            }}
   219→            onClick={handleAddToCart}
   220→          >
   221→            {addedFeedback ? 'Added!' : 'Add to Cart'}
   222→          </button>
   223→        )}
   224→      </div>
   225→    </div>
   226→  );
   227→}

   324→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   325→  const content = section.content as unknown as HeroContent;
   326→  const style = section.style;
   327→  const isSelected = selectedSectionId === section.id;
   328→
   329→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   330→  const alignMap = {
   331→    left: 'items-start text-left',
   332→    center: 'items-center text-center',
   333→    right: 'items-end text-right',
   334→  };
   335→
   336→  // Only apply theme gradient as default background when no custom color or image is set.
   337→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   338→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   339→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   340→    : undefined;
   341→
   342→  const outerClasses = [
   343→    'relative flex transition-all duration-200 cursor-pointer',
   344→    heightMap[content.height] || heightMap.md,
   345→    alignMap[content.alignment],
   346→    pyClass(style.paddingY),
   347→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   348→  ].join(' ');
   349→
   350→  return (
   351→    <div
   352→      className={outerClasses}
   353→      style={{
   354→        backgroundColor: style.backgroundColor || theme.colors.primary,
   355→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   356→        backgroundSize: 'cover',
   357→        backgroundPosition: 'center',
   358→        color: style.textColor || '#ffffff',
   359→      }}
   360→      onClick={(e) => {
   361→        e.stopPropagation();
   362→        onSelectSection?.(isSelected ? null : section.id);
   363→      }}
   364→      data-section-id={section.id}
   365→      data-section-type={section.type}
   366→    >
   367→      {/* Overlay */}
   368→      {style.backgroundImage && style.overlay && (
   369→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   370→      )}
   371→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   372→      {!style.backgroundImage && !style.backgroundColor && (
   373→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   374→      )}
   375→      {!style.backgroundImage && !style.backgroundColor && (
   376→        <>
   377→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   378→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />
   379→        </>
   380→      )}
   381→
   382→      <div className={`relative z-10 mx-auto w-full ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}>
   383→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   384→          {content.headline}
   385→        </h1>
   386→        {content.subheadline && (
   387→          <p className="mt-4 max-w-xl text-base opacity-80 sm:text-lg md:text-xl">
   388→            {content.subheadline}
   389→          </p>
   390→        )}
   391→        {content.ctaText && (
   392→          <button className="mt-8 inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-transform hover:scale-105 sm:text-base"
   393→            style={{
   394→              backgroundColor: section.style.buttonBackgroundColor || '#ffffff',
   395→              color: section.style.buttonTextColor || contrastTextColor(section.style.buttonBackgroundColor || '#ffffff'),
   396→            }}>
   397→            {content.ctaText}
   398→            <ArrowRight className="h-4 w-4" />
   399→          </button>
   400→        )}
   401→      </div>
   402→    </div>
   403→  );
   404→}

   408→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   409→  const content = section.content as unknown as FeaturedProductsContent;
   410→  const borderRadius = borderRadiusClass(theme.borderRadius);
   411→  const featuredProducts = useMemo(
   412→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   413→    [content.productIds, products],
   414→  );
   415→
   416→  return (
   417→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   418→      {content.headline && (
   419→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   420→          {content.headline}
   421→        </h2>
   422→      )}
   423→      {content.subtitle && (
   424→        <p className="mb-8 text-sm opacity-65">
   425→          {content.subtitle}
   426→        </p>
   427→      )}
   428→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   429→        {featuredProducts.map((product) => (
   430→          <ProductCard
   431→            key={product.id}
   432→            product={product}
   433→            theme={theme}
   434→            showAddToCart={content.showAddToCart && !forceHideAddToCart}
   435→            borderRadius={borderRadius}
   436→            buttonBgOverride={section.style.buttonBackgroundColor}
   437→            buttonTextOverride={section.style.buttonTextColor}
   438→            onViewProduct={onViewProduct}
   439→          />
   440→        ))}
   441→      </div>
   442→    </SectionWrapper>
   443→  );
   444→}

   446→// ─── 4. Product Grid ───────────────────────────────────────────────────
   447→
   448→export function ProductGridSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   449→  const content = section.content as unknown as ProductGridContent;
   450→  const borderRadius = borderRadiusClass(theme.borderRadius);
   451→
   452→  const filteredProducts = useMemo(() => {
   453→    let prods = products;
   454→    if (content.filterByCategory) {
   455→      prods = prods.filter((p) => p.category === content.filterByCategory);
   456→    }
   457→    return prods;
   458→  }, [products, content.filterByCategory]);
   459→
   460→  return (
   461→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   462→      {content.headline && (
   463→        <h2 className="mb-8 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   464→          {content.headline}
   465→        </h2>
   466→      )}
   467→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   468→        {filteredProducts.map((product) => (
   469→          <ProductCard
   470→            key={product.id}
   471→            product={product}
   472→            theme={theme}
   473→            showAddToCart={content.showAddToCart && !forceHideAddToCart}
   474→            borderRadius={borderRadius}
   475→            buttonBgOverride={section.style.buttonBackgroundColor}
   476→            buttonTextOverride={section.style.buttonTextColor}
   477→            onViewProduct={onViewProduct}
   478→          />
   479→        ))}
   480→      </div>
   481→      {filteredProducts.length === 0 && (
   482→        <div className="py-16 text-center opacity-65">
   483→          <p className="text-sm">No products found.</p>
   484→        </div>
   485→      )}
   486→    </SectionWrapper>
   487→  );
   488→}

   492→export function TextBannerSection({ section, theme, selectedSectionId, onSelectSection }: SectionRendererProps) {
   493→  const content = section.content as unknown as TextBannerContent;
   494→  const alignMap = {
   495→    left: 'text-left',
   496→    center: 'text-center',
   497→    right: 'text-right',

   500→    sm: { headline: 'text-xl sm:text-2xl', body: 'text-sm' },
   501→    md: { headline: 'text-2xl sm:text-3xl md:text-4xl', body: 'text-base' },
   502→    lg: { headline: 'text-3xl sm:text-4xl md:text-5xl lg:text-6xl', body: 'text-lg' },
   503→  };
   504→  const sizes = sizeMap[content.size] || sizeMap.md;

   509→        <h2 className={`font-bold leading-tight tracking-tight ${sizes.headline}`} style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   510→          {content.headline}
   511→        </h2>
   512→        {content.body && (
   513→          <p className={`mt-3 max-w-2xl ${sizes.body} opacity-65`}>
   514→            {content.body}
   515→          </p>
   516→        )}
   517→      </div>
   518→    </SectionWrapper>
   519→  );

   524→export function ImageGallerySection({ section, theme, selectedSectionId, onSelectSection }: SectionRendererProps) {
   525→  const content = section.content as unknown as ImageGalleryContent;
   526→  const borderRadius = borderRadiusClass(theme.borderRadius);

   531→      <div className={`grid ${gridCols(content.columns)} ${gapMap[content.gap] || gapMap.md}`}>
   532→        {content.images.map((img, i) => (

   534→            <div
   535→              className={`aspect-[4/3] w-full ${borderRadius} bg-gray-200 transition-transform duration-300 group-hover:scale-[1.02]`}
   536→              style={{
   537→                backgroundColor: stringToColor(img.src || `img-${i}`, theme),

   993→// ─── Main Section Router ────────────────────────────────────────────────
   994→
   995→export function renderSection(props: SectionRendererProps): React.ReactNode {
   996→  const { section } = props;
   997→
   998→  if (!section.visible) return null;
   999→
  1000→  switch (section.type) {
  1001→    case 'header':
  1002→      return <HeaderSection {...props} />;
  1003→    case 'hero':
  1004→      return <HeroSection {...props} />;
  1005→    case 'featured-products':
  1006→      return <FeaturedProductsSection {...props} />;
  1007→    case 'product-grid':
  1008→      return <ProductGridSection {...props} />;
  1009→    case 'text-banner':
  1010→      return <TextBannerSection {...props} />;
  1011→    case 'image-gallery':
  1012→      return <ImageGallerySection {...props} />;
  1013→    case 'testimonials':
  1014→      return <TestimonialsSection {...props} />;
  1015→    case 'newsletter':
  1016→      return <NewsletterSection {...props} />;
  1017→    case 'faq':
  1018→      return <FAQSection {...props} />;
  1019→    case 'cta':
  1020→      return <CTASection {...props} />;
  1021→    case 'categories':
  1022→      return <CategoriesSection {...props} />;
  1023→    case 'footer':
  1024→      return <FooterSection {...props} />;
  1025→    case 'rich-text':
  1026→      return <RichTextSection {...props} />;
  1027→    case 'spacer':
  1028→      return <SpacerSection {...props} />;
  1029→    case 'divider':
  1030→      return <DividerSection {...props} />;
  1031→    default:
  1032→      return null;
  1033→  }
  1034→}
