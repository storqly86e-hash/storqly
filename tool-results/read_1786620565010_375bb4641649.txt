    91→function LandingPage() {
    92→  const [promptText, setPromptText] = useState('')
    93→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    94→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    95→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
    96→  const abortRef = useRef<AbortController | null>(null)
    97→
    98→  const {
    99→    isGenerating,
   100→    setIsGenerating,
   101→    setStore,
   102→    setStoreWithFallback,
   103→  } = useStoreEditor()
   104→
   105→  // Local UI state (not in Zustand — this is view-level)
   106→  const [generationStatus, setGenerationStatus] = useState('')
   107→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   108→  const [error, setError] = useState<string | null>(null)
   109→
   110→  const clearTimers = useCallback(() => {
   111→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   112→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }
   113→  }, [])
   114→
   115→  const handleCancel = useCallback(() => {
   116→    abortRef.current?.abort()
   117→    clearTimers()
   118→    setIsGenerating(false)
   119→    setGenerationStatus('')
   120→    setElapsedSeconds(0)
   121→    setError(null)
   122→  }, [clearTimers, setIsGenerating])
   123→
   124→  const handleGenerate = useCallback(async () => {
   125→    const trimmed = promptText.trim()
   126→    if (!trimmed) {
   127→      setError('Please describe your store to get started.')
   128→      textareaRef.current?.focus()
   129→      return
   130→    }
   131→
   132→    // Reset state
   133→    setError(null)
   134→    setIsGenerating(true)
   135→    setGenerationStatus('Starting generation...')
   136→    setElapsedSeconds(0)
   137→
   138→    // Create abort controller for this request
   139→    const controller = new AbortController()
   140→    abortRef.current = controller
   141→
   142→    // Elapsed time counter
   143→    elapsedTimerRef.current = setInterval(() => {
   144→      setElapsedSeconds((prev) => prev + 1)
   145→    }, 1000)
   146→
   147→    // Fallback progress cycling (used only if SSE progress events stop)
   148→    let fallbackIdx = 0
   149→    let receivedSSEProgress = false
   150→    progressTimerRef.current = setInterval(() => {
   151→      if (!receivedSSEProgress) {
   152→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   153→        setGenerationStatus(progressMessages[fallbackIdx])
   154→      }
   155→    }, 3000)
   156→
   157→    try {
   158→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   159→
   160→      const res = await fetch('/api/store/generate', {
   161→        method: 'POST',
   162→        headers: { 'Content-Type': 'application/json' },
   163→        body: JSON.stringify({ prompt: trimmed }),
   164→        signal: controller.signal,
   165→      })
   166→
   167→      if (!res.ok) {
   168→        let errorMsg = `Server error (${res.status})`
   169→        try {
   170→          const errorData = await res.json()
   171→          errorMsg = errorData.error || errorMsg
   172→        } catch { /* ignore parse error */ }
   173→        throw new Error(errorMsg)
   174→      }
   175→
   176→      // ── Consume SSE stream ──
   177→      const reader = res.body?.getReader()
   178→      if (!reader) throw new Error('No response stream received.')
   179→
   180→      const decoder = new TextDecoder()
   181→      let buffer = ''
   182→      let resolved = false
   183→
   184→      while (true) {
   185→        const { done, value } = await reader.read()
   186→        if (done) break
   187→
   188→        buffer += decoder.decode(value, { stream: true })
   189→
   190→        // Parse SSE events from buffer
   191→        // SSE format: "event: <name>\ndata: <json>\n\n" or ": comment\n\n"
   192→        const lines = buffer.split('\n')
   193→        // Keep incomplete last line in buffer
   194→        buffer = lines.pop() || ''
   195→
   196→        let currentEvent = ''
   197→        for (const line of lines) {
   198→          if (line.startsWith(':')) {
   199→            // SSE comment (heartbeat) — skip
   200→            continue
   201→          }
   202→          if (line.startsWith('event: ')) {
   203→            currentEvent = line.slice(7).trim()
   204→          } else if (line.startsWith('data: ')) {
   205→            const dataStr = line.slice(6)
   206→
   207→            // Handle SSE events
   208→            if (currentEvent === 'progress') {
   209→              receivedSSEProgress = true
   210→              try {
   211→                const data = JSON.parse(dataStr)
   212→                setGenerationStatus(data.message || 'Processing...')
   213→              } catch { /* ignore */ }
   214→            } else if (currentEvent === 'result') {
   215→              try {
   216→                const data = JSON.parse(dataStr)
   217→                if (!data.store) {
   218→                  throw new Error('The AI response was missing store data.')
   219→                }
   220→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   221→                resolved = true
   222→                clearTimers()
   223→
   224→                if (data._isFallback) {
   225→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   226→                } else {
   227→                  setStore(data.store)
   228→                }
   229→              } catch (parseErr) {
   230→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   231→                console.error('[Storqly] Failed to parse result event:', msg)
   232→                throw new Error(msg)
   233→              }
   234→            } else if (currentEvent === 'error') {
   235→              try {
   236→                const data = JSON.parse(dataStr)
   237→                throw new Error(data.message || 'Generation failed')
   238→              } catch (e) {
   239→                if (e instanceof Error && e.message !== 'Generation failed') throw e
   240→                throw new Error('Generation failed')
   241→              }
   242→            }
   243→            // Reset event after processing its data
   244→            currentEvent = ''
   245→          }
   246→        }
   247→      }
   248→
   249→      if (!resolved) {
   250→        throw new Error('Stream ended without a result. The server may have disconnected.')
   251→      }
   252→    } catch (err: unknown) {
   253→      clearTimers()
   254→
   255→      let message = 'Something went wrong. Please try again.'
   256→      if (err instanceof Error) {
   257→        if (err.name === 'AbortError') {
   258→          message = 'Generation was cancelled.'
   259→        } else {
   260→          console.error('[Storqly] Generation error:', err)
   261→          message = err.message
   262→        }
   263→      }
   264→
   265→      setError(message)
   266→      setIsGenerating(false)
   267→      setGenerationStatus('')
   268→      setElapsedSeconds(0)
   269→      toast.error('Store generation failed', { description: message })
   270→    }
   271→  }, [promptText, setIsGenerating, setStore, setStoreWithFallback, clearTimers])
   272→
   273→  // Cleanup on unmount
   274→  useEffect(() => {
   275→    return () => {
   276→      clearTimers()
   277→      abortRef.current?.abort()
   278→    }
   279→  }, [clearTimers])
   280→
   281→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   282→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   283→      e.preventDefault()
   284→      handleGenerate()
   285→    }
   286→  }
   287→
   288→  const formatElapsed = (seconds: number) => {
   289→    if (seconds < 60) return `${seconds}s`
   290→    const m = Math.floor(seconds / 60)
   291→    const s = seconds % 60
   292→    return `${m}:${s.toString().padStart(2, '0')}`
   293→  }
   294→
   295→  return (
   296→    <>
   297→      {/* ── Fixed Nav ── */}
   298→      <header className="fixed top-0 left-0 right-0 z-50 border-b border-white/[0.06] bg-[#09090b]/80 backdrop-blur-xl">
   299→        <nav className="mx-auto flex h-16 max-w-6xl items-center justify-between px-5 sm:px-8">
   300→          <span className="text-2xl font-bold tracking-tight bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   301→            Storqly
   302→          </span>
   303→          <span className="hidden text-sm text-zinc-500 sm:block">
   304→            AI-Powered Store Builder
   305→          </span>
   306→        </nav>
   307→      </header>
   308→
   309→      {/* ── Hero Section ── */}
   310→      <section className="relative flex flex-1 flex-col items-center justify-center px-5 pt-32 pb-20 sm:pt-40 sm:pb-28">
   311→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   312→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />
   313→        </div>
   314→
   315→        <div className="relative z-10 mx-auto w-full max-w-3xl text-center">
   316→          <motion.h1
   317→            custom={0}
   318→            variants={fadeUp}
   319→            initial="hidden"
   320→            animate="visible"
   321→            className="text-4xl font-extrabold leading-[1.1] tracking-tight sm:text-5xl md:text-6xl lg:text-7xl"
   322→          >
   323→            Build Your Store{' '}
   324→            <span className="bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   325→              in Seconds
   326→            </span>
   327→          </motion.h1>
   328→
   329→          <motion.p
   330→            custom={1}
   331→            variants={fadeUp}
   332→            initial="hidden"
   333→            animate="visible"
   334→            className="mt-6 text-lg text-zinc-400 sm:text-xl"
   335→          >
   336→            Describe your store. AI builds it. You customize and publish.
   337→          </motion.p>
   338→
   339→          <motion.div
   340→            custom={2}
   341→            variants={fadeUp}
   342→            initial="hidden"
   343→            animate="visible"
   344→            className="mt-10"
   345→          >
   346→            <div className="group relative rounded-2xl border border-white/[0.08] bg-white/[0.03] shadow-2xl shadow-black/40 transition-all duration-300 focus-within:border-[#a855f7]/40 focus-within:shadow-[0_0_40px_-8px_rgba(168,85,247,0.25)]">
   347→              <Textarea
   348→                ref={textareaRef}
   349→                value={promptText}
   350→                onChange={(e) => { setPromptText(e.target.value); setError(null) }}
   351→                onKeyDown={handleKeyDown}
   352→                disabled={isGenerating}
   353→                placeholder='Describe the store you want to build... (e.g. "A modern minimalist jewelry brand with gold accents, selling handcrafted rings, necklaces, and bracelets")'
   354→                className="min-h-[120px] resize-none rounded-2xl border-0 bg-transparent px-5 py-5 text-base text-white placeholder:text-zinc-600 focus-visible:ring-0 focus-visible:border-0 sm:text-lg md:min-h-[140px]"
   355→              />

   361→                    ⌘↵

   382→                  >

   541→  const [showLeft, setShowLeft] = useState(true)

   632→function EditorToolbar({
   633→  onToggleLeft,
   634→  onToggleRight,
   635→  showLeft,
   636→  showRight,
   637→}: {
   638→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   639→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   640→  showLeft: boolean
   641→  showRight: boolean
   642→}) {
   643→  const store = useStoreEditor((s) => s.store)
   644→  const reset = useStoreEditor((s) => s.reset)
   645→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   646→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   647→  const isPublished = useStoreEditor((s) => s.isPublished)
   648→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   649→  const [isSaving, setIsSaving] = useState(false)
   650→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
   651→  const [copied, setCopied] = useState(false)
   652→
   653→  const handleBack = () => {
   654→    reset()
   655→  }
   656→
   657→  const handleSave = async () => {
   658→    if (!store || isSaving) return
   659→    setIsSaving(true)
   660→    try {
   661→      const res = await fetch('/api/store/save', {
   662→        method: 'POST',
   663→        headers: { 'Content-Type': 'application/json' },
   664→        body: JSON.stringify({ store }),
   665→      })
   666→      if (!res.ok) throw new Error('Save failed')
   667→      toast.success('Store saved')
   668→    } catch {
   669→      toast.error('Failed to save store')
   670→    } finally {
   671→      setIsSaving(false)
   672→    }
   673→  }
   674→
   675→  const handlePublish = async () => {
   676→    if (!store || isPublishing) return
   677→    setIsPublishing(true)
   678→    try {
   679→      const res = await fetch('/api/store/publish', {
   680→        method: 'POST',
   681→        headers: { 'Content-Type': 'application/json' },
   682→        body: JSON.stringify({ store }),
   683→      })
   684→      if (!res.ok) throw new Error('Publish failed')
   685→      const data = await res.json()
   686→      const slug = data.slug
   687→      setIsPublished(true)
   688→      // Build a real viewable URL using the current origin
   689→      const baseUrl = window.location.origin
   690→      const viewUrl = `${baseUrl}/?store=${slug}`
   691→      setPublishedUrl(viewUrl)
   692→      toast.success('Store published successfully!')
   693→    } catch {
   694→      toast.error('Failed to publish store')
   695→    } finally {
   696→      setIsPublishing(false)
   697→    }
   698→  }
   699→
   700→  const handleCopyUrl = useCallback(async () => {
   701→    if (!publishedUrl) return
   702→    try {
   703→      await navigator.clipboard.writeText(publishedUrl)
   704→      setCopied(true)
   705→      toast.success('URL copied to clipboard!')
   706→      setTimeout(() => setCopied(false), 2000)
   707→    } catch {
   708→      // Fallback for older browsers
   709→      const textarea = document.createElement('textarea')
   710→      textarea.value = publishedUrl
   711→      document.body.appendChild(textarea)
   712→      textarea.select()
   713→      document.execCommand('copy')
   714→      document.body.removeChild(textarea)
   715→      setCopied(true)
   716→      toast.success('URL copied to clipboard!')
   717→      setTimeout(() => setCopied(false), 2000)
   718→    }
   719→  }, [publishedUrl])
   720→
   721→  return (
   722→    <>
   723→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
   724→        <div className="flex items-center gap-1">
   725→          <Button
   726→            variant="ghost"
   727→            size="icon"
   728→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   729→            onClick={handleBack}
   730→            aria-label="Back to home"
   731→          >
   732→            <ArrowLeft className="h-4 w-4" />
   733→          </Button>
   734→          <div className="h-5 w-px bg-zinc-800" />
   735→          <Button
   736→            variant={showLeft ? 'secondary' : 'ghost'}
   737→            size="sm"
   738→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   739→            onClick={() => onToggleLeft((v) => !v)}
   740→          >
   741→            <Layers className="h-3.5 w-3.5" />
   742→            <span className="hidden sm:inline">Sections</span>
   743→          </Button>
   744→          <Button
   745→            variant={showRight ? 'secondary' : 'ghost'}
   746→            size="sm"
   747→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   748→            onClick={() => onToggleRight((v) => !v)}
   749→          >
   750→            <MessageSquare className="h-3.5 w-3.5" />
   751→            <span className="hidden sm:inline">Chat</span>
   752→          </Button>
   753→        </div>
   754→
   755→        <div className="flex items-center gap-1.5">
   756→          {publishedUrl && (
   757→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
   758→              <Check className="h-3 w-3 text-emerald-400" />
   759→              <span className="max-w-[180px] truncate text-xs font-medium text-emerald-300">
   760→                {publishedUrl}
   761→              </span>
   762→              <button
   763→                onClick={handleCopyUrl}
   764→                className="ml-1 rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
   765→                aria-label="Copy URL"
   766→              >
   767→                {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
   768→              </button>
   769→              <a
   770→                href={publishedUrl}
   771→                target="_blank"
   772→                rel="noopener noreferrer"
   773→                className="rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
   774→                aria-label="Open published store"
   775→              >
   776→                <ExternalLink className="h-3 w-3" />
   777→              </a>
   778→            </div>
   779→          )}
   780→          <span className="mr-2 text-xs font-medium text-zinc-500 hidden md:inline truncate max-w-[200px]">
   781→            {store?.name || 'Untitled Store'}
   782→          </span>
   783→          <Button
   784→            variant="ghost"
   785→            size="sm"
   786→            className="h-8 gap-1.5 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   787→            onClick={handleSave}
   788→            disabled={isSaving}
   789→          >
   790→            <Save className={`h-3.5 w-3.5 ${isSaving ? 'animate-pulse' : ''}`} />
   791→            <span className="hidden sm:inline">{isSaving ? 'Saving...' : 'Save'}</span>
   792→          </Button>
   793→          <Button
   794→            size="sm"
   795→            className="h-8 gap-1.5 bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] text-white shadow-none hover:opacity-90"
   796→            onClick={handlePublish}
   797→            disabled={isPublishing}
   798→          >
   799→            {isPublishing ? (
   800→              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   801→            ) : isPublished ? (
   802→              <Eye className="h-3.5 w-3.5" />
   803→            ) : (
   804→              <Globe className="h-3.5 w-3.5" />
   805→            )}
   806→            <span className="hidden sm:inline">{isPublishing ? 'Publishing...' : isPublished ? 'Published' : 'Publish'}</span>
   807→          </Button>
   808→        </div>
   809→      </div>
   810→
   811→      {/* Publish Success Dialog */}
   812→      <AnimatePresence>
   813→        {publishedUrl && !isPublishing && (
   814→          <motion.div
   815→            initial={{ opacity: 0 }}
   816→            animate={{ opacity: 1 }}
   817→            exit={{ opacity: 0 }}
   818→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 backdrop-blur-sm"
   819→            onClick={() => setPublishedUrl(null)}
   820→          >
   821→            <motion.div
   822→              initial={{ scale: 0.95, opacity: 0 }}
   823→              animate={{ scale: 1, opacity: 1 }}
   824→              exit={{ scale: 0.95, opacity: 0 }}
   825→              transition={{ duration: 0.2 }}
   826→              className="mx-4 w-full max-w-md rounded-2xl border border-zinc-800 bg-zinc-900 p-6 shadow-2xl"
   827→              onClick={(e) => e.stopPropagation()}
   828→            >
   829→              {/* Success icon */}
   830→              <div className="mx-auto mb-4 flex h-14 w-14 items-center justify-center rounded-full bg-emerald-500/15">
   831→                <Check className="h-7 w-7 text-emerald-400" />
   832→              </div>
   833→
   834→              <h3 className="mb-1 text-center text-lg font-semibold text-zinc-100">
   835→                Store Published!
   836→              </h3>
   837→              <p className="mb-5 text-center text-sm text-zinc-400">
   838→                Your store is now live and ready to share.
   839→              </p>
   840→
   841→              {/* URL Display */}
   842→              <div className="mb-5 rounded-xl border border-zinc-700/60 bg-zinc-800/60 p-3">

   844→                <div className="flex items-center gap-2">

   846→                    {publishedUrl}
   847→                  </div>
   848→                  <Button
   849→                    variant="ghost"
   850→                    size="icon"
   851→                    className="h-9 w-9 shrink-0 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-700"

   887→// ─── Published Store Viewer (read-only, shown via ?store=slug) ────────

   889→function PublishedStoreViewer({ slug }: { slug: string }) {
   890→  const [store, setStore] = useState<Store | null>(null)
   891→  const [loading, setLoading] = useState(true)
   892→  const [error, setError] = useState<string | null>(null)
   893→
   894→  useEffect(() => {
   895→    let cancelled = false
   896→    async function load() {
   897→      try {
   898→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
   899→        if (!res.ok) {
   900→          const data = await res.json().catch(() => ({}))
   901→          throw new Error(data.error || `Store not found (${res.status})`)
   902→        }
   903→        const data = await res.json()
   904→        if (!cancelled) setStore(data.store)
   905→      } catch (err: unknown) {
   906→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
   907→      } finally {
   908→        if (!cancelled) setLoading(false)
   909→      }
   910→    }
   911→    load()
   912→    return () => { cancelled = true }
   913→  }, [slug])
   914→
   915→  if (loading) {
   916→    return (
   917→      <div className="flex min-h-screen items-center justify-center bg-white">
   918→        <div className="text-center">
   919→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
   920→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
   921→        </div>
   922→      </div>
   923→    )
   924→  }
   925→
   926→  if (error || !store) {
   927→    return (
   928→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
   929→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
   930→          <AlertCircle className="h-8 w-8 text-red-400" />
   931→        </div>
   932→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
   933→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
   934→          {error || 'This store could not be found or has not been published.'}
   935→        </p>
   936→        <Button
   937→          variant="outline"
   938→          onClick={() => window.location.href = '/'}
   939→          className="gap-2"
   940→        >
   941→          <ArrowLeft className="h-4 w-4" />
   942→          Back to Storqly
   943→        </Button>
   944→      </div>
   945→    )
   946→  }
   947→
   948→  return (
   949→    <>
   950→      <StoreRenderer store={store} />
   951→      {/* Minimal footer for published stores */}
   952→      <div className="fixed bottom-3 right-3 z-40">
   953→        <a
   954→          href="/"
   955→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
   956→        >
   957→          <Sparkles className="h-3 w-3" />
   958→          Built with Storqly
   959→        </a>
   960→      </div>
   961→    </>
   962→  )
   963→}
