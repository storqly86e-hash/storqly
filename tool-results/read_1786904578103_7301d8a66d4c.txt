     1→// ========================================
     2→// Storqly Schema Normalization Layer
     3→// ========================================
     4→// Deterministic, testable function that takes ANY valid JSON object
     5→// (guaranteed by json_object mode) and reshapes it into a valid Store.
     6→//
     7→// This is NOT JSON repair (that fixes broken syntax).
     8→// This IS schema normalization (coercing valid JSON to match the Store type).
     9→//
    10→// Design principles:
    11→// - Never throws — always returns a valid Store
    12→// - Preserve AI output when valid — only fill defaults for missing/wrong fields
    13→// - Be lenient with types — coerce strings to numbers, etc.
    14→// - Log every normalization for debugging
    15→
    16→import type { Store, StoreTheme, StoreProduct, StorePage, Section, SectionStyle, SectionType, PageType } from './store-schema';
    17→import { defaultTheme } from './store-schema';
    18→
    19→// ─── Validation sets ─────────────────────────────────────────────
    20→
    21→const VALID_SECTION_TYPES = new Set<string>([
    22→  'hero', 'featured-products', 'product-grid', 'text-banner', 'image-gallery',
    23→  'testimonials', 'newsletter', 'faq', 'cta', 'categories', 'header',
    24→  'footer', 'rich-text', 'spacer', 'divider',
    25→]);
    26→
    27→const VALID_SPACING = new Set<string>(['compact', 'normal', 'spacious']);
    28→const VALID_BORDER_RADIUS = new Set<string>(['none', 'sm', 'md', 'lg', 'xl']);
    29→const VALID_PADDING_Y = new Set<string>(['sm', 'md', 'lg', 'xl']);
    30→const VALID_PADDING_X = new Set<string>(['sm', 'md', 'lg']);
    31→const VALID_MAX_WIDTH = new Set<string>(['sm', 'md', 'lg', 'xl', 'full']);
    32→const VALID_ALIGNMENT = new Set<string>(['left', 'center', 'right']);
    33→const VALID_HEIGHT = new Set<string>(['sm', 'md', 'lg', 'xl']);
    34→const VALID_HERO_LAYOUT = new Set<string>(['centered', 'split-left', 'split-right']);
    35→const VALID_SIZE = new Set<string>(['sm', 'md', 'lg']);
    36→const VALID_COLUMNS = new Set<number>([2, 3, 4]);
    37→const VALID_PAGE_TYPES = new Set<string>(['home', 'collection', 'product', 'cart', 'checkout']);
    38→const VALID_CTA_STYLE = new Set<string>(['solid', 'outline', 'gradient']);
    39→const VALID_GAP = new Set<string>(['sm', 'md', 'lg']);

    41→// ─── Logging ──────────────────────────────────────────────────────
    42→
    43→type NormLog = { field: string; action: 'missing' | 'coerced' | 'invalid' | 'defaulted'; from?: unknown; to?: unknown };
    44→
    45→function createLogger() {
    46→  const logs: NormLog[] = [];
    47→  return {
    48→    log(entry: NormLog) { logs.push(entry); },
    49→    getLogs() { return logs; },
    50→    summary(): string {
    51→      const counts = { missing: 0, coerced: 0, invalid: 0, defaulted: 0 };
    52→      for (const l of logs) counts[l.action]++;
    53→      return `Normalization: ${logs.length} fixes (${counts.missing} missing, ${counts.coerced} coerced, ${counts.invalid} invalid, ${counts.defaulted} defaulted)`;
    54→    },
    55→  };
    56→}

    58→// ─── Primitive helpers ────────────────────────────────────────────
    59→
    60→function uuid(): string {
    61→  return crypto.randomUUID();
    62→}
    63→
    64→function str(v: unknown, fallback: string): string {
    65→  if (typeof v === 'string' && v.length > 0) return v;
    66→  if (v === null || v === undefined) return fallback;
    67→  return String(v).slice(0, 500) || fallback;
    68→}
    69→
    70→function num(v: unknown, fallback: number, min?: number, max?: number): number {
    71→  if (typeof v === 'number' && isFinite(v)) {
    72→    let n = v;
    73→    if (min !== undefined) n = Math.max(min, n);
    74→    if (max !== undefined) n = Math.min(max, n);
    75→    return n;
    76→  }
    77→  // Coerce string numbers
    78→  if (typeof v === 'string') {
    79→    const n = parseFloat(v.replace(/[^\d.\-]/g, ''));
    80→    if (!isNaN(n) && isFinite(n)) {
    81→      if (min !== undefined) return Math.max(min, n);
    82→      if (max !== undefined) return Math.min(max, n);
    83→      return n;
    84→    }
    85→  }
    86→  return fallback;
    87→}
    88→
    89→function bool(v: unknown, fallback: boolean): boolean {
    90→  if (typeof v === 'boolean') return v;
    91→  if (typeof v === 'string') return v === 'true' || v === '1';
    92→  if (typeof v === 'number') return v !== 0;
    93→  return fallback;
    94→}
    95→
    96→function arr<T>(v: unknown, fallback: T[]): T[] {
    97→  if (Array.isArray(v)) return v;
    98→  return fallback;
    99→}
   100→
   101→function obj(v: unknown, fallback: Record<string, unknown>): Record<string, unknown> {
   102→  if (v !== null && typeof v === 'object' && !Array.isArray(v)) return v as Record<string, unknown>;
   103→  return fallback;
   104→}
   105→
   106→function enumVal(v: unknown, valid: Set<string>, fallback: string): string {
   107→  if (typeof v === 'string' && valid.has(v)) return v;
   108→  // Case-insensitive fallback
   109→  if (typeof v === 'string') {
   110→    const lower = v.toLowerCase();
   111→    for (const validVal of valid) {
   112→      if (validVal === lower) return validVal;
   113→    }
   114→  }
   115→  return fallback;
   116→}
   117→
   118→function oneLine(s: string): string {
   119→  return s.replace(/[\r\n]+/g, ' ').replace(/\s{2,}/g, ' ').trim();
   120→}
   121→
   122→function slugify(s: string): string {
   123→  return s.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '').substring(0, 60) || 'store';
   124→}
   125→
   126→function isUUID(s: string): boolean {
   127→  return /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(s);
   128→}

   130→// ─── Theme normalization ──────────────────────────────────────────
   131→
   132→function normalizeTheme(raw: unknown, log: ReturnType<typeof createLogger>): StoreTheme {
   133→  const t = obj(raw, {} as Record<string, unknown>);
   134→  const theme: StoreTheme = { ...defaultTheme };
   135→
   136→  // Colors
   137→  const colors = obj(t.colors, {} as Record<string, unknown>);
   138→  const colorKeys = ['primary', 'secondary', 'accent', 'background', 'surface', 'text', 'textMuted', 'border'] as const;
   139→  for (const key of colorKeys) {
   140→    const v = colors[key];
   141→    if (typeof v === 'string' && v.startsWith('#') && v.length >= 4) {
   142→      theme.colors[key] = v;
   143→    } else if (v !== undefined) {
   144→      log.log({ field: `theme.colors.${key}`, action: 'invalid', from: v, to: defaultTheme.colors[key] });
   145→    }
   146→  }
   147→
   148→  // Fonts
   149→  const fonts = obj(t.fonts, {} as Record<string, unknown>);
   150→  theme.fonts.heading = str(fonts.heading, 'Inter');
   151→  theme.fonts.body = str(fonts.body, 'Inter');
   152→
   153→  // Spacing
   154→  theme.spacing = enumVal(t.spacing, VALID_SPACING, 'normal') as StoreTheme['spacing'];
   155→  if (t.spacing && theme.spacing === 'normal' && !VALID_SPACING.has(t.spacing as string)) {
   156→    log.log({ field: 'theme.spacing', action: 'invalid', from: t.spacing, to: 'normal' });
   157→  }
   158→
   159→  // Border radius
   160→  theme.borderRadius = enumVal(t.borderRadius, VALID_BORDER_RADIUS, 'md') as StoreTheme['borderRadius'];
   161→
   162→  return theme;
   163→}

   165→// ─── Section style normalization ──────────────────────────────────
   166→
   167→function normalizeSectionStyle(raw: unknown, log: ReturnType<typeof createLogger>): SectionStyle {
   168→  const s = obj(raw, {} as Record<string, unknown>);
   169→  const style: SectionStyle = {};
   170→
   171→  if (typeof s.backgroundColor === 'string' && s.backgroundColor) style.backgroundColor = s.backgroundColor;
   172→  if (typeof s.textColor === 'string' && s.textColor) style.textColor = s.textColor;
   173→  if (typeof s.backgroundImage === 'string' && s.backgroundImage) style.backgroundImage = s.backgroundImage;
   174→  if (typeof s.overlay === 'boolean') style.overlay = s.overlay;
   175→
   176→  style.paddingY = enumVal(s.paddingY, VALID_PADDING_Y, 'md') as SectionStyle['paddingY'];
   177→  style.paddingX = enumVal(s.paddingX, VALID_PADDING_X, 'md') as SectionStyle['paddingX'];
   178→  style.maxWidth = enumVal(s.maxWidth, VALID_MAX_WIDTH, 'lg') as SectionStyle['maxWidth'];
   179→  style.borderRadius = enumVal(s.borderRadius, VALID_BORDER_RADIUS, 'none') as SectionStyle['borderRadius'];
   180→
   181→  return style;
   182→}

   366→// ─── Section normalization ────────────────────────────────────────
   367→
   368→function normalizeSection(raw: unknown, log: ReturnType<typeof createLogger>): Section {
   369→  const s = obj(raw, {} as Record<string, unknown>);
   370→
   371→  // Validate and normalize type
   372→  let type = str(s.type, 'text-banner').toLowerCase().trim() as SectionType;
   373→  if (!VALID_SECTION_TYPES.has(type)) {
   374→    log.log({ field: 'section.type', action: 'invalid', from: s.type, to: 'text-banner' });
   375→    type = 'text-banner';
   376→  }
   377→
   378→  return {
   379→    id: isUUID(str(s.id, '')) ? str(s.id, '') : uuid(),
   380→    type,
   381→    content: normalizeSectionContent(type, s.content, log),
   382→    style: normalizeSectionStyle(s.style, log),
   383→    visible: bool(s.visible, true),
   384→  };
   385→}

   387→// ─── Product normalization ────────────────────────────────────────

   389→function normalizeProduct(raw: unknown, log: ReturnType<typeof createLogger>): StoreProduct {
   390→  const p = obj(raw, {} as Record<string, unknown>);
   391→
   392→  const name = oneLine(str(p.name, 'Product'));
   393→  const price = num(p.price, 29.99, 0.01, 999999);
   394→  const images = arr<string>(p.images, ['https://images.unsplash.com/photo-1523275335684-37898b6baf30?w=600']);
   395→
   396→  // Normalize variants
   397→  const rawVariants = arr<Record<string, unknown>>(p.variants, []);
   398→  const variants = rawVariants.slice(0, 5).map((v) => ({
   399→    id: isUUID(str(v.id, '')) ? str(v.id, '') : uuid(),
   400→    name: str(v.name, 'Size'),
   401→    options: arr<Record<string, unknown>>(v.options, [{ label: 'M', value: 'm' }]).slice(0, 6).map((opt) => ({
   402→      label: str(opt.label, 'Option'),
   403→      value: str(opt.value, 'option'),
   404→    })),
   405→    price: v.price !== undefined ? num(v.price, undefined as unknown as number, 0) : undefined,
   406→    inStock: bool(v.inStock, true),
   407→  }));
   408→
   409→  return {
   410→    id: isUUID(str(p.id, '')) ? str(p.id, '') : uuid(),
   411→    name,
   412→    price,
   413→    compareAtPrice: p.compareAtPrice !== undefined && p.compareAtPrice !== null ? num(p.compareAtPrice, undefined as unknown as number, 0) : undefined,
   414→    images: images.length > 0 ? images : ['https://images.unsplash.com/photo-1523275335684-37898b6baf30?w=600'],
   415→    description: oneLine(str(p.description, '')),
   416→    category: p.category !== undefined ? str(p.category, '') : undefined,
   417→    variants: variants.length > 0 ? variants : undefined,
   418→    featured: bool(p.featured, false),
   419→    inStock: bool(p.inStock, true),
   420→  };
   421→}

   423→// ─── Page normalization ───────────────────────────────────────────
   424→
   425→/** Infer page type from slug/name if not explicitly set. */
   426→function normalizePageType(
   427→  rawType: unknown,
   428→  slug: string,
   429→  name: string,
   430→  isHomepage: boolean,
   431→  log: ReturnType<typeof createLogger>
   432→): PageType {
   433→  // Explicit type takes priority
   434→  if (typeof rawType === 'string' && VALID_PAGE_TYPES.has(rawType)) {
   435→    return rawType as PageType;
   436→  }
   437→  // Homepage is always 'home'
   438→  if (isHomepage) return 'home';
   439→  // Infer from slug
   440→  const slugLower = slug.toLowerCase();
   441→  if (slugLower === 'shop' || slugLower === 'products' || slugLower === 'collection') return 'collection';
   442→  if (slugLower === 'cart') return 'cart';
   443→  if (slugLower === 'checkout') return 'checkout';
   444→  // Infer from name
   445→  const nameLower = name.toLowerCase();
   446→  if (nameLower.includes('shop') || nameLower.includes('collection') || nameLower.includes('catalog')) return 'collection';
   447→  if (nameLower.includes('cart')) return 'cart';
   448→  if (nameLower.includes('checkout')) return 'checkout';
   449→  // Default
   450→  if (rawType !== undefined) {
   451→    log.log({ field: 'page.type', action: 'coerced', from: String(rawType), to: 'home' });
   452→  }
   453→  return 'home';
   454→}

   456→function normalizePage(raw: unknown, log: ReturnType<typeof createLogger>): StorePage {
   457→  const p = obj(raw, {} as Record<string, unknown>);
   458→  const isHomepage = bool(p.isHomepage, false);
   459→  const slug = str(p.slug, '');
   460→  const name = str(p.name, 'Home').substring(0, 100);
   461→  const pageType = normalizePageType(p.type, slug, name, isHomepage, log);
   462→
   463→  return {
   464→    id: isUUID(str(p.id, '')) ? str(p.id, '') : uuid(),
   465→    name,
   466→    slug,
   467→    type: pageType,
   468→    isHomepage,
   469→    productId: p.productId && typeof p.productId === 'string' ? str(p.productId, '') : undefined,
   470→    sections: arr<unknown>(p.sections, []).map((s) => normalizeSection(s, log)),
   471→  };
   472→}

   476→/** Hard-cap enforcement: truncate sections and products to safe output limits.
   477→ *  This is the safety net if the AI ignores the system prompt caps.
   478→ *  Layers of defense:
   479→ *    1. System prompt language (strong AI guidance)
   480→ *    2. THIS function: hard truncation (final safety net)
   481→ *
   482→ *  @param maxProducts - If provided, cap products to this number. If undefined, don't cap products.
   483→ */
   484→function enforceOutputCaps(store: Store, log: ReturnType<typeof createLogger>, maxProducts?: number): void {
   485→  const MAX_SECTIONS = 4;
   486→
   487→  // ── Cap products (only if maxProducts is specified) ──
   488→  if (maxProducts !== undefined && store.products.length > maxProducts) {
   489→    const before = store.products.length;
   490→    store.products = store.products.slice(0, maxProducts);
   491→    // Ensure at least 1 featured product after truncation
   492→    if (!store.products.some((p) => p.featured) && store.products.length > 0) {
   493→      store.products[0].featured = true;
   494→    }
   495→    log.log({
   496→      field: 'products',
   497→      action: 'coerced',
   498→      from: `${before} products`,
   499→      to: `${maxProducts} products (capped)`,
   500→    });
   501→  }
   502→
   503→  // ── Cap sections per page (only for 'home' type — template pages have 0 sections by design) ──
   504→  for (const page of store.pages) {
   505→    if (page.type && page.type !== 'home') continue; // Skip non-home pages
   506→    if (page.sections.length > MAX_SECTIONS) {
   507→      const before = page.sections.length;
   508→      page.sections = page.sections.slice(0, MAX_SECTIONS);
   509→      log.log({
   510→        field: `pages[${page.name}].sections`,
   511→        action: 'coerced',
   512→        from: `${before} sections`,
   513→        to: `${MAX_SECTIONS} sections (capped)`,
   514→      });
   515→    }
   516→  }
   517→}

   519→/** Fix productIds in featured-products sections to reference actual products */
   520→function fixProductReferences(store: Store, log: ReturnType<typeof createLogger>): void {
   521→  const validProductIds = new Set(store.products.map((p) => p.id));
   522→  const allProductIds = store.products.map((p) => p.id);
   523→
   524→  for (const page of store.pages) {
   525→    for (const section of page.sections) {
   526→      if (section.type === 'featured-products' && Array.isArray(section.content.productIds)) {
   527→        const before = section.content.productIds.length;
   528→        // Keep only IDs that reference actual products
   529→        const validIds = (section.content.productIds as string[]).filter((id) => validProductIds.has(id));
   530→
   531→        // Fill missing slots with other product IDs so all products are shown
   532→        const usedIds = new Set(validIds);
   533→        for (const pid of allProductIds) {
   534→          if (validIds.length >= store.products.length) break;
   535→          if (!usedIds.has(pid)) {
   536→            validIds.push(pid);
   537→            usedIds.add(pid);
   538→          }
   539→        }
   540→
   541→        if (validIds.length !== before) {
   542→          const invalidCount = before - (validIds.length - (store.products.length - before > 0 ? store.products.length - before : 0));
   543→          log.log({ field: 'featured-products.productIds', action: 'coerced', from: `${before} IDs (some invalid)`, to: `${validIds.length} valid IDs` });
   544→        }
   545→        section.content.productIds = validIds;
   546→      }
   547→    }
   548→  }
   549→}

   551→/** Strip junk pages: pages with type='home' that are NOT the homepage.
   552→ *  These are AI-generated pages (e.g. "About Us", "Shipping Policy") with no sections

   554→function stripJunkPages(store: Store, log: ReturnType<typeof createLogger>): void {

   561→    log.log({ field: 'pages', action: 'coerced', from: `${before} pages`, to: `${store.pages.length} pages (${removed} non-homepage 'home' pages removed)` });
   562→  }

   615→/** Fill in default metadata for template pages using store name for branding */
   616→function ensurePageMetadata(store: Store, log: ReturnType<typeof createLogger>): void {
   617→  const name = store.name || 'My Store';
   618→
   619→  // Generate branded metadata from store name (deterministic — no AI needed)
   620→  const branded: Record<string, Record<string, string>> = {
   621→    collection: {
   622→      headline: `The ${name} Collection`,
   623→      subtitle: `Curated pieces for every style`,
   624→    },
   625→    cart: {
   626→      headline: `Your ${name} Selection`,
   627→      emptyMessage: `Your ${name} collection awaits. Start exploring our pieces!`,
   628→    },
   629→    checkout: {
   630→      successHeadline: `Welcome to ${name}!`,
   631→      successMessage: `Thank you for your order from ${name}. Your items are on the way!`,
   632→    },
   633→  };
   634→
   635→  let fixed = 0;
   636→  for (const page of store.pages) {
   637→    const typeDefaults = branded[page.type || ''];
   638→    if (!typeDefaults) continue;
   639→
   640→    if (!page.metadata || typeof page.metadata !== 'object') {
   641→      page.metadata = { ...typeDefaults };
   642→      fixed += Object.keys(typeDefaults).length;
   643→      continue;
   644→    }
   645→
   646→    // Only fill in keys the AI explicitly generated; keep AI text if present
   647→    const missing = Object.keys(typeDefaults).filter((k) => !page.metadata[k] || typeof page.metadata[k] !== 'string' || page.metadata[k].trim().length === 0);
   648→    for (const key of missing) {
   649→      page.metadata[key] = typeDefaults[key];
   650→      fixed++;
   651→    }
   652→  }
   653→
   654→  if (fixed > 0) {
   655→    log.log({ field: 'page metadata', action: 'defaulted', to: `filled ${fixed} metadata values with branded defaults for ${name}` });
   656→  }
   657→}

   668→/** Pad product array to MIN_PRODUCTS if AI generated too few */
   669→const MIN_PRODUCTS = 1;
   670→const PAD_PRODUCT_TEMPLATES = [
   671→  { name: 'Essential Item', price: 39.99, description: 'A must-have for every collection.', category: 'Essentials', img: 'photo-1505740420928-5e560c06d30e' },

   673→];
   674→function padProducts(store: Store, log: ReturnType<typeof createLogger>): void {
   675→  const deficit = MIN_PRODUCTS - store.products.length;
   676→  if (deficit <= 0) return;
   677→  const before = store.products.length;
   678→  for (let i = 0; i < deficit; i++) {
   679→    const tmpl = PAD_PRODUCT_TEMPLATES[i % PAD_PRODUCT_TEMPLATES.length];
   680→    store.products.push({
   681→      id: uuid(),
   682→      name: tmpl.name + (store.products.length + 1),
   683→      price: tmpl.price + store.products.length * 5,
   684→      compareAtPrice: null,
   685→      images: [`https://images.unsplash.com/${tmpl.img}?w=600`],
   686→      description: tmpl.description,
   687→      category: tmpl.category,
   688→      featured: false,
   689→      inStock: true,
   690→    });
   691→  }
   692→  log.log({ field: 'products', action: 'defaulted', from: `${before} products`, to: `${before + deficit} products (padded to min ${MIN_PRODUCTS})` });
   693→}

   697→export interface NormalizeResult {
   698→  store: Store;
   699→  log: string[];        // Human-readable log lines
   700→  summary: string;      // One-line summary
   701→  isFallback: boolean;  // true if the input was too broken to normalize
   702→  normalizationCount: number; // Number of fixes applied
   703→}

   705→/**
   706→ * Normalize an array of raw product objects into valid StoreProduct[].
   707→ * Lightweight version — no store/page/section normalization.
   708→ * Used for Phase 2 batch products.
   709→ */
   710→export function normalizeProducts(rawProducts: unknown[]): StoreProduct[] {
   711→  const log = createLogger();
   712→  if (!Array.isArray(rawProducts)) return [];
   713→  return rawProducts.map((p) => normalizeProduct(p, log));
   714→}
   715→
   716→/**
   717→ * Normalize any valid JSON object into a conformant Store.
   718→ *
   719→ * This function NEVER throws. If the input is too malformed to recover,
   720→ * it returns null and the caller should use a hardcoded fallback store.
   721→ *
   722→ * @param raw - Parsed JSON object (JSON.parse already succeeded)
   723→ * @param prompt - Original user prompt (for fallback name extraction)
   724→ * @param maxProducts - If provided, cap products to this number. If undefined, don't cap.
   725→ */
   726→export function normalizeStore(raw: unknown, prompt?: string, maxProducts?: number): NormalizeResult | null {
   727→  const log = createLogger();
   728→
   729→  // ── Guard: must be a non-null object ──
   730→  if (raw === null || typeof raw !== 'object' || Array.isArray(raw)) {
   731→    console.warn('[normalizeStore] Input is not an object:', typeof raw);
   732→    return null;
   733→  }
   734→
   735→  const r = raw as Record<string, unknown>;
   736→
   737→  // ── Extract store name (used as fallback for missing name) ──
   738→  const storeName = oneLine(str(r.name, prompt ? extractStoreNameFromPrompt(prompt) : 'My Store'));
   739→
   740→  if (!r.name || typeof r.name !== 'string') {
   741→    log.log({ field: 'name', action: 'missing', to: storeName });
   742→  }
   743→
   744→  // ── Top-level fields ──
   745→  const now = new Date().toISOString();
   746→  const store: Store = {
   747→    id: isUUID(str(r.id, '')) ? str(r.id, '') : uuid(),
   748→    name: storeName.substring(0, 100),
   749→    slug: slugify(storeName),
   750→    description: r.description !== undefined ? oneLine(str(r.description, '')) : undefined,
   751→    theme: r.theme ? normalizeTheme(r.theme, log) : defaultTheme,
   752→    pages: Array.isArray(r.pages)
   753→      ? r.pages.map((p: unknown) => normalizePage(p, log))
   754→      : [createDefaultPage(storeName)],
   755→    products: Array.isArray(r.products)
   756→      ? (r.products as unknown[]).map((p: unknown) => normalizeProduct(p, log))
   757→      : [],
   758→    published: bool(r.published, false),
   759→    publishedAt: r.publishedAt !== undefined ? str(r.publishedAt, '') : undefined,
   760→    createdAt: str(r.createdAt, now),
   761→    updatedAt: str(r.updatedAt, now),
   762→  };
   763→
   764→  // Log missing arrays
   765→  if (!Array.isArray(r.pages)) log.log({ field: 'pages', action: 'missing' });
   766→  if (!Array.isArray(r.products)) log.log({ field: 'products', action: 'missing' });
   767→  if (!r.theme) log.log({ field: 'theme', action: 'missing' });
   768→
   769→  // ── Ensure at least one page exists ──
   770→  if (store.pages.length === 0) {
   771→    store.pages.push(createDefaultPage(storeName));
   772→    log.log({ field: 'pages', action: 'defaulted' });
   773→  }
   774→
   775→  // ── Ensure at least one product exists (renderer expects products array) ──
   776→  if (store.products.length === 0) {
   777→    store.products.push(createDefaultProduct());
   778→    log.log({ field: 'products', action: 'defaulted' });
   779→  }
   780→
   781→  // ── Cross-reference fixes ──
   782→  // IMPORTANT: enforceOutputCaps MUST run before fixProductReferences
   783→  // because product references depend on the final product list
   784→  enforceOutputCaps(store, log, maxProducts);
   785→  padProducts(store, log);
   786→  fixProductReferences(store, log);
   787→  stripJunkPages(store, log);
   788→  ensureHomepage(store, log);
   789→  ensureFeaturedProducts(store, log);
   790→  ensureTemplatePages(store, log);
   791→  ensurePageMetadata(store, log);
   792→
   793→  // ── Build result ──
   794→  const logEntries = log.getLogs();
   795→  const logLines = logEntries.map(
   796→    (l) => `[${l.action}] ${l.field}${l.from !== undefined ? ` (was: ${JSON.stringify(l.from).substring(0, 60)})` : ''}${l.to !== undefined ? ` → ${JSON.stringify(l.to).substring(0, 60)}` : ''}`
   797→  );
   798→
   799→  return {
   800→    store,
   801→    log: logLines,
   802→    summary: log.summary(),
   803→    isFallback: false,
   804→    normalizationCount: logEntries.length,
   805→  };
   806→}

   848→/** Extract a short store name from a user prompt (lightweight version for normalization) */
   849→function extractStoreNameFromPrompt(prompt: string): string {
   850→  const text = prompt.trim();
   851→
   852→  // Brand name after "called / named / known as"
   853→  const calledMatch = text.match(/(?:called|named|known\s+as)\s+([\w&'\-]+(?:\s+[\w&'\-]+){0,2})/i);
   854→  if (calledMatch?.[1]) {
   855→    const name = calledMatch[1].replace(/\s+(selling|with|that|for|using|featuring)\s*$/i, '').trim();
   856→    if (name.length >= 2 && name.length <= 40) return name;
   857→  }
   858→
   859→  // Quoted brand name
   860→  const quotedMatch = text.match(/["']([^"']{2,40})["']/);
   861→  if (quotedMatch?.[1]) return quotedMatch[1].trim();
   862→
   863→  // Strip common prefixes
   864→  const stripped = text
   865→    .replace(/^(build|create|make|design|set\s+up)\s+(a|an|the|my)\s+/i, '')
   866→    .replace(/\b(online|e-commerce|ecommerce)\s+(store|shop|boutique)\b/gi, '')
   867→    .replace(/\b(store|shop|boutique|website|site|brand)\b/gi, '')
   868→    .replace(/\b(selling|with|that|for|using|featuring)\b.*/i, '')
   869→    .trim();
   870→
   871→  // Find title-case run
   872→  const words = stripped.split(/\s+/).filter(w => w.length >= 2);
   873→  const isTitle = (w: string) => w[0] === w[0].toUpperCase() && w[0] !== w[0].toLowerCase();
   874→  let bestRun: string[] = [];
   875→  let currentRun: string[] = [];
   876→  for (const w of words) {
   877→    if (isTitle(w)) {
   878→      currentRun.push(w.replace(/[^a-zA-Z0-9&'-]/g, ''));
   879→    } else {
   880→      if (currentRun.length > bestRun.length) bestRun = currentRun;
   881→      currentRun = [];
   882→    }
   883→  }
   884→  if (currentRun.length > bestRun.length) bestRun = currentRun;
   885→
   886→  if (bestRun.length >= 1) {
   887→    const candidate = bestRun.slice(0, 3).join(' ');
   888→    if (candidate.length >= 2 && candidate.length <= 40) return candidate;
   889→  }
   890→
   891→  return 'My Store';
   892→}
