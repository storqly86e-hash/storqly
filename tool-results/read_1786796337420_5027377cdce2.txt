    42→type NormLog = { field: string; action: 'missing' | 'coerced' | 'invalid' | 'defaulted'; from?: unknown; to?: unknown };
    43→
    44→function createLogger() {
    45→  const logs: NormLog[] = [];
    46→  return {
    47→    log(entry: NormLog) { logs.push(entry); },
    48→    getLogs() { return logs; },
    49→    summary(): string {
    50→      const counts = { missing: 0, coerced: 0, invalid: 0, defaulted: 0 };
    51→      for (const l of logs) counts[l.action]++;
    52→      return `Normalization: ${logs.length} fixes (${counts.missing} missing, ${counts.coerced} coerced, ${counts.invalid} invalid, ${counts.defaulted} defaulted)`;
    53→    },
    54→  };
    55→}

    63→function str(v: unknown, fallback: string): string {
    64→  if (typeof v === 'string' && v.length > 0) return v;
    65→  if (v === null || v === undefined) return fallback;
    66→  return String(v).slice(0, 500) || fallback;

    69→function num(v: unknown, fallback: number, min?: number, max?: number): number {
    70→  if (typeof v === 'number' && isFinite(v)) {
    71→    let n = v;
    72→    if (min !== undefined) n = Math.max(min, n);
    73→    if (max !== undefined) n = Math.min(max, n);
    74→    return n;
    75→  }
    76→  // Coerce string numbers
    77→  if (typeof v === 'string') {
    78→    const n = parseFloat(v.replace(/[^\d.\-]/g, ''));
    79→    if (!isNaN(n) && isFinite(n)) {
    80→      if (min !== undefined) return Math.max(min, n);
    81→      if (max !== undefined) return Math.min(max, n);
    82→      return n;
    83→    }
    84→  }
    85→  return fallback;
    86→}

    88→function bool(v: unknown, fallback: boolean): boolean {
    89→  if (typeof v === 'boolean') return v;
    90→  if (typeof v === 'string') return v === 'true' || v === '1';
    91→  if (typeof v === 'number') return v !== 0;
    92→  return fallback;
    93→}

   100→function obj(v: unknown, fallback: Record<string, unknown>): Record<string, unknown> {
   101→  if (v !== null && typeof v === 'object' && !Array.isArray(v)) return v as Record<string, unknown>;
   102→  return fallback;

   105→function enumVal(v: unknown, valid: Set<string>, fallback: string): string {
   106→  if (typeof v === 'string' && valid.has(v)) return v;
   107→  // Case-insensitive fallback
   108→  if (typeof v === 'string') {
   109→    const lower = v.toLowerCase();
   110→    for (const validVal of valid) {
   111→      if (validVal === lower) return validVal;
   112→    }
   113→  }
   114→  return fallback;
   115→}

   410→  rawType: unknown,

   412→  name: string,
   413→  isHomepage: boolean,
   414→  log: ReturnType<typeof createLogger>
   415→): PageType {
   416→  // Explicit type takes priority
   417→  if (typeof rawType === 'string' && VALID_PAGE_TYPES.has(rawType)) {

   420→  // Homepage is always 'home'

   423→  const slugLower = slug.toLowerCase();
   424→  if (slugLower === 'shop' || slugLower === 'products' || slugLower === 'collection') return 'collection';
   425→  if (slugLower === 'cart') return 'cart';
   426→  if (slugLower === 'checkout') return 'checkout';
   427→  // Infer from name
   428→  const nameLower = name.toLowerCase();
   429→  if (nameLower.includes('shop') || nameLower.includes('collection') || nameLower.includes('catalog')) return 'collection';
   430→  if (nameLower.includes('cart')) return 'cart';
   431→  if (nameLower.includes('checkout')) return 'checkout';
   432→  // Default
   433→  if (rawType !== undefined) {
   434→    log.log({ field: 'page.type', action: 'coerced', from: String(rawType), to: 'home' });
   435→  }
   436→  return 'home';
   437→}

   457→// ─── Cross-reference fixes ────────────────────────────────────────
   458→
   459→/** Hard-cap enforcement: truncate sections and products to safe output limits.
   460→ *  This is the safety net if the AI ignores the system prompt caps.
   461→ *  Layers of defense:
   462→ *    1. sanitizePrompt() in route.ts strips count requests (prevents AI from trying)
   463→ *    2. System prompt ABSOLUTE CAPS language (strong AI guidance)
   464→ *    3. THIS function: hard truncation (final safety net)
   465→ */
   466→function enforceOutputCaps(store: Store, log: ReturnType<typeof createLogger>): void {
   467→  const MAX_PRODUCTS = 3;
   468→  const MAX_SECTIONS = 4;
   469→
   470→  // ── Cap products ──
   471→  if (store.products.length > MAX_PRODUCTS) {
   472→    const before = store.products.length;
   473→    store.products = store.products.slice(0, MAX_PRODUCTS);
   474→    // Ensure at least 1 featured product after truncation
   475→    if (!store.products.some((p) => p.featured) && store.products.length > 0) {
   476→      store.products[0].featured = true;
   477→    }
   478→    log.log({
   479→      field: 'products',
   480→      action: 'coerced',
   481→      from: `${before} products`,
   482→      to: `${MAX_PRODUCTS} products (capped)`,
   483→    });
   484→  }
   485→
   486→  // ── Cap sections per page (only for 'home' type — template pages have 0 sections by design) ──
   487→  for (const page of store.pages) {
   488→    if (page.type && page.type !== 'home') continue; // Skip non-home pages
   489→    if (page.sections.length > MAX_SECTIONS) {
   490→      const before = page.sections.length;
   491→      page.sections = page.sections.slice(0, MAX_SECTIONS);
   492→      log.log({
   493→        field: `pages[${page.name}].sections`,
   494→        action: 'coerced',
   495→        from: `${before} sections`,
   496→        to: `${MAX_SECTIONS} sections (capped)`,
   497→      });
   498→    }
   499→  }
   500→}

   502→/** Fix productIds in featured-products sections to reference actual products */
   503→function fixProductReferences(store: Store, log: ReturnType<typeof createLogger>): void {
   504→  const validProductIds = new Set(store.products.map((p) => p.id));
   505→  const allProductIds = store.products.map((p) => p.id);
   506→
   507→  for (const page of store.pages) {
   508→    for (const section of page.sections) {
   509→      if (section.type === 'featured-products' && Array.isArray(section.content.productIds)) {
   510→        const before = section.content.productIds.length;
   511→        // Keep only IDs that reference actual products
   512→        const validIds = (section.content.productIds as string[]).filter((id) => validProductIds.has(id));
   513→
   514→        // Fill missing slots with other product IDs so all products are shown
   515→        const usedIds = new Set(validIds);
   516→        for (const pid of allProductIds) {
   517→          if (validIds.length >= store.products.length) break;
   518→          if (!usedIds.has(pid)) {
   519→            validIds.push(pid);
   520→            usedIds.add(pid);
   521→          }
   522→        }
   523→
   524→        if (validIds.length !== before) {
   525→          const invalidCount = before - (validIds.length - (store.products.length - before > 0 ? store.products.length - before : 0));
   526→          log.log({ field: 'featured-products.productIds', action: 'coerced', from: `${before} IDs (some invalid)`, to: `${validIds.length} valid IDs` });
   527→        }
   528→        section.content.productIds = validIds;
   529→      }
   530→    }
   531→  }
   532→}

   598→/** Fill in default metadata for template pages using store name for branding */
   599→function ensurePageMetadata(store: Store, log: ReturnType<typeof createLogger>): void {
   600→  const name = store.name || 'My Store';
   601→
   602→  // Generate branded metadata from store name (deterministic — no AI needed)
   603→  const branded: Record<string, Record<string, string>> = {
   604→    collection: {
   605→      headline: `The ${name} Collection`,
   606→      subtitle: `Curated pieces for every style`,
   607→    },
   608→    cart: {
   609→      headline: `Your ${name} Selection`,
   610→      emptyMessage: `Your ${name} collection awaits. Start exploring our pieces!`,
   611→    },
   612→    checkout: {
   613→      successHeadline: `Welcome to ${name}!`,
   614→      successMessage: `Thank you for your order from ${name}. Your items are on the way!`,
   615→    },
   616→  };
   617→
   618→  let fixed = 0;
   619→  for (const page of store.pages) {
   620→    const typeDefaults = branded[page.type || ''];
   621→    if (!typeDefaults) continue;
   622→
   623→    if (!page.metadata || typeof page.metadata !== 'object') {
   624→      page.metadata = { ...typeDefaults };
   625→      fixed += Object.keys(typeDefaults).length;
   626→      continue;
   627→    }
   628→
   629→    // Only fill in keys the AI explicitly generated; keep AI text if present
   630→    const missing = Object.keys(typeDefaults).filter((k) => !page.metadata[k] || typeof page.metadata[k] !== 'string' || page.metadata[k].trim().length === 0);
   631→    for (const key of missing) {
   632→      page.metadata[key] = typeDefaults[key];
   633→      fixed++;
   634→    }
   635→  }
   636→
   637→  if (fixed > 0) {
   638→    log.log({ field: 'page metadata', action: 'defaulted', to: `filled ${fixed} metadata values with branded defaults for ${name}` });
   639→  }
   640→}

   680→export interface NormalizeResult {
   681→  store: Store;
   682→  log: string[];        // Human-readable log lines
   683→  summary: string;      // One-line summary
   684→  isFallback: boolean;  // true if the input was too broken to normalize
   685→  normalizationCount: number; // Number of fixes applied
   686→}
   687→
   688→/**
   689→ * Normalize any valid JSON object into a conformant Store.
   690→ *
   691→ * This function NEVER throws. If the input is too malformed to recover,
   692→ * it returns null and the caller should use a hardcoded fallback store.
   693→ *
   694→ * @param raw - Parsed JSON object (JSON.parse already succeeded)
   695→ * @param prompt - Original user prompt (for fallback name extraction)
   696→ */
   697→export function normalizeStore(raw: unknown, prompt?: string): NormalizeResult | null {
   698→  const log = createLogger();
   699→
   700→  // ── Guard: must be a non-null object ──
   701→  if (raw === null || typeof raw !== 'object' || Array.isArray(raw)) {
   702→    console.warn('[normalizeStore] Input is not an object:', typeof raw);
   703→    return null;
   704→  }
   705→
   706→  const r = raw as Record<string, unknown>;
   707→
   708→  // ── Extract store name (used as fallback for missing name) ──
   709→  const storeName = oneLine(str(r.name, prompt ? extractStoreNameFromPrompt(prompt) : 'My Store'));
   710→
   711→  if (!r.name || typeof r.name !== 'string') {
   712→    log.log({ field: 'name', action: 'missing', to: storeName });
   713→  }
   714→
   715→  // ── Top-level fields ──
   716→  const now = new Date().toISOString();
   717→  const store: Store = {
   718→    id: isUUID(str(r.id, '')) ? str(r.id, '') : uuid(),
   719→    name: storeName.substring(0, 100),
   720→    slug: slugify(storeName),
   721→    description: r.description !== undefined ? oneLine(str(r.description, '')) : undefined,
   722→    theme: r.theme ? normalizeTheme(r.theme, log) : defaultTheme,
   723→    pages: Array.isArray(r.pages)
   724→      ? r.pages.map((p: unknown) => normalizePage(p, log))
   725→      : [createDefaultPage(storeName)],
   726→    products: Array.isArray(r.products)
   727→      ? (r.products as unknown[]).map((p: unknown) => normalizeProduct(p, log))
   728→      : [],
   729→    published: bool(r.published, false),
   730→    publishedAt: r.publishedAt !== undefined ? str(r.publishedAt, '') : undefined,
   731→    createdAt: str(r.createdAt, now),
   732→    updatedAt: str(r.updatedAt, now),
   733→  };
   734→
   735→  // Log missing arrays
   736→  if (!Array.isArray(r.pages)) log.log({ field: 'pages', action: 'missing' });
   737→  if (!Array.isArray(r.products)) log.log({ field: 'products', action: 'missing' });
   738→  if (!r.theme) log.log({ field: 'theme', action: 'missing' });
   739→
   740→  // ── Ensure at least one page exists ──
   741→  if (store.pages.length === 0) {
   742→    store.pages.push(createDefaultPage(storeName));
   743→    log.log({ field: 'pages', action: 'defaulted' });
   744→  }
   745→
   746→  // ── Ensure at least one product exists (renderer expects products array) ──
   747→  if (store.products.length === 0) {
   748→    store.products.push(createDefaultProduct());
   749→    log.log({ field: 'products', action: 'defaulted' });
   750→  }
   751→
   752→  // ── Cross-reference fixes ──
   753→  // IMPORTANT: enforceOutputCaps MUST run before fixProductReferences
   754→  // because product references depend on the final product list
   755→  enforceOutputCaps(store, log);
   756→  padProducts(store, log);
   757→  fixProductReferences(store, log);
   758→  stripJunkPages(store, log);
   759→  ensureHomepage(store, log);
   760→  ensureFeaturedProducts(store, log);
   761→  ensureTemplatePages(store, log);
   762→  ensurePageMetadata(store, log);
   763→
   764→  // ── Build result ──
   765→  const logEntries = log.getLogs();
   766→  const logLines = logEntries.map(
   767→    (l) => `[${l.action}] ${l.field}${l.from !== undefined ? ` (was: ${JSON.stringify(l.from).substring(0, 60)})` : ''}${l.to !== undefined ? ` → ${JSON.stringify(l.to).substring(0, 60)}` : ''}`
   768→  );
   769→
   770→  return {
   771→    store,
   772→    log: logLines,
   773→    summary: log.summary(),
   774→    isFallback: false,
   775→    normalizationCount: logEntries.length,
   776→  };
   777→}

   819→/** Extract a short store name from a user prompt (lightweight version for normalization) */
   820→function extractStoreNameFromPrompt(prompt: string): string {
   821→  const text = prompt.trim();
   822→
   823→  // Brand name after "called / named / known as"
   824→  const calledMatch = text.match(/(?:called|named|known\s+as)\s+([\w&'\-]+(?:\s+[\w&'\-]+){0,2})/i);
   825→  if (calledMatch?.[1]) {
   826→    const name = calledMatch[1].replace(/\s+(selling|with|that|for|using|featuring)\s*$/i, '').trim();
   827→    if (name.length >= 2 && name.length <= 40) return name;
   828→  }
   829→
   830→  // Quoted brand name
   831→  const quotedMatch = text.match(/["']([^"']{2,40})["']/);
   832→  if (quotedMatch?.[1]) return quotedMatch[1].trim();
   833→
   834→  // Strip common prefixes
   835→  const stripped = text
   836→    .replace(/^(build|create|make|design|set\s+up)\s+(a|an|the|my)\s+/i, '')
   837→    .replace(/\b(online|e-commerce|ecommerce)\s+(store|shop|boutique)\b/gi, '')
   838→    .replace(/\b(store|shop|boutique|website|site|brand)\b/gi, '')
   839→    .replace(/\b(selling|with|that|for|using|featuring)\b.*/i, '')
   840→    .trim();
   841→
   842→  // Find title-case run
   843→  const words = stripped.split(/\s+/).filter(w => w.length >= 2);
   844→  const isTitle = (w: string) => w[0] === w[0].toUpperCase() && w[0] !== w[0].toLowerCase();
   845→  let bestRun: string[] = [];
   846→  let currentRun: string[] = [];
   847→  for (const w of words) {
   848→    if (isTitle(w)) {
   849→      currentRun.push(w.replace(/[^a-zA-Z0-9&'-]/g, ''));
   850→    } else {
   851→      if (currentRun.length > bestRun.length) bestRun = currentRun;
   852→      currentRun = [];
   853→    }
   854→  }
   855→  if (currentRun.length > bestRun.length) bestRun = currentRun;
   856→
   857→  if (bestRun.length >= 1) {
   858→    const candidate = bestRun.slice(0, 3).join(' ');
   859→    if (candidate.length >= 2 && candidate.length <= 40) return candidate;
   860→  }
   861→
   862→  return 'My Store';
   863→}
