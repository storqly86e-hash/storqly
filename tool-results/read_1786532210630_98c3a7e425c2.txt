    39→    39→function pyClass(paddingY?: SectionStyle['paddingY']) {
    40→    40→  switch (paddingY) {
    41→    41→    case 'sm': return 'py-6';
    42→    42→    case 'md': return 'py-10';
    43→    43→    case 'lg': return 'py-16';
    44→    44→    case 'xl': return 'py-24';
    45→    45→    default: return 'py-12';
    46→    46→  }
    47→    47→}
    48→    48→
    49→    49→/** Get padding-x class from style config */
    50→    50→function pxClass(paddingX?: SectionStyle['paddingX']) {
    51→    51→  switch (paddingX) {
    52→    52→    case 'sm': return 'px-4';
    53→    53→    case 'md': return 'px-6';
    54→    54→    case 'lg': return 'px-10';
    55→    55→    default: return 'px-6';

    75→    75→    case 'sm': return 'rounded-sm';
    76→    76→    case 'md': return 'rounded-md';

    83→    83→/** Generate a consistent color from a string for image placeholders */
    84→    84→function stringToColor(str: string, theme: StoreTheme): string {
    85→    85→  let hash = 0;
    86→    86→  for (let i = 0; i < str.length; i++) {
    87→    87→    hash = str.charCodeAt(i) + ((hash << 5) - hash);
    88→    88→  }
    89→    89→  const hue = Math.abs(hash) % 360;
    90→    90→  return `hsl(${hue}, 35%, 88%)`;
    91→    91→}
    92→    92→
    93→    93→/** Get a contrasting text color for a given background color */
    94→    94→function contrastTextColor(bgColor: string): string {
    95→    95→  const hex = bgColor.replace('#', '');
    96→    96→  if (hex.length !== 6) return '#111827';
    97→    97→  const r = parseInt(hex.substring(0, 2), 16);
    98→    98→  const g = parseInt(hex.substring(2, 4), 16);
    99→    99→  const b = parseInt(hex.substring(4, 6), 16);
   100→   100→  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   101→   101→  return luminance > 0.55 ? '#111827' : '#ffffff';
   102→   102→}

   104→   104→/** Format price */
   105→   105→function formatPrice(price: number): string {
   106→   106→  return new Intl.NumberFormat('en-US', {
   107→   107→    style: 'currency',
   108→   108→    currency: 'USD',
   109→   109→  }).format(price);
   110→   110→}

   112→   112→/** Get initials from a name */
   113→   113→function getInitials(name: string): string {
   114→   114→  return name
   115→   115→    .split(' ')
   116→   116→    .map((n) => n[0])
   117→   117→    .join('')
   118→   118→    .toUpperCase()
   119→   119→    .slice(0, 2);
   120→   120→}

   123→   123→
   124→   124→interface SectionWrapperProps {
   125→   125→  section: Section;
   126→   126→  theme: StoreTheme;
   127→   127→  selectedSectionId?: string | null;
   128→   128→  onSelectSection?: (sectionId: string | null) => void;
   129→   129→  children: React.ReactNode;

   131→   131→
   132→   132→function SectionWrapper({
   133→   133→  section,
   134→   134→  theme,
   135→   135→  selectedSectionId,
   136→   136→  onSelectSection,
   137→   137→  children,
   138→   138→}: SectionWrapperProps) {
   139→   139→  const isSelected = selectedSectionId === section.id;
   140→   140→  const style = section.style;
   141→   141→
   142→   142→  const sectionBg = style.backgroundColor || theme.colors.background;
   143→   143→  const sectionText = style.textColor || theme.colors.text;
   144→   144→
   145→   145→  const bgImage = style.backgroundImage
   146→   146→    ? `url(${style.backgroundImage})`
   147→   147→    : undefined;
   148→   148→
   149→   149→  const outerClasses = [
   150→   150→    'relative transition-all duration-200 cursor-pointer',
   151→   151→    pyClass(style.paddingY),
   152→   152→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   153→   153→  ].join(' ');
   154→   154→
   155→   155→  return (
   156→   156→    <div
   157→   157→      className={outerClasses}
   158→   158→      style={{
   159→   159→        backgroundColor: sectionBg,
   160→   160→        color: sectionText,
   161→   161→        backgroundImage: bgImage,
   162→   162→        backgroundSize: 'cover',
   163→   163→        backgroundPosition: 'center',
   164→   164→      }}
   165→   165→      onClick={(e) => {
   166→   166→        e.stopPropagation();
   167→   167→        onSelectSection?.(isSelected ? null : section.id);
   168→   168→      }}
   169→   169→      data-section-id={section.id}
   170→   170→      data-section-type={section.type}
   171→   171→    >
   172→   172→      {/* Overlay for background images */}
   173→   173→      {style.backgroundImage && style.overlay && (
   174→   174→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   175→   175→      )}
   176→   176→      <div
   177→   177→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   178→   178→      >
   179→   179→        {/* Selected section label */}
   180→   180→        {isSelected && (
   181→   181→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   182→   182→            {section.type.replace(/-/g, ' ')}
   183→   183→          </div>
   184→   184→        )}
   185→   185→        {children}
   186→   186→      </div>
   187→   187→    </div>
   188→   188→  );
   189→   189→}

   193→   193→function ProductCard({

   198→   198→}: {

   251→   251→          {product.name}
   252→   252→        </h3>
   253→   253→        <div className="mt-1.5 flex items-center gap-2">
   254→   254→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>

   276→   276→          </button>
   277→   277→        )}
   278→   278→      </div>
   279→   279→    </div>
   280→   280→  );
   281→   281→}

   395→   395→    left: 'items-start text-left',

   397→   397→    right: 'items-end text-right',

   400→   400→  // Only apply theme gradient as default background when no custom color or image is set.

   402→   402→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   403→   403→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`

   406→   406→  const outerClasses = [

   410→   410→    pyClass(style.paddingY),
   411→   411→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   412→   412→  ].join(' ');
   413→   413→
   414→   414→  return (
   415→   415→    <div

   417→   417→      style={{
   418→   418→        backgroundColor: style.backgroundColor || theme.colors.primary,

   421→   421→        backgroundPosition: 'center',
   422→   422→        color: style.textColor || '#ffffff',
   423→   423→      }}
   424→   424→      onClick={(e) => {
   425→   425→        e.stopPropagation();
   426→   426→        onSelectSection?.(isSelected ? null : section.id);
   427→   427→      }}
   428→   428→      data-section-id={section.id}

   430→   430→    >
   431→   431→      {/* Overlay */}

   433→   433→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   434→   434→      )}
   435→   435→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}

   437→   437→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />

   439→   439→      {!style.backgroundImage && !style.backgroundColor && (

   441→   441→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   442→   442→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />

   444→   444→      )}

   447→   447→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl">
   448→   448→          {content.headline}
   449→   449→        </h1>
