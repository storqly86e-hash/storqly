   628→function EditorToolbar({
   629→  onToggleLeft,
   630→  onToggleRight,
   631→  showLeft,
   632→  showRight,
   633→}: {
   634→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   635→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   636→  showLeft: boolean
   637→  showRight: boolean
   638→}) {
   639→  const store = useStoreEditor((s) => s.store)
   640→  const reset = useStoreEditor((s) => s.reset)
   641→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   642→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   643→  const isPublished = useStoreEditor((s) => s.isPublished)
   644→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   645→  const [isSaving, setIsSaving] = useState(false)
   646→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
   647→  const [copied, setCopied] = useState(false)
   648→
   649→  const handleBack = () => {
   650→    reset()
   651→  }
   652→
   653→  const handleSave = async () => {
   654→    if (!store || isSaving) return
   655→    setIsSaving(true)
   656→    try {
   657→      const res = await fetch('/api/store/save', {
   658→        method: 'POST',
   659→        headers: { 'Content-Type': 'application/json' },
   660→        body: JSON.stringify({ store }),
   661→      })
   662→      if (!res.ok) throw new Error('Save failed')
   663→      toast.success('Store saved')
   664→    } catch {
   665→      toast.error('Failed to save store')
   666→    } finally {
   667→      setIsSaving(false)
   668→    }
   669→  }
   670→
   671→  const handlePublish = async () => {
   672→    if (!store || isPublishing) return
   673→    setIsPublishing(true)
   674→    try {
   675→      const res = await fetch('/api/store/publish', {
   676→        method: 'POST',
   677→        headers: { 'Content-Type': 'application/json' },
   678→        body: JSON.stringify({ store }),
   679→      })
   680→      if (!res.ok) throw new Error('Publish failed')
   681→      const data = await res.json()
   682→      const slug = data.slug
   683→      setIsPublished(true)
   684→      // Build a real viewable URL using the current origin
   685→      const baseUrl = window.location.origin
   686→      const viewUrl = `${baseUrl}/?store=${slug}`
   687→      setPublishedUrl(viewUrl)
   688→      toast.success('Store published successfully!')
   689→    } catch {
   690→      toast.error('Failed to publish store')
   691→    } finally {
   692→      setIsPublishing(false)
   693→    }
   694→  }
   695→
   696→  const handleCopyUrl = useCallback(async () => {
   697→    if (!publishedUrl) return
   698→    try {
   699→      await navigator.clipboard.writeText(publishedUrl)
   700→      setCopied(true)
   701→      toast.success('URL copied to clipboard!')
   702→      setTimeout(() => setCopied(false), 2000)
   703→    } catch {
   704→      // Fallback for older browsers
   705→      const textarea = document.createElement('textarea')
   706→      textarea.value = publishedUrl
   707→      document.body.appendChild(textarea)
   708→      textarea.select()
   709→      document.execCommand('copy')
   710→      document.body.removeChild(textarea)
   711→      setCopied(true)
   712→      toast.success('URL copied to clipboard!')
   713→      setTimeout(() => setCopied(false), 2000)
   714→    }
   715→  }, [publishedUrl])
   716→
   717→  return (
   718→    <>
   719→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
   720→        <div className="flex items-center gap-1">
   721→          <Button
   722→            variant="ghost"
   723→            size="icon"
   724→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   725→            onClick={handleBack}
   726→            aria-label="Back to home"
   727→          >
   728→            <ArrowLeft className="h-4 w-4" />
   729→          </Button>
   730→          <div className="h-5 w-px bg-zinc-800" />
   731→          <Button
   732→            variant={showLeft ? 'secondary' : 'ghost'}
   733→            size="sm"
   734→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   735→            onClick={() => onToggleLeft((v) => !v)}
   736→          >
   737→            <Layers className="h-3.5 w-3.5" />
   738→            <span className="hidden sm:inline">Sections</span>
   739→          </Button>
   740→          <Button
   741→            variant={showRight ? 'secondary' : 'ghost'}
   742→            size="sm"
   743→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   744→            onClick={() => onToggleRight((v) => !v)}
   745→          >
   746→            <MessageSquare className="h-3.5 w-3.5" />
   747→            <span className="hidden sm:inline">Chat</span>
   748→          </Button>
   749→        </div>
   750→
   751→        <div className="flex items-center gap-1.5">
   752→          {publishedUrl && (
   753→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
   754→              <Check className="h-3 w-3 text-emerald-400" />
   755→              <span className="max-w-[180px] truncate text-xs font-medium text-emerald-300">
   756→                {publishedUrl}
   757→              </span>
   758→              <button
   759→                onClick={handleCopyUrl}
   760→                className="ml-1 rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
   761→                aria-label="Copy URL"
   762→              >
   763→                {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
   764→              </button>
   765→              <a
   766→                href={publishedUrl}
   767→                target="_blank"
   768→                rel="noopener noreferrer"
   769→                className="rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
   770→                aria-label="Open published store"
   771→              >
   772→                <ExternalLink className="h-3 w-3" />
   773→              </a>
   774→            </div>
   775→          )}
   776→          <span className="mr-2 text-xs font-medium text-zinc-500 hidden md:inline truncate max-w-[200px]">
   777→            {store?.name || 'Untitled Store'}
   778→          </span>
   779→          <Button
   780→            variant="ghost"
   781→            size="sm"
   782→            className="h-8 gap-1.5 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   783→            onClick={handleSave}
   784→            disabled={isSaving}
   785→          >
   786→            <Save className={`h-3.5 w-3.5 ${isSaving ? 'animate-pulse' : ''}`} />
   787→            <span className="hidden sm:inline">{isSaving ? 'Saving...' : 'Save'}</span>
   788→          </Button>
   789→          <Button
   790→            size="sm"
   791→            className="h-8 gap-1.5 bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] text-white shadow-none hover:opacity-90"
   792→            onClick={handlePublish}
   793→            disabled={isPublishing}
   794→          >
   795→            {isPublishing ? (
   796→              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   797→            ) : isPublished ? (
   798→              <Eye className="h-3.5 w-3.5" />
   799→            ) : (
   800→              <Globe className="h-3.5 w-3.5" />
   801→            )}
   802→            <span className="hidden sm:inline">{isPublishing ? 'Publishing...' : isPublished ? 'Published' : 'Publish'}</span>
   803→          </Button>
   804→        </div>
   805→      </div>
   806→
   807→      {/* Publish Success Dialog */}
   808→      <AnimatePresence>
   809→        {publishedUrl && !isPublishing && (
   810→          <motion.div
   811→            initial={{ opacity: 0 }}
   812→            animate={{ opacity: 1 }}

   883→// ─── Published Store Viewer (read-only, shown via ?store=slug) ────────
   884→
   885→function PublishedStoreViewer({ slug }: { slug: string }) {
   886→  const [store, setStore] = useState<Store | null>(null)
   887→  const [loading, setLoading] = useState(true)
   888→  const [error, setError] = useState<string | null>(null)
   889→
   890→  useEffect(() => {
   891→    let cancelled = false
   892→    async function load() {
   893→      try {
   894→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
   895→        if (!res.ok) {
   896→          const data = await res.json().catch(() => ({}))
   897→          throw new Error(data.error || `Store not found (${res.status})`)
   898→        }
   899→        const data = await res.json()
   900→        if (!cancelled) setStore(data.store)
   901→      } catch (err: unknown) {
   902→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
   903→      } finally {
   904→        if (!cancelled) setLoading(false)
   905→      }
   906→    }
   907→    load()
   908→    return () => { cancelled = true }
   909→  }, [slug])
   910→
   911→  if (loading) {
   912→    return (
   913→      <div className="flex min-h-screen items-center justify-center bg-white">
   914→        <div className="text-center">
   915→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
   916→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
   917→        </div>
   918→      </div>
   919→    )
   920→  }
   921→
   922→  if (error || !store) {
   923→    return (
   924→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
   925→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
   926→          <AlertCircle className="h-8 w-8 text-red-400" />
   927→        </div>
   928→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
   929→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
   930→          {error || 'This store could not be found or has not been published.'}
   931→        </p>
   932→        <Button
   933→          variant="outline"
   934→          onClick={() => window.location.href = '/'}
   935→          className="gap-2"
   936→        >
   937→          <ArrowLeft className="h-4 w-4" />
   938→          Back to Storqly
   939→        </Button>
   940→      </div>
   941→    )
   942→  }
   943→
   944→  return (
   945→    <>
   946→      <StoreRenderer store={store} />
   947→      {/* Minimal footer for published stores */}
   948→      <div className="fixed bottom-3 right-3 z-40">
   949→        <a
   950→          href="/"
   951→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
   952→        >
   953→          <Sparkles className="h-3 w-3" />
   954→          Built with Storqly
   955→        </a>
   956→      </div>
   957→    </>
   958→  )
   959→}
