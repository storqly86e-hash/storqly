    86→function LandingPage() {
    87→  const [promptText, setPromptText] = useState('')
    88→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    89→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    90→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
    91→  const abortRef = useRef<AbortController | null>(null)
    92→
    93→  const {
    94→    isGenerating,
    95→    setIsGenerating,
    96→    setStore,
    97→    setStoreWithFallback,
    98→  } = useStoreEditor()
    99→
   100→  // Local UI state (not in Zustand — this is view-level)
   101→  const [generationStatus, setGenerationStatus] = useState('')
   102→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   103→  const [error, setError] = useState<string | null>(null)
   104→
   105→  const clearTimers = useCallback(() => {
   106→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   107→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }
   108→  }, [])
   109→
   110→  const handleCancel = useCallback(() => {
   111→    abortRef.current?.abort()
   112→    clearTimers()
   113→    setIsGenerating(false)
   114→    setGenerationStatus('')
   115→    setElapsedSeconds(0)
   116→    setError(null)
   117→  }, [clearTimers, setIsGenerating])
   118→
   119→  const handleGenerate = useCallback(async () => {
   120→    const trimmed = promptText.trim()
   121→    if (!trimmed) {
   122→      setError('Please describe your store to get started.')
   123→      textareaRef.current?.focus()
   124→      return
   125→    }
   126→
   127→    // Reset state
   128→    setError(null)
   129→    setIsGenerating(true)
   130→    setGenerationStatus(progressMessages[0])
   131→    setElapsedSeconds(0)
   132→
   133→    // Create abort controller for this request
   134→    const controller = new AbortController()
   135→    abortRef.current = controller
   136→
   137→    // Cycle through progress messages (looping every ~17.5s)
   138→    let idx = 0
   139→    progressTimerRef.current = setInterval(() => {
   140→      idx = (idx + 1) % progressMessages.length
   141→      setGenerationStatus(progressMessages[idx])
   142→    }, 2500)
   143→
   144→    // Elapsed time counter
   145→    elapsedTimerRef.current = setInterval(() => {
   146→      setElapsedSeconds((prev) => prev + 1)
   147→    }, 1000)
   148→
   149→    try {
   150→      console.log('[Storqly] Starting store generation for prompt:', trimmed)
   151→
   152→      const res = await fetch('/api/store/generate', {
   153→        method: 'POST',
   154→        headers: { 'Content-Type': 'application/json' },
   155→        body: JSON.stringify({ prompt: trimmed }),
   156→        signal: controller.signal,
   157→      })
   158→
   159→      if (!res.ok) {
   160→        let errorMsg = `Server error (${res.status})`
   161→        try {
   162→          const errorData = await res.json()
   163→          errorMsg = errorData.error || errorMsg
   164→        } catch { /* ignore parse error */ }
   165→        throw new Error(errorMsg)
   166→      }
   167→
   168→      const data = await res.json()
   169→
   170→      if (!data.store) {
   171→        throw new Error('The AI response was missing store data. Please try again.')
   172→      }
   173→
   174→      console.log('[Storqly] Store generated. isFallback:', data._isFallback, 'name:', data.store.name)
   175→      clearTimers()
   176→
   177→      if (data._isFallback) {
   178→        setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   179→      } else {
   180→        setStore(data.store)
   181→      }
   182→    } catch (err: unknown) {
   183→      clearTimers()
   184→
   185→      let message = 'Something went wrong. Please try again.'
   186→      if (err instanceof Error) {
   187→        if (err.name === 'AbortError') {
   188→          // AbortError is expected when user cancels — don't log as error
   189→          message = 'Generation was cancelled.'
   190→        } else {
   191→          console.error('[Storqly] Generation error:', err)
   192→          message = err.message
   193→        }
   194→      }
   195→
   196→      setError(message)
   197→      setIsGenerating(false)
   198→      setGenerationStatus('')
   199→      setElapsedSeconds(0)
   200→      toast.error('Store generation failed', { description: message })
   201→    }
   202→  }, [promptText, setIsGenerating, setStore, setStoreWithFallback, clearTimers])
   203→
   204→  // Cleanup on unmount
   205→  useEffect(() => {
   206→    return () => {
   207→      clearTimers()
   208→      abortRef.current?.abort()
   209→    }
   210→  }, [clearTimers])
   211→
   212→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   213→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   214→      e.preventDefault()
   215→      handleGenerate()
   216→    }
   217→  }
   218→
   219→  const formatElapsed = (seconds: number) => {
   220→    if (seconds < 60) return `${seconds}s`
   221→    const m = Math.floor(seconds / 60)
   222→    const s = seconds % 60
   223→    return `${m}:${s.toString().padStart(2, '0')}`
   224→  }
   225→
   226→  return (
   227→    <>
   228→      {/* ── Fixed Nav ── */}
   229→      <header className="fixed top-0 left-0 right-0 z-50 border-b border-white/[0.06] bg-[#09090b]/80 backdrop-blur-xl">
   230→        <nav className="mx-auto flex h-16 max-w-6xl items-center justify-between px-5 sm:px-8">
   231→          <span className="text-2xl font-bold tracking-tight bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   232→            Storqly
   233→          </span>
   234→          <span className="hidden text-sm text-zinc-500 sm:block">
   235→            AI-Powered Store Builder
   236→          </span>
   237→        </nav>
   238→      </header>
   239→
   240→      {/* ── Hero Section ── */}
   241→      <section className="relative flex flex-1 flex-col items-center justify-center px-5 pt-32 pb-20 sm:pt-40 sm:pb-28">
   242→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   243→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />
   244→        </div>
   245→
   246→        <div className="relative z-10 mx-auto w-full max-w-3xl text-center">
   247→          <motion.h1
   248→            custom={0}
   249→            variants={fadeUp}
   250→            initial="hidden"
   251→            animate="visible"
   252→            className="text-4xl font-extrabold leading-[1.1] tracking-tight sm:text-5xl md:text-6xl lg:text-7xl"
   253→          >
   254→            Build Your Store{' '}
   255→            <span className="bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   256→              in Seconds
   257→            </span>
   258→          </motion.h1>
   259→
   260→          <motion.p
   261→            custom={1}
   262→            variants={fadeUp}
   263→            initial="hidden"
   264→            animate="visible"
   265→            className="mt-6 text-lg text-zinc-400 sm:text-xl"
   266→          >
   267→            Describe your store. AI builds it. You customize and publish.
   268→          </motion.p>
   269→
   270→          <motion.div
   271→            custom={2}
   272→            variants={fadeUp}
   273→            initial="hidden"
   274→            animate="visible"
   275→            className="mt-10"
   276→          >
   277→            <div className="group relative rounded-2xl border border-white/[0.08] bg-white/[0.03] shadow-2xl shadow-black/40 transition-all duration-300 focus-within:border-[#a855f7]/40 focus-within:shadow-[0_0_40px_-8px_rgba(168,85,247,0.25)]">
   278→              <Textarea
   279→                ref={textareaRef}
   280→                value={promptText}
   281→                onChange={(e) => { setPromptText(e.target.value); setError(null) }}
   282→                onKeyDown={handleKeyDown}
   283→                disabled={isGenerating}
   284→                placeholder='Describe the store you want to build... (e.g. "A modern minimalist jewelry brand with gold accents, selling handcrafted rings, necklaces, and bracelets")'
   285→                className="min-h-[120px] resize-none rounded-2xl border-0 bg-transparent px-5 py-5 text-base text-white placeholder:text-zinc-600 focus-visible:ring-0 focus-visible:border-0 sm:text-lg md:min-h-[140px]"
   286→              />
   287→
   288→              <div className="flex items-center justify-between border-t border-white/[0.06] px-5 py-3">
   289→                <span className="text-xs text-zinc-600">
   290→                  Press{' '}
   291→                  <kbd className="rounded border border-zinc-700 bg-zinc-800 px-1.5 py-0.5 font-mono text-[10px] text-zinc-400">
   292→                    ⌘↵
   293→                  </kbd>{' '}
   294→                  to generate
   295→                </span>
   296→
   297→                <div className="flex items-center gap-2">
   298→                  {isGenerating && (
   299→                    <Button
   300→                      variant="ghost"
   301→                      size="sm"
   302→                      onClick={handleCancel}
   303→                      className="h-8 gap-1.5 rounded-lg px-3 text-xs text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   304→                    >
   305→                      <X className="h-3.5 w-3.5" />
   306→                      Cancel
   307→                    </Button>
   308→                  )}
   309→                  <Button
   310→                    onClick={handleGenerate}
   311→                    disabled={isGenerating || !promptText.trim()}
   312→                    className="relative overflow-hidden rounded-xl bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] px-6 py-2.5 text-sm font-semibold text-white shadow-lg shadow-[#a855f7]/20 transition-all duration-300 hover:scale-105 hover:shadow-[#a855f7]/30 disabled:opacity-50 disabled:hover:scale-100"
   313→                  >
   314→                    {isGenerating ? (
   315→                      <span className="flex items-center gap-2">
   316→                        <Loader2 className="h-4 w-4 animate-spin" />
   317→                        Generating…
   318→                      </span>
   319→                    ) : (
   320→                      <span className="flex items-center gap-2">
   321→                        <Sparkles className="h-4 w-4" />
   322→                        Generate Store
   323→                      </span>
   324→                    )}
   325→                  </Button>
   326→                </div>
   327→              </div>
   328→            </div>
   329→
   330→            {/* Status feedback area */}
   331→            <div className="mt-4 min-h-[48px]">
   332→              <AnimatePresence mode="wait">
   333→                {isGenerating && !error && (
   334→                  <motion.div
   335→                    key="progress"
   336→                    initial={{ opacity: 0, y: 8 }}
   337→                    animate={{ opacity: 1, y: 0 }}
   338→                    exit={{ opacity: 0, y: -8 }}
   339→                    className="flex flex-col items-center gap-1"

   351→                {error && (

   403→              variants={fadeUp}
   404→              className="group relative rounded-2xl border border-white/[0.06] bg-white/[0.02] p-6 transition-all duration-300 hover:border-white/[0.12] hover:bg-white/[0.04] sm:p-8"
   405→            >
   406→              <div className="mb-4 flex h-11 w-11 items-center justify-center rounded-xl bg-gradient-to-br from-[#a855f7]/20 to-[#ec4899]/20">

   409→              <h3 className="text-base font-semibold text-white sm:text-lg">

   411→              </h3>

   440→// ─── Editor View ─────────────────────────────────────────────────────

   442→function EditorView() {
   443→  const store = useStoreEditor((s) => s.store)
   444→  const selectedSectionId = useStoreEditor((s) => s.selectedSectionId)
   445→  const setSelectedSectionId = useStoreEditor((s) => s.setSelectedSectionId)
   446→  const isFallbackStore = useStoreEditor((s) => s.isFallbackStore)
   447→  const [showLeft, setShowLeft] = useState(true)
   448→  const [showRight, setShowRight] = useState(true)
   449→
   450→  if (!store) {
   451→    return (
   452→      <div className="flex flex-1 items-center justify-center bg-zinc-950">
   453→        <div className="text-center">
   454→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
   455→          <p className="mt-3 text-sm text-zinc-500">Loading editor…</p>
   456→        </div>
   457→      </div>
   458→    )
   459→  }
   460→
   461→  return (
   462→    <div className="flex h-screen flex-col bg-zinc-950">
   463→      <EditorToolbar onToggleLeft={setShowLeft} onToggleRight={setShowRight} showLeft={showLeft} showRight={showRight} />
   464→
   465→      {isFallbackStore && <FallbackBanner />}
   466→
   467→      <div className="flex-1 overflow-hidden">
   468→        <PanelGroup direction="horizontal" autoSaveId="storqly-editor-layout">
   469→          <AnimatePresence mode="wait">
   470→            {showLeft && (
   471→              <Panel id="left" order={1} defaultSize={18} minSize={12} maxSize={28}>
   472→                <VisualEditor />
   473→              </Panel>
   474→            )}
   475→          </AnimatePresence>
   476→
   477→          {showLeft && showRight && <ResizeHandle direction="left" />}
   478→
   479→          <Panel id="center" order={2} defaultSize={showLeft && showRight ? 48 : showLeft || showRight ? 72 : 100} minSize={30}>
   480→            <div className="h-full overflow-auto bg-zinc-100">
   481→              <StoreRenderer
   482→                store={store}
   483→                selectedSectionId={selectedSectionId}
   484→                onSelectSection={setSelectedSectionId}
   485→              />
   486→            </div>
   487→          </Panel>
   488→
   489→          {showLeft && showRight && <ResizeHandle direction="right" />}
   490→
   491→          <AnimatePresence mode="wait">
   492→            {showRight && (
   493→              <Panel id="right" order={3} defaultSize={22} minSize={16} maxSize={32}>
   494→                <ChatPanel />
   495→              </Panel>
   496→            )}
   497→          </AnimatePresence>
   498→        </PanelGroup>
   499→      </div>
   500→    </div>
   501→  )
   502→}

   506→function FallbackBanner() {

   509→

   511→
   512→  return (
   513→    <div className="flex items-center gap-3 border-b border-amber-500/30 bg-amber-500/10 px-4 py-2.5">
   514→      <AlertTriangle className="h-4 w-4 shrink-0 text-amber-400" />

   516→        AI couldn't generate a custom store — you're viewing a starter template.
   517→        <span className="hidden sm:inline"> Edit it manually or try regenerating.</span>

   519→      <div className="flex items-center gap-2">
   520→        <Button
   521→          size="sm"
   522→          className="h-7 gap-1.5 rounded-lg bg-amber-500/20 px-3 text-xs font-medium text-amber-200 hover:bg-amber-500/30"

   525→          }}
   526→        >
   527→          <RotateCcw className="h-3 w-3" />
   528→          Regenerate with AI

   533→          aria-label="Dismiss"
   534→        >
   535→          <X className="h-3.5 w-3.5" />
   536→        </button>
   537→      </div>
   538→    </div>
   539→  )

   542→// ─── Editor Toolbar ──────────────────────────────────────────────────
   543→
   544→function EditorToolbar({
   545→  onToggleLeft,
   546→  onToggleRight,
   547→  showLeft,
   548→  showRight,
   549→}: {
   550→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   551→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   552→  showLeft: boolean
   553→  showRight: boolean
   554→}) {
   555→  const store = useStoreEditor((s) => s.store)
   556→  const reset = useStoreEditor((s) => s.reset)
   557→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   558→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   559→  const isPublished = useStoreEditor((s) => s.isPublished)
   560→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   561→  const [isSaving, setIsSaving] = useState(false)
   562→
   563→  const handleBack = () => {
   564→    reset()
   565→  }
   566→
   567→  const handleSave = async () => {
   568→    if (!store || isSaving) return
   569→    setIsSaving(true)
   570→    try {
   571→      const res = await fetch('/api/store/save', {
   572→        method: 'POST',
   573→        headers: { 'Content-Type': 'application/json' },
   574→        body: JSON.stringify({ store }),
   575→      })
   576→      if (!res.ok) throw new Error('Save failed')
   577→      toast.success('Store saved')
   578→    } catch {
   579→      toast.error('Failed to save store')
   580→    } finally {
   581→      setIsSaving(false)
   582→    }
   583→  }
   584→
   585→  const handlePublish = async () => {
   586→    if (!store || isPublishing) return
   587→    setIsPublishing(true)
   588→    try {
   589→      const res = await fetch('/api/store/publish', {
   590→        method: 'POST',
   591→        headers: { 'Content-Type': 'application/json' },
   592→        body: JSON.stringify({ store }),
   593→      })
   594→      if (!res.ok) throw new Error('Publish failed')
   595→      const data = await res.json()
   596→      setIsPublished(true)
   597→      toast.success(`Published! Live at ${data.slug}.storqly.com`)
   598→    } catch {
   599→      toast.error('Failed to publish store')
   600→    } finally {
   601→      setIsPublishing(false)
   602→    }
   603→  }
   604→
   605→  return (
   606→    <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
   607→      <div className="flex items-center gap-1">
   608→        <Button
   609→          variant="ghost"
   610→          size="icon"
   611→          className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   612→          onClick={handleBack}
   613→          aria-label="Back to home"
   614→        >
   615→          <ArrowLeft className="h-4 w-4" />
   616→        </Button>
   617→        <div className="h-5 w-px bg-zinc-800" />
   618→        <Button
   619→          variant={showLeft ? 'secondary' : 'ghost'}
   620→          size="sm"
   621→          className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   622→          onClick={() => onToggleLeft((v) => !v)}
   623→        >
   624→          <Layers className="h-3.5 w-3.5" />
   625→          <span className="hidden sm:inline">Sections</span>
   626→        </Button>
   627→        <Button
   628→          variant={showRight ? 'secondary' : 'ghost'}
   629→          size="sm"
   630→          className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   631→          onClick={() => onToggleRight((v) => !v)}
   632→        >
   633→          <MessageSquare className="h-3.5 w-3.5" />
   634→          <span className="hidden sm:inline">Chat</span>
   635→        </Button>
   636→      </div>
   637→
   638→      <div className="flex items-center gap-1.5">
   639→        <span className="mr-2 text-xs font-medium text-zinc-500 hidden md:inline truncate max-w-[200px]">
   640→          {store?.name || 'Untitled Store'}
   641→        </span>
   642→        <Button
   643→          variant="ghost"
   644→          size="sm"
   645→          className="h-8 gap-1.5 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   646→          onClick={handleSave}
   647→          disabled={isSaving}
   648→        >
   649→          <Save className={`h-3.5 w-3.5 ${isSaving ? 'animate-pulse' : ''}`} />
   650→          <span className="hidden sm:inline">{isSaving ? 'Saving...' : 'Save'}</span>
   651→        </Button>
   652→        <Button
   653→          size="sm"
   654→          className="h-8 gap-1.5 bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] text-white shadow-none hover:opacity-90"
   655→          onClick={handlePublish}
   656→          disabled={isPublishing}
   657→        >
   658→          {isPublishing ? (
   659→            <Loader2 className="h-3.5 w-3.5 animate-spin" />
   660→          ) : isPublished ? (
   661→            <Eye className="h-3.5 w-3.5" />
   662→          ) : (
   663→            <Globe className="h-3.5 w-3.5" />
   664→          )}
   665→          <span className="hidden sm:inline">{isPublishing ? 'Publishing...' : isPublished ? 'Published' : 'Publish'}</span>
   666→        </Button>
   667→      </div>
   668→    </div>
   669→  )
   670→}
