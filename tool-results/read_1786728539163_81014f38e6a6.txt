    95→function LandingPage() {

    99→  const [authOpen, setAuthOpen] = useState(false)
   100→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   101→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   102→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   103→  const abortRef = useRef<AbortController | null>(null)
   104→
   105→  const {
   106→    isGenerating,
   107→    setIsGenerating,
   108→    setStore,
   109→    setStoreWithFallback,
   110→  } = useStoreEditor()
   111→
   112→  // Local UI state (not in Zustand — this is view-level)
   113→  const [generationStatus, setGenerationStatus] = useState('')
   114→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   115→  const [error, setError] = useState<string | null>(null)
   116→
   117→  const clearTimers = useCallback(() => {
   118→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   119→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }

   121→
   122→  const handleCancel = useCallback(() => {
   123→    abortRef.current?.abort()
   124→    clearTimers()

   126→    setGenerationStatus('')
   127→    setElapsedSeconds(0)
   128→    setError(null)
   129→  }, [clearTimers, setIsGenerating])
   130→
   131→  const handleGenerate = useCallback(async () => {
   132→    const trimmed = promptText.trim()
   133→    if (!trimmed) {
   134→      setError('Please describe your store to get started.')
   135→      textareaRef.current?.focus()

   138→
   139→    // Auth gate — block generation for logged-out users (show modal instead)

   145→    // Reset state

   155→    // Elapsed time counter

   158→    }, 1000)
   159→
   160→    // Fallback progress cycling (used only if SSE progress events stop)
   161→    let fallbackIdx = 0
   162→    let receivedSSEProgress = false
   163→    progressTimerRef.current = setInterval(() => {
   164→      if (!receivedSSEProgress) {
   165→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   166→        setGenerationStatus(progressMessages[fallbackIdx])
   167→      }
   168→    }, 3000)
   169→
   170→    try {
   171→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   172→
   173→      const res = await fetch('/api/store/generate', {
   174→        method: 'POST',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({ prompt: trimmed }),
   177→        signal: controller.signal,
   178→      })
   179→
   180→      if (!res.ok) {
   181→        let errorMsg = `Server error (${res.status})`
   182→        try {
   183→          const errorData = await res.json()
   184→          errorMsg = errorData.error || errorMsg
   185→        } catch { /* ignore parse error */ }
   186→        throw new Error(errorMsg)
   187→      }
   188→
   189→      // ── Consume SSE stream ──
   190→      const reader = res.body?.getReader()
   191→      if (!reader) throw new Error('No response stream received.')
   192→
   193→      const decoder = new TextDecoder()
   194→      let buffer = ''
   195→      let resolved = false
   196→
   197→      while (true) {
   198→        const { done, value } = await reader.read()
   199→        if (done) break
   200→
   201→        buffer += decoder.decode(value, { stream: true })
   202→
   203→        // Parse SSE events from buffer
   204→        // SSE format: "event: <name>\ndata: <json>\n\n" or ": comment\n\n"
   205→        const lines = buffer.split('\n')
   206→        // Keep incomplete last line in buffer
   207→        buffer = lines.pop() || ''
   208→
   209→        let currentEvent = ''
   210→        for (const line of lines) {
   211→          if (line.startsWith(':')) {
   212→            // SSE comment (heartbeat) — skip
   213→            continue
   214→          }
   215→          if (line.startsWith('event: ')) {
   216→            currentEvent = line.slice(7).trim()
   217→          } else if (line.startsWith('data: ')) {
   218→            const dataStr = line.slice(6)
   219→
   220→            // Handle SSE events
   221→            if (currentEvent === 'progress') {
   222→              receivedSSEProgress = true
   223→              try {
   224→                const data = JSON.parse(dataStr)
   225→                setGenerationStatus(data.message || 'Processing...')
   226→              } catch { /* ignore */ }
   227→            } else if (currentEvent === 'result') {
   228→              try {
   229→                const data = JSON.parse(dataStr)
   230→                if (!data.store) {
   231→                  throw new Error('The AI response was missing store data.')
   232→                }
   233→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   234→                resolved = true
   235→                clearTimers()
   236→
   237→                if (data._isFallback) {
   238→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   239→                } else {
   240→                  setStore(data.store)
   241→                }
   242→              } catch (parseErr) {
   243→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   244→                console.error('[Storqly] Failed to parse result event:', msg)
   245→                throw new Error(msg)
   246→              }
   247→            } else if (currentEvent === 'error') {
   248→              try {
   249→                const data = JSON.parse(dataStr)
   250→                throw new Error(data.message || 'Generation failed')
   251→              } catch (e) {
   252→                if (e instanceof Error && e.message !== 'Generation failed') throw e
   253→                throw new Error('Generation failed')
   254→              }
   255→            }
   256→            // Reset event after processing its data
   257→            currentEvent = ''
   258→          }
   259→        }
   260→      }
   261→
   262→      if (!resolved) {
   263→        throw new Error('Stream ended without a result. The server may have disconnected.')
   264→      }
   265→    } catch (err: unknown) {
   266→      clearTimers()
   267→
   268→      let message = 'Something went wrong. Please try again.'
   269→      if (err instanceof Error) {
   270→        if (err.name === 'AbortError') {
   271→          message = 'Generation was cancelled.'
   272→        } else {
   273→          console.error('[Storqly] Generation error:', err)
   274→          message = err.message
   275→        }
   276→      }
   277→
   278→      setError(message)
   279→      setIsGenerating(false)
   280→      setGenerationStatus('')
   281→      setElapsedSeconds(0)
   282→      toast.error('Store generation failed', { description: message })
   283→    }
   284→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])
   285→
   286→  // Cleanup on unmount

   288→    return () => {
   289→      clearTimers()
   290→      abortRef.current?.abort()
   291→    }
   292→  }, [clearTimers])
   293→
   294→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {

   306→  }

   344→            Build Your Store{' '}

   346→              in Seconds

   670→function EditorToolbar({
   671→  onToggleLeft,
   672→  onToggleRight,
   673→  showLeft,
   674→  showRight,
   675→}: {
   676→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   677→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   678→  showLeft: boolean
   679→  showRight: boolean
   680→}) {
   681→  const store = useStoreEditor((s) => s.store)
   682→  const reset = useStoreEditor((s) => s.reset)
   683→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   684→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   685→  const isPublished = useStoreEditor((s) => s.isPublished)
   686→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   687→  const [isSaving, setIsSaving] = useState(false)
   688→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
   689→  const [copied, setCopied] = useState(false)
   690→  const [authOpen, setAuthOpen] = useState(false)
   691→
   692→  const handleBack = () => {
   693→    reset()
   694→  }
   695→
   696→  const handleSave = async () => {
   697→    if (!store || isSaving) return
   698→    setIsSaving(true)
   699→    try {
   700→      const res = await fetch('/api/store/save', {
   701→        method: 'POST',
   702→        headers: { 'Content-Type': 'application/json' },
   703→        body: JSON.stringify({ store }),
   704→      })
   705→      if (!res.ok) throw new Error('Save failed')
   706→      toast.success('Store saved')
   707→    } catch {
   708→      toast.error('Failed to save store')
   709→    } finally {
   710→      setIsSaving(false)
   711→    }
   712→  }
   713→
   714→  const handlePublish = async () => {
   715→    if (!store || isPublishing) return
   716→    setIsPublishing(true)
   717→    try {
   718→      const res = await fetch('/api/store/publish', {
   719→        method: 'POST',
   720→        headers: { 'Content-Type': 'application/json' },
   721→        body: JSON.stringify({ store }),
   722→      })
   723→      if (!res.ok) throw new Error('Publish failed')
   724→      const data = await res.json()
   725→      const slug = data.slug
   726→      setIsPublished(true)
   727→      // Build a real viewable URL using the current origin
   728→      const baseUrl = window.location.origin
   729→      const viewUrl = `${baseUrl}/?store=${slug}`
   730→      setPublishedUrl(viewUrl)
   731→      toast.success('Store published successfully!')
   732→    } catch {
   733→      toast.error('Failed to publish store')
   734→    } finally {
   735→      setIsPublishing(false)
   736→    }
   737→  }
   738→
   739→  const handleCopyUrl = useCallback(async () => {
   740→    if (!publishedUrl) return
   741→    try {
   742→      await navigator.clipboard.writeText(publishedUrl)
   743→      setCopied(true)
   744→      toast.success('URL copied to clipboard!')
   745→      setTimeout(() => setCopied(false), 2000)
   746→    } catch {
   747→      // Fallback for older browsers
   748→      const textarea = document.createElement('textarea')
   749→      textarea.value = publishedUrl
   750→      document.body.appendChild(textarea)
   751→      textarea.select()
   752→      document.execCommand('copy')
   753→      document.body.removeChild(textarea)
   754→      setCopied(true)
   755→      toast.success('URL copied to clipboard!')
   756→      setTimeout(() => setCopied(false), 2000)
   757→    }
   758→  }, [publishedUrl])
   759→
   760→  return (
   761→    <>
   762→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
   763→        <div className="flex items-center gap-1">
   764→          <Button
   765→            variant="ghost"
   766→            size="icon"
   767→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   768→            onClick={handleBack}
   769→            aria-label="Back to home"
   770→          >
   771→            <ArrowLeft className="h-4 w-4" />
   772→          </Button>
   773→          <div className="h-5 w-px bg-zinc-800" />
   774→          <Button
   775→            variant={showLeft ? 'secondary' : 'ghost'}
   776→            size="sm"
   777→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   778→            onClick={() => onToggleLeft((v) => !v)}
   779→          >
   780→            <Layers className="h-3.5 w-3.5" />
   781→            <span className="hidden sm:inline">Sections</span>
   782→          </Button>
   783→          <Button
   784→            variant={showRight ? 'secondary' : 'ghost'}
   785→            size="sm"
   786→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   787→            onClick={() => onToggleRight((v) => !v)}
   788→          >
   789→            <MessageSquare className="h-3.5 w-3.5" />
   790→            <span className="hidden sm:inline">Chat</span>
   791→          </Button>
   792→        </div>
   793→
   794→        <div className="flex items-center gap-1.5">
   795→          {publishedUrl && (
   796→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
   797→              <Check className="h-3 w-3 text-emerald-400" />
   798→              <span className="max-w-[180px] truncate text-xs font-medium text-emerald-300">
   799→                {publishedUrl}
   800→              </span>
   801→              <button
   802→                onClick={handleCopyUrl}
   803→                className="ml-1 rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
   804→                aria-label="Copy URL"
   805→              >
   806→                {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
   807→              </button>
   808→              <a
   809→                href={publishedUrl}
   810→                target="_blank"
   811→                rel="noopener noreferrer"
   812→                className="rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
   813→                aria-label="Open published store"
   814→              >
   815→                <ExternalLink className="h-3 w-3" />
   816→              </a>
   817→            </div>
   818→          )}
   819→          <AuthButton onSignIn={() => setAuthOpen(true)} />
   820→          <div className="h-5 w-px bg-zinc-800" />
   821→          <span className="mr-2 text-xs font-medium text-zinc-500 hidden md:inline truncate max-w-[200px]">
   822→            {store?.name || 'Untitled Store'}
   823→          </span>
   824→          <Button
   825→            variant="ghost"
   826→            size="sm"
   827→            className="h-8 gap-1.5 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   828→            onClick={handleSave}
   829→            disabled={isSaving}
   830→          >
   831→            <Save className={`h-3.5 w-3.5 ${isSaving ? 'animate-pulse' : ''}`} />
   832→            <span className="hidden sm:inline">{isSaving ? 'Saving...' : 'Save'}</span>
   833→          </Button>
   834→          <Button
   835→            size="sm"
   836→            className="h-8 gap-1.5 bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] text-white shadow-none hover:opacity-90"
   837→            onClick={handlePublish}
   838→            disabled={isPublishing}
   839→          >
   840→            {isPublishing ? (
   841→              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   842→            ) : isPublished ? (
   843→              <Eye className="h-3.5 w-3.5" />
   844→            ) : (
   845→              <Globe className="h-3.5 w-3.5" />

   847→            <span className="hidden sm:inline">{isPublishing ? 'Publishing...' : isPublished ? 'Published' : 'Publish'}</span>
   848→          </Button>
   849→        </div>
   850→      </div>
   851→
   852→      {/* Publish Success Dialog */}
   853→      <AnimatePresence>
   854→        {publishedUrl && !isPublishing && (
   855→          <motion.div
   856→            initial={{ opacity: 0 }}
   857→            animate={{ opacity: 1 }}
   858→            exit={{ opacity: 0 }}
   859→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 backdrop-blur-sm"
   860→            onClick={() => setPublishedUrl(null)}
   861→          >
   862→            <motion.div
   863→              initial={{ scale: 0.95, opacity: 0 }}
   864→              animate={{ scale: 1, opacity: 1 }}
   865→              exit={{ scale: 0.95, opacity: 0 }}
   866→              transition={{ duration: 0.2 }}
   867→              className="mx-4 w-full max-w-md rounded-2xl border border-zinc-800 bg-zinc-900 p-6 shadow-2xl"
   868→              onClick={(e) => e.stopPropagation()}
   869→            >
   870→              {/* Success icon */}
   871→              <div className="mx-auto mb-4 flex h-14 w-14 items-center justify-center rounded-full bg-emerald-500/15">
   872→                <Check className="h-7 w-7 text-emerald-400" />
   873→              </div>
   874→
   875→              <h3 className="mb-1 text-center text-lg font-semibold text-zinc-100">

   929→// ─── Published Store Viewer (read-only, shown via ?store=slug) ────────

   931→function PublishedStoreViewer({ slug }: { slug: string }) {
   932→  const [store, setStore] = useState<Store | null>(null)
   933→  const [loading, setLoading] = useState(true)
   934→  const [error, setError] = useState<string | null>(null)
   935→
   936→  useEffect(() => {
   937→    let cancelled = false
   938→    async function load() {
   939→      try {
   940→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
   941→        if (!res.ok) {
   942→          const data = await res.json().catch(() => ({}))
   943→          throw new Error(data.error || `Store not found (${res.status})`)
   944→        }
   945→        const data = await res.json()
   946→        if (!cancelled) setStore(data.store)
   947→      } catch (err: unknown) {
   948→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
   949→      } finally {
   950→        if (!cancelled) setLoading(false)
   951→      }
   952→    }
   953→    load()
   954→    return () => { cancelled = true }
   955→  }, [slug])
   956→
   957→  if (loading) {
   958→    return (
   959→      <div className="flex min-h-screen items-center justify-center bg-white">
   960→        <div className="text-center">
   961→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
   962→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
   963→        </div>
   964→      </div>
   965→    )
   966→  }
   967→
   968→  if (error || !store) {
   969→    return (
   970→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
   971→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
   972→          <AlertCircle className="h-8 w-8 text-red-400" />
   973→        </div>
   974→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
   975→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
   976→          {error || 'This store could not be found or has not been published.'}
   977→        </p>
   978→        <Button
   979→          variant="outline"
   980→          onClick={() => window.location.href = '/'}
   981→          className="gap-2"
   982→        >
   983→          <ArrowLeft className="h-4 w-4" />
   984→          Back to Storqly
   985→        </Button>
   986→      </div>
   987→    )
   988→  }
   989→
   990→  return (
   991→    <>
   992→      <StoreRenderer store={store} />
   993→      {/* Minimal footer for published stores */}
   994→      <div className="fixed bottom-3 right-3 z-40">
   995→        <a
   996→          href="/"
   997→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
   998→        >
   999→          <Sparkles className="h-3 w-3" />
  1000→          Built with Storqly
  1001→        </a>
  1002→      </div>
  1003→    </>
  1004→  )
  1005→}
