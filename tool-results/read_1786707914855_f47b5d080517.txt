    93→    93→function LandingPage() {
    94→    94→  const [promptText, setPromptText] = useState('')
    95→    95→  const [mkOpen, setMkOpen] = useState(false)
    96→    96→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    97→    97→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    98→    98→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
    99→    99→  const abortRef = useRef<AbortController | null>(null)
   100→   100→
   101→   101→  const {
   102→   102→    isGenerating,
   103→   103→    setIsGenerating,
   104→   104→    setStore,
   105→   105→    setStoreWithFallback,
   106→   106→  } = useStoreEditor()
   107→   107→
   108→   108→  // Local UI state (not in Zustand — this is view-level)
   109→   109→  const [generationStatus, setGenerationStatus] = useState('')
   110→   110→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   111→   111→  const [error, setError] = useState<string | null>(null)
   112→   112→
   113→   113→  const clearTimers = useCallback(() => {
   114→   114→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   115→   115→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }
   116→   116→  }, [])
   117→   117→
   118→   118→  const handleCancel = useCallback(() => {
   119→   119→    abortRef.current?.abort()
   120→   120→    clearTimers()
   121→   121→    setIsGenerating(false)
   122→   122→    setGenerationStatus('')
   123→   123→    setElapsedSeconds(0)
   124→   124→    setError(null)
   125→   125→  }, [clearTimers, setIsGenerating])
   126→   126→
   127→   127→  const handleGenerate = useCallback(async () => {
   128→   128→    const trimmed = promptText.trim()
   129→   129→    if (!trimmed) {
   130→   130→      setError('Please describe your store to get started.')
   131→   131→      textareaRef.current?.focus()
   132→   132→      return
   133→   133→    }
   134→   134→
   135→   135→    // Reset state
   136→   136→    setError(null)
   137→   137→    setIsGenerating(true)
   138→   138→    setGenerationStatus('Starting generation...')
   139→   139→    setElapsedSeconds(0)
   140→   140→
   141→   141→    // Create abort controller for this request
   142→   142→    const controller = new AbortController()
   143→   143→    abortRef.current = controller
   144→   144→
   145→   145→    // Elapsed time counter
   146→   146→    elapsedTimerRef.current = setInterval(() => {
   147→   147→      setElapsedSeconds((prev) => prev + 1)
   148→   148→    }, 1000)
   149→   149→
   150→   150→    // Fallback progress cycling (used only if SSE progress events stop)
   151→   151→    let fallbackIdx = 0
   152→   152→    let receivedSSEProgress = false
   153→   153→    progressTimerRef.current = setInterval(() => {
   154→   154→      if (!receivedSSEProgress) {
   155→   155→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   156→   156→        setGenerationStatus(progressMessages[fallbackIdx])
   157→   157→      }
   158→   158→    }, 3000)
   159→   159→
   160→   160→    try {
   161→   161→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   162→   162→
   163→   163→      const res = await fetch('/api/store/generate', {
   164→   164→        method: 'POST',
   165→   165→        headers: { 'Content-Type': 'application/json' },
   166→   166→        body: JSON.stringify({ prompt: trimmed }),
   167→   167→        signal: controller.signal,
   168→   168→      })
   169→   169→
   170→   170→      if (!res.ok) {
   171→   171→        let errorMsg = `Server error (${res.status})`
   172→   172→        try {
   173→   173→          const errorData = await res.json()
   174→   174→          errorMsg = errorData.error || errorMsg
   175→   175→        } catch { /* ignore parse error */ }
   176→   176→        throw new Error(errorMsg)
   177→   177→      }
   178→   178→
   179→   179→      // ── Consume SSE stream ──
   180→   180→      const reader = res.body?.getReader()
   181→   181→      if (!reader) throw new Error('No response stream received.')
   182→   182→
   183→   183→      const decoder = new TextDecoder()
   184→   184→      let buffer = ''
   185→   185→      let resolved = false
   186→   186→
   187→   187→      while (true) {
   188→   188→        const { done, value } = await reader.read()
   189→   189→        if (done) break
   190→   190→
   191→   191→        buffer += decoder.decode(value, { stream: true })
   192→   192→
   193→   193→        // Parse SSE events from buffer
   194→   194→        // SSE format: "event: <name>\ndata: <json>\n\n" or ": comment\n\n"
   195→   195→        const lines = buffer.split('\n')
   196→   196→        // Keep incomplete last line in buffer
   197→   197→        buffer = lines.pop() || ''
   198→   198→
   199→   199→        let currentEvent = ''
   200→   200→        for (const line of lines) {
   201→   201→          if (line.startsWith(':')) {
   202→   202→            // SSE comment (heartbeat) — skip
   203→   203→            continue
   204→   204→          }
   205→   205→          if (line.startsWith('event: ')) {
   206→   206→            currentEvent = line.slice(7).trim()
   207→   207→          } else if (line.startsWith('data: ')) {
   208→   208→            const dataStr = line.slice(6)
   209→   209→
   210→   210→            // Handle SSE events
   211→   211→            if (currentEvent === 'progress') {
   212→   212→              receivedSSEProgress = true
   213→   213→              try {
   214→   214→                const data = JSON.parse(dataStr)
   215→   215→                setGenerationStatus(data.message || 'Processing...')
   216→   216→              } catch { /* ignore */ }
   217→   217→            } else if (currentEvent === 'result') {
   218→   218→              try {
   219→   219→                const data = JSON.parse(dataStr)
   220→   220→                if (!data.store) {
   221→   221→                  throw new Error('The AI response was missing store data.')
   222→   222→                }
   223→   223→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   224→   224→                resolved = true
   225→   225→                clearTimers()
   226→   226→
   227→   227→                if (data._isFallback) {
   228→   228→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   229→   229→                } else {
   230→   230→                  setStore(data.store)
   231→   231→                }
   232→   232→              } catch (parseErr) {
   233→   233→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   234→   234→                console.error('[Storqly] Failed to parse result event:', msg)
   235→   235→                throw new Error(msg)
   236→   236→              }
   237→   237→            } else if (currentEvent === 'error') {
   238→   238→              try {
   239→   239→                const data = JSON.parse(dataStr)
   240→   240→                throw new Error(data.message || 'Generation failed')
   241→   241→              } catch (e) {
   242→   242→                if (e instanceof Error && e.message !== 'Generation failed') throw e
   243→   243→                throw new Error('Generation failed')
   244→   244→              }
   245→   245→            }
   246→   246→            // Reset event after processing its data
   247→   247→            currentEvent = ''
   248→   248→          }
   249→   249→        }
   250→   250→      }
   251→   251→
   252→   252→      if (!resolved) {
   253→   253→        throw new Error('Stream ended without a result. The server may have disconnected.')
   254→   254→      }
   255→   255→    } catch (err: unknown) {
   256→   256→      clearTimers()
   257→   257→
   258→   258→      let message = 'Something went wrong. Please try again.'
   259→   259→      if (err instanceof Error) {
   260→   260→        if (err.name === 'AbortError') {
   261→   261→          message = 'Generation was cancelled.'
   262→   262→        } else {
   263→   263→          console.error('[Storqly] Generation error:', err)
   264→   264→          message = err.message
   265→   265→        }
   266→   266→      }
   267→   267→
   268→   268→      setError(message)
   269→   269→      setIsGenerating(false)
   270→   270→      setGenerationStatus('')
   271→   271→      setElapsedSeconds(0)
   272→   272→      toast.error('Store generation failed', { description: message })
   273→   273→    }
   274→   274→  }, [promptText, setIsGenerating, setStore, setStoreWithFallback, clearTimers])
   275→   275→
   276→   276→  // Cleanup on unmount
   277→   277→  useEffect(() => {
   278→   278→    return () => {
   279→   279→      clearTimers()
   280→   280→      abortRef.current?.abort()
   281→   281→    }
   282→   282→  }, [clearTimers])
   283→   283→
   284→   284→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   285→   285→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   286→   286→      e.preventDefault()
   287→   287→      handleGenerate()
   288→   288→    }
   289→   289→  }
   290→   290→
   291→   291→  const formatElapsed = (seconds: number) => {
   292→   292→    if (seconds < 60) return `${seconds}s`
   293→   293→    const m = Math.floor(seconds / 60)
   294→   294→    const s = seconds % 60
   295→   295→    return `${m}:${s.toString().padStart(2, '0')}`
   296→   296→  }

   302→   302→        <nav className="mx-auto flex h-16 max-w-6xl items-center justify-between px-5 sm:px-8">
   303→   303→          <span className="text-2xl font-bold tracking-tight bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   304→   304→            Storqly
   305→   305→          </span>
   306→   306→          <span className="hidden text-sm text-zinc-500 sm:block">
   307→   307→            AI-Powered Store Builder
   308→   308→          </span>
   309→   309→        </nav>
   310→   310→      </header>
   311→   311→
   312→   312→      {/* ── Hero Section ── */}
   313→   313→      <section className="relative flex flex-1 flex-col items-center justify-center px-5 pt-32 pb-20 sm:pt-40 sm:pb-28">
   314→   314→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   315→   315→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />
   316→   316→        </div>
   317→   317→
   318→   318→        <div className="relative z-10 mx-auto w-full max-w-3xl text-center">
   319→   319→          <motion.h1
   320→   320→            custom={0}
   321→   321→            variants={fadeUp}
   322→   322→            initial="hidden"
   323→   323→            animate="visible"
   324→   324→            className="text-4xl font-extrabold leading-[1.1] tracking-tight sm:text-5xl md:text-6xl lg:text-7xl"
   325→   325→          >
   326→   326→            Build Your Store{' '}
   327→   327→            <span className="bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   328→   328→              in Seconds
   329→   329→            </span>
   330→   330→          </motion.h1>
   331→   331→
   332→   332→          <motion.p
   333→   333→            custom={1}
   334→   334→            variants={fadeUp}
   335→   335→            initial="hidden"
   336→   336→            animate="visible"
   337→   337→            className="mt-6 text-lg text-zinc-400 sm:text-xl"
   338→   338→          >
   339→   339→            Describe your store. AI builds it. You customize and publish.
   340→   340→          </motion.p>
   341→   341→
   342→   342→          <motion.div
   343→   343→            custom={2}
   344→   344→            variants={fadeUp}
   345→   345→            initial="hidden"
   346→   346→            animate="visible"
   347→   347→            className="mt-10"
   348→   348→          >
   349→   349→            <div className="group relative rounded-2xl border border-white/[0.08] bg-white/[0.03] shadow-2xl shadow-black/40 transition-all duration-300 focus-within:border-[#a855f7]/40 focus-within:shadow-[0_0_40px_-8px_rgba(168,85,247,0.25)]">
   350→   350→              <Textarea
   351→   351→                ref={textareaRef}

   353→   353→                onChange={(e) => { setPromptText(e.target.value); setError(null) }}

   355→   355→                disabled={isGenerating}

   357→   357→                className="min-h-[120px] resize-none rounded-2xl border-0 bg-transparent px-5 py-5 text-base text-white placeholder:text-zinc-600 focus-visible:ring-0 focus-visible:border-0 sm:text-lg md:min-h-[140px]"
   358→   358→              />
   359→   359→
   360→   360→              <div className="flex items-center justify-between border-t border-white/[0.06] px-5 py-3">
   361→   361→                <span className="text-xs text-zinc-600">
   362→   362→                  Press{' '}
   363→   363→                  <kbd className="rounded border border-zinc-700 bg-zinc-800 px-1.5 py-0.5 font-mono text-[10px] text-zinc-400">
   364→   364→                    ⌘↵

   418→   418→                      Elapsed: {formatElapsed(elapsedSeconds)} · AI is building your store, this may take up to 2 minutes

   420→   420→                  </motion.div>

   422→   422→
   423→   423→                {error && (

   652→   652→function EditorToolbar({
   653→   653→  onToggleLeft,
   654→   654→  onToggleRight,
   655→   655→  showLeft,
   656→   656→  showRight,
   657→   657→}: {
   658→   658→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   659→   659→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   660→   660→  showLeft: boolean
   661→   661→  showRight: boolean
   662→   662→}) {
   663→   663→  const store = useStoreEditor((s) => s.store)
   664→   664→  const reset = useStoreEditor((s) => s.reset)
   665→   665→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   666→   666→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   667→   667→  const isPublished = useStoreEditor((s) => s.isPublished)
   668→   668→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   669→   669→  const [isSaving, setIsSaving] = useState(false)
   670→   670→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
   671→   671→  const [copied, setCopied] = useState(false)
   672→   672→
   673→   673→  const handleBack = () => {
   674→   674→    reset()
   675→   675→  }
   676→   676→
   677→   677→  const handleSave = async () => {
   678→   678→    if (!store || isSaving) return
   679→   679→    setIsSaving(true)
   680→   680→    try {
   681→   681→      const res = await fetch('/api/store/save', {
   682→   682→        method: 'POST',
   683→   683→        headers: { 'Content-Type': 'application/json' },
   684→   684→        body: JSON.stringify({ store }),
   685→   685→      })
   686→   686→      if (!res.ok) throw new Error('Save failed')
   687→   687→      toast.success('Store saved')
   688→   688→    } catch {
   689→   689→      toast.error('Failed to save store')
   690→   690→    } finally {
   691→   691→      setIsSaving(false)
   692→   692→    }
   693→   693→  }
   694→   694→
   695→   695→  const handlePublish = async () => {
   696→   696→    if (!store || isPublishing) return
   697→   697→    setIsPublishing(true)
   698→   698→    try {
   699→   699→      const res = await fetch('/api/store/publish', {
   700→   700→        method: 'POST',
   701→   701→        headers: { 'Content-Type': 'application/json' },
   702→   702→        body: JSON.stringify({ store }),
   703→   703→      })
   704→   704→      if (!res.ok) throw new Error('Publish failed')
   705→   705→      const data = await res.json()
   706→   706→      const slug = data.slug
   707→   707→      setIsPublished(true)
   708→   708→      // Build a real viewable URL using the current origin
   709→   709→      const baseUrl = window.location.origin
   710→   710→      const viewUrl = `${baseUrl}/?store=${slug}`
   711→   711→      setPublishedUrl(viewUrl)
   712→   712→      toast.success('Store published successfully!')
   713→   713→    } catch {
   714→   714→      toast.error('Failed to publish store')
   715→   715→    } finally {
   716→   716→      setIsPublishing(false)
   717→   717→    }
   718→   718→  }
   719→   719→
   720→   720→  const handleCopyUrl = useCallback(async () => {
   721→   721→    if (!publishedUrl) return
   722→   722→    try {
   723→   723→      await navigator.clipboard.writeText(publishedUrl)
   724→   724→      setCopied(true)
   725→   725→      toast.success('URL copied to clipboard!')
   726→   726→      setTimeout(() => setCopied(false), 2000)
   727→   727→    } catch {
   728→   728→      // Fallback for older browsers
   729→   729→      const textarea = document.createElement('textarea')
   730→   730→      textarea.value = publishedUrl
   731→   731→      document.body.appendChild(textarea)
   732→   732→      textarea.select()
   733→   733→      document.execCommand('copy')
   734→   734→      document.body.removeChild(textarea)
   735→   735→      setCopied(true)
   736→   736→      toast.success('URL copied to clipboard!')
   737→   737→      setTimeout(() => setCopied(false), 2000)
   738→   738→    }
   739→   739→  }, [publishedUrl])
   740→   740→
   741→   741→  return (
   742→   742→    <>
   743→   743→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
   744→   744→        <div className="flex items-center gap-1">
   745→   745→          <Button
   746→   746→            variant="ghost"
   747→   747→            size="icon"
   748→   748→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   749→   749→            onClick={handleBack}
   750→   750→            aria-label="Back to home"
   751→   751→          >
   752→   752→            <ArrowLeft className="h-4 w-4" />
   753→   753→          </Button>
   754→   754→          <div className="h-5 w-px bg-zinc-800" />
   755→   755→          <Button
   756→   756→            variant={showLeft ? 'secondary' : 'ghost'}
   757→   757→            size="sm"
   758→   758→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   759→   759→            onClick={() => onToggleLeft((v) => !v)}
   760→   760→          >
   761→   761→            <Layers className="h-3.5 w-3.5" />
   762→   762→            <span className="hidden sm:inline">Sections</span>
   763→   763→          </Button>
   764→   764→          <Button
   765→   765→            variant={showRight ? 'secondary' : 'ghost'}
   766→   766→            size="sm"
   767→   767→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   768→   768→            onClick={() => onToggleRight((v) => !v)}
   769→   769→          >
   770→   770→            <MessageSquare className="h-3.5 w-3.5" />
   771→   771→            <span className="hidden sm:inline">Chat</span>
   772→   772→          </Button>
   773→   773→        </div>
   774→   774→
   775→   775→        <div className="flex items-center gap-1.5">
   776→   776→          {publishedUrl && (
   777→   777→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
   778→   778→              <Check className="h-3 w-3 text-emerald-400" />
   779→   779→              <span className="max-w-[180px] truncate text-xs font-medium text-emerald-300">
   780→   780→                {publishedUrl}
   781→   781→              </span>
   782→   782→              <button
   783→   783→                onClick={handleCopyUrl}
   784→   784→                className="ml-1 rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"

   909→   909→function PublishedStoreViewer({ slug }: { slug: string }) {
   910→   910→  const [store, setStore] = useState<Store | null>(null)
   911→   911→  const [loading, setLoading] = useState(true)
   912→   912→  const [error, setError] = useState<string | null>(null)
   913→   913→
   914→   914→  useEffect(() => {
   915→   915→    let cancelled = false
   916→   916→    async function load() {
   917→   917→      try {
   918→   918→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
   919→   919→        if (!res.ok) {
   920→   920→          const data = await res.json().catch(() => ({}))
   921→   921→          throw new Error(data.error || `Store not found (${res.status})`)
   922→   922→        }
   923→   923→        const data = await res.json()
   924→   924→        if (!cancelled) setStore(data.store)
   925→   925→      } catch (err: unknown) {
   926→   926→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
   927→   927→      } finally {
   928→   928→        if (!cancelled) setLoading(false)
   929→   929→      }
   930→   930→    }
   931→   931→    load()
   932→   932→    return () => { cancelled = true }
   933→   933→  }, [slug])
