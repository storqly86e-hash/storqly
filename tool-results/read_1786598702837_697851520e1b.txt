    50→    50→
    51→    51→interface SectionWrapperProps {
    52→    52→  section: Section;
    53→    53→  theme: StoreTheme;
    54→    54→  selectedSectionId?: string | null;
    55→    55→  onSelectSection?: (sectionId: string | null) => void;
    56→    56→  children: React.ReactNode;
    57→    57→}
    58→    58→
    59→    59→function SectionWrapper({
    60→    60→  section,
    61→    61→  theme,
    62→    62→  selectedSectionId,
    63→    63→  onSelectSection,
    64→    64→  children,
    65→    65→}: SectionWrapperProps) {
    66→    66→  const isSelected = selectedSectionId === section.id;
    67→    67→  const style = section.style;
    68→    68→
    69→    69→  const sectionBg = style.backgroundColor || theme.colors.background;
    70→    70→  const sectionText = style.textColor || theme.colors.text;
    71→    71→
    72→    72→  const bgImage = style.backgroundImage
    73→    73→    ? `url(${style.backgroundImage})`
    74→    74→    : undefined;
    75→    75→
    76→    76→  const outerClasses = [
    77→    77→    'relative transition-all duration-200 cursor-pointer',
    78→    78→    pyClass(style.paddingY),
    79→    79→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
    80→    80→  ].join(' ');
    81→    81→
    82→    82→  return (
    83→    83→    <div
    84→    84→      className={outerClasses}
    85→    85→      style={{
    86→    86→        backgroundColor: sectionBg,
    87→    87→        color: sectionText,
    88→    88→        backgroundImage: bgImage,
    89→    89→        backgroundSize: 'cover',
    90→    90→        backgroundPosition: 'center',
    91→    91→      }}
    92→    92→      onClick={(e) => {
    93→    93→        e.stopPropagation();
    94→    94→        onSelectSection?.(isSelected ? null : section.id);
    95→    95→      }}
    96→    96→      data-section-id={section.id}
    97→    97→      data-section-type={section.type}
    98→    98→    >
    99→    99→      {/* Overlay for background images */}
   100→   100→      {style.backgroundImage && style.overlay && (
   101→   101→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   102→   102→      )}
   103→   103→      <div
   104→   104→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   105→   105→      >
   106→   106→        {/* Selected section label */}
   107→   107→        {isSelected && (
   108→   108→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   109→   109→            {section.type.replace(/-/g, ' ')}
   110→   110→          </div>
   111→   111→        )}
   112→   112→        {children}
   113→   113→      </div>
   114→   114→    </div>
   115→   115→  );
   116→   116→}

   120→   120→function ProductCard({
   121→   121→  product,
   122→   122→  theme,
   123→   123→  showAddToCart,
   124→   124→  borderRadius,
   125→   125→  buttonBgOverride,
   126→   126→  buttonTextOverride,
   127→   127→  onViewProduct,
   128→   128→}: {
   129→   129→  product: StoreProduct;
   130→   130→  theme: StoreTheme;
   131→   131→  showAddToCart: boolean;
   132→   132→  borderRadius: string;
   133→   133→  buttonBgOverride?: string;
   134→   134→  buttonTextOverride?: string;
   135→   135→  onViewProduct?: (productId: string) => void;
   136→   136→}) {
   137→   137→  const [hovered, setHovered] = useState(false);
   138→   138→  const addToCart = useCartStore((s) => s.addItem);
   139→   139→  const imgColor = stringToColor(product.id, theme);
   140→   140→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   141→   141→
   142→   142→  return (
   143→   143→    <div
   144→   144→      className={`${borderRadius} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   145→   145→      style={{
   146→   146→        borderColor: theme.colors.border,
   147→   147→      }}
   148→   148→      onMouseEnter={() => setHovered(true)}
   149→   149→      onMouseLeave={() => setHovered(false)}
   150→   150→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   151→   151→      role={onViewProduct ? 'button' : undefined}
   152→   152→      tabIndex={onViewProduct ? 0 : undefined}
   153→   153→    >
   154→   154→ {/* Image placeholder */}
   155→   155→      <div
   156→   156→        className="relative aspect-square w-full overflow-hidden"
   157→   157→        style={{ backgroundColor: imgColor }}
   158→   158→      >
   159→   159→        <div className="absolute inset-0 flex items-center justify-center">
   160→   160→          <div
   161→   161→            className="h-16 w-16 rounded-full opacity-30"
   162→   162→            style={{ backgroundColor: theme.colors.primary }}
   163→   163→          />
   164→   164→        </div>
   165→   165→        {/* Out of stock overlay */}
   166→   166→        {!product.inStock && (
   167→   167→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   168→   168→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   169→   169→              Sold Out
   170→   170→            </span>
   171→   171→          </div>
   172→   172→        )}
   173→   173→        {/* Discount badge */}
   174→   174→        {hasDiscount && product.compareAtPrice && (
   175→   175→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   176→   176→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   177→   177→          </div>
   178→   178→        )}
   179→   179→      </div>
   180→   180→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   181→   181→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   182→   182→        {product.category && (
   183→   183→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   184→   184→            {product.category}
   185→   185→          </p>
   186→   186→        )}
   187→   187→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   188→   188→          {product.name}
   189→   189→        </h3>
   190→   190→        <div className="mt-1.5 flex items-center gap-2">
   191→   191→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   192→   192→            {formatPrice(product.price)}
   193→   193→          </span>
   194→   194→          {hasDiscount && product.compareAtPrice && (
   195→   195→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   196→   196→              {formatPrice(product.compareAtPrice)}
   197→   197→            </span>
   198→   198→          )}
   199→   199→        </div>
   200→   200→        {showAddToCart && product.inStock && (
   201→   201→          <button
   202→   202→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 ${
   203→   203→              hovered
   204→   204→                ? 'opacity-100 translate-y-0'
   205→   205→                : 'opacity-0 translate-y-1'
   206→   206→            }`}
   207→   207→            style={{
   208→   208→              backgroundColor: buttonBgOverride || theme.colors.primary,
   209→   209→              color: buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary),
   210→   210→            }}
   211→   211→            onClick={(e) => {
   212→   212→              e.stopPropagation();
   213→   213→              addToCart(product);
   214→   214→            }}
   215→   215→          >
   216→   216→            Add to Cart
   217→   217→          </button>
   218→   218→        )}
   219→   219→      </div>
   220→   220→    </div>
   221→   221→  );
   222→   222→}

   319→   319→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   320→   320→  const content = section.content as unknown as HeroContent;
   321→   321→  const style = section.style;
   322→   322→  const isSelected = selectedSectionId === section.id;
   323→   323→
   324→   324→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   325→   325→  const alignMap = {
   326→   326→    left: 'items-start text-left',
   327→   327→    center: 'items-center text-center',
   328→   328→    right: 'items-end text-right',
   329→   329→  };
   330→   330→
   331→   331→  // Only apply theme gradient as default background when no custom color or image is set.
   332→   332→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   333→   333→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   334→   334→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   335→   335→    : undefined;
   336→   336→
   337→   337→  const outerClasses = [
   338→   338→    'relative flex transition-all duration-200 cursor-pointer',
   339→   339→    heightMap[content.height] || heightMap.md,
   340→   340→    alignMap[content.alignment],
   341→   341→    pyClass(style.paddingY),
   342→   342→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   343→   343→  ].join(' ');
   344→   344→
   345→   345→  return (
   346→   346→    <div
   347→   347→      className={outerClasses}
   348→   348→      style={{
   349→   349→        backgroundColor: style.backgroundColor || theme.colors.primary,
   350→   350→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   351→   351→        backgroundSize: 'cover',
   352→   352→        backgroundPosition: 'center',
   353→   353→        color: style.textColor || '#ffffff',
   354→   354→      }}
   355→   355→      onClick={(e) => {
   356→   356→        e.stopPropagation();
   357→   357→        onSelectSection?.(isSelected ? null : section.id);
   358→   358→      }}
   359→   359→      data-section-id={section.id}
   360→   360→      data-section-type={section.type}
   361→   361→    >
   362→   362→      {/* Overlay */}
   363→   363→      {style.backgroundImage && style.overlay && (
   364→   364→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   365→   365→      )}
   366→   366→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   367→   367→      {!style.backgroundImage && !style.backgroundColor && (
   368→   368→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   369→   369→      )}
   370→   370→      {!style.backgroundImage && !style.backgroundColor && (
   371→   371→        <>
   372→   372→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   373→   373→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />
   374→   374→        </>
   375→   375→      )}
   376→   376→
   377→   377→      <div className={`relative z-10 mx-auto w-full ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}>
   378→   378→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   379→   379→          {content.headline}
   380→   380→        </h1>
   381→   381→        {content.subheadline && (
   382→   382→          <p className="mt-4 max-w-xl text-base opacity-80 sm:text-lg md:text-xl">
   383→   383→            {content.subheadline}
   384→   384→          </p>
   385→   385→        )}
   386→   386→        {content.ctaText && (
   387→   387→          <button className="mt-8 inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-transform hover:scale-105 sm:text-base"
   388→   388→            style={{
   389→   389→              backgroundColor: section.style.buttonBackgroundColor || '#ffffff',
   390→   390→              color: section.style.buttonTextColor || contrastTextColor(section.style.buttonBackgroundColor || '#ffffff'),
   391→   391→            }}>
   392→   392→            {content.ctaText}
   393→   393→            <ArrowRight className="h-4 w-4" />
   394→   394→          </button>
   395→   395→        )}
   396→   396→      </div>
   397→   397→    </div>
   398→   398→  );
   399→   399→}

   403→   403→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct }: SectionRendererProps) {
   404→   404→  const content = section.content as unknown as FeaturedProductsContent;
   405→   405→  const borderRadius = borderRadiusClass(theme.borderRadius);
   406→   406→  const featuredProducts = useMemo(
   407→   407→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   408→   408→    [content.productIds, products],
   409→   409→  );
   410→   410→
   411→   411→  return (
   412→   412→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   413→   413→      {content.headline && (
   414→   414→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   415→   415→          {content.headline}
   416→   416→        </h2>
   417→   417→      )}
   418→   418→      {content.subtitle && (
   419→   419→        <p className="mb-8 text-sm opacity-65">
   420→   420→          {content.subtitle}
   421→   421→        </p>
   422→   422→      )}
   423→   423→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   424→   424→        {featuredProducts.map((product) => (
   425→   425→          <ProductCard
   426→   426→            key={product.id}
   427→   427→            product={product}
   428→   428→            theme={theme}
   429→   429→            showAddToCart={content.showAddToCart}
   430→   430→            borderRadius={borderRadius}
   431→   431→            buttonBgOverride={section.style.buttonBackgroundColor}
   432→   432→            buttonTextOverride={section.style.buttonTextColor}
   433→   433→            onViewProduct={onViewProduct}
   434→   434→          />
   435→   435→        ))}
   436→   436→      </div>
   437→   437→    </SectionWrapper>
   438→   438→  );
   439→   439→}

   443→   443→export function ProductGridSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct }: SectionRendererProps) {
   444→   444→  const content = section.content as unknown as ProductGridContent;
   445→   445→  const borderRadius = borderRadiusClass(theme.borderRadius);
   446→   446→
   447→   447→  const filteredProducts = useMemo(() => {
   448→   448→    let prods = products;
   449→   449→    if (content.filterByCategory) {
   450→   450→      prods = prods.filter((p) => p.category === content.filterByCategory);
   451→   451→    }
   452→   452→    return prods;
   453→   453→  }, [products, content.filterByCategory]);
   454→   454→
   455→   455→  return (
   456→   456→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   457→   457→      {content.headline && (
   458→   458→        <h2 className="mb-8 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   459→   459→          {content.headline}
   460→   460→        </h2>
   461→   461→      )}
   462→   462→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   463→   463→        {filteredProducts.map((product) => (
   464→   464→          <ProductCard
   465→   465→            key={product.id}
   466→   466→            product={product}
   467→   467→            theme={theme}
   468→   468→            showAddToCart={content.showAddToCart}
   469→   469→            borderRadius={borderRadius}

   928→   928→      className={`relative transition-all duration-200 ${heightMap[content.height] || heightMap.md} ${

   930→   930→          ? 'ring-2 ring-[#a855f7] ring-offset-2'

   932→   932→      }`}
   933→   933→      onClick={(e) => {
   934→   934→        e.stopPropagation();
   935→   935→        onSelectSection?.(isSelected ? null : section.id);

   937→   937→      data-section-id={section.id}
   938→   938→      data-section-type={section.type}
   939→   939→    >
   940→   940→      {/* Dashed guide when selected */}
   941→   941→      {isSelected && (
   942→   942→        <div className="absolute inset-x-0 top-1/2 -translate-y-1/2 border-t-2 border-dashed border-[#a855f7]/50" />
   943→   943→      )}
   944→   944→    </div>
   945→   945→  );
   946→   946→}
