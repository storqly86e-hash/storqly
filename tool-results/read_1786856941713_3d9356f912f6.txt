   109→function LandingPage() {
   110→  const { data: session } = useSession()
   111→  const [promptText, setPromptText] = useState('')
   112→  const [mkOpen, setMkOpen] = useState(false)
   113→  const [authOpen, setAuthOpen] = useState(false)
   114→  const [myStores, setMyStores] = useState<StoreListItem[]>([])
   115→  const [storesLoading, setStoresLoading] = useState(false)
   116→  const [editingSlug, setEditingSlug] = useState<string | null>(null)
   117→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   118→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   119→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   120→  const abortRef = useRef<AbortController | null>(null)
   121→
   122→  const {
   123→    isGenerating,
   124→    setIsGenerating,
   125→    setStore,
   126→    setStoreWithFallback,
   127→  } = useStoreEditor()
   128→
   129→  // Local UI state (not in Zustand — this is view-level)
   130→  const [generationStatus, setGenerationStatus] = useState('')
   131→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   132→  const [error, setError] = useState<string | null>(null)
   133→
   134→  const clearTimers = useCallback(() => {
   135→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   136→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }
   137→  }, [])
   138→
   139→  // ── Fetch user's stores on login ──
   140→  useEffect(() => {
   141→    if (!session?.user?.id) {
   142→      setMyStores([])
   143→      return
   144→    }
   145→    let cancelled = false
   146→    setStoresLoading(true)
   147→    fetch('/api/store/list')
   148→      .then((r) => { if (!r.ok) throw new Error(); return r.json() })
   149→      .then((data) => { if (!cancelled) setMyStores(data.stores ?? []) })
   150→      .catch(() => { if (!cancelled) setMyStores([]) })
   151→      .finally(() => { if (!cancelled) setStoresLoading(false) })
   152→    return () => { cancelled = true }
   153→  }, [session?.user?.id])
   154→
   155→  // ── Edit store: fetch full data via lookup, then load into editor ──
   156→  const handleEditStore = useCallback(async (slug: string) => {
   157→    setEditingSlug(slug)
   158→    try {
   159→      const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
   160→      if (!res.ok) throw new Error()
   161→      const data = await res.json()
   162→      setStore(data.store)
   163→    } catch {
   164→      toast.error('Failed to load store. It may have been deleted.')
   165→    } finally {
   166→      setEditingSlug(null)
   167→    }
   168→  }, [setStore])
   169→
   170→  // Format relative time
   171→  const formatTimeAgo = useCallback((iso: string) => {
   172→    const diff = Date.now() - new Date(iso).getTime()
   173→    const mins = Math.floor(diff / 60000)
   174→    if (mins < 1) return 'just now'
   175→    if (mins < 60) return `${mins}m ago`
   176→    const hours = Math.floor(mins / 60)
   177→    if (hours < 24) return `${hours}h ago`
   178→    const days = Math.floor(hours / 24)
   179→    if (days < 30) return `${days}d ago`
   180→    return new Date(iso).toLocaleDateString()
   181→  }, [])
   182→
   183→  const handleCancel = useCallback(() => {
   184→    abortRef.current?.abort()
   185→    clearTimers()
   186→    setIsGenerating(false)
   187→    setGenerationStatus('')
   188→    setElapsedSeconds(0)
   189→    setError(null)
   190→  }, [clearTimers, setIsGenerating])
   191→
   192→  const handleGenerate = useCallback(async () => {
   193→    const trimmed = promptText.trim()
   194→    if (!trimmed) {
   195→      setError('Please describe your store to get started.')
   196→      textareaRef.current?.focus()
   197→      return
   198→    }
   199→
   200→    // Auth gate — block generation for logged-out users (show modal instead)
   201→    if (!session?.user?.id) {
   202→      setAuthOpen(true)
   203→      return
   204→    }
   205→
   206→    // Reset state
   207→    setError(null)
   208→    setIsGenerating(true)
   209→    setGenerationStatus('Starting generation...')
   210→    setElapsedSeconds(0)
   211→
   212→    // Create abort controller for this request
   213→    const controller = new AbortController()
   214→    abortRef.current = controller
   215→
   216→    // Hard timeout: 120 seconds — no request should ever take longer than this.
   217→    // If the server is hung or the proxy drops the connection, the user gets a
   218→    // clear message instead of waiting indefinitely.
   219→    const HARD_TIMEOUT_MS = 120_000
   220→    let timedOut = false
   221→    const timeoutId = setTimeout(() => {
   222→      timedOut = true
   223→      controller.abort()
   224→      console.warn(`[Storqly] Hard timeout (${HARD_TIMEOUT_MS / 1000}s) reached — aborting generation`)
   225→    }, HARD_TIMEOUT_MS)
   226→
   227→    // Elapsed time counter
   228→    elapsedTimerRef.current = setInterval(() => {
   229→      setElapsedSeconds((prev) => prev + 1)
   230→    }, 1000)
   231→
   232→    // Fallback progress cycling (used only if SSE progress events stop)
   233→    let fallbackIdx = 0
   234→    let receivedSSEProgress = false
   235→    progressTimerRef.current = setInterval(() => {
   236→      if (!receivedSSEProgress) {
   237→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   238→        setGenerationStatus(progressMessages[fallbackIdx])
   239→      }
   240→    }, 3000)
   241→
   242→    // ── Helper: clean up on any error path (no throw, no crash overlay) ──
   243→    const finishWithError = (message: string) => {
   244→      clearTimeout(timeoutId)
   245→      clearTimers()
   246→      setError(message)
   247→      setIsGenerating(false)
   248→      setGenerationStatus('')
   249→      setElapsedSeconds(0)
   250→      toast.error('Store generation failed', { description: message })
   251→    }
   252→
   253→    const finishOk = () => {
   254→      clearTimeout(timeoutId)
   255→      clearTimers()
   256→      setIsGenerating(false)
   257→      setGenerationStatus('')
   258→      setElapsedSeconds(0)
   259→    }
   260→
   261→    try {
   262→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   263→
   264→      // ── Fetch with auto-retry for transient gateway errors (502/503/504) ──
   265→      const RETRYABLE_STATUSES = [502, 503, 504]
   266→      let res: Response | undefined
   267→      let retryCount = 0
   268→      const maxRetries = 2
   269→
   270→      while (retryCount <= maxRetries) {
   271→        try {
   272→          res = await fetch('/api/store/generate', {
   273→            method: 'POST',
   274→            headers: { 'Content-Type': 'application/json' },
   275→            body: JSON.stringify({ prompt: trimmed }),
   276→            signal: controller.signal,
   277→          })
   278→
   279→          // If the status is retryable, wait and loop again
   280→          if (!res.ok && RETRYABLE_STATUSES.includes(res.status) && retryCount < maxRetries) {
   281→            retryCount++
   282→            console.warn(`[Storqly] Gateway error ${res.status}, retrying (${retryCount}/${maxRetries}) in 3s...`)
   283→            setGenerationStatus('Connection issue — retrying...')
   284→            await new Promise(r => setTimeout(r, 3000))
   285→            continue
   286→          }
   287→          break // Success or non-retryable error — exit loop
   288→        } catch (fetchErr: unknown) {
   289→          // Network-level error (DNS failure, connection refused, etc.)
   290→          if (fetchErr instanceof TypeError && retryCount < maxRetries) {
   291→            retryCount++
   292→            console.warn(`[Storqly] Network error, retrying (${retryCount}/${maxRetries}) in 3s...`)
   293→            setGenerationStatus('Connection issue — retrying...')
   294→            await new Promise(r => setTimeout(r, 3000))
   295→            continue
   296→          }
   297→          // Out of retries or not a network error — handle gracefully
   298→          const msg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr)
   299→          if (controller.signal.aborted) {
   300→            finishWithError('Generation was cancelled.')
   301→          } else {
   302→            console.warn('[Storqly] Network error after retries:', msg)
   303→            finishWithError('Could not connect to the server. Please check your connection and try again.')
   304→          }
   305→          return
   306→        }
   307→      }
   308→
   309→      if (!res!.ok) {
   310→        let errorMsg = `Server error (${res!.status})`
   311→        try {
   312→          const errorData = await res!.json()
   313→          errorMsg = errorData.error || errorMsg
   314→        } catch { /* ignore parse error */ }
   315→        console.warn('[Storqly] HTTP error:', errorMsg)
   316→        finishWithError(errorMsg)
   317→        return
   318→      }
   319→
   320→      // ── Consume SSE stream ──
   321→      const reader = res.body?.getReader()
   322→      if (!reader) {
   323→        finishWithError('No response stream received from server.')
   324→        return
   325→      }
   326→
   327→      const decoder = new TextDecoder()
   328→      let buffer = ''
   329→      let resolved = false
   330→
   331→      while (true) {
   332→        const { done, value } = await reader.read()
   333→        if (done) break
   334→
   335→        buffer += decoder.decode(value, { stream: true })
   336→
   337→        // Parse SSE events from buffer
   338→        const lines = buffer.split('\n')
   339→        buffer = lines.pop() || ''
   340→
   341→        let currentEvent = ''
   342→        for (const line of lines) {
   343→          if (line.startsWith(':')) continue
   344→          if (line.startsWith('event: ')) {
   345→            currentEvent = line.slice(7).trim()
   346→          } else if (line.startsWith('data: ')) {
   347→            const dataStr = line.slice(6)
   348→
   349→            if (currentEvent === 'progress') {
   350→              receivedSSEProgress = true
   351→              try {
   352→                const data = JSON.parse(dataStr)
   353→                setGenerationStatus(data.message || 'Processing...')
   354→              } catch { /* ignore */ }
   355→            } else if (currentEvent === 'result') {
   356→              try {
   357→                const data = JSON.parse(dataStr)
   358→                if (!data.store) {
   359→                  console.warn('[Storqly] Result event missing store data')
   360→                  finishWithError('The AI response was missing store data.')
   361→                  return
   362→                }
   363→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   364→                resolved = true
   365→
   366→                if (data._isFallback) {
   367→                  toast.warning('AI service unavailable — showing starter template. Try again in a moment.', { duration: 6000 })
   368→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   369→                } else {
   370→                  setStore(data.store)
   371→                }
   372→
   373→                // Soft cap toast
   374→                if (data._productCapHit) {
   375→                  toast.info(
   376→                    `Generated ${data._generatedCount} products — for larger catalogs, you can add more via the chat editor.`,
   377→                    { duration: 8000 }
   378→                  )
   379→                }
   380→
   381→                finishOk()
   382→                return
   383→              } catch (parseErr) {
   384→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   385→                console.warn('[Storqly] Failed to parse result event:', msg)
   386→                finishWithError(msg)
   387→                return
   388→              }
   389→            } else if (currentEvent === 'error') {
   390→              try {
   391→                const data = JSON.parse(dataStr)
   392→                const serverMsg = data.message || 'Generation failed'
   393→                console.warn('[Storqly] Server error event:', serverMsg)
   394→                finishWithError(serverMsg)
   395→                return
   396→              } catch {
   397→                finishWithError('Generation failed')
   398→                return
   399→              }
   400→            }
   401→            currentEvent = ''
   402→          }
   403→        }
   404→      }
   405→
   406→      if (!resolved) {
   407→        console.warn('[Storqly] Stream ended without a result event')
   408→        finishWithError('Stream ended without a result. The server may have disconnected.')
   409→      }
   410→    } catch (err: unknown) {
   411→      // Last-resort safety net — should rarely be reached now
   412→      clearTimeout(timeoutId)
   413→      clearTimers()
   414→      let message = 'Something went wrong. Please try again.'
   415→      if (err instanceof Error) {
   416→        if (err.name === 'AbortError') {
   417→          message = timedOut
   418→            ? 'Generation timed out after 2 minutes. The server may be overloaded — please try again.'
   419→            : 'Generation was cancelled.'
   420→          console.warn(`[Storqly] Request aborted: ${message}`)
   421→        } else {
   422→          console.warn('[Storqly] Unexpected error (caught by safety net):', err.message)
   423→          message = err.message
   424→        }
   425→      }
   426→      setError(message)
   427→      setIsGenerating(false)
   428→      setGenerationStatus('')
   429→      setElapsedSeconds(0)
   430→      toast.error('Store generation failed', { description: message })
   431→    }
   432→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])
   433→
   434→  // Cleanup on unmount
   435→  useEffect(() => {
   436→    return () => {
   437→      clearTimers()
   438→      abortRef.current?.abort()
   439→    }
   440→  }, [clearTimers])
   441→
   442→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   443→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   444→      e.preventDefault()
   445→      handleGenerate()
   446→    }
   447→  }
   448→
   449→  const formatElapsed = (seconds: number) => {
   450→    if (seconds < 60) return `${seconds}s`
   451→    const m = Math.floor(seconds / 60)
   452→    const s = seconds % 60
   453→    return `${m}:${s.toString().padStart(2, '0')}`
   454→  }
   455→
   456→  return (
   457→    <>
   458→      {/* ── Fixed Nav ── */}
   459→      <header className="fixed top-0 left-0 right-0 z-50 border-b border-white/[0.06] bg-[#09090b]/80 backdrop-blur-xl">
   460→        <nav className="mx-auto flex h-16 max-w-6xl items-center justify-between px-5 sm:px-8">
   461→          <span className="text-2xl font-bold tracking-tight bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   462→            Storqly
   463→          </span>
   464→          <div className="hidden sm:flex items-center gap-4">
   465→            <span className="text-sm text-zinc-500">
   466→              AI-Powered Store Builder
   467→            </span>
   468→            <AuthButton onSignIn={() => setAuthOpen(true)} />
   469→          </div>
   470→          <div className="flex sm:hidden">
   471→            <AuthButton onSignIn={() => setAuthOpen(true)} />
   472→          </div>
   473→        </nav>
   474→      </header>
   475→
   476→      <AuthModal open={authOpen} onOpenChange={setAuthOpen} />
   477→
   478→      {/* ── Hero Section ── */}
   479→      <section className="relative flex flex-1 flex-col items-center justify-center px-5 pt-32 pb-20 sm:pt-40 sm:pb-28">
   480→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   481→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />
   482→        </div>
   483→
   484→        <div className="relative z-10 mx-auto w-full max-w-3xl text-center">
   485→          <motion.h1
   486→            custom={0}
   487→            variants={fadeUp}
   488→            initial="hidden"
   489→            animate="visible"
   490→            className="text-4xl font-extrabold leading-[1.1] tracking-tight sm:text-5xl md:text-6xl lg:text-7xl"
   491→          >
   492→            Build Your Store{' '}
   493→            <span className="bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   494→              in Seconds
   495→            </span>
   496→          </motion.h1>
   497→
   498→          <motion.p
   499→            custom={1}
   500→            variants={fadeUp}
   501→            initial="hidden"
   502→            animate="visible"
   503→            className="mt-6 text-lg text-zinc-400 sm:text-xl"
   504→          >
   505→            Describe your store. AI builds it. You customize and publish.
   506→          </motion.p>
   507→
   508→          <motion.div
   509→            custom={2}
   510→            variants={fadeUp}
   511→            initial="hidden"
   512→            animate="visible"
   513→            className="mt-10"
   514→          >
   515→            <div className="group relative rounded-2xl border border-white/[0.08] bg-white/[0.03] shadow-2xl shadow-black/40 transition-all duration-300 focus-within:border-[#a855f7]/40 focus-within:shadow-[0_0_40px_-8px_rgba(168,85,247,0.25)]">
   516→              <Textarea
   517→                ref={textareaRef}
   518→                value={promptText}
   519→                onChange={(e) => { setPromptText(e.target.value); setError(null) }}
   520→                onKeyDown={handleKeyDown}
   521→                disabled={isGenerating}
   522→                placeholder='Describe the store you want to build... (e.g. "A modern minimalist jewelry brand with gold accents, selling handcrafted rings, necklaces, and bracelets")'
   523→                className="min-h-[120px] resize-none rounded-2xl border-0 bg-transparent px-5 py-5 text-base text-white placeholder:text-zinc-600 focus-visible:ring-0 focus-visible:border-0 sm:text-lg md:min-h-[140px]"
   524→              />
   525→
   526→              <div className="flex items-center justify-between border-t border-white/[0.06] px-5 py-3">
   527→                <span className="text-xs text-zinc-600">
   528→                  Press{' '}
   529→                  <kbd className="rounded border border-zinc-700 bg-zinc-800 px-1.5 py-0.5 font-mono text-[10px] text-zinc-400">
   530→                    ⌘↵
   531→                  </kbd>{' '}
   532→                  to generate
   533→                </span>
   534→
   535→                <div className="flex items-center gap-2">
   536→                  {isGenerating && (
   537→                    <Button
   538→                      variant="ghost"
   539→                      size="sm"
   540→                      onClick={handleCancel}
   541→                      className="h-8 gap-1.5 rounded-lg px-3 text-xs text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   542→                    >
   543→                      <X className="h-3.5 w-3.5" />
   544→                      Cancel
   545→                    </Button>
   546→                  )}
   547→                  <Button
   548→                    onClick={handleGenerate}
   549→                    disabled={isGenerating || !promptText.trim()}
   550→                    className="relative overflow-hidden rounded-xl bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] px-6 py-2.5 text-sm font-semibold text-white shadow-lg shadow-[#a855f7]/20 transition-all duration-300 hover:scale-105 hover:shadow-[#a855f7]/30 disabled:opacity-50 disabled:hover:scale-100"
   551→                  >
   552→                    {isGenerating ? (

   555→                        Generating…

   565→              </div>

   920→// ─── Editor Toolbar ──────────────────────────────────────────────────
   921→
   922→function EditorToolbar({
   923→  onToggleLeft,
   924→  onToggleRight,
   925→  showLeft,
   926→  showRight,
   927→}: {
   928→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   929→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   930→  showLeft: boolean
   931→  showRight: boolean
   932→}) {
   933→  const store = useStoreEditor((s) => s.store)
   934→  const reset = useStoreEditor((s) => s.reset)
   935→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   936→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   937→  const isPublished = useStoreEditor((s) => s.isPublished)
   938→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   939→  const [isSaving, setIsSaving] = useState(false)
   940→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
   941→  const [copied, setCopied] = useState(false)
   942→  const [authOpen, setAuthOpen] = useState(false)
   943→
   944→  const handleBack = () => {
   945→    reset()
   946→  }
   947→
   948→  const handleSave = async () => {
   949→    if (!store || isSaving) return
   950→    setIsSaving(true)
   951→    try {
   952→      const res = await fetch('/api/store/save', {
   953→        method: 'POST',
   954→        headers: { 'Content-Type': 'application/json' },
   955→        body: JSON.stringify({ store }),
   956→      })
   957→      if (!res.ok) throw new Error('Save failed')
   958→      const data = await res.json()
   959→      toast.success(`Draft saved — ID: ${data.id}`, {
   960→        description: `Slug: ${data.slug}`,
   961→        duration: 5000,
   962→      })
   963→    } catch {
   964→      toast.error('Failed to save store')
   965→    } finally {
   966→      setIsSaving(false)
   967→    }
   968→  }
   969→
   970→  const handlePublish = async () => {
   971→    if (!store || isPublishing) return
   972→    setIsPublishing(true)
   973→    try {
   974→      const res = await fetch('/api/store/publish', {
   975→        method: 'POST',
   976→        headers: { 'Content-Type': 'application/json' },
   977→        body: JSON.stringify({ store }),
   978→      })
   979→      if (!res.ok) throw new Error('Publish failed')
   980→      const data = await res.json()
   981→      const slug = data.slug
   982→      setIsPublished(true)
   983→      // Build a real viewable URL using the current origin
   984→      const baseUrl = window.location.origin
   985→      const viewUrl = `${baseUrl}/?store=${slug}`
   986→      setPublishedUrl(viewUrl)
   987→      toast.success('Store published successfully!')
   988→    } catch {
   989→      toast.error('Failed to publish store')
   990→    } finally {
   991→      setIsPublishing(false)
   992→    }
   993→  }
   994→
   995→  const handleCopyUrl = useCallback(async () => {
   996→    if (!publishedUrl) return
   997→    try {
   998→      await navigator.clipboard.writeText(publishedUrl)
   999→      setCopied(true)
  1000→      toast.success('URL copied to clipboard!')
  1001→      setTimeout(() => setCopied(false), 2000)
  1002→    } catch {
  1003→      // Fallback for older browsers
  1004→      const textarea = document.createElement('textarea')
  1005→      textarea.value = publishedUrl
  1006→      document.body.appendChild(textarea)
  1007→      textarea.select()
  1008→      document.execCommand('copy')
  1009→      document.body.removeChild(textarea)
  1010→      setCopied(true)
  1011→      toast.success('URL copied to clipboard!')
  1012→      setTimeout(() => setCopied(false), 2000)
  1013→    }
  1014→  }, [publishedUrl])
  1015→
  1016→  return (
  1017→    <>
  1018→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
  1019→        <div className="flex items-center gap-1">
  1020→          <Button
  1021→            variant="ghost"
  1022→            size="icon"
  1023→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
  1024→            onClick={handleBack}
  1025→            aria-label="Back to home"
  1026→          >
  1027→            <ArrowLeft className="h-4 w-4" />
  1028→          </Button>
  1029→          <div className="h-5 w-px bg-zinc-800" />
  1030→          <Button
  1031→            variant={showLeft ? 'secondary' : 'ghost'}
  1032→            size="sm"
  1033→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
  1034→            onClick={() => onToggleLeft((v) => !v)}
  1035→          >
  1036→            <Layers className="h-3.5 w-3.5" />
  1037→            <span className="hidden sm:inline">Sections</span>
  1038→          </Button>
  1039→          <Button
  1040→            variant={showRight ? 'secondary' : 'ghost'}
  1041→            size="sm"
  1042→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
  1043→            onClick={() => onToggleRight((v) => !v)}
  1044→          >
  1045→            <MessageSquare className="h-3.5 w-3.5" />
  1046→            <span className="hidden sm:inline">Chat</span>
  1047→          </Button>
  1048→        </div>
  1049→
  1050→        <div className="flex items-center gap-1.5">
  1051→          {publishedUrl && (
  1052→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
  1053→              <Check className="h-3 w-3 text-emerald-400" />
  1054→              <span className="max-w-[180px] truncate text-xs font-medium text-emerald-300">
  1055→                {publishedUrl}
  1056→              </span>
  1057→              <button
  1058→                onClick={handleCopyUrl}
  1059→                className="ml-1 rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
  1060→                aria-label="Copy URL"
  1061→              >
  1062→                {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
  1063→              </button>
  1064→              <a
  1065→                href={publishedUrl}
  1066→                target="_blank"
  1067→                rel="noopener noreferrer"
  1068→                className="rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
  1069→                aria-label="Open published store"
  1070→              >
  1071→                <ExternalLink className="h-3 w-3" />
  1072→              </a>
  1073→            </div>
  1074→          )}

  1185→// ─── Published Store Viewer (read-only, shown via ?store=slug) ────────

  1187→function PublishedStoreViewer({ slug }: { slug: string }) {
  1188→  const [store, setStore] = useState<Store | null>(null)
  1189→  const [loading, setLoading] = useState(true)
  1190→  const [error, setError] = useState<string | null>(null)
  1191→
  1192→  useEffect(() => {
  1193→    let cancelled = false
  1194→    async function load() {
  1195→      try {
  1196→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
  1197→        if (!res.ok) {
  1198→          const data = await res.json().catch(() => ({}))
  1199→          throw new Error(data.error || `Store not found (${res.status})`)
  1200→        }
  1201→        const data = await res.json()
  1202→        if (!cancelled) setStore(data.store)
  1203→      } catch (err: unknown) {
  1204→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
  1205→      } finally {
  1206→        if (!cancelled) setLoading(false)
  1207→      }
  1208→    }
  1209→    load()
  1210→    return () => { cancelled = true }
  1211→  }, [slug])
  1212→
  1213→  if (loading) {
  1214→    return (
  1215→      <div className="flex min-h-screen items-center justify-center bg-white">
  1216→        <div className="text-center">
  1217→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
  1218→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
  1219→        </div>
  1220→      </div>
  1221→    )
  1222→  }
  1223→
  1224→  if (error || !store) {
  1225→    return (
  1226→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
  1227→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
  1228→          <AlertCircle className="h-8 w-8 text-red-400" />
  1229→        </div>
  1230→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
  1231→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
  1232→          {error || 'This store could not be found or has not been published.'}
  1233→        </p>
  1234→        <Button
  1235→          variant="outline"
  1236→          onClick={() => window.location.href = '/'}
  1237→          className="gap-2"
  1238→        >
  1239→          <ArrowLeft className="h-4 w-4" />
  1240→          Back to Storqly
  1241→        </Button>
  1242→      </div>
  1243→    )
  1244→  }
  1245→
  1246→  return (
  1247→    <>
  1248→      <StoreRenderer store={store} />
  1249→      <Toaster />
  1250→      {/* Minimal footer for published stores */}
  1251→      <div className="fixed bottom-3 right-3 z-40">
  1252→        <a
  1253→          href="/"
  1254→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
  1255→        >
  1256→          <Sparkles className="h-3 w-3" />
  1257→          Built with Storqly
  1258→        </a>
  1259→      </div>
  1260→    </>
  1261→  )
  1262→}
