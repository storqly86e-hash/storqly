    49→    49→// ─── Section Wrapper ────────────────────────────────────────────────────
    50→    50→
    51→    51→interface SectionWrapperProps {
    52→    52→  section: Section;
    53→    53→  theme: StoreTheme;
    54→    54→  selectedSectionId?: string | null;
    55→    55→  onSelectSection?: (sectionId: string | null) => void;
    56→    56→  children: React.ReactNode;
    57→    57→}
    58→    58→
    59→    59→function SectionWrapper({
    60→    60→  section,
    61→    61→  theme,
    62→    62→  selectedSectionId,
    63→    63→  onSelectSection,
    64→    64→  children,
    65→    65→}: SectionWrapperProps) {
    66→    66→  const isSelected = selectedSectionId === section.id;
    67→    67→  const style = section.style;
    68→    68→
    69→    69→  const sectionBg = style.backgroundColor || theme.colors.background;
    70→    70→  const sectionText = style.textColor || theme.colors.text;
    71→    71→
    72→    72→  const bgImage = style.backgroundImage
    73→    73→    ? `url(${style.backgroundImage})`
    74→    74→    : undefined;
    75→    75→
    76→    76→  const outerClasses = [
    77→    77→    'relative transition-all duration-200 cursor-pointer',
    78→    78→    pyClass(style.paddingY),
    79→    79→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
    80→    80→  ].join(' ');
    81→    81→
    82→    82→  return (
    83→    83→    <div
    84→    84→      className={outerClasses}
    85→    85→      style={{
    86→    86→        backgroundColor: sectionBg,
    87→    87→        color: sectionText,
    88→    88→        backgroundImage: bgImage,
    89→    89→        backgroundSize: 'cover',
    90→    90→        backgroundPosition: 'center',
    91→    91→      }}
    92→    92→      onClick={(e) => {
    93→    93→        e.stopPropagation();
    94→    94→        onSelectSection?.(isSelected ? null : section.id);
    95→    95→      }}
    96→    96→      data-section-id={section.id}
    97→    97→      data-section-type={section.type}
    98→    98→    >
    99→    99→      {/* Overlay for background images */}
   100→   100→      {style.backgroundImage && style.overlay && (
   101→   101→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   102→   102→      )}
   103→   103→      <div
   104→   104→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   105→   105→      >
   106→   106→        {/* Selected section label */}
   107→   107→        {isSelected && (
   108→   108→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   109→   109→            {section.type.replace(/-/g, ' ')}
   110→   110→          </div>
   111→   111→        )}
   112→   112→        {children}
   113→   113→      </div>
   114→   114→    </div>
   115→   115→  );
   116→   116→}

   120→   120→function ProductCard({
   121→   121→  product,
   122→   122→  theme,
   123→   123→  showAddToCart,
   124→   124→  borderRadius,
   125→   125→  buttonBgOverride,
   126→   126→  buttonTextOverride,
   127→   127→  onViewProduct,
   128→   128→}: {
   129→   129→  product: StoreProduct;
   130→   130→  theme: StoreTheme;
   131→   131→  showAddToCart: boolean;
   132→   132→  borderRadius: string;
   133→   133→  buttonBgOverride?: string;
   134→   134→  buttonTextOverride?: string;
   135→   135→  onViewProduct?: (productId: string) => void;
   136→   136→}) {
   137→   137→  const [hovered, setHovered] = useState(false);
   138→   138→  const [addedFeedback, setAddedFeedback] = useState(false);
   139→   139→  const addToCart = useCartStore((s) => s.addItem);
   140→   140→  const imgColor = stringToColor(product.id, theme);
   141→   141→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   142→   142→
   143→   143→  const handleAddToCart = (e: React.MouseEvent) => {
   144→   144→    e.stopPropagation();
   145→   145→    addToCart(product);
   146→   146→    setAddedFeedback(true);
   147→   147→    setTimeout(() => setAddedFeedback(false), 1200);
   148→   148→  };
   149→   149→
   150→   150→  return (
   151→   151→    <div
   152→   152→      className={`${borderRadius} ${onViewProduct ? 'cursor-pointer' : ''} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   153→   153→      style={{
   154→   154→        borderColor: theme.colors.border,
   155→   155→      }}
   156→   156→      onMouseEnter={() => setHovered(true)}
   157→   157→      onMouseLeave={() => setHovered(false)}
   158→   158→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   159→   159→      role={onViewProduct ? 'button' : undefined}
   160→   160→      tabIndex={onViewProduct ? 0 : undefined}
   161→   161→    >
   162→   162→ {/* Image placeholder */}
   163→   163→      <div
   164→   164→        className="relative aspect-square w-full overflow-hidden"
   165→   165→        style={{ backgroundColor: imgColor }}
   166→   166→      >
   167→   167→        <div className="absolute inset-0 flex items-center justify-center">
   168→   168→          <div
   169→   169→            className="h-16 w-16 rounded-full opacity-30"
   170→   170→            style={{ backgroundColor: theme.colors.primary }}
   171→   171→          />
   172→   172→        </div>
   173→   173→        {/* Out of stock overlay */}
   174→   174→        {!product.inStock && (
   175→   175→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   176→   176→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   177→   177→              Sold Out
   178→   178→            </span>
   179→   179→          </div>
   180→   180→        )}
   181→   181→        {/* Discount badge */}
   182→   182→        {hasDiscount && product.compareAtPrice && (
   183→   183→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   184→   184→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   185→   185→          </div>
   186→   186→        )}
   187→   187→      </div>
   188→   188→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   189→   189→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   190→   190→        {product.category && (
   191→   191→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   192→   192→            {product.category}
   193→   193→          </p>
   194→   194→        )}
   195→   195→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   196→   196→          {product.name}
   197→   197→        </h3>
   198→   198→        <div className="mt-1.5 flex items-center gap-2">
   199→   199→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   200→   200→            {formatPrice(product.price)}
   201→   201→          </span>
   202→   202→          {hasDiscount && product.compareAtPrice && (
   203→   203→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   204→   204→              {formatPrice(product.compareAtPrice)}
   205→   205→            </span>
   206→   206→          )}
   207→   207→        </div>
   208→   208→        {showAddToCart && product.inStock && (
   209→   209→          <button
   210→   210→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 cursor-pointer ${
   211→   211→              hovered
   212→   212→                ? 'opacity-100 translate-y-0'
   213→   213→                : 'opacity-0 translate-y-1 pointer-events-none'
   214→   214→            }`}
   215→   215→            style={{
   216→   216→              backgroundColor: addedFeedback ? '#16a34a' : (buttonBgOverride || theme.colors.primary),
   217→   217→              color: addedFeedback ? '#ffffff' : (buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary)),
   218→   218→            }}
   219→   219→            onClick={handleAddToCart}
   220→   220→          >
   221→   221→            {addedFeedback ? 'Added!' : 'Add to Cart'}
   222→   222→          </button>
   223→   223→        )}
   224→   224→      </div>
   225→   225→    </div>
   226→   226→  );
   227→   227→}

   324→   324→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   325→   325→  const content = section.content as unknown as HeroContent;
   326→   326→  const style = section.style;
   327→   327→  const isSelected = selectedSectionId === section.id;
   328→   328→
   329→   329→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   330→   330→  const alignMap = {
   331→   331→    left: 'items-start text-left',
   332→   332→    center: 'items-center text-center',
   333→   333→    right: 'items-end text-right',
   334→   334→  };
   335→   335→
   336→   336→  // Only apply theme gradient as default background when no custom color or image is set.
   337→   337→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   338→   338→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   339→   339→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   340→   340→    : undefined;
   341→   341→
   342→   342→  const outerClasses = [
   343→   343→    'relative flex transition-all duration-200 cursor-pointer',
   344→   344→    heightMap[content.height] || heightMap.md,
   345→   345→    alignMap[content.alignment],
   346→   346→    pyClass(style.paddingY),
   347→   347→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   348→   348→  ].join(' ');
   349→   349→
   350→   350→  return (
   351→   351→    <div
   352→   352→      className={outerClasses}
   353→   353→      style={{
   354→   354→        backgroundColor: style.backgroundColor || theme.colors.primary,
   355→   355→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   356→   356→        backgroundSize: 'cover',
   357→   357→        backgroundPosition: 'center',
   358→   358→        color: style.textColor || '#ffffff',
   359→   359→      }}
   360→   360→      onClick={(e) => {
   361→   361→        e.stopPropagation();
   362→   362→        onSelectSection?.(isSelected ? null : section.id);
   363→   363→      }}
   364→   364→      data-section-id={section.id}
   365→   365→      data-section-type={section.type}
   366→   366→    >
   367→   367→      {/* Overlay */}
   368→   368→      {style.backgroundImage && style.overlay && (
   369→   369→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   370→   370→      )}
   371→   371→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   372→   372→      {!style.backgroundImage && !style.backgroundColor && (
   373→   373→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   374→   374→      )}
   375→   375→      {!style.backgroundImage && !style.backgroundColor && (
   376→   376→        <>
   377→   377→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   378→   378→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />
   379→   379→        </>
   380→   380→      )}
   381→   381→
   382→   382→      <div className={`relative z-10 mx-auto w-full ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}>
   383→   383→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   384→   384→          {content.headline}
   385→   385→        </h1>
   386→   386→        {content.subheadline && (
   387→   387→          <p className="mt-4 max-w-xl text-base opacity-80 sm:text-lg md:text-xl">
   388→   388→            {content.subheadline}
   389→   389→          </p>
   390→   390→        )}
   391→   391→        {content.ctaText && (
   392→   392→          <button className="mt-8 inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-transform hover:scale-105 sm:text-base"
   393→   393→            style={{
   394→   394→              backgroundColor: section.style.buttonBackgroundColor || '#ffffff',
   395→   395→              color: section.style.buttonTextColor || contrastTextColor(section.style.buttonBackgroundColor || '#ffffff'),
   396→   396→            }}>
   397→   397→            {content.ctaText}
   398→   398→            <ArrowRight className="h-4 w-4" />
   399→   399→          </button>
   400→   400→        )}
   401→   401→      </div>
   402→   402→    </div>
   403→   403→  );
   404→   404→}

   408→   408→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   409→   409→  const content = section.content as unknown as FeaturedProductsContent;
   410→   410→  const borderRadius = borderRadiusClass(theme.borderRadius);
   411→   411→  const featuredProducts = useMemo(
   412→   412→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   413→   413→    [content.productIds, products],
   414→   414→  );
   415→   415→
   416→   416→  return (
   417→   417→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   418→   418→      {content.headline && (
   419→   419→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   420→   420→          {content.headline}
   421→   421→        </h2>
   422→   422→      )}
   423→   423→      {content.subtitle && (
   424→   424→        <p className="mb-8 text-sm opacity-65">
   425→   425→          {content.subtitle}
   426→   426→        </p>
   427→   427→      )}
   428→   428→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   429→   429→        {featuredProducts.map((product) => (
   430→   430→          <ProductCard
   431→   431→            key={product.id}
   432→   432→            product={product}
   433→   433→            theme={theme}
   434→   434→            showAddToCart={content.showAddToCart && !forceHideAddToCart}
   435→   435→            borderRadius={borderRadius}
   436→   436→            buttonBgOverride={section.style.buttonBackgroundColor}
   437→   437→            buttonTextOverride={section.style.buttonTextColor}
   438→   438→            onViewProduct={onViewProduct}
   439→   439→          />
   440→   440→        ))}
   441→   441→      </div>
   442→   442→    </SectionWrapper>
   443→   443→  );
   444→   444→}

   448→   448→export function ProductGridSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   449→   449→  const content = section.content as unknown as ProductGridContent;
   450→   450→  const borderRadius = borderRadiusClass(theme.borderRadius);
   451→   451→
   452→   452→  const filteredProducts = useMemo(() => {
   453→   453→    let prods = products;
   454→   454→    if (content.filterByCategory) {
   455→   455→      prods = prods.filter((p) => p.category === content.filterByCategory);
   456→   456→    }
   457→   457→    return prods;
   458→   458→  }, [products, content.filterByCategory]);
   459→   459→
   460→   460→  return (
   461→   461→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   462→   462→      {content.headline && (
   463→   463→        <h2 className="mb-8 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   464→   464→          {content.headline}

   890→   890→                <span
   891→   891→                  key={social.platform}

   894→   894→                  title={social.platform}
   895→   895→                >
   896→   896→                  <svg className="h-4 w-4" fill="currentColor" viewBox="0 0 24 24">

   935→   935→          ? 'ring-2 ring-[#a855f7] ring-offset-2'

   937→   937→      }`}
   938→   938→      onClick={(e) => {
   939→   939→        e.stopPropagation();
   940→   940→        onSelectSection?.(isSelected ? null : section.id);

   943→   943→      data-section-type={section.type}

   945→   945→      {/* Dashed guide when selected */}

   995→   995→export function renderSection(props: SectionRendererProps): React.ReactNode {
   996→   996→  const { section } = props;
   997→   997→
   998→   998→  if (!section.visible) return null;
   999→   999→
  1000→  1000→  switch (section.type) {
  1001→  1001→    case 'header':
  1002→  1002→      return <HeaderSection {...props} />;

  1013→  1013→    case 'testimonials':
  1014→  1014→      return <TestimonialsSection {...props} />;

  1016→  1016→      return <NewsletterSection {...props} />;
  1017→  1017→    case 'faq':

  1019→  1019→    case 'cta':
  1020→  1020→      return <CTASection {...props} />;

  1022→  1022→      return <CategoriesSection {...props} />;
  1023→  1023→    case 'footer':

  1034→  1034→}
