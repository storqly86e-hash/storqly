   118→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)

   122→  const {

   134→  const clearTimers = useCallback(() => {

   140→  useEffect(() => {

   144→    }

   150→      .catch(() => { if (!cancelled) setMyStores([]) })
   151→      .finally(() => { if (!cancelled) setStoresLoading(false) })
   152→    return () => { cancelled = true }
   153→  }, [session?.user?.id])
   154→
   155→  // ── Edit store: fetch full data via lookup, then load into editor ──
   156→  const handleEditStore = useCallback(async (slug: string) => {
   157→    setEditingSlug(slug)
   158→    try {
   159→      const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)

   163→    } catch {
   164→      toast.error('Failed to load store. It may have been deleted.')

   166→      setEditingSlug(null)

   168→  }, [setStore])
   169→
   170→  // Format relative time
   171→  const formatTimeAgo = useCallback((iso: string) => {
   172→    const diff = Date.now() - new Date(iso).getTime()
   173→    const mins = Math.floor(diff / 60000)
   174→    if (mins < 1) return 'just now'
   175→    if (mins < 60) return `${mins}m ago`
   176→    const hours = Math.floor(mins / 60)
   177→    if (hours < 24) return `${hours}h ago`
   178→    const days = Math.floor(hours / 24)
   179→    if (days < 30) return `${days}d ago`
   180→    return new Date(iso).toLocaleDateString()
   181→  }, [])
   182→
   183→  const handleCancel = useCallback(() => {
   184→    abortRef.current?.abort()
   185→    clearTimers()
   186→    setIsGenerating(false)
   187→    setGenerationStatus('')
   188→    setElapsedSeconds(0)
   189→    setError(null)
   190→  }, [clearTimers, setIsGenerating])
   191→
   192→  const handleGenerate = useCallback(async () => {
   193→    const trimmed = promptText.trim()
   194→    if (!trimmed) {
   195→      setError('Please describe your store to get started.')
   196→      textareaRef.current?.focus()
   197→      return
   198→    }
   199→
   200→    // Auth gate — block generation for logged-out users (show modal instead)
   201→    if (!session?.user?.id) {
   202→      setAuthOpen(true)
   203→      return
   204→    }
   205→
   206→    // Reset state
   207→    setError(null)
   208→    setIsGenerating(true)
   209→    setGenerationStatus('Starting generation...')
   210→    setElapsedSeconds(0)
   211→
   212→    // Create abort controller for this request
   213→    const controller = new AbortController()
   214→    abortRef.current = controller
   215→
   216→    // Elapsed time counter
   217→    elapsedTimerRef.current = setInterval(() => {
   218→      setElapsedSeconds((prev) => prev + 1)
   219→    }, 1000)
   220→
   221→    // Fallback progress cycling (used only if SSE progress events stop)
   222→    let fallbackIdx = 0
   223→    let receivedSSEProgress = false
   224→    progressTimerRef.current = setInterval(() => {
   225→      if (!receivedSSEProgress) {
   226→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   227→        setGenerationStatus(progressMessages[fallbackIdx])
   228→      }
   229→    }, 3000)
   230→
   231→    try {
   232→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   233→
   234→      // ── Fetch with auto-retry for transient gateway errors (502/503/504) ──
   235→      // Caddy returns 502 when Turbopack is briefly unavailable during recompilation.
   236→      // One retry with a 2s pause is enough to let the server finish recompiling.
   237→      const RETRYABLE_STATUSES = [502, 503, 504]
   238→      let res: Response | undefined
   239→      let retryCount = 0
   240→      const maxRetries = 1
   241→
   242→      while (retryCount <= maxRetries) {
   243→        try {
   244→          res = await fetch('/api/store/generate', {
   245→            method: 'POST',
   246→            headers: { 'Content-Type': 'application/json' },
   247→            body: JSON.stringify({ prompt: trimmed }),
   248→            signal: controller.signal,
   249→          })
   250→
   251→          // If the status is retryable, wait and loop again
   252→          if (!res.ok && RETRYABLE_STATUSES.includes(res.status) && retryCount < maxRetries) {
   253→            retryCount++
   254→            console.warn(`[Storqly] Gateway error ${res.status}, retrying (${retryCount}/${maxRetries}) in 2s...`)
   255→            setGenerationStatus('Connection issue — retrying...')
   256→            await new Promise(r => setTimeout(r, 2000))
   257→            continue
   258→          }
   259→          break // Success or non-retryable error — exit loop
   260→        } catch (fetchErr: unknown) {
   261→          // Network-level error (DNS failure, connection refused, etc.)
   262→          if (fetchErr instanceof TypeError && retryCount < maxRetries) {
   263→            retryCount++
   264→            console.warn(`[Storqly] Network error, retrying (${retryCount}/${maxRetries}) in 2s...`)
   265→            setGenerationStatus('Connection issue — retrying...')
   266→            await new Promise(r => setTimeout(r, 2000))
   267→            continue
   268→          }
   269→          throw fetchErr // Re-throw if out of retries or not a network error
   270→        }
   271→      }
   272→
   273→      if (!res!.ok) {
   274→        let errorMsg = `Server error (${res!.status})`
   275→        try {
   276→          const errorData = await res!.json()
   277→          errorMsg = errorData.error || errorMsg
   278→        } catch { /* ignore parse error */ }
   279→        throw new Error(errorMsg)
   280→      }
   281→
   282→      // ── Consume SSE stream ──
   283→      const reader = res.body?.getReader()
   284→      if (!reader) throw new Error('No response stream received.')
   285→
   286→      const decoder = new TextDecoder()
   287→      let buffer = ''
   288→      let resolved = false
   289→
   290→      while (true) {
   291→        const { done, value } = await reader.read()
   292→        if (done) break
   293→
   294→        buffer += decoder.decode(value, { stream: true })
   295→
   296→        // Parse SSE events from buffer
   297→        // SSE format: "event: <name>\ndata: <json>\n\n" or ": comment\n\n"
   298→        const lines = buffer.split('\n')
   299→        // Keep incomplete last line in buffer
   300→        buffer = lines.pop() || ''
   301→
   302→        let currentEvent = ''
   303→        for (const line of lines) {
   304→          if (line.startsWith(':')) {
   305→            // SSE comment (heartbeat) — skip
   306→            continue
   307→          }
   308→          if (line.startsWith('event: ')) {
   309→            currentEvent = line.slice(7).trim()
   310→          } else if (line.startsWith('data: ')) {
   311→            const dataStr = line.slice(6)
   312→
   313→            // Handle SSE events
   314→            if (currentEvent === 'progress') {
   315→              receivedSSEProgress = true
   316→              try {
   317→                const data = JSON.parse(dataStr)
   318→                setGenerationStatus(data.message || 'Processing...')
   319→              } catch { /* ignore */ }
   320→            } else if (currentEvent === 'result') {
   321→              try {
   322→                const data = JSON.parse(dataStr)
   323→                if (!data.store) {
   324→                  throw new Error('The AI response was missing store data.')
   325→                }
   326→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   327→                resolved = true
   328→                clearTimers()
   329→
   330→                if (data._isFallback) {
   331→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   332→                } else {
   333→                  setStore(data.store)
   334→                }
   335→              } catch (parseErr) {
   336→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   337→                console.error('[Storqly] Failed to parse result event:', msg)
   338→                throw new Error(msg)
   339→              }
   340→            } else if (currentEvent === 'error') {
   341→              try {
   342→                const data = JSON.parse(dataStr)
   343→                throw new Error(data.message || 'Generation failed')
   344→              } catch (e) {
   345→                if (e instanceof Error && e.message !== 'Generation failed') throw e
   346→                throw new Error('Generation failed')
   347→              }
   348→            }
   349→            // Reset event after processing its data
   350→            currentEvent = ''
   351→          }
   352→        }
   353→      }
   354→
   355→      if (!resolved) {
   356→        throw new Error('Stream ended without a result. The server may have disconnected.')
   357→      }
   358→    } catch (err: unknown) {
   359→      clearTimers()
   360→
   361→      let message = 'Something went wrong. Please try again.'
   362→      if (err instanceof Error) {
   363→        if (err.name === 'AbortError') {
   364→          message = 'Generation was cancelled.'
   365→        } else {
   366→          console.error('[Storqly] Generation error:', err)
   367→          message = err.message
   368→        }
   369→      }
   370→
   371→      setError(message)
   372→      setIsGenerating(false)
   373→      setGenerationStatus('')
   374→      setElapsedSeconds(0)
   375→      toast.error('Store generation failed', { description: message })
   376→    }
   377→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])
   378→

   381→    return () => {
   382→      clearTimers()
   383→      abortRef.current?.abort()
   384→    }
   385→  }, [clearTimers])
   386→
   387→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   388→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   389→      e.preventDefault()
   390→      handleGenerate()
   391→    }
   392→  }
   393→

   397→    const s = seconds % 60
   398→    return `${m}:${s.toString().padStart(2, '0')}`
   399→  }

   402→    <>
   403→      {/* ── Fixed Nav ── */}
   404→      <header className="fixed top-0 left-0 right-0 z-50 border-b border-white/[0.06] bg-[#09090b]/80 backdrop-blur-xl">
   405→        <nav className="mx-auto flex h-16 max-w-6xl items-center justify-between px-5 sm:px-8">
   406→          <span className="text-2xl font-bold tracking-tight bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   407→            Storqly
   408→          </span>
   409→          <div className="hidden sm:flex items-center gap-4">
   410→            <span className="text-sm text-zinc-500">
   411→              AI-Powered Store Builder

   425→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   426→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />
   427→        </div>

   430→          <motion.h1
   431→            custom={0}
   432→            variants={fadeUp}
   433→            initial="hidden"
   434→            animate="visible"
   435→            className="text-4xl font-extrabold leading-[1.1] tracking-tight sm:text-5xl md:text-6xl lg:text-7xl"
   436→          >
   437→            Build Your Store{' '}
   438→            <span className="bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   439→              in Seconds
   440→            </span>
   441→          </motion.h1>
   442→
   443→          <motion.p
   444→            custom={1}
   445→            variants={fadeUp}
   446→            initial="hidden"
   447→            animate="visible"
   448→            className="mt-6 text-lg text-zinc-400 sm:text-xl"

   461→              <Textarea
   462→                ref={textareaRef}
   463→                value={promptText}
   464→                onChange={(e) => { setPromptText(e.target.value); setError(null) }}
   465→                onKeyDown={handleKeyDown}

   492→                  <Button
   493→                    onClick={handleGenerate}
   494→                    disabled={isGenerating || !promptText.trim()}
   495→                    className="relative overflow-hidden rounded-xl bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] px-6 py-2.5 text-sm font-semibold text-white shadow-lg shadow-[#a855f7]/20 transition-all duration-300 hover:scale-105 hover:shadow-[#a855f7]/30 disabled:opacity-50 disabled:hover:scale-100"
   496→                  >

   513→            {/* Status feedback area */}

   515→              <AnimatePresence mode="wait">
   516→                {isGenerating && !error && (
   517→                  <motion.div
   518→                    key="progress"
   519→                    initial={{ opacity: 0, y: 8 }}

   522→                    className="flex flex-col items-center gap-1"

   531→                  </motion.div>

   533→
   534→                {error && (
   535→                  <motion.div
   536→                    key="error"
   537→                    initial={{ opacity: 0, y: 8 }}
   538→                    animate={{ opacity: 1, y: 0 }}
   539→                    exit={{ opacity: 0, y: -8 }}
   540→                    className="flex flex-col items-center gap-2"
   541→                  >
   542→                    <div className="flex items-start gap-2 rounded-xl border border-red-500/20 bg-red-500/5 px-4 py-3 text-left max-w-lg">
   543→                      <AlertCircle className="mt-0.5 h-4 w-4 shrink-0 text-red-400" />
   544→                      <div className="flex-1 min-w-0">
   545→                        <p className="text-sm font-medium text-red-300">Generation failed</p>
   546→                        <p className="mt-1 text-xs text-red-400/80 leading-relaxed">{error}</p>
   547→                      </div>
   548→                      <button
   549→                        onClick={() => setError(null)}
   550→                        className="shrink-0 text-zinc-500 hover:text-zinc-300"
   551→                        aria-label="Dismiss error"
   552→                      >
   553→                        <X className="h-3.5 w-3.5" />

   867→function EditorToolbar({
   868→  onToggleLeft,
   869→  onToggleRight,
   870→  showLeft,
   871→  showRight,
   872→}: {
   873→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   874→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   875→  showLeft: boolean
   876→  showRight: boolean
   877→}) {
   878→  const store = useStoreEditor((s) => s.store)
   879→  const reset = useStoreEditor((s) => s.reset)
   880→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   881→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   882→  const isPublished = useStoreEditor((s) => s.isPublished)
   883→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   884→  const [isSaving, setIsSaving] = useState(false)
   885→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
   886→  const [copied, setCopied] = useState(false)
   887→  const [authOpen, setAuthOpen] = useState(false)
   888→
   889→  const handleBack = () => {
   890→    reset()
   891→  }
   892→
   893→  const handleSave = async () => {
   894→    if (!store || isSaving) return
   895→    setIsSaving(true)
   896→    try {
   897→      const res = await fetch('/api/store/save', {
   898→        method: 'POST',
   899→        headers: { 'Content-Type': 'application/json' },
   900→        body: JSON.stringify({ store }),
   901→      })
   902→      if (!res.ok) throw new Error('Save failed')
   903→      const data = await res.json()
   904→      toast.success(`Draft saved — ID: ${data.id}`, {
   905→        description: `Slug: ${data.slug}`,
   906→        duration: 5000,
   907→      })
   908→    } catch {
   909→      toast.error('Failed to save store')
   910→    } finally {
   911→      setIsSaving(false)
   912→    }
   913→  }
   914→
   915→  const handlePublish = async () => {
   916→    if (!store || isPublishing) return
   917→    setIsPublishing(true)
   918→    try {
   919→      const res = await fetch('/api/store/publish', {
   920→        method: 'POST',
   921→        headers: { 'Content-Type': 'application/json' },
   922→        body: JSON.stringify({ store }),
   923→      })
   924→      if (!res.ok) throw new Error('Publish failed')
   925→      const data = await res.json()
   926→      const slug = data.slug
   927→      setIsPublished(true)
   928→      // Build a real viewable URL using the current origin
   929→      const baseUrl = window.location.origin
   930→      const viewUrl = `${baseUrl}/?store=${slug}`
   931→      setPublishedUrl(viewUrl)
   932→      toast.success('Store published successfully!')
   933→    } catch {
   934→      toast.error('Failed to publish store')
   935→    } finally {
   936→      setIsPublishing(false)
   937→    }
   938→  }
   939→
   940→  const handleCopyUrl = useCallback(async () => {
   941→    if (!publishedUrl) return
   942→    try {
   943→      await navigator.clipboard.writeText(publishedUrl)
   944→      setCopied(true)
   945→      toast.success('URL copied to clipboard!')
   946→      setTimeout(() => setCopied(false), 2000)
   947→    } catch {
   948→      // Fallback for older browsers
   949→      const textarea = document.createElement('textarea')
   950→      textarea.value = publishedUrl
   951→      document.body.appendChild(textarea)
   952→      textarea.select()
   953→      document.execCommand('copy')
   954→      document.body.removeChild(textarea)
   955→      setCopied(true)
   956→      toast.success('URL copied to clipboard!')
   957→      setTimeout(() => setCopied(false), 2000)
   958→    }
   959→  }, [publishedUrl])
   960→
   961→  return (
   962→    <>
   963→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
   964→        <div className="flex items-center gap-1">
   965→          <Button
   966→            variant="ghost"
   967→            size="icon"
   968→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   969→            onClick={handleBack}
   970→            aria-label="Back to home"
   971→          >
   972→            <ArrowLeft className="h-4 w-4" />
   973→          </Button>
   974→          <div className="h-5 w-px bg-zinc-800" />
   975→          <Button
   976→            variant={showLeft ? 'secondary' : 'ghost'}
   977→            size="sm"
   978→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   979→            onClick={() => onToggleLeft((v) => !v)}
   980→          >
   981→            <Layers className="h-3.5 w-3.5" />
   982→            <span className="hidden sm:inline">Sections</span>
   983→          </Button>
   984→          <Button
   985→            variant={showRight ? 'secondary' : 'ghost'}
   986→            size="sm"
   987→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   988→            onClick={() => onToggleRight((v) => !v)}
   989→          >
   990→            <MessageSquare className="h-3.5 w-3.5" />
   991→            <span className="hidden sm:inline">Chat</span>
   992→          </Button>
   993→        </div>
   994→
   995→        <div className="flex items-center gap-1.5">
   996→          {publishedUrl && (
   997→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
   998→              <Check className="h-3 w-3 text-emerald-400" />
   999→              <span className="max-w-[180px] truncate text-xs font-medium text-emerald-300">
  1000→                {publishedUrl}
  1001→              </span>
  1002→              <button
  1003→                onClick={handleCopyUrl}
  1004→                className="ml-1 rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
  1005→                aria-label="Copy URL"
  1006→              >
  1007→                {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
  1008→              </button>
  1009→              <a
  1010→                href={publishedUrl}
  1011→                target="_blank"
  1012→                rel="noopener noreferrer"
  1013→                className="rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
  1014→                aria-label="Open published store"
  1015→              >
  1016→                <ExternalLink className="h-3 w-3" />
  1017→              </a>
  1018→            </div>
  1019→          )}
  1020→          <AuthButton onSignIn={() => setAuthOpen(true)} />
  1021→          <div className="h-5 w-px bg-zinc-800" />
  1022→          <span className="mr-2 text-xs font-medium text-zinc-500 hidden md:inline truncate max-w-[200px]">
  1023→            {store?.name || 'Untitled Store'}
  1024→          </span>
  1025→          <Button
  1026→            variant="ghost"
  1027→            size="sm"
  1028→            className="h-8 gap-1.5 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
  1029→            onClick={handleSave}
  1030→            disabled={isSaving}
  1031→          >
  1032→            <Save className={`h-3.5 w-3.5 ${isSaving ? 'animate-pulse' : ''}`} />
  1033→            <span className="hidden sm:inline">{isSaving ? 'Saving...' : 'Save'}</span>
  1034→          </Button>
  1035→          <Button
  1036→            size="sm"
  1037→            className="h-8 gap-1.5 bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] text-white shadow-none hover:opacity-90"
  1038→            onClick={handlePublish}
  1039→            disabled={isPublishing}
  1040→          >
  1041→            {isPublishing ? (
  1042→              <Loader2 className="h-3.5 w-3.5 animate-spin" />
  1043→            ) : isPublished ? (
  1044→              <Eye className="h-3.5 w-3.5" />
  1045→            ) : (
  1046→              <Globe className="h-3.5 w-3.5" />
  1047→            )}
  1048→            <span className="hidden sm:inline">{isPublishing ? 'Publishing...' : isPublished ? 'Published' : 'Publish'}</span>
  1049→          </Button>
  1050→        </div>
  1051→      </div>
  1052→
  1053→      {/* Publish Success Dialog */}
  1054→      <AnimatePresence>
  1055→        {publishedUrl && !isPublishing && (
  1056→          <motion.div
  1057→            initial={{ opacity: 0 }}
  1058→            animate={{ opacity: 1 }}
  1059→            exit={{ opacity: 0 }}
  1060→            className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 backdrop-blur-sm"
  1061→            onClick={() => setPublishedUrl(null)}
  1062→          >
  1063→            <motion.div
  1064→              initial={{ scale: 0.95, opacity: 0 }}

  1130→// ─── Published Store Viewer (read-only, shown via ?store=slug) ────────
  1131→
  1132→function PublishedStoreViewer({ slug }: { slug: string }) {
  1133→  const [store, setStore] = useState<Store | null>(null)
  1134→  const [loading, setLoading] = useState(true)
  1135→  const [error, setError] = useState<string | null>(null)
  1136→
  1137→  useEffect(() => {
  1138→    let cancelled = false
  1139→    async function load() {
  1140→      try {
  1141→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
  1142→        if (!res.ok) {
  1143→          const data = await res.json().catch(() => ({}))
  1144→          throw new Error(data.error || `Store not found (${res.status})`)
  1145→        }
  1146→        const data = await res.json()
  1147→        if (!cancelled) setStore(data.store)
  1148→      } catch (err: unknown) {
  1149→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
  1150→      } finally {
  1151→        if (!cancelled) setLoading(false)
  1152→      }
  1153→    }
  1154→    load()
  1155→    return () => { cancelled = true }
  1156→  }, [slug])
  1157→
  1158→  if (loading) {
  1159→    return (
  1160→      <div className="flex min-h-screen items-center justify-center bg-white">
  1161→        <div className="text-center">
  1162→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
  1163→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
  1164→        </div>
  1165→      </div>
  1166→    )
  1167→  }
  1168→
  1169→  if (error || !store) {
  1170→    return (
  1171→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
  1172→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
  1173→          <AlertCircle className="h-8 w-8 text-red-400" />
  1174→        </div>
  1175→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
  1176→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
  1177→          {error || 'This store could not be found or has not been published.'}
  1178→        </p>
  1179→        <Button
  1180→          variant="outline"
  1181→          onClick={() => window.location.href = '/'}
  1182→          className="gap-2"
  1183→        >
  1184→          <ArrowLeft className="h-4 w-4" />
  1185→          Back to Storqly
  1186→        </Button>
  1187→      </div>
  1188→    )
  1189→  }
  1190→
  1191→  return (
  1192→    <>
  1193→      <StoreRenderer store={store} />
  1194→      <Toaster />
  1195→      {/* Minimal footer for published stores */}
  1196→      <div className="fixed bottom-3 right-3 z-40">
  1197→        <a
  1198→          href="/"
  1199→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
  1200→        >
  1201→          <Sparkles className="h-3 w-3" />
  1202→          Built with Storqly
  1203→        </a>
  1204→      </div>
  1205→    </>
  1206→  )
  1207→}
