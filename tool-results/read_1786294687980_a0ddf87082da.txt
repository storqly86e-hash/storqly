   565→   565→  onToggleLeft,
   566→   566→  onToggleRight,
   567→   567→  showLeft,
   568→   568→  showRight,
   569→   569→}: {

   573→   573→  showRight: boolean

   575→   575→  const store = useStoreEditor((s) => s.store)
   576→   576→  const reset = useStoreEditor((s) => s.reset)
   577→   577→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   578→   578→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   579→   579→  const isPublished = useStoreEditor((s) => s.isPublished)

   583→   583→  const [copied, setCopied] = useState(false)
   584→   584→
   585→   585→  const handleBack = () => {
   586→   586→    reset()
   587→   587→  }
   588→   588→
   589→   589→  const handleSave = async () => {
   590→   590→    if (!store || isSaving) return
   591→   591→    setIsSaving(true)
   592→   592→    try {
   593→   593→      const res = await fetch('/api/store/save', {
   594→   594→        method: 'POST',
   595→   595→        headers: { 'Content-Type': 'application/json' },
   596→   596→        body: JSON.stringify({ store }),
   597→   597→      })

   601→   601→      toast.error('Failed to save store')

   603→   603→      setIsSaving(false)
   604→   604→    }
   605→   605→  }
   606→   606→
   607→   607→  const handlePublish = async () => {
   608→   608→    if (!store || isPublishing) return

   610→   610→    try {
   611→   611→      const res = await fetch('/api/store/publish', {
   612→   612→        method: 'POST',
   613→   613→        headers: { 'Content-Type': 'application/json' },
   614→   614→        body: JSON.stringify({ store }),
   615→   615→      })

   617→   617→      const data = await res.json()
   618→   618→      const slug = data.slug
   619→   619→      setIsPublished(true)
   620→   620→      // Build a real viewable URL using the current origin
   621→   621→      const baseUrl = window.location.origin
   622→   622→      const viewUrl = `${baseUrl}/?store=${slug}`

   628→   628→      setIsPublishing(false)
   629→   629→    }
   630→   630→  }

   642→   642→      textarea.value = publishedUrl
   643→   643→      document.body.appendChild(textarea)
   644→   644→      textarea.select()
   645→   645→      document.execCommand('copy')
   646→   646→      document.body.removeChild(textarea)
   647→   647→      setCopied(true)
   648→   648→      toast.success('URL copied to clipboard!')
   649→   649→      setTimeout(() => setCopied(false), 2000)
   650→   650→    }
   651→   651→  }, [publishedUrl])
   652→   652→
   653→   653→  return (
   654→   654→    <>
   655→   655→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
   656→   656→        <div className="flex items-center gap-1">
   657→   657→          <Button
   658→   658→            variant="ghost"
   659→   659→            size="icon"
   660→   660→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   661→   661→            onClick={handleBack}
   662→   662→            aria-label="Back to home"
   663→   663→          >
   664→   664→            <ArrowLeft className="h-4 w-4" />
   665→   665→          </Button>
   666→   666→          <div className="h-5 w-px bg-zinc-800" />
   667→   667→          <Button
   668→   668→            variant={showLeft ? 'secondary' : 'ghost'}
   669→   669→            size="sm"
   670→   670→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   671→   671→            onClick={() => onToggleLeft((v) => !v)}
   672→   672→          >
   673→   673→            <Layers className="h-3.5 w-3.5" />
