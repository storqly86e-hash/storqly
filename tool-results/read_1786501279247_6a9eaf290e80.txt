   256→function SortableSectionItem({
   257→  section,
   258→  isSelected,
   259→  onSelect,
   260→  onToggleVisibility,
   261→  onDelete,
   262→}: SortableSectionItemProps) {
   263→  const {
   264→    attributes,
   265→    listeners,
   266→    setNodeRef,
   267→    transform,
   268→    transition,
   269→    isDragging,
   270→  } = useSortable({ id: section.id });
   271→
   272→  const style = {
   273→    transform: CSS.Transform.toString(transform),
   274→    transition,
   275→  };
   276→
   277→  const Icon = SECTION_TYPE_ICONS[section.type];
   278→  const label = SECTION_TYPE_LABELS[section.type];
   279→
   280→  return (
   281→    <div
   282→      ref={setNodeRef}
   283→      style={style}
   284→      className={[
   285→        'group flex items-center gap-2 rounded-lg px-3 py-2 text-sm cursor-pointer transition-colors',
   286→        isDragging
   287→          ? 'z-50 bg-zinc-700/80 shadow-xl ring-1 ring-white/10'
   288→          : isSelected
   289→            ? 'bg-amber-500/15 text-amber-200 ring-1 ring-amber-500/40'
   290→            : 'text-zinc-400 hover:bg-zinc-800 hover:text-zinc-200',
   291→      ].join(' ')}
   292→      onClick={onSelect}
   293→    >
   294→      {/* Drag handle */}
   295→      <button
   296→        className="flex-shrink-0 cursor-grab text-zinc-600 hover:text-zinc-400 active:cursor-grabbing focus:outline-none"
   297→        {...attributes}
   298→        {...listeners}
   299→        onClick={(e) => e.stopPropagation()}
   300→        aria-label="Drag to reorder"
   301→      >
   302→        <GripVertical className="h-4 w-4" />
   303→      </button>
   304→
   305→      {/* Icon */}
   306→      <Icon className="h-4 w-4 flex-shrink-0 opacity-60" />
   307→
   308→      {/* Label */}
   309→      <span className="flex-1 truncate font-medium">{label}</span>
   310→
   311→      {/* Visibility toggle */}
   312→      <button
   313→        className="flex-shrink-0 rounded p-1 text-zinc-500 hover:text-zinc-300 hover:bg-zinc-700/50 focus:outline-none"
   314→        onClick={(e) => {
   315→          e.stopPropagation();
   316→          onToggleVisibility();
   317→        }}

   326→
   327→      {/* Delete */}
   328→      <button
   329→        className="flex-shrink-0 rounded p-1 text-zinc-600 opacity-0 group-hover:opacity-100 hover:text-red-400 hover:bg-red-400/10 transition-opacity focus:outline-none"
   330→        onClick={(e) => {
   331→          e.stopPropagation();

   394→// ── Content Field Renderer ───────────────────────────────────────────
   395→
   396→function ContentFieldRenderer({
   397→  keyName,
   398→  value,
   399→  onChange,
   400→  depth = 0,
   401→}: {
   402→  keyName: string;
   403→  value: unknown;
   404→  onChange: (key: string, value: unknown) => void;
   405→  depth?: number;
   406→}) {
   407→  const label = keyName
   408→    .replace(/([A-Z])/g, ' $1')
   409→    .replace(/^./, (s) => s.toUpperCase())
   410→    .trim();
   411→
   412→  if (value === null || value === undefined) {
   413→    return null;
   414→  }
   415→
   416→  if (typeof value === 'string') {
   417→    if (keyName === 'html' && value.includes('<')) {
   418→      return (
   419→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   420→          <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   421→          <Textarea
   422→            value={value}
   423→            onChange={(e) => onChange(keyName, e.target.value)}
   424→            rows={4}
   425→            className="bg-zinc-800 border-zinc-700 text-zinc-100 text-xs font-mono resize-none"
   426→          />
   427→        </div>
   428→      );
   429→    }
   430→    return (
   431→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   432→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   433→        <Input
   434→          value={value}
   435→          onChange={(e) => onChange(keyName, e.target.value)}
   436→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   437→        />
   438→      </div>
   439→    );
   440→  }
   441→
   442→  if (typeof value === 'number') {
   443→    return (
   444→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   445→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   446→        <Input
   447→          type="number"
   448→          value={value}
   449→          onChange={(e) => onChange(keyName, Number(e.target.value))}
   450→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   451→        />
   452→      </div>
   453→    );
   454→  }
   455→
   456→  if (typeof value === 'boolean') {
   457→    return (
   458→      <div
   459→        key={keyName}
   460→        className={`flex items-center justify-between ${depth > 0 ? 'ml-4' : ''}`}
   461→      >
   462→        <Label className="text-xs text-zinc-400">{label}</Label>
   463→        <Switch
   464→          checked={value}
   465→          onCheckedChange={(checked) => onChange(keyName, checked)}
   466→        />
   467→      </div>
   468→    );
   469→  }
   470→
   471→  if (Array.isArray(value)) {
   472→    return (
   473→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   474→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   475→        <div className="rounded-lg bg-zinc-800/60 border border-zinc-700/50 px-3 py-2 text-xs text-zinc-500">
   476→          {value.length} item{value.length !== 1 ? 's' : ''}
   477→        </div>
   478→      </div>
   479→    );
   480→  }
   481→
   482→  if (typeof value === 'object') {
   483→    const obj = value as Record<string, unknown>;
   484→    const keys = Object.keys(obj);
   485→
   486→    const allPrimitive = keys.every(
   487→      (k) =>
   488→        typeof obj[k] === 'string' ||
   489→        typeof obj[k] === 'number' ||
   490→        typeof obj[k] === 'boolean' ||
   491→        obj[k] === null ||
   492→        obj[k] === undefined
   493→    );
   494→
   495→    if (allPrimitive && keys.length > 0) {
   496→      return (
   497→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   498→          <p className="text-xs font-medium text-zinc-300 mb-2 mt-1">{label}</p>
   499→          <div className="space-y-3 border-l border-zinc-700/50 pl-3 ml-1">
   500→            {keys.map((k) => (
   501→              <ContentFieldRenderer
   502→                key={k}
   503→                keyName={k}
   504→                value={obj[k]}
   505→                onChange={(subKey, subVal) => {
   506→                  onChange(keyName, { ...obj, [subKey]: subVal });
   507→                }}
   508→                depth={depth + 1}
   509→              />
   510→            ))}
   511→          </div>
   512→        </div>
   513→      );
   514→    }
   515→
   516→    return (
   517→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   518→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   519→        <Textarea
   520→          value={JSON.stringify(value, null, 2)}
   521→          readOnly
   522→          rows={3}
   523→          className="bg-zinc-800/60 border-zinc-700/50 text-zinc-500 text-xs font-mono resize-none"
   524→        />
   525→      </div>
   526→    );
   527→  }
   528→
   529→  return null;
   530→}

   539→function PropertiesPanel({ section, pageId }: PropertiesPanelProps) {
   540→  const {
   541→    updateSectionContent,
   542→    updateSectionStyle,
   543→    removeSection,
   544→    setSelectedSectionId,
   545→    store,
   546→    setStore,
   547→  } = useStoreEditor();
   548→
   549→  const [localStyle, setLocalStyle] = useState<SectionStyle>(section.style);
   550→
   551→  // Sync style from store when section changes
   552→  if (section.style !== localStyle) {
   553→    setLocalStyle(section.style);
   554→  }
   555→
   556→  const handleContentChange = useCallback(
   557→    (key: string, value: unknown) => {
   558→      updateSectionContent(pageId, section.id, { [key]: value });
   559→    },
   560→    [pageId, section.id, updateSectionContent]
   561→  );
   562→
   563→  const handleStyleChange = useCallback(
   564→    (patch: Partial<SectionStyle>) => {
   565→      const updated = { ...localStyle, ...patch };
   566→      setLocalStyle(updated);
   567→      updateSectionStyle(pageId, section.id, patch);
   568→    },
   569→    [pageId, section.id, localStyle, updateSectionStyle]
   570→  );
   571→
   572→  const handleDelete = useCallback(() => {
   573→    removeSection(pageId, section.id);
   574→    setSelectedSectionId(null);
   575→    toast.success('Section deleted');
   576→  }, [pageId, section.id, removeSection, setSelectedSectionId]);
   577→
   578→  const handleVisibilityToggle = useCallback(() => {
   579→    if (!store) return;
   580→    const updatedStore = {
   581→      ...store,
   582→      updatedAt: new Date().toISOString(),
   583→      pages: store.pages.map((page) => ({
   584→        ...page,
   585→        sections: page.sections.map((s) =>
   586→          s.id === section.id ? { ...s, visible: !s.visible } : s
   587→        ),
   588→      })),
   589→    };
   590→    setStore(updatedStore);
   591→  }, [store, section.id, setStore]);
   592→
   593→  const Icon = SECTION_TYPE_ICONS[section.type];
   594→  const typeLabel = SECTION_TYPE_LABELS[section.type];
   595→
   596→  const renderSelect = (
   597→    lbl: string,
   598→    value: string | undefined,
   599→    options: { label: string; value: string }[],
   600→    onChange: (val: string) => void
   601→  ) => (
   602→    <div>
   603→      <Label className="text-xs text-zinc-400 mb-1.5 block">{lbl}</Label>
   604→      <Select value={value ?? ''} onValueChange={onChange}>
   605→        <SelectTrigger className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm">
   606→          <SelectValue />
   607→        </SelectTrigger>
   608→        <SelectContent className="bg-zinc-900 border-zinc-700">
   609→          {options.map((opt) => (
   610→            <SelectItem
   611→              key={opt.value}
   612→              value={opt.value}
   613→              className="text-zinc-300 focus:bg-zinc-800 focus:text-zinc-100"
   614→            >
   615→              {opt.label}
   616→            </SelectItem>
   617→          ))}
   618→        </SelectContent>
   619→      </Select>
   620→    </div>
   621→  );
   622→
   623→  return (
   624→    <div className="flex h-full flex-col">
   625→      {/* Header */}
   626→      <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-3">
   627→        <div className="flex items-center gap-2">
   628→          <Settings2 className="h-4 w-4 text-zinc-500" />
   629→          <span className="text-sm font-semibold text-zinc-200">Properties</span>
   630→        </div>
   631→        <button
   632→          className="rounded-md p-1 text-zinc-500 hover:text-zinc-300 hover:bg-zinc-800 focus:outline-none"
   633→          onClick={() => setSelectedSectionId(null)}
   634→          aria-label="Close properties"
   635→        >
   636→          <X className="h-4 w-4" />
   637→        </button>
   638→      </div>
   639→
   640→      <ScrollArea className="flex-1">

   829→// ── Main VisualEditor Component ─────────────────────────────────────
   830→
   831→export function VisualEditor() {
   832→  const {
   833→    store,
   834→    setStore,
   835→    selectedSectionId,
   836→    setSelectedSectionId,
   837→    moveSection,
   838→    addSection,
   839→    removeSection,
   840→  } = useStoreEditor();
   841→
   842→  // Get the current page (homepage for now)
   843→  const currentPage = useMemo(
   844→    () => store?.pages.find((p) => p.isHomepage) ?? store?.pages[0],
   845→    [store?.pages]
   846→  );
   847→
   848→  const sections = currentPage?.sections ?? [];
   849→  const pageId = currentPage?.id ?? '';
   850→
   851→  const selectedSection = useMemo(
   852→    () => sections.find((s) => s.id === selectedSectionId) ?? null,
   853→    [sections, selectedSectionId]
   854→  );
   855→
   856→  // Drag end handler
   857→  const handleDragEnd = useCallback(
   858→    (event: DragEndEvent) => {
   859→      const { active, over } = event;
   860→      if (!over || active.id === over.id || !pageId) return;
   861→
   862→      const oldIndex = sections.findIndex((s) => s.id === active.id);
   863→      const newIndex = sections.findIndex((s) => s.id === over.id);
   864→
   865→      if (oldIndex !== -1 && newIndex !== -1 && oldIndex !== newIndex) {
   866→        moveSection(pageId, oldIndex, newIndex);
   867→      }
   868→    },
   869→    [sections, pageId, moveSection]
   870→  );
   871→
   872→  // Add section handler
   873→  const handleAddSection = useCallback(
   874→    (type: SectionType) => {
   875→      if (!pageId) return;
   876→      const newSection = createDefaultSection(type);
   877→      addSection(pageId, newSection);
   878→      setSelectedSectionId(newSection.id);
   879→      toast.success(`Added ${SECTION_TYPE_LABELS[type]} section`);
   880→    },
   881→    [pageId, addSection, setSelectedSectionId]
   882→  );
   883→
   884→  // Toggle visibility for section list items
   885→  const handleToggleVisibility = useCallback(
   886→    (sectionId: string) => {
   887→      if (!store) return;
   888→      const updatedStore = {
   889→        ...store,
   890→        updatedAt: new Date().toISOString(),
   891→        pages: store.pages.map((page) => ({
   892→          ...page,
   893→          sections: page.sections.map((s) =>
   894→            s.id === sectionId ? { ...s, visible: !s.visible } : s
   895→          ),
   896→        })),
   897→      };
   898→      setStore(updatedStore);
   899→    },
   900→    [store, setStore]
   901→  );
   902→
   903→  const handleDeleteSection = useCallback(
   904→    (sectionId: string) => {
   905→      if (!pageId) return;
   906→      const section = sections.find((s) => s.id === sectionId);
   907→      removeSection(pageId, sectionId);
   908→      if (selectedSectionId === sectionId) {
   909→        setSelectedSectionId(null);
   910→      }
   911→      if (section) {
   912→        toast.success(`Deleted ${SECTION_TYPE_LABELS[section.type]} section`);
   913→      }
   914→    },
   915→    [pageId, sections, selectedSectionId, removeSection, setSelectedSectionId]
   916→  );
   917→
   918→  if (!store || !currentPage) {
   919→    return (
   920→      <div className="flex h-full items-center justify-center bg-zinc-950">
   921→        <p className="text-sm text-zinc-600">No store loaded</p>
   922→      </div>
   923→    );
   924→  }
   925→
   926→  return (
   927→    <div className="flex h-full overflow-hidden bg-zinc-950">
   928→      {/* Section List Panel */}
   929→      <div
   930→        className={[
   931→          'flex flex-col border-r border-zinc-800 min-h-0',
   932→          selectedSection ? 'w-64 lg:w-72' : 'w-full',
   933→        ].join(' ')}
   934→      >
   935→        {/* Panel header */}
   936→        <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-3">
   937→          <h3 className="text-sm font-semibold text-zinc-200">Sections</h3>
   938→          <span className="text-xs text-zinc-600 tabular-nums">
   939→            {sections.length}
   940→          </span>
   941→        </div>
   942→
   943→        {/* Section list with DnD */}
   944→        <ScrollArea className="flex-1 min-h-0">
   945→          <div className="p-2">
   946→            <DndContext
   947→              collisionDetection={closestCenter}
   948→              onDragEnd={handleDragEnd}
   949→            >
   950→              <SortableContext
   951→                items={sections.map((s) => s.id)}
   952→                strategy={verticalListSortingStrategy}
   953→              >
   954→                <div className="space-y-0.5">
   955→                  {sections.map((section) => (
   956→                    <SortableSectionItem
   957→                      key={section.id}
   958→                      section={section}
   959→                      isSelected={section.id === selectedSectionId}
   960→                      onSelect={() =>
   961→                        setSelectedSectionId(
   962→                          section.id === selectedSectionId ? null : section.id
   963→                        )
   964→                      }
   965→                      onToggleVisibility={() =>
   966→                        handleToggleVisibility(section.id)
   967→                      }
   968→                      onDelete={() => handleDeleteSection(section.id)}
   969→                    />
   970→                  ))}
   971→                </div>
   972→              </SortableContext>
   973→            </DndContext>
   974→          </div>
   975→        </ScrollArea>
   976→
   977→        {/* Add section button */}
   978→        <div className="border-t border-zinc-800 p-2">
   979→          <AddSectionPopover onAdd={handleAddSection} />
   980→        </div>
   981→      </div>
   982→
   983→      {/* Properties Panel */}
   984→      {selectedSection && (
   985→        <div className="hidden md:flex w-72 lg:w-80 flex-shrink-0 border-r border-zinc-800">
   986→          <PropertiesPanel section={selectedSection} pageId={pageId} />
   987→        </div>
   988→      )}
   989→    </div>
   990→  );
   991→}
   992→
   993→export default VisualEditor;
