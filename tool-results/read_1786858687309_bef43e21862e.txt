    50→// ─── Error boundary for individual sections ──────────────────────────
    51→// Prevents a malformed section from crashing the entire page.
    52→
    53→interface SectionErrorBoundaryState { hasError: boolean; error: Error | null }
    54→class SectionErrorBoundary extends Component<
    55→  { children: React.ReactNode; sectionType: string },
    56→  SectionErrorBoundaryState
    57→> {
    58→  constructor(props: { children: React.ReactNode; sectionType: string }) {
    59→    super(props);
    60→    this.state = { hasError: false, error: null };
    61→  }
    62→  static getDerivedStateFromError(error: Error) {
    63→    return { hasError: true, error };
    64→  }
    65→  render() {
    66→    if (this.state.hasError) {
    67→      return (
    68→        <div className="flex items-center justify-center py-8 px-6 text-center">
    69→          <div className="max-w-sm">
    70→            <p className="text-sm font-medium" style={{ color: '#888' }}>
    71→              Section render error ({this.props.sectionType})
    72→            </p>
    73→            <p className="mt-1 text-xs" style={{ color: '#aaa' }}>
    74→              {this.state.error?.message || 'Unknown error'}
    75→            </p>
    76→          </div>
    77→        </div>
    78→      );
    79→    }
    80→    return this.props.children;
    81→  }
    82→}

    86→interface SectionWrapperProps {
    87→  section: Section;
    88→  theme: StoreTheme;
    89→  selectedSectionId?: string | null;
    90→  onSelectSection?: (sectionId: string | null) => void;
    91→  children: React.ReactNode;
    92→}
    93→
    94→function SectionWrapper({
    95→  section,
    96→  theme,
    97→  selectedSectionId,
    98→  onSelectSection,
    99→  children,
   100→}: SectionWrapperProps) {
   101→  const isSelected = selectedSectionId === section.id;
   102→  const style = section.style;
   103→
   104→  const sectionBg = style.backgroundColor || theme.colors.background;
   105→  const sectionText = style.textColor || theme.colors.text;
   106→
   107→  const bgImage = style.backgroundImage
   108→    ? `url(${style.backgroundImage})`
   109→    : undefined;
   110→
   111→  const outerClasses = [
   112→    'relative transition-all duration-200 cursor-pointer',
   113→    pyClass(style.paddingY),
   114→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   115→  ].join(' ');
   116→
   117→  return (
   118→    <div
   119→      className={outerClasses}
   120→      style={{
   121→        backgroundColor: sectionBg,
   122→        color: sectionText,
   123→        backgroundImage: bgImage,
   124→        backgroundSize: 'cover',
   125→        backgroundPosition: 'center',
   126→      }}
   127→      onClick={(e) => {
   128→        e.stopPropagation();
   129→        onSelectSection?.(isSelected ? null : section.id);
   130→      }}
   131→      data-section-id={section.id}
   132→      data-section-type={section.type}
   133→    >
   134→      {/* Overlay for background images */}
   135→      {style.backgroundImage && style.overlay && (
   136→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   137→      )}
   138→      <div
   139→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   140→      >
   141→        {/* Selected section label */}
   142→        {isSelected && (
   143→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   144→            {section.type.replace(/-/g, ' ')}
   145→          </div>
   146→        )}
   147→        {children}
   148→      </div>
   149→    </div>
   150→  );
   151→}

   153→// ─── Product Card (shared) ─────────────────────────────────────────────

   155→function ProductCard({
   156→  product,
   157→  theme,
   158→  showAddToCart,
   159→  borderRadius,
   160→  buttonBgOverride,
   161→  buttonTextOverride,
   162→  onViewProduct,
   163→}: {
   164→  product: StoreProduct;
   165→  theme: StoreTheme;
   166→  showAddToCart: boolean;
   167→  borderRadius: string;
   168→  buttonBgOverride?: string;
   169→  buttonTextOverride?: string;
   170→  onViewProduct?: (productId: string) => void;
   171→}) {
   172→  const [hovered, setHovered] = useState(false);
   173→  const [addedFeedback, setAddedFeedback] = useState(false);
   174→  const addToCart = useCartStore((s) => s.addItem);
   175→  const imgColor = stringToColor(product.id, theme);
   176→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   177→
   178→  const handleAddToCart = (e: React.MouseEvent) => {
   179→    e.stopPropagation();
   180→    addToCart(product);
   181→    setAddedFeedback(true);
   182→    setTimeout(() => setAddedFeedback(false), 1200);
   183→  };
   184→
   185→  return (
   186→    <div
   187→      className={`${borderRadius} ${onViewProduct ? 'cursor-pointer' : ''} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   188→      style={{
   189→        borderColor: theme.colors.border,
   190→      }}
   191→      onMouseEnter={() => setHovered(true)}
   192→      onMouseLeave={() => setHovered(false)}
   193→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   194→      role={onViewProduct ? 'button' : undefined}
   195→      tabIndex={onViewProduct ? 0 : undefined}
   196→    >
   197→      {/* Product image with real image + fallback */}
   198→      <div
   199→        className="relative aspect-square w-full overflow-hidden"
   200→        style={{ backgroundColor: imgColor }}
   201→      >
   202→        <StoreImage
   203→          src={product.images[0] || ''}
   204→          alt={product.name}
   205→          fallbackColor={imgColor}
   206→          className="absolute inset-0 h-full w-full object-cover transition-transform duration-300 group-hover:scale-105"
   207→          iconSize="lg"
   208→        />
   209→        {/* Out of stock overlay */}
   210→        {!product.inStock && (
   211→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   212→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   213→              Sold Out
   214→            </span>
   215→          </div>
   216→        )}
   217→        {/* Discount badge */}
   218→        {hasDiscount && product.compareAtPrice && (
   219→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   220→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   221→          </div>
   222→        )}
   223→      </div>
   224→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   225→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   226→        {product.category && (
   227→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   228→            {product.category}
   229→          </p>
   230→        )}
   231→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   232→          {product.name}
   233→        </h3>
   234→        <div className="mt-1.5 flex items-center gap-2">
   235→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   236→            {formatPrice(product.price)}
   237→          </span>
   238→          {hasDiscount && product.compareAtPrice && (
   239→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   240→              {formatPrice(product.compareAtPrice)}
   241→            </span>
   242→          )}
   243→        </div>
   244→        {showAddToCart && product.inStock && (
   245→          <button
   246→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 cursor-pointer ${
   247→              hovered
   248→                ? 'opacity-100 translate-y-0'
   249→                : 'opacity-0 translate-y-1 pointer-events-none'
   250→            }`}
   251→            style={{
   252→              backgroundColor: addedFeedback ? '#16a34a' : (buttonBgOverride || theme.colors.primary),
   253→              color: addedFeedback ? '#ffffff' : (buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary)),
   254→            }}
   255→            onClick={handleAddToCart}
   256→          >
   257→            {addedFeedback ? 'Added!' : 'Add to Cart'}
   258→          </button>
   259→        )}
   260→      </div>
   261→    </div>
   262→  );
   263→}

   360→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   361→  const content = section.content as unknown as HeroContent;
   362→  const style = section.style;
   363→  const isSelected = selectedSectionId === section.id;
   364→
   365→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   366→  const alignMap = {
   367→    left: 'items-start text-left',
   368→    center: 'items-center text-center',
   369→    right: 'items-end text-right',
   370→  };
   371→
   372→  // Only apply theme gradient as default background when no custom color or image is set.
   373→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   374→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   375→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   376→    : undefined;
   377→
   378→  const outerClasses = [
   379→    'relative flex transition-all duration-200 cursor-pointer',
   380→    heightMap[content.height] || heightMap.md,
   381→    alignMap[content.alignment],
   382→    pyClass(style.paddingY),
   383→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   384→  ].join(' ');
   385→
   386→  return (
   387→    <div
   388→      className={outerClasses}
   389→      style={{
   390→        backgroundColor: style.backgroundColor || theme.colors.primary,
   391→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   392→        backgroundSize: 'cover',
   393→        backgroundPosition: 'center',
   394→        color: style.textColor || '#ffffff',
   395→      }}
   396→      onClick={(e) => {
   397→        e.stopPropagation();
   398→        onSelectSection?.(isSelected ? null : section.id);
   399→      }}
   400→      data-section-id={section.id}
   401→      data-section-type={section.type}
   402→    >
   403→      {/* Overlay */}
   404→      {style.backgroundImage && style.overlay && (
   405→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   406→      )}
   407→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   408→      {!style.backgroundImage && !style.backgroundColor && (
   409→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   410→      )}
   411→      {!style.backgroundImage && !style.backgroundColor && (
   412→        <>
   413→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   414→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />
   415→        </>
   416→      )}
   417→
   418→      <div className={`relative z-10 mx-auto w-full ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}>
   419→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   420→          {content.headline}
   421→        </h1>
   422→        {content.subheadline && (
   423→          <p className="mt-4 max-w-xl text-base opacity-80 sm:text-lg md:text-xl">
   424→            {content.subheadline}
   425→          </p>
   426→        )}
   427→        {content.ctaText && (
   428→          <button className="mt-8 inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-transform hover:scale-105 sm:text-base"
   429→            style={{
   430→              backgroundColor: section.style.buttonBackgroundColor || '#ffffff',
   431→              color: section.style.buttonTextColor || contrastTextColor(section.style.buttonBackgroundColor || '#ffffff'),
   432→            }}>
   433→            {content.ctaText}
   434→            <ArrowRight className="h-4 w-4" />
   435→          </button>
   436→        )}
   437→      </div>
   438→    </div>
   439→  );
   440→}

   444→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   445→  const content = section.content as unknown as FeaturedProductsContent;
   446→  const borderRadius = borderRadiusClass(theme.borderRadius);
   447→  const featuredProducts = useMemo(
   448→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   449→    [content.productIds, products],
   450→  );
   451→
   452→  return (
   453→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   454→      {content.headline && (
   455→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   456→          {content.headline}
   457→        </h2>
   458→      )}
   459→      {content.subtitle && (
   460→        <p className="mb-8 text-sm opacity-65">
   461→          {content.subtitle}
   462→        </p>
   463→      )}
   464→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   465→        {featuredProducts.map((product) => (
   466→          <ProductCard
   467→            key={product.id}
   468→            product={product}
   469→            theme={theme}
   470→            showAddToCart={content.showAddToCart && !forceHideAddToCart}
   471→            borderRadius={borderRadius}
   472→            buttonBgOverride={section.style.buttonBackgroundColor}
   473→            buttonTextOverride={section.style.buttonTextColor}
   474→            onViewProduct={onViewProduct}
   475→          />
   476→        ))}
   477→      </div>
   478→    </SectionWrapper>
   479→  );
   480→}

   484→export function ProductGridSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   485→  const content = section.content as unknown as ProductGridContent;
   486→  const borderRadius = borderRadiusClass(theme.borderRadius);
   487→
   488→  const filteredProducts = useMemo(() => {
   489→    let prods = products;
   490→    if (content.filterByCategory) {
   491→      prods = prods.filter((p) => p.category === content.filterByCategory);
   492→    }
   493→    return prods;
   494→  }, [products, content.filterByCategory]);
   495→
   496→  return (
   497→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   498→      {content.headline && (
   499→        <h2 className="mb-8 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   500→          {content.headline}
   501→        </h2>
   502→      )}
   503→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   504→        {filteredProducts.map((product) => (
   505→          <ProductCard
   506→            key={product.id}
   507→            product={product}
   508→            theme={theme}
   509→            showAddToCart={content.showAddToCart && !forceHideAddToCart}
   510→            borderRadius={borderRadius}
   511→            buttonBgOverride={section.style.buttonBackgroundColor}
   512→            buttonTextOverride={section.style.buttonTextColor}
   513→            onViewProduct={onViewProduct}
   514→          />
   515→        ))}
   516→      </div>
   517→      {filteredProducts.length === 0 && (
   518→        <div className="py-16 text-center opacity-65">
   519→          <p className="text-sm">No products found.</p>
   520→        </div>
   521→      )}
   522→    </SectionWrapper>
   523→  );
   524→}

   528→export function TextBannerSection({ section, theme, selectedSectionId, onSelectSection }: SectionRendererProps) {
   529→  const content = section.content as unknown as TextBannerContent;
   530→  const alignMap = {
   531→    left: 'text-left',
   532→    center: 'text-center',
   533→    right: 'text-right',
   534→  };
   535→  const sizeMap = {
   536→    sm: { headline: 'text-xl sm:text-2xl', body: 'text-sm' },
   537→    md: { headline: 'text-2xl sm:text-3xl md:text-4xl', body: 'text-base' },
   538→    lg: { headline: 'text-3xl sm:text-4xl md:text-5xl lg:text-6xl', body: 'text-lg' },
   539→  };
   540→  const sizes = sizeMap[content.size] || sizeMap.md;
   541→
   542→  return (
   543→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   544→      <div className={`${alignMap[content.alignment]}`}>
   545→        <h2 className={`font-bold leading-tight tracking-tight ${sizes.headline}`} style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   546→          {content.headline}
   547→        </h2>
   548→        {content.body && (
   549→          <p className={`mt-3 max-w-2xl ${sizes.body} opacity-65`}>
   550→            {content.body}
   551→          </p>
   552→        )}
   553→      </div>
   554→    </SectionWrapper>
   555→  );

  1033→// ─── Main Section Router ────────────────────────────────────────────────
  1034→
  1035→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1036→  const { section } = props;
  1037→
  1038→  if (!section.visible) return null;
  1039→
  1040→  switch (section.type) {
  1041→    case 'header':
  1042→      return <HeaderSection {...props} />;
  1043→    case 'hero':
  1044→      return <HeroSection {...props} />;
  1045→    case 'featured-products':
  1046→      return <FeaturedProductsSection {...props} />;
  1047→    case 'product-grid':
  1048→      return <ProductGridSection {...props} />;
  1049→    case 'text-banner':
  1050→      return <TextBannerSection {...props} />;
  1051→    case 'image-gallery':
  1052→      return <ImageGallerySection {...props} />;
  1053→    case 'testimonials':
  1054→      return <TestimonialsSection {...props} />;
  1055→    case 'newsletter':
  1056→      return <NewsletterSection {...props} />;
  1057→    case 'faq':
  1058→      return <FAQSection {...props} />;
  1059→    case 'cta':
  1060→      return <CTASection {...props} />;
  1061→    case 'categories':
  1062→      return <CategoriesSection {...props} />;
  1063→    case 'footer':
  1064→      return <FooterSection {...props} />;
  1065→    case 'rich-text':
  1066→      return <RichTextSection {...props} />;
  1067→    case 'spacer':
  1068→      return <SpacerSection {...props} />;
  1069→    case 'divider':
  1070→      return <DividerSection {...props} />;
  1071→    default:
  1072→      return null;
  1073→  }
  1074→}
