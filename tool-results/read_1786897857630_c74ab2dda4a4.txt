    66→// ─── Background Image Enrichment (lazy, non-blocking) ─────
    67→// Triggered after store generation completes. Fetches real product
    68→// images sequentially without blocking the user or exhausting rate limits.
    69→
    70→function triggerBackgroundImageEnrichment(store: Store) {
    71→  if (!store.products || store.products.length === 0) return
    72→
    73→  // Only enrich products that still have AI-hallucinated Unsplash URLs
    74→  // (Real enriched URLs from the image search service will have different patterns)
    75→  const needsEnrichment = store.products.filter(p =>
    76→    p.images.length > 0 && p.images[0].includes('unsplash.com/photo-')
    77→  )
    78→
    79→  if (needsEnrichment.length === 0) return
    80→
    81→  console.log(`[Storqly] Background image enrichment: ${needsEnrichment.length}/${store.products.length} products need enrichment`)
    82→
    83→  fetch('/api/store/enrich-images', {
    84→    method: 'POST',
    85→    headers: { 'Content-Type': 'application/json' },
    86→    body: JSON.stringify({
    87→      products: needsEnrichment.map(p => ({
    88→        id: p.id,
    89→        name: p.name,
    90→        images: [...p.images],
    91→        category: p.category,
    92→        description: p.description,
    93→      })),
    94→      storeName: store.name,
    95→    }),
    96→  })
    97→    .then(res => res.json())
    98→    .then(data => {
    99→      if (data.enriched > 0) {
   100→        console.log(`[Storqly] Background enrichment complete: ${data.enriched} enriched, ${data.kept} kept, ${data.failed} failed in ${data.latencyMs}ms`)
   101→        // Update product images in the store via the Zustand store
   102→        // The enrich-images endpoint mutated the products array in-place,
   103→        // but we need to update the Zustand store to trigger re-renders.
   104→        // Dynamic import to avoid circular dependency
   105→        import('@/lib/store').then(({ useStoreEditor }) => {
   106→          const currentStore = useStoreEditor.getState().store
   107→          if (currentStore) {
   108→            const updatedProducts = currentStore.products.map(prod => {
   109→              const enriched = data.products?.find((ep: { id: string; images: string[] }) => ep.id === prod.id)
   110→              if (enriched && enriched.images[0] !== prod.images[0]) {
   111→                return { ...prod, images: enriched.images }
   112→              }
   113→              return prod
   114→            })
   115→            useStoreEditor.getState().setStore({ ...currentStore, products: updatedProducts })
   116→          }
   117→        })
   118→      }
   119→    })
   120→    .catch(err => {
   121→      // Non-fatal — products still show AI placeholder images
   122→      console.warn('[Storqly] Background enrichment failed (non-fatal):', err)
   123→    })
   124→}

   182→  const abortRef = useRef<AbortController | null>(null)

   184→  const {

   205→  const clearTimers = useCallback(() => {

   282→
   283→    // Create abort controller for this request
   284→    const controller = new AbortController()
   285→    abortRef.current = controller
   286→
   287→    // Hard timeout: 120 seconds — no request should ever take longer than this.
   288→    // If the server is hung or the proxy drops the connection, the user gets a
   289→    // clear message instead of waiting indefinitely.
   290→    const HARD_TIMEOUT_MS = 120_000
   291→    let timedOut = false
   292→    const timeoutId = setTimeout(() => {
   293→      timedOut = true
   294→      controller.abort()

   298→    // Elapsed time counter
   299→    elapsedTimerRef.current = setInterval(() => {
   300→      setElapsedSeconds((prev) => prev + 1)

   302→
   303→    // Fallback progress cycling (used only if SSE progress events stop)
   304→    let fallbackIdx = 0
   305→    let receivedSSEProgress = false
   306→    progressTimerRef.current = setInterval(() => {
   307→      if (!receivedSSEProgress) {
   308→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   309→        setGenerationStatus(progressMessages[fallbackIdx])
   310→      }
   311→    }, 3000)
   312→
   313→    // ── Helper: clean up on any error path (no throw, no crash overlay) ──
   314→    const finishWithError = (message: string) => {
   315→      clearTimeout(timeoutId)
   316→      clearTimers()
   317→      setError(message)
   318→      setIsGenerating(false)
   319→      setGenerationStatus('')
   320→      setElapsedSeconds(0)
   321→      toast.error('Store generation failed', { description: message })
   322→    }
   323→
   324→    const finishOk = () => {
   325→      clearTimeout(timeoutId)
   326→      clearTimers()
   327→      setIsGenerating(false)
   328→      setGenerationStatus('')
   329→      setElapsedSeconds(0)
   330→    }
   331→
   332→    try {
   333→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   334→
   335→      // ── Fetch with auto-retry for transient gateway errors (502/503/504) ──
   336→      const RETRYABLE_STATUSES = [502, 503, 504]
   337→      let res: Response | undefined
   338→      let retryCount = 0
   339→      const maxRetries = 2
   340→
   341→      while (retryCount <= maxRetries) {
   342→        try {
   343→          res = await fetch('/api/store/generate', {
   344→            method: 'POST',
   345→            headers: { 'Content-Type': 'application/json' },
   346→            body: JSON.stringify({ prompt: trimmed }),
   347→            signal: controller.signal,
   348→          })
   349→
   350→          // If the status is retryable, wait and loop again
   351→          if (!res.ok && RETRYABLE_STATUSES.includes(res.status) && retryCount < maxRetries) {
   352→            retryCount++
   353→            console.warn(`[Storqly] Gateway error ${res.status}, retrying (${retryCount}/${maxRetries}) in 3s...`)
   354→            setGenerationStatus('Connection issue — retrying...')
   355→            await new Promise(r => setTimeout(r, 3000))
   356→            continue
   357→          }
   358→          break // Success or non-retryable error — exit loop
   359→        } catch (fetchErr: unknown) {
   360→          // Network-level error (DNS failure, connection refused, etc.)
   361→          if (fetchErr instanceof TypeError && retryCount < maxRetries) {
   362→            retryCount++
   363→            console.warn(`[Storqly] Network error, retrying (${retryCount}/${maxRetries}) in 3s...`)
   364→            setGenerationStatus('Connection issue — retrying...')
   365→            await new Promise(r => setTimeout(r, 3000))
   366→            continue
   367→          }
   368→          // Out of retries or not a network error — handle gracefully
   369→          const msg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr)
   370→          if (controller.signal.aborted) {
   371→            finishWithError('Generation was cancelled.')
   372→          } else {
   373→            console.warn('[Storqly] Network error after retries:', msg)
   374→            finishWithError('Could not connect to the server. Please check your connection and try again.')
   375→          }
   376→          return
   377→        }
   378→      }
   379→
   380→      if (!res!.ok) {
   381→        let errorMsg = `Server error (${res!.status})`
   382→        try {
   383→          const errorData = await res!.json()
   384→          errorMsg = errorData.error || errorMsg
   385→        } catch { /* ignore parse error */ }
   386→        console.warn('[Storqly] HTTP error:', errorMsg)
   387→        finishWithError(errorMsg)
   388→        return
   389→      }
   390→
   391→      // ── Consume SSE stream ──
   392→      const reader = res.body?.getReader()
   393→      if (!reader) {
   394→        finishWithError('No response stream received from server.')
   395→        return
   396→      }
   397→
   398→      const decoder = new TextDecoder()
   399→      let buffer = ''
   400→      let resolved = false
   401→
   402→      while (true) {
   403→        const { done, value } = await reader.read()
   404→        if (done) break
   405→
   406→        buffer += decoder.decode(value, { stream: true })
   407→
   408→        // Parse SSE events from buffer
   409→        const lines = buffer.split('\n')
   410→        buffer = lines.pop() || ''
   411→
   412→        let currentEvent = ''
   413→        for (const line of lines) {
   414→          if (line.startsWith(':')) continue
   415→          if (line.startsWith('event: ')) {
   416→            currentEvent = line.slice(7).trim()
   417→          } else if (line.startsWith('data: ')) {
   418→            const dataStr = line.slice(6)
   419→
   420→            if (currentEvent === 'progress') {
   421→              receivedSSEProgress = true
   422→              try {
   423→                const data = JSON.parse(dataStr)
   424→                setGenerationStatus(data.message || 'Processing...')
   425→              } catch { /* ignore */ }
   426→            } else if (currentEvent === 'result') {
   427→              try {
   428→                const data = JSON.parse(dataStr)
   429→                if (!data.store) {
   430→                  console.warn('[Storqly] Result event missing store data')
   431→                  finishWithError('The AI response was missing store data.')
   432→                  return
   433→                }
   434→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   435→                resolved = true
   436→
   437→                if (data._isFallback) {
   438→                  // Differentiate "all providers down" from other fallback reasons
   439→                  const reason = data._fallbackReason || 'AI generation failed'
   440→                  const isProviderFailure = reason.includes('All') || reason.includes('429') || reason.includes('403') || reason.includes('404') || reason.includes('providers failed')
   441→                  toast.warning(
   442→                    isProviderFailure
   443→                      ? 'All AI providers are currently unavailable. You are viewing a starter template. Try again later.'
   444→                      : 'AI service returned invalid data — showing starter template. Try again.',
   445→                    { duration: 8000 }
   446→                  )
   447→                  setStoreWithFallback(data.store, true, reason)
   448→                } else {
   449→                  setStore(data.store)
   450→                  // Trigger lazy background image enrichment (non-blocking)
   451→                  triggerBackgroundImageEnrichment(data.store)
   452→                }
   453→
   454→                // Soft cap toast
   455→                if (data._productCapHit) {
   456→                  toast.info(
   457→                    `Generated ${data._generatedCount} products — for larger catalogs, you can add more via the chat editor.`,
   458→                    { duration: 8000 }
   459→                  )
   460→                }
   461→
   462→                finishOk()
   463→                return
   464→              } catch (parseErr) {
   465→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   466→                console.warn('[Storqly] Failed to parse result event:', msg)
   467→                finishWithError(msg)
   468→                return
   469→              }
   470→            } else if (currentEvent === 'error') {
   471→              try {
   472→                const data = JSON.parse(dataStr)
   473→                const serverMsg = data.message || 'Generation failed'
   474→                console.warn('[Storqly] Server error event:', serverMsg)
   475→                finishWithError(serverMsg)
   476→                return
   477→              } catch {
   478→                finishWithError('Generation failed')
   479→                return
   480→              }
   481→            }
   482→            currentEvent = ''
   483→          }
   484→        }
   485→      }
   486→
   487→      if (!resolved) {
   488→        console.warn('[Storqly] Stream ended without a result event')
   489→        finishWithError('Stream ended without a result. The server may have disconnected.')
   490→      }
   491→    } catch (err: unknown) {
   492→      // Last-resort safety net — should rarely be reached now
   493→      clearTimeout(timeoutId)
   494→      clearTimers()
   495→      let message = 'Something went wrong. Please try again.'
   496→      if (err instanceof Error) {
   497→        if (err.name === 'AbortError') {
   498→          message = timedOut
   499→            ? 'Generation timed out after 2 minutes. The server may be overloaded — please try again.'
   500→            : 'Generation was cancelled.'
   501→          console.warn(`[Storqly] Request aborted: ${message}`)
   502→        } else {
   503→          console.warn('[Storqly] Unexpected error (caught by safety net):', err.message)
   504→          message = err.message
   505→        }
   506→      }
   507→      setError(message)
   508→      setIsGenerating(false)
   509→      setGenerationStatus('')
   510→      setElapsedSeconds(0)
   511→      toast.error('Store generation failed', { description: message })
   512→    }
   513→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])
   514→
   515→  // Cleanup on unmount
   516→  useEffect(() => {
   517→    return () => {
   518→      clearTimers()
   519→      abortRef.current?.abort()
   520→    }
   521→  }, [clearTimers])

   525→      e.preventDefault()

   527→    }
   528→  }

   533→    const s = seconds % 60
   534→    return `${m}:${s.toString().padStart(2, '0')}`
   535→  }

   543→            Storqly
   544→          </span>
   545→          <div className="hidden sm:flex items-center gap-4">
   546→            <span className="text-sm text-zinc-500">
   547→              AI-Powered Store Builder
   548→            </span>
   549→            <AuthButton onSignIn={() => setAuthOpen(true)} />
   550→          </div>

   553→          </div>
   554→        </nav>
   555→      </header>
   556→
   557→      <AuthModal open={authOpen} onOpenChange={setAuthOpen} />
   558→
   559→      {/* ── Hero Section ── */}
   560→      <section className="relative flex flex-1 flex-col items-center justify-center px-5 pt-32 pb-20 sm:pt-40 sm:pb-28">

   562→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />

   991→// ─── Fallback Banner ──────────────────────────────────────────────
   992→// Shows when: (1) AI returned a fallback template, or
   993→// (2) the store ended up in a broken/incomplete state (e.g. server
   994→//     crashed mid-generation leaving only placeholder sections).
   995→
   996→function FallbackBanner() {
   997→  const reset = useStoreEditor((s) => s.reset)
   998→  const isFallbackStore = useStoreEditor((s) => s.isFallbackStore)
   999→  const fallbackReason = useStoreEditor((s) => s.fallbackReason)
  1000→  const [dismissed, setDismissed] = useState(false)
  1001→
  1002→  if (dismissed || !isFallbackStore) return null
  1003→
  1004→  const isIncomplete = fallbackReason.includes('interrupted') || fallbackReason.includes('incomplete')
  1005→  const isProviderDown = fallbackReason.includes('All') || fallbackReason.includes('429') || fallbackReason.includes('403') || fallbackReason.includes('404') || fallbackReason.includes('providers failed')
  1006→
  1007→  // Determine message based on failure type
  1008→  let bannerMessage: string
  1009→  if (isIncomplete) {
  1010→    bannerMessage = 'Generation was interrupted — your store is incomplete. Please try regenerating.'
  1011→  } else if (isProviderDown) {
  1012→    bannerMessage = 'All AI providers are currently unavailable — you are viewing a starter template.'
  1013→  } else {
  1014→    bannerMessage = 'AI couldn\'t generate a custom store — you\'re viewing a starter template.'
  1015→  }
  1016→
  1017→  return (
  1018→    <div className={`flex items-center gap-3 border-b px-4 py-2.5 ${isIncomplete ? 'border-red-500/30 bg-red-500/10' : 'border-amber-500/30 bg-amber-500/10'}`}>
  1019→      {isIncomplete ? (
  1020→        <AlertCircle className="h-4 w-4 shrink-0 text-red-400" />
  1021→      ) : (
  1022→        <AlertTriangle className="h-4 w-4 shrink-0 text-amber-400" />
  1023→      )}
  1024→      <p className={`flex-1 text-sm ${isIncomplete ? 'text-red-200' : 'text-amber-200'}`}>
  1025→        {bannerMessage}
  1026→        <span className="hidden sm:inline"> Edit it manually or try regenerating later.</span>
  1027→      </p>
  1028→      <div className="flex items-center gap-2">
  1029→        <Button
  1030→          size="sm"
  1031→          className={`h-7 gap-1.5 rounded-lg px-3 text-xs font-medium ${isIncomplete ? 'bg-red-500/20 text-red-200 hover:bg-red-500/30' : 'bg-amber-500/20 text-amber-200 hover:bg-amber-500/30'}`}
  1032→          onClick={() => { reset() }}
  1033→        >
  1034→          <RotateCcw className="h-3 w-3" />
  1035→          Regenerate with AI
  1036→        </Button>
  1037→        <button
  1038→          onClick={() => setDismissed(true)}
  1039→          className="text-zinc-500 hover:text-zinc-300"
  1040→          aria-label="Dismiss"
  1041→        >
  1042→          <X className="h-3.5 w-3.5" />
  1043→        </button>
  1044→      </div>
  1045→    </div>
  1046→  )
  1047→}

  1365→function PublishedStoreViewer({ slug }: { slug: string }) {
  1366→  const [store, setStore] = useState<Store | null>(null)
  1367→  const [loading, setLoading] = useState(true)
  1368→  const [error, setError] = useState<string | null>(null)
  1369→
  1370→  useEffect(() => {
  1371→    let cancelled = false
  1372→    async function load() {
  1373→      try {
  1374→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
  1375→        if (!res.ok) {
  1376→          const data = await res.json().catch(() => ({}))
  1377→          throw new Error(data.error || `Store not found (${res.status})`)
  1378→        }
  1379→        const data = await res.json()
  1380→        if (!cancelled) setStore(data.store)
  1381→      } catch (err: unknown) {
  1382→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
  1383→      } finally {
  1384→        if (!cancelled) setLoading(false)
  1385→      }
  1386→    }
  1387→    load()
  1388→    return () => { cancelled = true }
  1389→  }, [slug])
  1390→
  1391→  if (loading) {
  1392→    return (
  1393→      <div className="flex min-h-screen items-center justify-center bg-white">
  1394→        <div className="text-center">
  1395→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
  1396→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
  1397→        </div>
  1398→      </div>
  1399→    )
  1400→  }
  1401→
  1402→  if (error || !store) {
  1403→    return (
  1404→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
  1405→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
  1406→          <AlertCircle className="h-8 w-8 text-red-400" />
  1407→        </div>
  1408→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
  1409→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
  1410→          {error || 'This store could not be found or has not been published.'}
  1411→        </p>
  1412→        <Button
  1413→          variant="outline"
  1414→          onClick={() => window.location.href = '/'}
  1415→          className="gap-2"
  1416→        >
  1417→          <ArrowLeft className="h-4 w-4" />
  1418→          Back to Storqly
  1419→        </Button>
  1420→      </div>
  1421→    )
  1422→  }
  1423→
  1424→  return (
  1425→    <>
  1426→      <StoreRenderer store={store} />
  1427→      <Toaster />
  1428→      {/* Minimal footer for published stores */}

  1430→        <a
  1431→          href="/"
  1432→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
  1433→        >
  1434→          <Sparkles className="h-3 w-3" />
  1435→          Built with Storqly
  1436→        </a>
  1437→      </div>
  1438→    </>
  1439→  )
