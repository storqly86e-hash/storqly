   396→   396→function ContentFieldRenderer({
   397→   397→  keyName,
   398→   398→  value,
   399→   399→  onChange,
   400→   400→  depth = 0,
   401→   401→}: {
   402→   402→  keyName: string;
   403→   403→  value: unknown;
   404→   404→  onChange: (key: string, value: unknown) => void;
   405→   405→  depth?: number;
   406→   406→}) {
   407→   407→  const label = keyName
   408→   408→    .replace(/([A-Z])/g, ' $1')
   409→   409→    .replace(/^./, (s) => s.toUpperCase())
   410→   410→    .trim();
   411→   411→
   412→   412→  if (value === null || value === undefined) {
   413→   413→    return null;
   414→   414→  }
   415→   415→
   416→   416→  if (typeof value === 'string') {
   417→   417→    if (keyName === 'html' && value.includes('<')) {
   418→   418→      return (
   419→   419→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   420→   420→          <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   421→   421→          <Textarea
   422→   422→            value={value}
   423→   423→            onChange={(e) => onChange(keyName, e.target.value)}
   424→   424→            rows={4}
   425→   425→            className="bg-zinc-800 border-zinc-700 text-zinc-100 text-xs font-mono resize-none"
   426→   426→          />
   427→   427→        </div>
   428→   428→      );
   429→   429→    }
   430→   430→    return (
   431→   431→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   432→   432→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   433→   433→        <Input
   434→   434→          value={value}
   435→   435→          onChange={(e) => onChange(keyName, e.target.value)}
   436→   436→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   437→   437→        />
   438→   438→      </div>
   439→   439→    );
   440→   440→  }
   441→   441→
   442→   442→  if (typeof value === 'number') {
   443→   443→    return (
   444→   444→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   445→   445→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   446→   446→        <Input
   447→   447→          type="number"
   448→   448→          value={value}
   449→   449→          onChange={(e) => onChange(keyName, Number(e.target.value))}
   450→   450→          className="bg-zinc-800 border-zinc-700 text-zinc-100 h-8 text-sm"
   451→   451→        />
   452→   452→      </div>
   453→   453→    );
   454→   454→  }
   455→   455→
   456→   456→  if (typeof value === 'boolean') {
   457→   457→    return (
   458→   458→      <div
   459→   459→        key={keyName}
   460→   460→        className={`flex items-center justify-between ${depth > 0 ? 'ml-4' : ''}`}
   461→   461→      >
   462→   462→        <Label className="text-xs text-zinc-400">{label}</Label>
   463→   463→        <Switch
   464→   464→          checked={value}
   465→   465→          onCheckedChange={(checked) => onChange(keyName, checked)}
   466→   466→        />
   467→   467→      </div>
   468→   468→    );
   469→   469→  }
   470→   470→
   471→   471→  if (Array.isArray(value)) {
   472→   472→    return (
   473→   473→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   474→   474→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   475→   475→        <div className="rounded-lg bg-zinc-800/60 border border-zinc-700/50 px-3 py-2 text-xs text-zinc-500">
   476→   476→          {value.length} item{value.length !== 1 ? 's' : ''}
   477→   477→        </div>
   478→   478→      </div>
   479→   479→    );
   480→   480→  }
   481→   481→
   482→   482→  if (typeof value === 'object') {
   483→   483→    const obj = value as Record<string, unknown>;
   484→   484→    const keys = Object.keys(obj);
   485→   485→
   486→   486→    const allPrimitive = keys.every(
   487→   487→      (k) =>
   488→   488→        typeof obj[k] === 'string' ||
   489→   489→        typeof obj[k] === 'number' ||
   490→   490→        typeof obj[k] === 'boolean' ||
   491→   491→        obj[k] === null ||
   492→   492→        obj[k] === undefined
   493→   493→    );
   494→   494→
   495→   495→    if (allPrimitive && keys.length > 0) {
   496→   496→      return (
   497→   497→        <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   498→   498→          <p className="text-xs font-medium text-zinc-300 mb-2 mt-1">{label}</p>
   499→   499→          <div className="space-y-3 border-l border-zinc-700/50 pl-3 ml-1">
   500→   500→            {keys.map((k) => (
   501→   501→              <ContentFieldRenderer
   502→   502→                key={k}
   503→   503→                keyName={k}
   504→   504→                value={obj[k]}
   505→   505→                onChange={(subKey, subVal) => {
   506→   506→                  onChange(keyName, { ...obj, [subKey]: subVal });
   507→   507→                }}
   508→   508→                depth={depth + 1}
   509→   509→              />
   510→   510→            ))}
   511→   511→          </div>
   512→   512→        </div>
   513→   513→      );
   514→   514→    }
   515→   515→
   516→   516→    return (
   517→   517→      <div key={keyName} className={depth > 0 ? 'ml-4' : ''}>
   518→   518→        <Label className="text-xs text-zinc-400 mb-1.5 block">{label}</Label>
   519→   519→        <Textarea
   520→   520→          value={JSON.stringify(value, null, 2)}
   521→   521→          readOnly
   522→   522→          rows={3}
   523→   523→          className="bg-zinc-800/60 border-zinc-700/50 text-zinc-500 text-xs font-mono resize-none"
   524→   524→        />
   525→   525→      </div>
   526→   526→    );
   527→   527→  }
   528→   528→
   529→   529→  return null;
   530→   530→}

   831→   831→export function VisualEditor() {
   832→   832→  const {
   833→   833→    store,
   834→   834→    setStore,
   835→   835→    selectedSectionId,
   836→   836→    setSelectedSectionId,
   837→   837→    moveSection,
   838→   838→    addSection,
   839→   839→    removeSection,
   840→   840→  } = useStoreEditor();
   841→   841→
   842→   842→  // Get the current page (homepage for now)
   843→   843→  const currentPage = useMemo(
   844→   844→    () => store?.pages.find((p) => p.isHomepage) ?? store?.pages[0],
   845→   845→    [store?.pages]
   846→   846→  );
   847→   847→
   848→   848→  const sections = currentPage?.sections ?? [];
   849→   849→  const pageId = currentPage?.id ?? '';
   850→   850→
   851→   851→  const selectedSection = useMemo(
   852→   852→    () => sections.find((s) => s.id === selectedSectionId) ?? null,
   853→   853→    [sections, selectedSectionId]
   854→   854→  );
   855→   855→
   856→   856→  // Drag end handler
   857→   857→  const handleDragEnd = useCallback(
   858→   858→    (event: DragEndEvent) => {
   859→   859→      const { active, over } = event;
   860→   860→      if (!over || active.id === over.id || !pageId) return;
   861→   861→
   862→   862→      const oldIndex = sections.findIndex((s) => s.id === active.id);
   863→   863→      const newIndex = sections.findIndex((s) => s.id === over.id);
   864→   864→
   865→   865→      if (oldIndex !== -1 && newIndex !== -1 && oldIndex !== newIndex) {
   866→   866→        moveSection(pageId, oldIndex, newIndex);
   867→   867→      }
   868→   868→    },
   869→   869→    [sections, pageId, moveSection]
   870→   870→  );
   871→   871→
   872→   872→  // Add section handler
   873→   873→  const handleAddSection = useCallback(
   874→   874→    (type: SectionType) => {
   875→   875→      if (!pageId) return;
   876→   876→      const newSection = createDefaultSection(type);
   877→   877→      addSection(pageId, newSection);
   878→   878→      setSelectedSectionId(newSection.id);
   879→   879→      toast.success(`Added ${SECTION_TYPE_LABELS[type]} section`);
   880→   880→    },
   881→   881→    [pageId, addSection, setSelectedSectionId]
   882→   882→  );
   883→   883→
   884→   884→  // Toggle visibility for section list items
   885→   885→  const handleToggleVisibility = useCallback(
   886→   886→    (sectionId: string) => {
   887→   887→      if (!store) return;
   888→   888→      const updatedStore = {
   889→   889→        ...store,
   890→   890→        updatedAt: new Date().toISOString(),
   891→   891→        pages: store.pages.map((page) => ({
   892→   892→          ...page,
   893→   893→          sections: page.sections.map((s) =>
   894→   894→            s.id === sectionId ? { ...s, visible: !s.visible } : s
   895→   895→          ),
   896→   896→        })),
   897→   897→      };
   898→   898→      setStore(updatedStore);
   899→   899→    },
   900→   900→    [store, setStore]
   901→   901→  );
   902→   902→
   903→   903→  const handleDeleteSection = useCallback(
   904→   904→    (sectionId: string) => {
   905→   905→      if (!pageId) return;
   906→   906→      const section = sections.find((s) => s.id === sectionId);
   907→   907→      removeSection(pageId, sectionId);
   908→   908→      if (selectedSectionId === sectionId) {
   909→   909→        setSelectedSectionId(null);
   910→   910→      }
   911→   911→      if (section) {
   912→   912→        toast.success(`Deleted ${SECTION_TYPE_LABELS[section.type]} section`);
   913→   913→      }
   914→   914→    },
   915→   915→    [pageId, sections, selectedSectionId, removeSection, setSelectedSectionId]
   916→   916→  );
   917→   917→
   918→   918→  if (!store || !currentPage) {
   919→   919→    return (
   920→   920→      <div className="flex h-full items-center justify-center bg-zinc-950">
   921→   921→        <p className="text-sm text-zinc-600">No store loaded</p>
   922→   922→      </div>
   923→   923→    );
   924→   924→  }
   925→   925→
   926→   926→  return (
   927→   927→    <div className="flex h-full overflow-hidden bg-zinc-950">
   928→   928→      {/* Section List Panel */}
   929→   929→      <div
   930→   930→        className={[
   931→   931→          'flex flex-col border-r border-zinc-800 min-h-0',
   932→   932→          selectedSection ? 'w-64 lg:w-72' : 'w-full',
   933→   933→        ].join(' ')}
   934→   934→      >
   935→   935→        {/* Panel header */}
   936→   936→        <div className="flex items-center justify-between border-b border-zinc-800 px-4 py-3">
   937→   937→          <h3 className="text-sm font-semibold text-zinc-200">Sections</h3>
   938→   938→          <span className="text-xs text-zinc-600 tabular-nums">
   939→   939→            {sections.length}
   940→   940→          </span>
   941→   941→        </div>
   942→   942→
   943→   943→        {/* Section list with DnD */}
   944→   944→        <ScrollArea className="flex-1">
   945→   945→          <div className="p-2">
   946→   946→            <DndContext
   947→   947→              collisionDetection={closestCenter}
   948→   948→              onDragEnd={handleDragEnd}
   949→   949→            >
   950→   950→              <SortableContext
   951→   951→                items={sections.map((s) => s.id)}
   952→   952→                strategy={verticalListSortingStrategy}
   953→   953→              >
   954→   954→                <div className="space-y-0.5">
   955→   955→                  {sections.map((section) => (
   956→   956→                    <SortableSectionItem
   957→   957→                      key={section.id}
   958→   958→                      section={section}
   959→   959→                      isSelected={section.id === selectedSectionId}
   960→   960→                      onSelect={() =>
   961→   961→                        setSelectedSectionId(
   962→   962→                          section.id === selectedSectionId ? null : section.id
   963→   963→                        )
   964→   964→                      }
   965→   965→                      onToggleVisibility={() =>
   966→   966→                        handleToggleVisibility(section.id)
   967→   967→                      }
   968→   968→                      onDelete={() => handleDeleteSection(section.id)}
   969→   969→                    />
   970→   970→                  ))}
   971→   971→                </div>
   972→   972→              </SortableContext>
   973→   973→            </DndContext>
   974→   974→          </div>
   975→   975→        </ScrollArea>
   976→   976→
   977→   977→        {/* Add section button */}
   978→   978→        <div className="border-t border-zinc-800 p-2">
   979→   979→          <AddSectionPopover onAdd={handleAddSection} />
   980→   980→        </div>
   981→   981→      </div>
   982→   982→
   983→   983→      {/* Properties Panel */}
   984→   984→      {selectedSection && (
   985→   985→        <div className="hidden md:flex w-72 lg:w-80 flex-shrink-0 border-r border-zinc-800">
   986→   986→          <PropertiesPanel section={selectedSection} pageId={pageId} />
   987→   987→        </div>
   988→   988→      )}
   989→   989→    </div>
   990→   990→  );
   991→   991→}
