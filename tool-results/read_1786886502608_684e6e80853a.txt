    65→// ─── Background Image Enrichment (lazy, non-blocking) ─────
    66→// Triggered after store generation completes. Fetches real product
    67→// images sequentially without blocking the user or exhausting rate limits.
    68→
    69→function triggerBackgroundImageEnrichment(store: Store) {
    70→  if (!store.products || store.products.length === 0) return
    71→
    72→  // Only enrich products that still have AI-hallucinated Unsplash URLs
    73→  // (Real enriched URLs from the image search service will have different patterns)
    74→  const needsEnrichment = store.products.filter(p =>
    75→    p.images.length > 0 && p.images[0].includes('unsplash.com/photo-')
    76→  )
    77→
    78→  if (needsEnrichment.length === 0) return
    79→
    80→  console.log(`[Storqly] Background image enrichment: ${needsEnrichment.length}/${store.products.length} products need enrichment`)
    81→
    82→  fetch('/api/store/enrich-images', {
    83→    method: 'POST',
    84→    headers: { 'Content-Type': 'application/json' },
    85→    body: JSON.stringify({
    86→      products: needsEnrichment.map(p => ({
    87→        id: p.id,
    88→        name: p.name,
    89→        images: [...p.images],
    90→        category: p.category,
    91→        description: p.description,
    92→      })),
    93→      storeName: store.name,
    94→    }),
    95→  })
    96→    .then(res => res.json())
    97→    .then(data => {
    98→      if (data.enriched > 0) {
    99→        console.log(`[Storqly] Background enrichment complete: ${data.enriched} enriched, ${data.kept} kept, ${data.failed} failed in ${data.latencyMs}ms`)
   100→        // Update product images in the store via the Zustand store
   101→        // The enrich-images endpoint mutated the products array in-place,
   102→        // but we need to update the Zustand store to trigger re-renders.
   103→        // Dynamic import to avoid circular dependency
   104→        import('@/lib/store').then(({ useStoreEditor }) => {
   105→          const currentStore = useStoreEditor.getState().store
   106→          if (currentStore) {
   107→            const updatedProducts = currentStore.products.map(prod => {
   108→              const enriched = data.products?.find((ep: { id: string; images: string[] }) => ep.id === prod.id)
   109→              if (enriched && enriched.images[0] !== prod.images[0]) {
   110→                return { ...prod, images: enriched.images }
   111→              }
   112→              return prod
   113→            })
   114→            useStoreEditor.getState().setStore({ ...currentStore, products: updatedProducts })
   115→          }
   116→        })
   117→      }
   118→    })
   119→    .catch(err => {
   120→      // Non-fatal — products still show AI placeholder images
   121→      console.warn('[Storqly] Background enrichment failed (non-fatal):', err)
   122→    })
   123→}

   179→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   180→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   181→  const abortRef = useRef<AbortController | null>(null)

   183→  const {
   184→    isGenerating,

   192→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   193→  const [error, setError] = useState<string | null>(null)

   195→  const clearTimers = useCallback(() => {
   196→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   197→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }

   235→    if (mins < 1) return 'just now'
   236→    if (mins < 60) return `${mins}m ago`

   239→    const days = Math.floor(hours / 24)
   240→    if (days < 30) return `${days}d ago`

   243→

   254→    const trimmed = promptText.trim()
   255→    if (!trimmed) {
   256→      setError('Please describe your store to get started.')
   257→      textareaRef.current?.focus()
   258→      return
   259→    }
   260→
   261→    // Auth gate — block generation for logged-out users (show modal instead)
   262→    if (!session?.user?.id) {
   263→      setAuthOpen(true)

   265→    }
   266→

   268→    setError(null)
   269→    setIsGenerating(true)
   270→    setGenerationStatus('Starting generation...')
   271→    setElapsedSeconds(0)
   272→
   273→    // Create abort controller for this request
   274→    const controller = new AbortController()
   275→    abortRef.current = controller
   276→
   277→    // Hard timeout: 120 seconds — no request should ever take longer than this.
   278→    // If the server is hung or the proxy drops the connection, the user gets a
   279→    // clear message instead of waiting indefinitely.
   280→    const HARD_TIMEOUT_MS = 120_000
   281→    let timedOut = false
   282→    const timeoutId = setTimeout(() => {
   283→      timedOut = true
   284→      controller.abort()

   286→    }, HARD_TIMEOUT_MS)
   287→
   288→    // Elapsed time counter
   289→    elapsedTimerRef.current = setInterval(() => {
   290→      setElapsedSeconds((prev) => prev + 1)
   291→    }, 1000)
   292→
   293→    // Fallback progress cycling (used only if SSE progress events stop)
   294→    let fallbackIdx = 0
   295→    let receivedSSEProgress = false
   296→    progressTimerRef.current = setInterval(() => {
   297→      if (!receivedSSEProgress) {
   298→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   299→        setGenerationStatus(progressMessages[fallbackIdx])
   300→      }
   301→    }, 3000)
   302→
   303→    // ── Helper: clean up on any error path (no throw, no crash overlay) ──
   304→    const finishWithError = (message: string) => {
   305→      clearTimeout(timeoutId)
   306→      clearTimers()
   307→      setError(message)
   308→      setIsGenerating(false)
   309→      setGenerationStatus('')
   310→      setElapsedSeconds(0)
   311→      toast.error('Store generation failed', { description: message })
   312→    }
   313→
   314→    const finishOk = () => {
   315→      clearTimeout(timeoutId)
   316→      clearTimers()
   317→      setIsGenerating(false)
   318→      setGenerationStatus('')
   319→      setElapsedSeconds(0)
   320→    }
   321→
   322→    try {
   323→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   324→
   325→      // ── Fetch with auto-retry for transient gateway errors (502/503/504) ──
   326→      const RETRYABLE_STATUSES = [502, 503, 504]
   327→      let res: Response | undefined
   328→      let retryCount = 0
   329→      const maxRetries = 2
   330→
   331→      while (retryCount <= maxRetries) {
   332→        try {
   333→          res = await fetch('/api/store/generate', {
   334→            method: 'POST',
   335→            headers: { 'Content-Type': 'application/json' },
   336→            body: JSON.stringify({ prompt: trimmed }),
   337→            signal: controller.signal,
   338→          })
   339→
   340→          // If the status is retryable, wait and loop again
   341→          if (!res.ok && RETRYABLE_STATUSES.includes(res.status) && retryCount < maxRetries) {
   342→            retryCount++
   343→            console.warn(`[Storqly] Gateway error ${res.status}, retrying (${retryCount}/${maxRetries}) in 3s...`)
   344→            setGenerationStatus('Connection issue — retrying...')
   345→            await new Promise(r => setTimeout(r, 3000))
   346→            continue
   347→          }
   348→          break // Success or non-retryable error — exit loop
   349→        } catch (fetchErr: unknown) {
   350→          // Network-level error (DNS failure, connection refused, etc.)
   351→          if (fetchErr instanceof TypeError && retryCount < maxRetries) {
   352→            retryCount++
   353→            console.warn(`[Storqly] Network error, retrying (${retryCount}/${maxRetries}) in 3s...`)
   354→            setGenerationStatus('Connection issue — retrying...')
   355→            await new Promise(r => setTimeout(r, 3000))
   356→            continue
   357→          }
   358→          // Out of retries or not a network error — handle gracefully
   359→          const msg = fetchErr instanceof Error ? fetchErr.message : String(fetchErr)
   360→          if (controller.signal.aborted) {
   361→            finishWithError('Generation was cancelled.')
   362→          } else {
   363→            console.warn('[Storqly] Network error after retries:', msg)
   364→            finishWithError('Could not connect to the server. Please check your connection and try again.')
   365→          }
   366→          return
   367→        }
   368→      }
   369→
   370→      if (!res!.ok) {
   371→        let errorMsg = `Server error (${res!.status})`
   372→        try {
   373→          const errorData = await res!.json()
   374→          errorMsg = errorData.error || errorMsg
   375→        } catch { /* ignore parse error */ }
   376→        console.warn('[Storqly] HTTP error:', errorMsg)
   377→        finishWithError(errorMsg)
   378→        return
   379→      }
   380→
   381→      // ── Consume SSE stream ──
   382→      const reader = res.body?.getReader()
   383→      if (!reader) {
   384→        finishWithError('No response stream received from server.')
   385→        return
   386→      }
   387→
   388→      const decoder = new TextDecoder()
   389→      let buffer = ''
   390→      let resolved = false
   391→
   392→      while (true) {
   393→        const { done, value } = await reader.read()
   394→        if (done) break
   395→
   396→        buffer += decoder.decode(value, { stream: true })
   397→
   398→        // Parse SSE events from buffer
   399→        const lines = buffer.split('\n')
   400→        buffer = lines.pop() || ''
   401→
   402→        let currentEvent = ''
   403→        for (const line of lines) {
   404→          if (line.startsWith(':')) continue
   405→          if (line.startsWith('event: ')) {
   406→            currentEvent = line.slice(7).trim()
   407→          } else if (line.startsWith('data: ')) {
   408→            const dataStr = line.slice(6)
   409→
   410→            if (currentEvent === 'progress') {
   411→              receivedSSEProgress = true
   412→              try {
   413→                const data = JSON.parse(dataStr)
   414→                setGenerationStatus(data.message || 'Processing...')
   415→              } catch { /* ignore */ }
   416→            } else if (currentEvent === 'result') {
   417→              try {
   418→                const data = JSON.parse(dataStr)
   419→                if (!data.store) {
   420→                  console.warn('[Storqly] Result event missing store data')
   421→                  finishWithError('The AI response was missing store data.')
   422→                  return
   423→                }
   424→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   425→                resolved = true
   426→
   427→                if (data._isFallback) {
   428→                  toast.warning('AI service unavailable — showing starter template. Try again in a moment.', { duration: 6000 })
   429→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   430→                } else {
   431→                  setStore(data.store)
   432→                  // Trigger lazy background image enrichment (non-blocking)
   433→                  triggerBackgroundImageEnrichment(data.store)
   434→                }
   435→
   436→                // Soft cap toast
   437→                if (data._productCapHit) {
   438→                  toast.info(
   439→                    `Generated ${data._generatedCount} products — for larger catalogs, you can add more via the chat editor.`,
   440→                    { duration: 8000 }
   441→                  )
   442→                }
   443→
   444→                finishOk()
   445→                return
   446→              } catch (parseErr) {
   447→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   448→                console.warn('[Storqly] Failed to parse result event:', msg)
   449→                finishWithError(msg)
   450→                return
   451→              }
   452→            } else if (currentEvent === 'error') {
   453→              try {
   454→                const data = JSON.parse(dataStr)
   455→                const serverMsg = data.message || 'Generation failed'
   456→                console.warn('[Storqly] Server error event:', serverMsg)
   457→                finishWithError(serverMsg)
   458→                return
   459→              } catch {
   460→                finishWithError('Generation failed')
   461→                return
   462→              }
   463→            }
   464→            currentEvent = ''
   465→          }
   466→        }
   467→      }
   468→
   469→      if (!resolved) {
   470→        console.warn('[Storqly] Stream ended without a result event')
   471→        finishWithError('Stream ended without a result. The server may have disconnected.')
   472→      }
   473→    } catch (err: unknown) {
   474→      // Last-resort safety net — should rarely be reached now
   475→      clearTimeout(timeoutId)
   476→      clearTimers()
   477→      let message = 'Something went wrong. Please try again.'
   478→      if (err instanceof Error) {
   479→        if (err.name === 'AbortError') {
   480→          message = timedOut
   481→            ? 'Generation timed out after 2 minutes. The server may be overloaded — please try again.'
   482→            : 'Generation was cancelled.'
   483→          console.warn(`[Storqly] Request aborted: ${message}`)
   484→        } else {
   485→          console.warn('[Storqly] Unexpected error (caught by safety net):', err.message)
   486→          message = err.message
   487→        }
   488→      }
   489→      setError(message)
   490→      setIsGenerating(false)
   491→      setGenerationStatus('')
   492→      setElapsedSeconds(0)
   493→      toast.error('Store generation failed', { description: message })
   494→    }
   495→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])
   496→

   499→    return () => {
   500→      clearTimers()
   501→      abortRef.current?.abort()
   502→    }
   503→  }, [clearTimers])
   504→

   525→            Storqly
   526→          </span>

   544→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />

  1029→function EditorToolbar({
  1030→  onToggleLeft,
  1031→  onToggleRight,
  1032→  showLeft,
  1033→  showRight,
  1034→}: {
  1035→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
  1036→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
  1037→  showLeft: boolean
  1038→  showRight: boolean
  1039→}) {
  1040→  const store = useStoreEditor((s) => s.store)
  1041→  const reset = useStoreEditor((s) => s.reset)
  1042→  const isPublishing = useStoreEditor((s) => s.isPublishing)
  1043→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
  1044→  const isPublished = useStoreEditor((s) => s.isPublished)
  1045→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
  1046→  const [isSaving, setIsSaving] = useState(false)
  1047→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
  1048→  const [copied, setCopied] = useState(false)
  1049→  const [authOpen, setAuthOpen] = useState(false)
  1050→
  1051→  const handleBack = () => {
  1052→    reset()
  1053→  }
  1054→
  1055→  const handleSave = async () => {
  1056→    if (!store || isSaving) return
  1057→    setIsSaving(true)
  1058→    try {
  1059→      const res = await fetch('/api/store/save', {
  1060→        method: 'POST',
  1061→        headers: { 'Content-Type': 'application/json' },
  1062→        body: JSON.stringify({ store }),
  1063→      })
  1064→      if (!res.ok) throw new Error(`Save failed (${res.status})`)
  1065→      const data = await res.json()

  1108→          throw new Error('GATEWAY_ERROR')

  1132→          description: 'The server seems to be restarting. Please wait a moment and try again.',
  1133→          duration: 6000,

  1136→        toast.error('Connection lost while publishing', {
  1137→          description: 'Your store data is safe. Please check your connection and try again.',
  1138→          duration: 6000,
  1139→        })
  1140→      } else {
  1141→        toast.error('Failed to publish store', {
  1142→          description: msg || 'An unexpected error occurred. Please try again.',
  1143→          duration: 5000,
  1144→        })
  1145→      }
  1146→    } finally {
  1147→      setIsPublishing(false)
  1148→    }

  1163→      textarea.select()

  1165→      document.body.removeChild(textarea)
  1166→      setCopied(true)
  1167→      toast.success('URL copied to clipboard!')

  1171→

  1178→            size="icon"

  1180→            onClick={handleBack}
  1181→            aria-label="Back to home"
  1182→          >
  1183→            <ArrowLeft className="h-4 w-4" />
  1184→          </Button>
  1185→          <div className="h-5 w-px bg-zinc-800" />

  1188→            size="sm"
  1189→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
  1190→            onClick={() => onToggleLeft((v) => !v)}
  1191→          >
  1192→            <Layers className="h-3.5 w-3.5" />
  1193→            <span className="hidden sm:inline">Sections</span>
  1194→          </Button>
  1195→          <Button
  1196→            variant={showRight ? 'secondary' : 'ghost'}
  1197→            size="sm"
  1198→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
  1199→            onClick={() => onToggleRight((v) => !v)}

  1201→            <MessageSquare className="h-3.5 w-3.5" />

  1343→function PublishedStoreViewer({ slug }: { slug: string }) {
  1344→  const [store, setStore] = useState<Store | null>(null)
  1345→  const [loading, setLoading] = useState(true)
  1346→  const [error, setError] = useState<string | null>(null)
  1347→
  1348→  useEffect(() => {
  1349→    let cancelled = false
  1350→    async function load() {
  1351→      try {
  1352→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
  1353→        if (!res.ok) {
  1354→          const data = await res.json().catch(() => ({}))
  1355→          throw new Error(data.error || `Store not found (${res.status})`)
  1356→        }
  1357→        const data = await res.json()
  1358→        if (!cancelled) setStore(data.store)
  1359→      } catch (err: unknown) {
  1360→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
  1361→      } finally {
  1362→        if (!cancelled) setLoading(false)
  1363→      }
  1364→    }
  1365→    load()
  1366→    return () => { cancelled = true }
  1367→  }, [slug])
  1368→
  1369→  if (loading) {
  1370→    return (
  1371→      <div className="flex min-h-screen items-center justify-center bg-white">
  1372→        <div className="text-center">
  1373→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
  1374→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
  1375→        </div>
  1376→      </div>
  1377→    )
  1378→  }
  1379→
  1380→  if (error || !store) {
  1381→    return (
  1382→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
  1383→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
  1384→          <AlertCircle className="h-8 w-8 text-red-400" />
  1385→        </div>
  1386→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
  1387→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
  1388→          {error || 'This store could not be found or has not been published.'}
  1389→        </p>
  1390→        <Button
  1391→          variant="outline"
  1392→          onClick={() => window.location.href = '/'}
  1393→          className="gap-2"
  1394→        >
  1395→          <ArrowLeft className="h-4 w-4" />
  1396→          Back to Storqly
  1397→        </Button>
  1398→      </div>
  1399→    )
  1400→  }
  1401→
  1402→  return (
  1403→    <>
  1404→      <StoreRenderer store={store} />
  1405→      <Toaster />
  1406→      {/* Minimal footer for published stores */}
  1407→      <div className="fixed bottom-3 right-3 z-40">
  1408→        <a
  1409→          href="/"
  1410→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
  1411→        >
  1412→          <Sparkles className="h-3 w-3" />
  1413→          Built with Storqly
  1414→        </a>
  1415→      </div>
  1416→    </>
