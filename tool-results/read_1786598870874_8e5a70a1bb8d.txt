    80→      if (min !== undefined) return Math.max(min, n);
    81→      if (max !== undefined) return Math.min(max, n);
    82→      return n;
    83→    }
    84→  }
    85→  return fallback;
    86→}

    92→  return fallback;
    93→}

   105→function enumVal(v: unknown, valid: Set<string>, fallback: string): string {
   106→  if (typeof v === 'string' && valid.has(v)) return v;

   109→    const lower = v.toLowerCase();
   110→    for (const validVal of valid) {
   111→      if (validVal === lower) return validVal;
   112→    }
   113→  }
   114→  return fallback;
   115→}

   434→    log.log({ field: 'page.type', action: 'coerced', from: String(rawType), to: 'home' });

   459→/** Hard-cap enforcement: truncate sections and products to safe output limits.
   460→ *  This is the safety net if the AI ignores the system prompt caps.
   461→ *  Layers of defense:
   462→ *    1. sanitizePrompt() in route.ts strips count requests (prevents AI from trying)
   463→ *    2. System prompt ABSOLUTE CAPS language (strong AI guidance)
   464→ *    3. THIS function: hard truncation (final safety net)
   465→ */
   466→function enforceOutputCaps(store: Store, log: ReturnType<typeof createLogger>): void {
   467→  const MAX_PRODUCTS = 3;
   468→  const MAX_SECTIONS = 4;
   469→
   470→  // ── Cap products ──
   471→  if (store.products.length > MAX_PRODUCTS) {
   472→    const before = store.products.length;
   473→    store.products = store.products.slice(0, MAX_PRODUCTS);
   474→    // Ensure at least 1 featured product after truncation
   475→    if (!store.products.some((p) => p.featured) && store.products.length > 0) {
   476→      store.products[0].featured = true;
   477→    }
   478→    log.log({
   479→      field: 'products',
   480→      action: 'coerced',
   481→      from: `${before} products`,
   482→      to: `${MAX_PRODUCTS} products (capped)`,
   483→    });
   484→  }
   485→
   486→  // ── Cap sections per page (only for 'home' type — template pages have 0 sections by design) ──
   487→  for (const page of store.pages) {
   488→    if (page.type && page.type !== 'home') continue; // Skip non-home pages
   489→    if (page.sections.length > MAX_SECTIONS) {
   490→      const before = page.sections.length;
   491→      page.sections = page.sections.slice(0, MAX_SECTIONS);
   492→      log.log({
   493→        field: `pages[${page.name}].sections`,
   494→        action: 'coerced',
   495→        from: `${before} sections`,
   496→        to: `${MAX_SECTIONS} sections (capped)`,
   497→      });
   498→    }
   499→  }
   500→}

   502→/** Fix productIds in featured-products sections to reference actual products */
   503→function fixProductReferences(store: Store, log: ReturnType<typeof createLogger>): void {
   504→  const validProductIds = new Set(store.products.map((p) => p.id));
   505→  const allProductIds = store.products.map((p) => p.id);
   506→
   507→  for (const page of store.pages) {
   508→    for (const section of page.sections) {
   509→      if (section.type === 'featured-products' && Array.isArray(section.content.productIds)) {
   510→        const before = section.content.productIds.length;
   511→        // Keep only IDs that reference actual products
   512→        const validIds = (section.content.productIds as string[]).filter((id) => validProductIds.has(id));
   513→
   514→        // Fill missing slots with other product IDs so all products are shown
   515→        const usedIds = new Set(validIds);
   516→        for (const pid of allProductIds) {
   517→          if (validIds.length >= store.products.length) break;
   518→          if (!usedIds.has(pid)) {
   519→            validIds.push(pid);
   520→            usedIds.add(pid);
   521→          }
   522→        }
   523→
   524→        if (validIds.length !== before) {
   525→          const invalidCount = before - (validIds.length - (store.products.length - before > 0 ? store.products.length - before : 0));
   526→          log.log({ field: 'featured-products.productIds', action: 'coerced', from: `${before} IDs (some invalid)`, to: `${validIds.length} valid IDs` });
   527→        }
   528→        section.content.productIds = validIds;
   529→      }
   530→    }
   531→  }
   532→}

   575→      slug: 'checkout',
   576→      type: 'checkout',
   577→      isHomepage: false,
   578→      sections: [],
   579→    });
   580→    log.log({ field: 'pages', action: 'defaulted', to: 'added checkout page' });
   581→  }

   584→/** Ensure at least 1 product is marked as featured (for featured-products sections) */
   585→function ensureFeaturedProducts(store: Store, log: ReturnType<typeof createLogger>): void {
   586→  const featuredCount = store.products.filter((p) => p.featured).length;

   590→  }

   593→/** Pad product array to MIN_PRODUCTS if AI generated too few */
   594→const MIN_PRODUCTS = 3;
   595→const PAD_PRODUCT_TEMPLATES = [
   596→  { name: 'Essential Item', price: 39.99, description: 'A must-have for every collection.', category: 'Essentials', img: 'photo-1505740420928-5e560c06d30e' },
   597→  { name: 'Popular Choice', price: 59.99, description: 'Customer favorite, highly rated.', category: 'Popular', img: 'photo-1526170375885-4d8ecf77b99f' },
   598→];
   599→function padProducts(store: Store, log: ReturnType<typeof createLogger>): void {
   600→  const deficit = MIN_PRODUCTS - store.products.length;
   601→  if (deficit <= 0) return;
   602→  const before = store.products.length;
   603→  for (let i = 0; i < deficit; i++) {
   604→    const tmpl = PAD_PRODUCT_TEMPLATES[i % PAD_PRODUCT_TEMPLATES.length];
   605→    store.products.push({
   606→      id: uuid(),
   607→      name: tmpl.name + (store.products.length + 1),
   608→      price: tmpl.price + store.products.length * 5,
   609→      compareAtPrice: null,
   610→      images: [`https://images.unsplash.com/${tmpl.img}?w=600`],
   611→      description: tmpl.description,
   612→      category: tmpl.category,
   613→      featured: false,
   614→      inStock: true,
   615→    });
   616→  }
   617→  log.log({ field: 'products', action: 'defaulted', from: `${before} products`, to: `${MIN_PRODUCTS} products (padded)` });
   618→}

   622→export interface NormalizeResult {
   623→  store: Store;
   624→  log: string[];        // Human-readable log lines
   625→  summary: string;      // One-line summary
   626→  isFallback: boolean;  // true if the input was too broken to normalize
   627→  normalizationCount: number; // Number of fixes applied
   628→}
   629→
   630→/**
   631→ * Normalize any valid JSON object into a conformant Store.
   632→ *
   633→ * This function NEVER throws. If the input is too malformed to recover,
   634→ * it returns null and the caller should use a hardcoded fallback store.
   635→ *
   636→ * @param raw - Parsed JSON object (JSON.parse already succeeded)
   637→ * @param prompt - Original user prompt (for fallback name extraction)
   638→ */
   639→export function normalizeStore(raw: unknown, prompt?: string): NormalizeResult | null {
   640→  const log = createLogger();
   641→
   642→  // ── Guard: must be a non-null object ──
   643→  if (raw === null || typeof raw !== 'object' || Array.isArray(raw)) {
   644→    console.warn('[normalizeStore] Input is not an object:', typeof raw);
   645→    return null;
   646→  }
   647→
   648→  const r = raw as Record<string, unknown>;
   649→
   650→  // ── Extract store name (used as fallback for missing name) ──
   651→  const storeName = oneLine(str(r.name, prompt ? extractStoreNameFromPrompt(prompt) : 'My Store'));
   652→
   653→  if (!r.name || typeof r.name !== 'string') {
   654→    log.log({ field: 'name', action: 'missing', to: storeName });
   655→  }
   656→
   657→  // ── Top-level fields ──
   658→  const now = new Date().toISOString();
   659→  const store: Store = {
   660→    id: isUUID(str(r.id, '')) ? str(r.id, '') : uuid(),
   661→    name: storeName.substring(0, 100),
   662→    slug: slugify(storeName),
   663→    description: r.description !== undefined ? oneLine(str(r.description, '')) : undefined,
   664→    theme: r.theme ? normalizeTheme(r.theme, log) : defaultTheme,
   665→    pages: Array.isArray(r.pages)
   666→      ? r.pages.map((p: unknown) => normalizePage(p, log))
   667→      : [createDefaultPage(storeName)],
   668→    products: Array.isArray(r.products)
   669→      ? (r.products as unknown[]).map((p: unknown) => normalizeProduct(p, log))
   670→      : [],
   671→    published: bool(r.published, false),
   672→    publishedAt: r.publishedAt !== undefined ? str(r.publishedAt, '') : undefined,
   673→    createdAt: str(r.createdAt, now),
   674→    updatedAt: str(r.updatedAt, now),
   675→  };
   676→
   677→  // Log missing arrays
   678→  if (!Array.isArray(r.pages)) log.log({ field: 'pages', action: 'missing' });
   679→  if (!Array.isArray(r.products)) log.log({ field: 'products', action: 'missing' });
   680→  if (!r.theme) log.log({ field: 'theme', action: 'missing' });
   681→
   682→  // ── Ensure at least one page exists ──
   683→  if (store.pages.length === 0) {
   684→    store.pages.push(createDefaultPage(storeName));
   685→    log.log({ field: 'pages', action: 'defaulted' });
   686→  }
   687→
   688→  // ── Ensure at least one product exists (renderer expects products array) ──
   689→  if (store.products.length === 0) {
   690→    store.products.push(createDefaultProduct());
   691→    log.log({ field: 'products', action: 'defaulted' });
   692→  }
   693→
   694→  // ── Cross-reference fixes ──
   695→  // IMPORTANT: enforceOutputCaps MUST run before fixProductReferences
   696→  // because product references depend on the final product list
   697→  enforceOutputCaps(store, log);
   698→  padProducts(store, log);
   699→  fixProductReferences(store, log);
   700→  ensureHomepage(store, log);
   701→  ensureFeaturedProducts(store, log);
   702→  ensureTemplatePages(store, log);
   703→
   704→  // ── Build result ──
   705→  const logEntries = log.getLogs();
   706→  const logLines = logEntries.map(
   707→    (l) => `[${l.action}] ${l.field}${l.from !== undefined ? ` (was: ${JSON.stringify(l.from).substring(0, 60)})` : ''}${l.to !== undefined ? ` → ${JSON.stringify(l.to).substring(0, 60)}` : ''}`
   708→  );
   709→
   710→  return {
   711→    store,
   712→    log: logLines,
   713→    summary: log.summary(),
   714→    isFallback: false,
   715→    normalizationCount: logEntries.length,
   716→  };
   717→}

   759→/** Extract a short store name from a user prompt (lightweight version for normalization) */
   760→function extractStoreNameFromPrompt(prompt: string): string {
   761→  const text = prompt.trim();
   762→
   763→  // Brand name after "called / named / known as"
   764→  const calledMatch = text.match(/(?:called|named|known\s+as)\s+([\w&'\-]+(?:\s+[\w&'\-]+){0,2})/i);
   765→  if (calledMatch?.[1]) {
   766→    const name = calledMatch[1].replace(/\s+(selling|with|that|for|using|featuring)\s*$/i, '').trim();
   767→    if (name.length >= 2 && name.length <= 40) return name;
   768→  }
   769→
   770→  // Quoted brand name
   771→  const quotedMatch = text.match(/["']([^"']{2,40})["']/);
   772→  if (quotedMatch?.[1]) return quotedMatch[1].trim();
   773→
   774→  // Strip common prefixes
   775→  const stripped = text
   776→    .replace(/^(build|create|make|design|set\s+up)\s+(a|an|the|my)\s+/i, '')
   777→    .replace(/\b(online|e-commerce|ecommerce)\s+(store|shop|boutique)\b/gi, '')
   778→    .replace(/\b(store|shop|boutique|website|site|brand)\b/gi, '')
   779→    .replace(/\b(selling|with|that|for|using|featuring)\b.*/i, '')
   780→    .trim();
   781→
   782→  // Find title-case run
   783→  const words = stripped.split(/\s+/).filter(w => w.length >= 2);
   784→  const isTitle = (w: string) => w[0] === w[0].toUpperCase() && w[0] !== w[0].toLowerCase();
   785→  let bestRun: string[] = [];
   786→  let currentRun: string[] = [];
   787→  for (const w of words) {
   788→    if (isTitle(w)) {
   789→      currentRun.push(w.replace(/[^a-zA-Z0-9&'-]/g, ''));
   790→    } else {
   791→      if (currentRun.length > bestRun.length) bestRun = currentRun;
   792→      currentRun = [];
   793→    }
   794→  }
   795→  if (currentRun.length > bestRun.length) bestRun = currentRun;
   796→
   797→  if (bestRun.length >= 1) {
   798→    const candidate = bestRun.slice(0, 3).join(' ');
   799→    if (candidate.length >= 2 && candidate.length <= 40) return candidate;
   800→  }
   801→
   802→  return 'My Store';
   803→}
