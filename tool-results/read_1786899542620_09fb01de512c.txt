    49→    49→// ─── Error boundary for individual sections ──────────────────────────
    50→    50→// Prevents a malformed section from crashing the entire page.
    51→    51→
    52→    52→interface SectionErrorBoundaryState { hasError: boolean; error: Error | null }
    53→    53→class SectionErrorBoundary extends Component<
    54→    54→  { children: React.ReactNode; sectionType: string },
    55→    55→  SectionErrorBoundaryState
    56→    56→> {
    57→    57→  constructor(props: { children: React.ReactNode; sectionType: string }) {
    58→    58→    super(props);
    59→    59→    this.state = { hasError: false, error: null };
    60→    60→  }
    61→    61→  static getDerivedStateFromError(error: Error) {
    62→    62→    return { hasError: true, error };
    63→    63→  }
    64→    64→  render() {
    65→    65→    if (this.state.hasError) {
    66→    66→      return (
    67→    67→        <div className="flex items-center justify-center py-8 px-6 text-center">
    68→    68→          <div className="max-w-sm">
    69→    69→            <p className="text-sm font-medium" style={{ color: '#888' }}>
    70→    70→              Section render error ({this.props.sectionType})
    71→    71→            </p>
    72→    72→            <p className="mt-1 text-xs" style={{ color: '#aaa' }}>
    73→    73→              {this.state.error?.message || 'Unknown error'}
    74→    74→            </p>
    75→    75→          </div>
    76→    76→        </div>
    77→    77→      );
    78→    78→    }
    79→    79→    return this.props.children;
    80→    80→  }
    81→    81→}

    85→    85→interface SectionWrapperProps {
    86→    86→  section: Section;

    93→    93→function SectionWrapper({
    94→    94→  section,
    95→    95→  theme,
    96→    96→  selectedSectionId,
    97→    97→  onSelectSection,
    98→    98→  children,
    99→    99→}: SectionWrapperProps) {
   100→   100→  const isSelected = selectedSectionId === section.id;
   101→   101→  const style = section.style;
   102→   102→
   103→   103→  const sectionBg = style.backgroundColor || theme.colors.background;
   104→   104→  const sectionText = style.textColor || theme.colors.text;
   105→   105→
   106→   106→  const bgImage = style.backgroundImage
   107→   107→    ? `url(${style.backgroundImage})`
   108→   108→    : undefined;
   109→   109→
   110→   110→  const outerClasses = [
   111→   111→    'relative transition-all duration-200 cursor-pointer',
   112→   112→    pyClass(style.paddingY),
   113→   113→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   114→   114→  ].join(' ');
   115→   115→
   116→   116→  return (
   117→   117→    <div
   118→   118→      className={outerClasses}
   119→   119→      style={{
   120→   120→        backgroundColor: sectionBg,
   121→   121→        color: sectionText,
   122→   122→        backgroundImage: bgImage,
   123→   123→        backgroundSize: 'cover',
   124→   124→        backgroundPosition: 'center',
   125→   125→      }}
   126→   126→      onClick={(e) => {
   127→   127→        e.stopPropagation();
   128→   128→        onSelectSection?.(isSelected ? null : section.id);
   129→   129→      }}
   130→   130→      data-section-id={section.id}
   131→   131→      data-section-type={section.type}
   132→   132→    >
   133→   133→      {/* Overlay for background images */}
   134→   134→      {style.backgroundImage && style.overlay && (
   135→   135→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   136→   136→      )}
   137→   137→      <div
   138→   138→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}

   143→   143→            {section.type.replace(/-/g, ' ')}
   144→   144→          </div>
   145→   145→        )}

   147→   147→      </div>

   154→   154→function ProductCard({
   155→   155→  product,
   156→   156→  theme,
   157→   157→  showAddToCart,
   158→   158→  borderRadius,
   159→   159→  buttonBgOverride,
   160→   160→  buttonTextOverride,
   161→   161→  onViewProduct,
   162→   162→}: {
   163→   163→  product: StoreProduct;
   164→   164→  theme: StoreTheme;
   165→   165→  showAddToCart: boolean;
   166→   166→  borderRadius: string;
   167→   167→  buttonBgOverride?: string;
   168→   168→  buttonTextOverride?: string;
   169→   169→  onViewProduct?: (productId: string) => void;
   170→   170→}) {
   171→   171→  const [hovered, setHovered] = useState(false);
   172→   172→  const [addedFeedback, setAddedFeedback] = useState(false);
   173→   173→  const addToCart = useCartStore((s) => s.addItem);
   174→   174→  const imgColor = stringToColor(product.id, theme);
   175→   175→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   176→   176→
   177→   177→  const handleAddToCart = (e: React.MouseEvent) => {
   178→   178→    e.stopPropagation();
   179→   179→    addToCart(product);
   180→   180→    setAddedFeedback(true);
   181→   181→    setTimeout(() => setAddedFeedback(false), 1200);
   182→   182→  };
   183→   183→
   184→   184→  return (
   185→   185→    <div
   186→   186→      className={`${borderRadius} ${onViewProduct ? 'cursor-pointer' : ''} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   187→   187→      style={{
   188→   188→        borderColor: theme.colors.border,
   189→   189→      }}
   190→   190→      onMouseEnter={() => setHovered(true)}
   191→   191→      onMouseLeave={() => setHovered(false)}
   192→   192→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   193→   193→      role={onViewProduct ? 'button' : undefined}
   194→   194→      tabIndex={onViewProduct ? 0 : undefined}
   195→   195→    >
   196→   196→      {/* Product image with real image + fallback */}
   197→   197→      <div
   198→   198→        className="relative aspect-square w-full overflow-hidden"
   199→   199→        style={{ backgroundColor: imgColor }}
   200→   200→      >
   201→   201→        <StoreImage
   202→   202→          src={product.images[0] || ''}
   203→   203→          alt={product.name}
   204→   204→          fallbackColor={imgColor}
   205→   205→          className="absolute inset-0 h-full w-full object-cover transition-transform duration-300 group-hover:scale-105"
   206→   206→          iconSize="lg"
   207→   207→        />
   208→   208→        {/* Out of stock overlay */}
   209→   209→        {!product.inStock && (
   210→   210→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   211→   211→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   212→   212→              Sold Out
   213→   213→            </span>
   214→   214→          </div>
   215→   215→        )}
   216→   216→        {/* Discount badge */}
   217→   217→        {hasDiscount && product.compareAtPrice && (
   218→   218→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   219→   219→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   220→   220→          </div>
   221→   221→        )}
   222→   222→      </div>
   223→   223→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   224→   224→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   225→   225→        {product.category && (
   226→   226→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   227→   227→            {product.category}
   228→   228→          </p>
   229→   229→        )}
   230→   230→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   231→   231→          {product.name}
   232→   232→        </h3>
   233→   233→        <div className="mt-1.5 flex items-center gap-2">
   234→   234→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   235→   235→            {formatPrice(product.price)}
   236→   236→          </span>
   237→   237→          {hasDiscount && product.compareAtPrice && (
   238→   238→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   239→   239→              {formatPrice(product.compareAtPrice)}
   240→   240→            </span>
   241→   241→          )}
   242→   242→        </div>
   243→   243→        {showAddToCart && product.inStock && (
   244→   244→          <button
   245→   245→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 cursor-pointer ${
   246→   246→              hovered
   247→   247→                ? 'opacity-100 translate-y-0'
   248→   248→                : 'opacity-0 translate-y-1 pointer-events-none'
   249→   249→            }`}
   250→   250→            style={{
   251→   251→              backgroundColor: addedFeedback ? '#16a34a' : (buttonBgOverride || theme.colors.primary),
   252→   252→              color: addedFeedback ? '#ffffff' : (buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary)),
   253→   253→            }}
   254→   254→            onClick={handleAddToCart}
   255→   255→          >
   256→   256→            {addedFeedback ? 'Added!' : 'Add to Cart'}
   257→   257→          </button>
   258→   258→        )}
   259→   259→      </div>
   260→   260→    </div>
   261→   261→  );
   262→   262→}

   359→   359→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   360→   360→  const content = section.content as unknown as HeroContent;
   361→   361→  const style = section.style;
   362→   362→  const isSelected = selectedSectionId === section.id;
   363→   363→
   364→   364→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   365→   365→  const alignMap = {
   366→   366→    left: 'items-start text-left',
   367→   367→    center: 'items-center text-center',
   368→   368→    right: 'items-end text-right',
   369→   369→  };
   370→   370→
   371→   371→  // Only apply theme gradient as default background when no custom color or image is set.
   372→   372→  // If style.backgroundColor is set, respect it — don't cover it with a gradient.
   373→   373→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   374→   374→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   375→   375→    : undefined;
   376→   376→
   377→   377→  const outerClasses = [
   378→   378→    'relative flex transition-all duration-200 cursor-pointer',
   379→   379→    heightMap[content.height] || heightMap.md,
   380→   380→    alignMap[content.alignment],
   381→   381→    pyClass(style.paddingY),
   382→   382→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   383→   383→  ].join(' ');
   384→   384→
   385→   385→  return (
   386→   386→    <div
   387→   387→      className={outerClasses}
   388→   388→      style={{
   389→   389→        backgroundColor: style.backgroundColor || theme.colors.primary,
   390→   390→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   391→   391→        backgroundSize: 'cover',
   392→   392→        backgroundPosition: 'center',
   393→   393→        color: style.textColor || '#ffffff',
   394→   394→      }}
   395→   395→      onClick={(e) => {
   396→   396→        e.stopPropagation();
   397→   397→        onSelectSection?.(isSelected ? null : section.id);
   398→   398→      }}
   399→   399→      data-section-id={section.id}
   400→   400→      data-section-type={section.type}
   401→   401→    >
   402→   402→      {/* Overlay */}
   403→   403→      {style.backgroundImage && style.overlay && (
   404→   404→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   405→   405→      )}
   406→   406→      {/* Subtle overlay and decorative elements — only with gradient or image backgrounds */}
   407→   407→      {!style.backgroundImage && !style.backgroundColor && (
   408→   408→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   409→   409→      )}
   410→   410→      {!style.backgroundImage && !style.backgroundColor && (
   411→   411→        <>
   412→   412→          <div className="absolute right-10 top-10 h-40 w-40 rounded-full bg-white/10 blur-3xl" />
   413→   413→          <div className="absolute bottom-10 left-10 h-32 w-32 rounded-full bg-white/5 blur-2xl" />
   414→   414→        </>
   415→   415→      )}
   416→   416→
   417→   417→      <div className={`relative z-10 mx-auto w-full ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}>
   418→   418→        <h1 className="text-3xl font-extrabold leading-tight tracking-tight sm:text-4xl md:text-5xl lg:text-6xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   419→   419→          {content.headline}
   420→   420→        </h1>
   421→   421→        {content.subheadline && (
   422→   422→          <p className="mt-4 max-w-xl text-base opacity-80 sm:text-lg md:text-xl">
   423→   423→            {content.subheadline}
   424→   424→          </p>
   425→   425→        )}
   426→   426→        {content.ctaText && (
   427→   427→          <button className="mt-8 inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-transform hover:scale-105 sm:text-base"
   428→   428→            style={{
   429→   429→              backgroundColor: section.style.buttonBackgroundColor || '#ffffff',
   430→   430→              color: section.style.buttonTextColor || contrastTextColor(section.style.buttonBackgroundColor || '#ffffff'),
   431→   431→            }}>
   432→   432→            {content.ctaText}
   433→   433→            <ArrowRight className="h-4 w-4" />
   434→   434→          </button>
   435→   435→        )}
   436→   436→      </div>
   437→   437→    </div>
   438→   438→  );
   439→   439→}

   443→   443→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   444→   444→  const content = section.content as unknown as FeaturedProductsContent;
   445→   445→  const borderRadius = borderRadiusClass(theme.borderRadius);
   446→   446→  const featuredProducts = useMemo(
   447→   447→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   448→   448→    [content.productIds, products],
   449→   449→  );
   450→   450→
   451→   451→  return (
   452→   452→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   453→   453→      {content.headline && (
   454→   454→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   455→   455→          {content.headline}
   456→   456→        </h2>
   457→   457→      )}
   458→   458→      {content.subtitle && (
   459→   459→        <p className="mb-8 text-sm opacity-65">
   460→   460→          {content.subtitle}
   461→   461→        </p>
   462→   462→      )}

  1032→  1032→// ─── Main Section Router ────────────────────────────────────────────────
  1033→  1033→
  1034→  1034→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1035→  1035→  const { section } = props;
  1036→  1036→
  1037→  1037→  if (!section.visible) return null;
  1038→  1038→
  1039→  1039→  let element: React.ReactNode = null;
  1040→  1040→
  1041→  1041→  switch (section.type) {
  1042→  1042→    case 'header':
  1043→  1043→      element = <HeaderSection {...props} />; break;
  1044→  1044→    case 'hero':
  1045→  1045→      element = <HeroSection {...props} />; break;
  1046→  1046→    case 'featured-products':
  1047→  1047→      element = <FeaturedProductsSection {...props} />; break;
  1048→  1048→    case 'product-grid':
  1049→  1049→      element = <ProductGridSection {...props} />; break;
  1050→  1050→    case 'text-banner':
  1051→  1051→      element = <TextBannerSection {...props} />; break;
  1052→  1052→    case 'image-gallery':
  1053→  1053→      element = <ImageGallerySection {...props} />; break;
  1054→  1054→    case 'testimonials':
  1055→  1055→      element = <TestimonialsSection {...props} />; break;
  1056→  1056→    case 'newsletter':
  1057→  1057→      element = <NewsletterSection {...props} />; break;
  1058→  1058→    case 'faq':
  1059→  1059→      element = <FAQSection {...props} />; break;
  1060→  1060→    case 'cta':
  1061→  1061→      element = <CTASection {...props} />; break;
  1062→  1062→    case 'categories':
  1063→  1063→      element = <CategoriesSection {...props} />; break;
  1064→  1064→    case 'footer':
  1065→  1065→      element = <FooterSection {...props} />; break;
  1066→  1066→    case 'rich-text':
  1067→  1067→      element = <RichTextSection {...props} />; break;
  1068→  1068→    case 'spacer':
  1069→  1069→      element = <SpacerSection {...props} />; break;
  1070→  1070→    case 'divider':
  1071→  1071→      element = <DividerSection {...props} />; break;
  1072→  1072→    default:
  1073→  1073→      // Unknown section type — render a visible placeholder instead of invisible null
  1074→  1074→      return (
  1075→  1075→        <SectionErrorBoundary sectionType={section.type}>
  1076→  1076→          <div className="py-8 text-center text-sm text-gray-400">
  1077→  1077→            <p className="font-medium">Unsupported section type: {section.type}</p>
  1078→  1078→            <p className="mt-1 text-xs text-gray-300">This section can be edited or removed.</p>
  1079→  1079→          </div>
  1080→  1080→        </SectionErrorBoundary>
  1081→  1081→      );
  1082→  1082→  }
  1083→  1083→
  1084→  1084→  // Wrap each section in its own error boundary so a malformed
  1085→  1085→  // section (e.g. AI-generated content with wrong shape) can never
  1086→  1086→  // crash the entire page.
  1087→  1087→  return (
  1088→  1088→    <SectionErrorBoundary sectionType={section.type}>
  1089→  1089→      {element}
  1090→  1090→    </SectionErrorBoundary>
  1091→  1091→  );
  1092→  1092→}
