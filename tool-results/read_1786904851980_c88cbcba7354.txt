    49→// ─── Error boundary for individual sections ──────────────────────────
    50→// Prevents a malformed section from crashing the entire page.
    51→
    52→interface SectionErrorBoundaryState { hasError: boolean; error: Error | null }
    53→class SectionErrorBoundary extends Component<
    54→  { children: React.ReactNode; sectionType: string },
    55→  SectionErrorBoundaryState
    56→> {
    57→  constructor(props: { children: React.ReactNode; sectionType: string }) {
    58→    super(props);
    59→    this.state = { hasError: false, error: null };
    60→  }
    61→  static getDerivedStateFromError(error: Error) {
    62→    return { hasError: true, error };
    63→  }
    64→  render() {
    65→    if (this.state.hasError) {
    66→      return (
    67→        <div className="flex items-center justify-center py-8 px-6 text-center">
    68→          <div className="max-w-sm">
    69→            <p className="text-sm font-medium" style={{ color: '#888' }}>
    70→              Section render error ({this.props.sectionType})
    71→            </p>
    72→            <p className="mt-1 text-xs" style={{ color: '#aaa' }}>
    73→              {this.state.error?.message || 'Unknown error'}
    74→            </p>
    75→          </div>
    76→        </div>
    77→      );
    78→    }
    79→    return this.props.children;
    80→  }
    81→}

    85→interface SectionWrapperProps {
    86→  section: Section;
    87→  theme: StoreTheme;
    88→  selectedSectionId?: string | null;
    89→  onSelectSection?: (sectionId: string | null) => void;

    93→function SectionWrapper({
    94→  section,
    95→  theme,
    96→  selectedSectionId,
    97→  onSelectSection,
    98→  children,
    99→}: SectionWrapperProps) {
   100→  const isSelected = selectedSectionId === section.id;
   101→  const style = section.style;
   102→
   103→  const sectionBg = style.backgroundColor || theme.colors.background;
   104→  const sectionText = style.textColor || theme.colors.text;
   105→
   106→  const bgImage = style.backgroundImage
   107→    ? `url(${style.backgroundImage})`
   108→    : undefined;
   109→
   110→  const outerClasses = [
   111→    'relative transition-all duration-200 cursor-pointer',
   112→    pyClass(style.paddingY),
   113→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   114→  ].join(' ');
   115→
   116→  return (
   117→    <div
   118→      className={outerClasses}
   119→      style={{
   120→        backgroundColor: sectionBg,
   121→        color: sectionText,
   122→        backgroundImage: bgImage,
   123→        backgroundSize: 'cover',
   124→        backgroundPosition: 'center',
   125→      }}
   126→      onClick={(e) => {
   127→        e.stopPropagation();
   128→        onSelectSection?.(isSelected ? null : section.id);
   129→      }}
   130→      data-section-id={section.id}
   131→      data-section-type={section.type}
   132→    >
   133→      {/* Overlay for background images */}
   134→      {style.backgroundImage && style.overlay && (
   135→        <div className="absolute inset-0 bg-black/50 pointer-events-none" />
   136→      )}
   137→      <div
   138→        className={`relative z-10 mx-auto ${maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)}`}
   139→      >
   140→        {/* Selected section label */}
   141→        {isSelected && (
   142→          <div className="absolute -top-3 left-4 z-20 rounded-full bg-[#a855f7] px-2.5 py-0.5 text-[10px] font-semibold uppercase tracking-wider text-white shadow-lg">
   143→            {section.type.replace(/-/g, ' ')}
   144→          </div>
   145→        )}
   146→        {children}
   147→      </div>
   148→    </div>
   149→  );
   150→}

   154→function ProductCard({
   155→  product,
   156→  theme,
   157→  showAddToCart,
   158→  borderRadius,
   159→  buttonBgOverride,
   160→  buttonTextOverride,
   161→  onViewProduct,
   162→}: {
   163→  product: StoreProduct;
   164→  theme: StoreTheme;
   165→  showAddToCart: boolean;
   166→  borderRadius: string;
   167→  buttonBgOverride?: string;
   168→  buttonTextOverride?: string;
   169→  onViewProduct?: (productId: string) => void;
   170→}) {
   171→  const [hovered, setHovered] = useState(false);
   172→  const [addedFeedback, setAddedFeedback] = useState(false);
   173→  const addToCart = useCartStore((s) => s.addItem);
   174→  const imgColor = stringToColor(product.id, theme);
   175→  const hasDiscount = product.compareAtPrice && product.compareAtPrice > product.price;
   176→
   177→  const handleAddToCart = (e: React.MouseEvent) => {
   178→    e.stopPropagation();
   179→    addToCart(product);
   180→    setAddedFeedback(true);
   181→    setTimeout(() => setAddedFeedback(false), 1200);
   182→  };
   183→
   184→  return (
   185→    <div
   186→      className={`${borderRadius} ${onViewProduct ? 'cursor-pointer' : ''} overflow-hidden bg-white border border-gray-100 shadow-sm transition-shadow duration-200 group`}
   187→      style={{
   188→        borderColor: theme.colors.border,
   189→      }}
   190→      onMouseEnter={() => setHovered(true)}
   191→      onMouseLeave={() => setHovered(false)}
   192→      onClick={onViewProduct ? () => onViewProduct(product.id) : undefined}
   193→      role={onViewProduct ? 'button' : undefined}
   194→      tabIndex={onViewProduct ? 0 : undefined}
   195→    >
   196→      {/* Product image with real image + fallback */}
   197→      <div
   198→        className="relative aspect-square w-full overflow-hidden"
   199→        style={{ backgroundColor: imgColor }}
   200→      >
   201→        <StoreImage
   202→          src={product.images[0] || ''}
   203→          alt={product.name}
   204→          fallbackColor={imgColor}
   205→          className="absolute inset-0 h-full w-full object-cover transition-transform duration-300 group-hover:scale-105"
   206→          iconSize="lg"
   207→        />
   208→        {/* Out of stock overlay */}
   209→        {!product.inStock && (
   210→          <div className="absolute inset-0 flex items-center justify-center bg-white/60">
   211→            <span className="rounded-full bg-gray-900 px-3 py-1 text-xs font-medium text-white">
   212→              Sold Out
   213→            </span>
   214→          </div>
   215→        )}
   216→        {/* Discount badge */}
   217→        {hasDiscount && product.compareAtPrice && (
   218→          <div className="absolute left-2 top-2 rounded-full bg-red-500 px-2 py-0.5 text-[10px] font-bold uppercase tracking-wide text-white">
   219→            {Math.round(((product.compareAtPrice - product.price) / product.compareAtPrice) * 100)}% Off
   220→          </div>
   221→        )}
   222→      </div>
   223→      {/* Info — explicitly reset color to prevent section textColor bleeding into white card */}
   224→      <div className="p-3 sm:p-4" style={{ color: theme.colors.text }}>
   225→        {product.category && (
   226→          <p className="mb-1 text-[10px] font-medium uppercase tracking-wider" style={{ color: theme.colors.textMuted }}>
   227→            {product.category}
   228→          </p>
   229→        )}
   230→        <h3 className="text-sm font-semibold leading-snug line-clamp-2" style={{ color: theme.colors.text }}>
   231→          {product.name}
   232→        </h3>
   233→        <div className="mt-1.5 flex items-center gap-2">
   234→          <span className="text-sm font-bold" style={{ color: theme.colors.primary }}>
   235→            {formatPrice(product.price)}
   236→          </span>
   237→          {hasDiscount && product.compareAtPrice && (
   238→            <span className="text-xs line-through" style={{ color: theme.colors.textMuted }}>
   239→              {formatPrice(product.compareAtPrice)}
   240→            </span>
   241→          )}
   242→        </div>
   243→        {showAddToCart && product.inStock && (
   244→          <button
   245→            className={`mt-3 w-full rounded-md py-2 text-xs font-semibold transition-all duration-200 cursor-pointer ${
   246→              hovered
   247→                ? 'opacity-100 translate-y-0'
   248→                : 'opacity-0 translate-y-1 pointer-events-none'
   249→            }`}
   250→            style={{
   251→              backgroundColor: addedFeedback ? '#16a34a' : (buttonBgOverride || theme.colors.primary),
   252→              color: addedFeedback ? '#ffffff' : (buttonTextOverride || contrastTextColor(buttonBgOverride || theme.colors.primary)),
   253→            }}
   254→            onClick={handleAddToCart}
   255→          >
   256→            {addedFeedback ? 'Added!' : 'Add to Cart'}
   257→          </button>
   258→        )}
   259→      </div>
   260→    </div>
   261→  );
   262→}

   360→export function HeroSection({ section, theme, selectedSectionId, onSelectSection, products }: SectionRendererProps) {
   361→  const content = section.content as unknown as HeroContent;
   362→  const style = section.style;
   363→  const isSelected = selectedSectionId === section.id;
   364→
   365→  const heightMap = { sm: 'min-h-[300px]', md: 'min-h-[420px]', lg: 'min-h-[540px]', xl: 'min-h-[640px]' };
   366→  const alignMap = {
   367→    left: 'items-start text-left',
   368→    center: 'items-center text-center',
   369→    right: 'items-end text-right',
   370→  };
   371→
   372→  // Phase 3A: layout mode
   373→  const layout = content.layout || 'centered';
   374→  const isSplit = layout !== 'centered';
   375→  const imageOnRight = layout === 'split-left'; // text left, image right
   376→  const imageOnLeft = layout === 'split-right'; // image left, text right
   377→
   378→  // Resolve hero image: explicit content.heroImage, or first featured product image
   379→  const resolvedHeroImage = content.heroImage
   380→    || (isSplit ? products.find(p => p.featured)?.images[0] || products[0]?.images[0] : undefined);
   381→
   382→  // Background: gradient default, or custom color/image
   383→  const bgGradient = !style.backgroundImage && !style.backgroundColor
   384→    ? `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 100%)`
   385→    : undefined;
   386→
   387→  const outerClasses = [
   388→    'relative flex transition-all duration-200 cursor-pointer overflow-hidden',
   389→    heightMap[content.height] || heightMap.md,
   390→    isSplit ? 'items-center' : alignMap[content.alignment],
   391→    pyClass(style.paddingY),
   392→    isSelected ? 'ring-2 ring-[#a855f7] ring-offset-2' : 'hover:ring-1 hover:ring-[#a855f7]/40',
   393→  ].join(' ');
   394→
   395→  // CTA click handler: stop propagation to prevent section selection.
   396→  // Navigation to specific pages is handled by the parent via ctaLink.
   397→  const handleCtaClick = (e: React.MouseEvent) => {
   398→    e.stopPropagation();
   399→  };
   400→
   401→  const textColor = style.textColor || '#ffffff';
   402→  const buttonBg = section.style.buttonBackgroundColor || '#ffffff';
   403→  const buttonFg = section.style.buttonTextColor || contrastTextColor(buttonBg);
   404→  const headlineColor = section.style.headlineColor || undefined;
   405→
   406→  return (
   407→    <div
   408→      className={outerClasses}
   409→      style={{
   410→        backgroundColor: style.backgroundColor || theme.colors.primary,
   411→        backgroundImage: style.backgroundImage ? `url(${style.backgroundImage})` : bgGradient,
   412→        backgroundSize: 'cover',
   413→        backgroundPosition: 'center',
   414→        color: textColor,
   415→      }}
   416→      onClick={(e) => {
   417→        e.stopPropagation();
   418→        onSelectSection?.(isSelected ? null : section.id);
   419→      }}
   420→      data-section-id={section.id}
   421→      data-section-type={section.type}
   422→    >
   423→      {/* Image background overlay */}
   424→      {style.backgroundImage && style.overlay && (
   425→        <div className="absolute inset-0 pointer-events-none" style={{ background: 'linear-gradient(to bottom, rgba(0,0,0,0.15) 0%, rgba(0,0,0,0.55) 100%)' }} />
   426→      )}
   427→      {/* Subtle gradient overlay for depth */}
   428→      {bgGradient && (
   429→        <div className="absolute inset-0 bg-gradient-to-br from-black/10 to-black/30 pointer-events-none" />
   430→      )}
   431→      {/* Decorative blur orbs — always present for visual richness */}
   432→      <div className="absolute right-[10%] top-[10%] h-40 w-40 rounded-full bg-white/10 blur-3xl pointer-events-none" />
   433→      <div className="absolute bottom-[10%] left-[10%] h-32 w-32 rounded-full bg-white/5 blur-2xl pointer-events-none" />
   434→      {/* Extra decorative orb for split layouts */}
   435→      {isSplit && (
   436→        <div className="absolute left-[30%] top-[20%] h-24 w-24 rounded-full bg-white/5 blur-2xl pointer-events-none" />
   437→      )}
   438→
   439→      {/* Content area */}
   440→      <div className={`relative z-10 mx-auto w-full ${isSplit ? 'max-w-7xl' : maxWidthClass(style.maxWidth)} ${pxClass(style.paddingX)} ${isSplit ? 'grid grid-cols-1 gap-8 lg:grid-cols-2 lg:gap-12 items-center' : ''}`}>
   441→        {/* Text column */}
   442→        <div className={isSplit ? (imageOnLeft ? 'lg:order-2' : 'lg:order-1') : ''}>
   443→          {/* Badge */}
   444→          {content.badge && (
   445→            <span
   446→              className="mb-4 inline-block rounded-full border border-white/25 px-3 py-1 text-xs font-semibold uppercase tracking-widest backdrop-blur-sm"

   456→          >
   457→            {content.headline}
   458→          </h1>

   461→            <p
   462→              className={`mt-4 text-base opacity-80 sm:text-lg md:text-xl ${isSplit ? 'max-w-xl' : 'max-w-2xl'} mx-${isSplit ? '0' : 'auto'}`}

   465→              {content.subheadline}
   466→            </p>
   467→          )}
   468→
   469→          {/* CTA buttons */}
   470→          {(content.ctaText || content.secondaryCtaText) && (
   471→            <div className={`mt-8 flex flex-wrap gap-3 ${!isSplit && content.alignment === 'center' ? 'justify-center' : ''} ${!isSplit && content.alignment === 'right' ? 'justify-end' : ''}`}>
   472→              {content.ctaText && (
   473→                <button
   474→                  className="inline-flex items-center gap-2 rounded-lg px-6 py-3 text-sm font-semibold shadow-lg transition-all hover:scale-105 hover:shadow-xl sm:text-base"
   475→                  style={{ backgroundColor: buttonBg, color: buttonFg }}
   476→                  onClick={handleCtaClick}
   477→                >
   478→                  {content.ctaText}
   479→                  <ArrowRight className="h-4 w-4" />
   480→                </button>
   481→              )}
   482→              {content.secondaryCtaText && (
   483→                <button
   484→                  className="inline-flex items-center gap-2 rounded-lg border border-white/30 px-6 py-3 text-sm font-semibold backdrop-blur-sm transition-all hover:bg-white/10 hover:scale-105 sm:text-base"
   485→                  style={{ color: textColor, borderColor: textColor + '40' }}
   486→                  onClick={handleCtaClick}
   487→                >
   488→                  {content.secondaryCtaText}
   489→                </button>
   490→              )}
   491→            </div>
   492→          )}
   493→        </div>
   494→
   495→        {/* Image column (split layouts only) */}
   496→        {isSplit && resolvedHeroImage && (
   497→          <div className={`relative ${imageOnLeft ? 'lg:order-1' : 'lg:order-2'} hidden lg:block`}>
   498→            <div className="relative">
   499→              {/* Decorative background shape behind image */}
   500→              <div
   501→                className="absolute -inset-4 rounded-2xl opacity-20 blur-xl"
   502→                style={{ backgroundColor: textColor }}
   503→              />
   504→              <StoreImage
   505→                src={resolvedHeroImage}
   506→                alt={content.headline}
   507→                className="relative rounded-2xl shadow-2xl object-cover w-full aspect-[4/3]"
   508→              />
   509→            </div>
   510→          </div>
   511→        )}
   512→      </div>
   513→    </div>
   514→  );
   515→}

   519→export function FeaturedProductsSection({ section, theme, selectedSectionId, onSelectSection, products, onViewProduct, forceHideAddToCart }: SectionRendererProps) {
   520→  const content = section.content as unknown as FeaturedProductsContent;
   521→  const borderRadius = borderRadiusClass(theme.borderRadius);
   522→  const featuredProducts = useMemo(
   523→    () => content.productIds.map((id) => products.find((p) => p.id === id)).filter(Boolean) as StoreProduct[],
   524→    [content.productIds, products],
   525→  );
   526→
   527→  return (
   528→    <SectionWrapper section={section} theme={theme} selectedSectionId={selectedSectionId} onSelectSection={onSelectSection}>
   529→      {content.headline && (
   530→        <h2 className="mb-2 text-2xl font-bold sm:text-3xl" style={section.style.headlineColor ? { color: section.style.headlineColor } : undefined}>
   531→          {content.headline}
   532→        </h2>
   533→      )}
   534→      {content.subtitle && (
   535→        <p className="mb-8 text-sm opacity-65">
   536→          {content.subtitle}
   537→        </p>
   538→      )}
   539→      <div className={`grid ${gridCols(content.columns)} gap-4 sm:gap-6`}>
   540→        {featuredProducts.map((product) => (
   541→          <ProductCard

  1242→// ─── Main Section Router ────────────────────────────────────────────────
  1243→
  1244→export function renderSection(props: SectionRendererProps): React.ReactNode {
  1245→  const { section } = props;
  1246→
  1247→  if (!section.visible) return null;
  1248→
  1249→  let element: React.ReactNode = null;
  1250→
  1251→  switch (section.type) {
  1252→    case 'header':
  1253→      element = <HeaderSection {...props} />; break;
  1254→    case 'hero':
  1255→      element = <HeroSection {...props} />; break;
  1256→    case 'featured-products':
  1257→      element = <FeaturedProductsSection {...props} />; break;
  1258→    case 'product-grid':
  1259→      element = <ProductGridSection {...props} />; break;
  1260→    case 'text-banner':
  1261→      element = <TextBannerSection {...props} />; break;
  1262→    case 'image-gallery':
  1263→      element = <ImageGallerySection {...props} />; break;
  1264→    case 'testimonials':
  1265→      element = <TestimonialsSection {...props} />; break;
  1266→    case 'newsletter':
  1267→      element = <NewsletterSection {...props} />; break;
  1268→    case 'faq':
  1269→      element = <FAQSection {...props} />; break;
  1270→    case 'cta':
  1271→      element = <CTASection {...props} />; break;
  1272→    case 'categories':
  1273→      element = <CategoriesSection {...props} />; break;
  1274→    case 'footer':
  1275→      element = <FooterSection {...props} />; break;
  1276→    case 'rich-text':
  1277→      element = <RichTextSection {...props} />; break;
  1278→    case 'spacer':
  1279→      element = <SpacerSection {...props} />; break;
  1280→    case 'divider':
  1281→      element = <DividerSection {...props} />; break;
  1282→    default:
  1283→      // Unknown section type — render a visible placeholder instead of invisible null
  1284→      return (
  1285→        <SectionErrorBoundary sectionType={section.type}>
  1286→          <div className="py-8 text-center text-sm text-gray-400">
  1287→            <p className="font-medium">Unsupported section type: {section.type}</p>
  1288→            <p className="mt-1 text-xs text-gray-300">This section can be edited or removed.</p>
  1289→          </div>
  1290→        </SectionErrorBoundary>
  1291→      );
  1292→  }
  1293→
  1294→  // Wrap each section in its own error boundary so a malformed
  1295→  // section (e.g. AI-generated content with wrong shape) can never
  1296→  // crash the entire page.
  1297→  return (
  1298→    <SectionErrorBoundary sectionType={section.type}>
  1299→      {element}
  1300→    </SectionErrorBoundary>
  1301→  );
  1302→}
