    95→    95→function LandingPage() {
    96→    96→  const { data: session } = useSession()
    97→    97→  const [promptText, setPromptText] = useState('')
    98→    98→  const [mkOpen, setMkOpen] = useState(false)
    99→    99→  const [authOpen, setAuthOpen] = useState(false)
   100→   100→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   101→   101→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   102→   102→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   103→   103→  const abortRef = useRef<AbortController | null>(null)
   104→   104→
   105→   105→  const {
   106→   106→    isGenerating,
   107→   107→    setIsGenerating,
   108→   108→    setStore,
   109→   109→    setStoreWithFallback,
   110→   110→  } = useStoreEditor()
   111→   111→
   112→   112→  // Local UI state (not in Zustand — this is view-level)
   113→   113→  const [generationStatus, setGenerationStatus] = useState('')
   114→   114→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   115→   115→  const [error, setError] = useState<string | null>(null)
   116→   116→
   117→   117→  const clearTimers = useCallback(() => {
   118→   118→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   119→   119→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }
   120→   120→  }, [])
   121→   121→
   122→   122→  const handleCancel = useCallback(() => {
   123→   123→    abortRef.current?.abort()
   124→   124→    clearTimers()
   125→   125→    setIsGenerating(false)
   126→   126→    setGenerationStatus('')
   127→   127→    setElapsedSeconds(0)
   128→   128→    setError(null)
   129→   129→  }, [clearTimers, setIsGenerating])
   130→   130→
   131→   131→  const handleGenerate = useCallback(async () => {
   132→   132→    const trimmed = promptText.trim()
   133→   133→    if (!trimmed) {
   134→   134→      setError('Please describe your store to get started.')
   135→   135→      textareaRef.current?.focus()
   136→   136→      return
   137→   137→    }
   138→   138→
   139→   139→    // Auth gate — block generation for logged-out users (show modal instead)
   140→   140→    if (!session?.user?.id) {
   141→   141→      setAuthOpen(true)
   142→   142→      return
   143→   143→    }
   144→   144→
   145→   145→    // Reset state
   146→   146→    setError(null)
   147→   147→    setIsGenerating(true)
   148→   148→    setGenerationStatus('Starting generation...')
   149→   149→    setElapsedSeconds(0)
   150→   150→
   151→   151→    // Create abort controller for this request
   152→   152→    const controller = new AbortController()
   153→   153→    abortRef.current = controller
   154→   154→
   155→   155→    // Elapsed time counter
   156→   156→    elapsedTimerRef.current = setInterval(() => {
   157→   157→      setElapsedSeconds((prev) => prev + 1)
   158→   158→    }, 1000)
   159→   159→
   160→   160→    // Fallback progress cycling (used only if SSE progress events stop)
   161→   161→    let fallbackIdx = 0
   162→   162→    let receivedSSEProgress = false
   163→   163→    progressTimerRef.current = setInterval(() => {
   164→   164→      if (!receivedSSEProgress) {
   165→   165→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   166→   166→        setGenerationStatus(progressMessages[fallbackIdx])
   167→   167→      }
   168→   168→    }, 3000)
   169→   169→
   170→   170→    try {
   171→   171→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   172→   172→
   173→   173→      const res = await fetch('/api/store/generate', {
   174→   174→        method: 'POST',
   175→   175→        headers: { 'Content-Type': 'application/json' },
   176→   176→        body: JSON.stringify({ prompt: trimmed }),
   177→   177→        signal: controller.signal,
   178→   178→      })
   179→   179→
   180→   180→      if (!res.ok) {
   181→   181→        let errorMsg = `Server error (${res.status})`
   182→   182→        try {
   183→   183→          const errorData = await res.json()
   184→   184→          errorMsg = errorData.error || errorMsg
   185→   185→        } catch { /* ignore parse error */ }
   186→   186→        throw new Error(errorMsg)
   187→   187→      }
   188→   188→
   189→   189→      // ── Consume SSE stream ──
   190→   190→      const reader = res.body?.getReader()
   191→   191→      if (!reader) throw new Error('No response stream received.')
   192→   192→
   193→   193→      const decoder = new TextDecoder()
   194→   194→      let buffer = ''
   195→   195→      let resolved = false
   196→   196→
   197→   197→      while (true) {
   198→   198→        const { done, value } = await reader.read()
   199→   199→        if (done) break
   200→   200→
   201→   201→        buffer += decoder.decode(value, { stream: true })
   202→   202→
   203→   203→        // Parse SSE events from buffer
   204→   204→        // SSE format: "event: <name>\ndata: <json>\n\n" or ": comment\n\n"
   205→   205→        const lines = buffer.split('\n')
   206→   206→        // Keep incomplete last line in buffer
   207→   207→        buffer = lines.pop() || ''
   208→   208→
   209→   209→        let currentEvent = ''
   210→   210→        for (const line of lines) {
   211→   211→          if (line.startsWith(':')) {
   212→   212→            // SSE comment (heartbeat) — skip
   213→   213→            continue
   214→   214→          }
   215→   215→          if (line.startsWith('event: ')) {
   216→   216→            currentEvent = line.slice(7).trim()
   217→   217→          } else if (line.startsWith('data: ')) {
   218→   218→            const dataStr = line.slice(6)
   219→   219→
   220→   220→            // Handle SSE events
   221→   221→            if (currentEvent === 'progress') {
   222→   222→              receivedSSEProgress = true
   223→   223→              try {
   224→   224→                const data = JSON.parse(dataStr)
   225→   225→                setGenerationStatus(data.message || 'Processing...')
   226→   226→              } catch { /* ignore */ }
   227→   227→            } else if (currentEvent === 'result') {
   228→   228→              try {
   229→   229→                const data = JSON.parse(dataStr)
   230→   230→                if (!data.store) {
   231→   231→                  throw new Error('The AI response was missing store data.')
   232→   232→                }
   233→   233→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   234→   234→                resolved = true
   235→   235→                clearTimers()
   236→   236→
   237→   237→                if (data._isFallback) {
   238→   238→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   239→   239→                } else {
   240→   240→                  setStore(data.store)
   241→   241→                }
   242→   242→              } catch (parseErr) {
   243→   243→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   244→   244→                console.error('[Storqly] Failed to parse result event:', msg)
   245→   245→                throw new Error(msg)
   246→   246→              }
   247→   247→            } else if (currentEvent === 'error') {
   248→   248→              try {
   249→   249→                const data = JSON.parse(dataStr)
   250→   250→                throw new Error(data.message || 'Generation failed')
   251→   251→              } catch (e) {
   252→   252→                if (e instanceof Error && e.message !== 'Generation failed') throw e
   253→   253→                throw new Error('Generation failed')
   254→   254→              }
   255→   255→            }
   256→   256→            // Reset event after processing its data
   257→   257→            currentEvent = ''
   258→   258→          }
   259→   259→        }
   260→   260→      }
   261→   261→
   262→   262→      if (!resolved) {
   263→   263→        throw new Error('Stream ended without a result. The server may have disconnected.')
   264→   264→      }
   265→   265→    } catch (err: unknown) {
   266→   266→      clearTimers()
   267→   267→
   268→   268→      let message = 'Something went wrong. Please try again.'
   269→   269→      if (err instanceof Error) {
   270→   270→        if (err.name === 'AbortError') {
   271→   271→          message = 'Generation was cancelled.'
   272→   272→        } else {
   273→   273→          console.error('[Storqly] Generation error:', err)
   274→   274→          message = err.message
   275→   275→        }
   276→   276→      }
   277→   277→
   278→   278→      setError(message)
   279→   279→      setIsGenerating(false)
   280→   280→      setGenerationStatus('')
   281→   281→      setElapsedSeconds(0)
   282→   282→      toast.error('Store generation failed', { description: message })
   283→   283→    }
   284→   284→  }, [promptText, session, setIsGenerating, setStore, setStoreWithFallback, clearTimers, setAuthOpen])
   285→   285→
   286→   286→  // Cleanup on unmount
   287→   287→  useEffect(() => {
   288→   288→    return () => {
   289→   289→      clearTimers()
   290→   290→      abortRef.current?.abort()
   291→   291→    }
   292→   292→  }, [clearTimers])
   293→   293→
   294→   294→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   295→   295→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   296→   296→      e.preventDefault()
   297→   297→      handleGenerate()
   298→   298→    }
   299→   299→  }
   300→   300→
   301→   301→  const formatElapsed = (seconds: number) => {

   305→   305→    return `${m}:${s.toString().padStart(2, '0')}`
   306→   306→  }

   313→   313→          <span className="text-2xl font-bold tracking-tight bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   314→   314→            Storqly
   315→   315→          </span>
   316→   316→          <div className="hidden sm:flex items-center gap-4">
   317→   317→            <span className="text-sm text-zinc-500">
   318→   318→              AI-Powered Store Builder
   319→   319→            </span>
   320→   320→            <AuthButton onSignIn={() => setAuthOpen(true)} />
   321→   321→          </div>
   322→   322→          <div className="flex sm:hidden">
   323→   323→            <AuthButton onSignIn={() => setAuthOpen(true)} />
   324→   324→          </div>
   325→   325→        </nav>
   326→   326→      </header>
   327→   327→
   328→   328→      <AuthModal open={authOpen} onOpenChange={setAuthOpen} />
   329→   329→
   330→   330→      {/* ── Hero Section ── */}
   331→   331→      <section className="relative flex flex-1 flex-col items-center justify-center px-5 pt-32 pb-20 sm:pt-40 sm:pb-28">
   332→   332→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   333→   333→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />
   334→   334→        </div>
   335→   335→
   336→   336→        <div className="relative z-10 mx-auto w-full max-w-3xl text-center">
   337→   337→          <motion.h1
   338→   338→            custom={0}
   339→   339→            variants={fadeUp}
   340→   340→            initial="hidden"
   341→   341→            animate="visible"
   342→   342→            className="text-4xl font-extrabold leading-[1.1] tracking-tight sm:text-5xl md:text-6xl lg:text-7xl"
   343→   343→          >
   344→   344→            Build Your Store{' '}

   670→   670→function EditorToolbar({
   671→   671→  onToggleLeft,
   672→   672→  onToggleRight,
   673→   673→  showLeft,
   674→   674→  showRight,
   675→   675→}: {
   676→   676→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   677→   677→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   678→   678→  showLeft: boolean
   679→   679→  showRight: boolean
   680→   680→}) {
   681→   681→  const store = useStoreEditor((s) => s.store)
   682→   682→  const reset = useStoreEditor((s) => s.reset)
   683→   683→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   684→   684→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   685→   685→  const isPublished = useStoreEditor((s) => s.isPublished)
   686→   686→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   687→   687→  const [isSaving, setIsSaving] = useState(false)
   688→   688→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
   689→   689→  const [copied, setCopied] = useState(false)
   690→   690→  const [authOpen, setAuthOpen] = useState(false)
   691→   691→
   692→   692→  const handleBack = () => {
   693→   693→    reset()
   694→   694→  }
   695→   695→
   696→   696→  const handleSave = async () => {
   697→   697→    if (!store || isSaving) return
   698→   698→    setIsSaving(true)
   699→   699→    try {
   700→   700→      const res = await fetch('/api/store/save', {
   701→   701→        method: 'POST',
   702→   702→        headers: { 'Content-Type': 'application/json' },
   703→   703→        body: JSON.stringify({ store }),
   704→   704→      })
   705→   705→      if (!res.ok) throw new Error('Save failed')
   706→   706→      toast.success('Store saved')
   707→   707→    } catch {
   708→   708→      toast.error('Failed to save store')
   709→   709→    } finally {
   710→   710→      setIsSaving(false)
   711→   711→    }
   712→   712→  }
   713→   713→
   714→   714→  const handlePublish = async () => {
   715→   715→    if (!store || isPublishing) return
   716→   716→    setIsPublishing(true)
   717→   717→    try {
   718→   718→      const res = await fetch('/api/store/publish', {
   719→   719→        method: 'POST',
   720→   720→        headers: { 'Content-Type': 'application/json' },
   721→   721→        body: JSON.stringify({ store }),
   722→   722→      })
   723→   723→      if (!res.ok) throw new Error('Publish failed')
   724→   724→      const data = await res.json()
   725→   725→      const slug = data.slug
   726→   726→      setIsPublished(true)
   727→   727→      // Build a real viewable URL using the current origin
   728→   728→      const baseUrl = window.location.origin
   729→   729→      const viewUrl = `${baseUrl}/?store=${slug}`
   730→   730→      setPublishedUrl(viewUrl)
   731→   731→      toast.success('Store published successfully!')
   732→   732→    } catch {
   733→   733→      toast.error('Failed to publish store')
   734→   734→    } finally {
   735→   735→      setIsPublishing(false)
   736→   736→    }
   737→   737→  }
   738→   738→
   739→   739→  const handleCopyUrl = useCallback(async () => {
   740→   740→    if (!publishedUrl) return
   741→   741→    try {
   742→   742→      await navigator.clipboard.writeText(publishedUrl)
   743→   743→      setCopied(true)
   744→   744→      toast.success('URL copied to clipboard!')
   745→   745→      setTimeout(() => setCopied(false), 2000)
   746→   746→    } catch {
   747→   747→      // Fallback for older browsers
   748→   748→      const textarea = document.createElement('textarea')
   749→   749→      textarea.value = publishedUrl
   750→   750→      document.body.appendChild(textarea)
   751→   751→      textarea.select()
   752→   752→      document.execCommand('copy')
   753→   753→      document.body.removeChild(textarea)
   754→   754→      setCopied(true)
   755→   755→      toast.success('URL copied to clipboard!')
   756→   756→      setTimeout(() => setCopied(false), 2000)
   757→   757→    }
   758→   758→  }, [publishedUrl])
   759→   759→
   760→   760→  return (
   761→   761→    <>
   762→   762→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
   763→   763→        <div className="flex items-center gap-1">
   764→   764→          <Button
   765→   765→            variant="ghost"
   766→   766→            size="icon"
   767→   767→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
   768→   768→            onClick={handleBack}
   769→   769→            aria-label="Back to home"
   770→   770→          >
   771→   771→            <ArrowLeft className="h-4 w-4" />
   772→   772→          </Button>
   773→   773→          <div className="h-5 w-px bg-zinc-800" />
   774→   774→          <Button
   775→   775→            variant={showLeft ? 'secondary' : 'ghost'}
   776→   776→            size="sm"
   777→   777→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
   778→   778→            onClick={() => onToggleLeft((v) => !v)}
   779→   779→          >
   780→   780→            <Layers className="h-3.5 w-3.5" />
   781→   781→            <span className="hidden sm:inline">Sections</span>
   782→   782→          </Button>
   783→   783→          <Button
   784→   784→            variant={showRight ? 'secondary' : 'ghost'}
   785→   785→            size="sm"
   786→   786→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}

   789→   789→            <MessageSquare className="h-3.5 w-3.5" />
   790→   790→            <span className="hidden sm:inline">Chat</span>
   791→   791→          </Button>
   792→   792→        </div>
   793→   793→
   794→   794→        <div className="flex items-center gap-1.5">
   795→   795→          {publishedUrl && (
   796→   796→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
   797→   797→              <Check className="h-3 w-3 text-emerald-400" />

   935→   935→

   939→   939→      try {
   940→   940→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
   941→   941→        if (!res.ok) {

   943→   943→          throw new Error(data.error || `Store not found (${res.status})`)
   944→   944→        }

   947→   947→      } catch (err: unknown) {

   949→   949→      } finally {
