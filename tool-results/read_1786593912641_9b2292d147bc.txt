    41→type NormLog = { field: string; action: 'missing' | 'coerced' | 'invalid' | 'defaulted'; from?: unknown; to?: unknown };

    43→function createLogger() {
    44→  const logs: NormLog[] = [];
    45→  return {
    46→    log(entry: NormLog) { logs.push(entry); },
    47→    getLogs() { return logs; },
    48→    summary(): string {
    49→      const counts = { missing: 0, coerced: 0, invalid: 0, defaulted: 0 };
    50→      for (const l of logs) counts[l.action]++;
    51→      return `Normalization: ${logs.length} fixes (${counts.missing} missing, ${counts.coerced} coerced, ${counts.invalid} invalid, ${counts.defaulted} defaulted)`;
    52→    },
    53→  };
    54→}

    62→function str(v: unknown, fallback: string): string {
    63→  if (typeof v === 'string' && v.length > 0) return v;
    64→  if (v === null || v === undefined) return fallback;
    65→  return String(v).slice(0, 500) || fallback;

    68→function num(v: unknown, fallback: number, min?: number, max?: number): number {
    69→  if (typeof v === 'number' && isFinite(v)) {
    70→    let n = v;
    71→    if (min !== undefined) n = Math.max(min, n);
    72→    if (max !== undefined) n = Math.min(max, n);
    73→    return n;
    74→  }
    75→  // Coerce string numbers
    76→  if (typeof v === 'string') {
    77→    const n = parseFloat(v.replace(/[^\d.\-]/g, ''));
    78→    if (!isNaN(n) && isFinite(n)) {
    79→      if (min !== undefined) return Math.max(min, n);
    80→      if (max !== undefined) return Math.min(max, n);
    81→      return n;
    82→    }
    83→  }
    84→  return fallback;
    85→}

    87→function bool(v: unknown, fallback: boolean): boolean {
    88→  if (typeof v === 'boolean') return v;
    89→  if (typeof v === 'string') return v === 'true' || v === '1';
    90→  if (typeof v === 'number') return v !== 0;
    91→  return fallback;
    92→}

    99→function obj(v: unknown, fallback: Record<string, unknown>): Record<string, unknown> {
   100→  if (v !== null && typeof v === 'object' && !Array.isArray(v)) return v as Record<string, unknown>;
   101→  return fallback;

   104→function enumVal(v: unknown, valid: Set<string>, fallback: string): string {
   105→  if (typeof v === 'string' && valid.has(v)) return v;
   106→  // Case-insensitive fallback
   107→  if (typeof v === 'string') {
   108→    const lower = v.toLowerCase();
   109→    for (const validVal of valid) {
   110→      if (validVal === lower) return validVal;
   111→    }
   112→  }
   113→  return fallback;
   114→}

   419→// ─── Cross-reference fixes ────────────────────────────────────────

   421→/** Hard-cap enforcement: truncate sections and products to safe output limits.
   422→ *  This is the safety net if the AI ignores the system prompt caps.
   423→ *  Layers of defense:
   424→ *    1. sanitizePrompt() in route.ts strips count requests (prevents AI from trying)
   425→ *    2. System prompt ABSOLUTE CAPS language (strong AI guidance)
   426→ *    3. THIS function: hard truncation (final safety net)
   427→ */
   428→function enforceOutputCaps(store: Store, log: ReturnType<typeof createLogger>): void {
   429→  const MAX_PRODUCTS = 3;
   430→  const MAX_SECTIONS = 4;
   431→
   432→  // ── Cap products ──
   433→  if (store.products.length > MAX_PRODUCTS) {
   434→    const before = store.products.length;
   435→    store.products = store.products.slice(0, MAX_PRODUCTS);
   436→    // Ensure at least 1 featured product after truncation
   437→    if (!store.products.some((p) => p.featured) && store.products.length > 0) {
   438→      store.products[0].featured = true;
   439→    }
   440→    log.log({
   441→      field: 'products',
   442→      action: 'coerced',
   443→      from: `${before} products`,
   444→      to: `${MAX_PRODUCTS} products (capped)`,
   445→    });
   446→  }
   447→
   448→  // ── Cap sections per page ──
   449→  for (const page of store.pages) {
   450→    if (page.sections.length > MAX_SECTIONS) {
   451→      const before = page.sections.length;
   452→      page.sections = page.sections.slice(0, MAX_SECTIONS);
   453→      log.log({
   454→        field: `pages[${page.name}].sections`,
   455→        action: 'coerced',
   456→        from: `${before} sections`,
   457→        to: `${MAX_SECTIONS} sections (capped)`,
   458→      });
   459→    }
   460→  }
   461→}

   463→/** Fix productIds in featured-products sections to reference actual products */
   464→function fixProductReferences(store: Store, log: ReturnType<typeof createLogger>): void {
   465→  const validProductIds = new Set(store.products.map((p) => p.id));
   466→  const allProductIds = store.products.map((p) => p.id);
   467→
   468→  for (const page of store.pages) {
   469→    for (const section of page.sections) {
   470→      if (section.type === 'featured-products' && Array.isArray(section.content.productIds)) {
   471→        const before = section.content.productIds.length;
   472→        // Keep only IDs that reference actual products
   473→        const validIds = (section.content.productIds as string[]).filter((id) => validProductIds.has(id));
   474→
   475→        // Fill missing slots with other product IDs so all products are shown
   476→        const usedIds = new Set(validIds);
   477→        for (const pid of allProductIds) {
   478→          if (validIds.length >= store.products.length) break;
   479→          if (!usedIds.has(pid)) {
   480→            validIds.push(pid);
   481→            usedIds.add(pid);
   482→          }
   483→        }
   484→
   485→        if (validIds.length !== before) {
   486→          const invalidCount = before - (validIds.length - (store.products.length - before > 0 ? store.products.length - before : 0));
   487→          log.log({ field: 'featured-products.productIds', action: 'coerced', from: `${before} IDs (some invalid)`, to: `${validIds.length} valid IDs` });
   488→        }
   489→        section.content.productIds = validIds;
   490→      }
   491→    }
   492→  }
   493→}

   519→function padProducts(store: Store, log: ReturnType<typeof createLogger>): void {
   520→  const deficit = MIN_PRODUCTS - store.products.length;
   521→  if (deficit <= 0) return;
   522→  const before = store.products.length;
   523→  for (let i = 0; i < deficit; i++) {
   524→    const tmpl = PAD_PRODUCT_TEMPLATES[i % PAD_PRODUCT_TEMPLATES.length];
   525→    store.products.push({
   526→      id: uuid(),
   527→      name: tmpl.name + (store.products.length + 1),
   528→      price: tmpl.price + store.products.length * 5,
   529→      compareAtPrice: null,
   530→      images: [`https://images.unsplash.com/${tmpl.img}?w=600`],
   531→      description: tmpl.description,
   532→      category: tmpl.category,
   533→      featured: false,
   534→      inStock: true,
   535→    });
   536→  }
   537→  log.log({ field: 'products', action: 'defaulted', from: `${before} products`, to: `${MIN_PRODUCTS} products (padded)` });
   538→}

   542→export interface NormalizeResult {
   543→  store: Store;
   544→  log: string[];        // Human-readable log lines
   545→  summary: string;      // One-line summary
   546→  isFallback: boolean;  // true if the input was too broken to normalize
   547→  normalizationCount: number; // Number of fixes applied
   548→}
   549→
   550→/**
   551→ * Normalize any valid JSON object into a conformant Store.
   552→ *
   553→ * This function NEVER throws. If the input is too malformed to recover,
   554→ * it returns null and the caller should use a hardcoded fallback store.
   555→ *
   556→ * @param raw - Parsed JSON object (JSON.parse already succeeded)
   557→ * @param prompt - Original user prompt (for fallback name extraction)
   558→ */
   559→export function normalizeStore(raw: unknown, prompt?: string): NormalizeResult | null {
   560→  const log = createLogger();
   561→
   562→  // ── Guard: must be a non-null object ──
   563→  if (raw === null || typeof raw !== 'object' || Array.isArray(raw)) {
   564→    console.warn('[normalizeStore] Input is not an object:', typeof raw);
   565→    return null;
   566→  }
   567→
   568→  const r = raw as Record<string, unknown>;
   569→
   570→  // ── Extract store name (used as fallback for missing name) ──
   571→  const storeName = oneLine(str(r.name, prompt ? extractStoreNameFromPrompt(prompt) : 'My Store'));
   572→
   573→  if (!r.name || typeof r.name !== 'string') {
   574→    log.log({ field: 'name', action: 'missing', to: storeName });
   575→  }
   576→
   577→  // ── Top-level fields ──
   578→  const now = new Date().toISOString();
   579→  const store: Store = {
   580→    id: isUUID(str(r.id, '')) ? str(r.id, '') : uuid(),
   581→    name: storeName.substring(0, 100),
   582→    slug: slugify(storeName),
   583→    description: r.description !== undefined ? oneLine(str(r.description, '')) : undefined,
   584→    theme: r.theme ? normalizeTheme(r.theme, log) : defaultTheme,
   585→    pages: Array.isArray(r.pages)
   586→      ? r.pages.map((p: unknown) => normalizePage(p, log))
   587→      : [createDefaultPage(storeName)],
   588→    products: Array.isArray(r.products)
   589→      ? (r.products as unknown[]).map((p: unknown) => normalizeProduct(p, log))
   590→      : [],
   591→    published: bool(r.published, false),
   592→    publishedAt: r.publishedAt !== undefined ? str(r.publishedAt, '') : undefined,
   593→    createdAt: str(r.createdAt, now),
   594→    updatedAt: str(r.updatedAt, now),
   595→  };
   596→
   597→  // Log missing arrays
   598→  if (!Array.isArray(r.pages)) log.log({ field: 'pages', action: 'missing' });
   599→  if (!Array.isArray(r.products)) log.log({ field: 'products', action: 'missing' });
   600→  if (!r.theme) log.log({ field: 'theme', action: 'missing' });
   601→
   602→  // ── Ensure at least one page exists ──
   603→  if (store.pages.length === 0) {
   604→    store.pages.push(createDefaultPage(storeName));
   605→    log.log({ field: 'pages', action: 'defaulted' });
   606→  }
   607→
   608→  // ── Ensure at least one product exists (renderer expects products array) ──
   609→  if (store.products.length === 0) {
   610→    store.products.push(createDefaultProduct());
   611→    log.log({ field: 'products', action: 'defaulted' });
   612→  }
   613→
   614→  // ── Cross-reference fixes ──
   615→  // IMPORTANT: enforceOutputCaps MUST run before fixProductReferences
   616→  // because product references depend on the final product list
   617→  enforceOutputCaps(store, log);
   618→  padProducts(store, log);
   619→  fixProductReferences(store, log);
   620→  ensureHomepage(store, log);
   621→  ensureFeaturedProducts(store, log);
   622→
   623→  // ── Build result ──
   624→  const logEntries = log.getLogs();
   625→  const logLines = logEntries.map(
   626→    (l) => `[${l.action}] ${l.field}${l.from !== undefined ? ` (was: ${JSON.stringify(l.from).substring(0, 60)})` : ''}${l.to !== undefined ? ` → ${JSON.stringify(l.to).substring(0, 60)}` : ''}`
   627→  );
   628→
   629→  return {
   630→    store,
   631→    log: logLines,
   632→    summary: log.summary(),
   633→    isFallback: false,
   634→    normalizationCount: logEntries.length,
   635→  };
   636→}

   677→/** Extract a short store name from a user prompt (lightweight version for normalization) */
   678→function extractStoreNameFromPrompt(prompt: string): string {
   679→  const text = prompt.trim();
   680→
   681→  // Brand name after "called / named / known as"
   682→  const calledMatch = text.match(/(?:called|named|known\s+as)\s+([\w&'\-]+(?:\s+[\w&'\-]+){0,2})/i);
   683→  if (calledMatch?.[1]) {
   684→    const name = calledMatch[1].replace(/\s+(selling|with|that|for|using|featuring)\s*$/i, '').trim();
   685→    if (name.length >= 2 && name.length <= 40) return name;
   686→  }
   687→
   688→  // Quoted brand name
   689→  const quotedMatch = text.match(/["']([^"']{2,40})["']/);
   690→  if (quotedMatch?.[1]) return quotedMatch[1].trim();
   691→
   692→  // Strip common prefixes
   693→  const stripped = text
   694→    .replace(/^(build|create|make|design|set\s+up)\s+(a|an|the|my)\s+/i, '')
   695→    .replace(/\b(online|e-commerce|ecommerce)\s+(store|shop|boutique)\b/gi, '')
   696→    .replace(/\b(store|shop|boutique|website|site|brand)\b/gi, '')
   697→    .replace(/\b(selling|with|that|for|using|featuring)\b.*/i, '')
   698→    .trim();
   699→
   700→  // Find title-case run
   701→  const words = stripped.split(/\s+/).filter(w => w.length >= 2);
   702→  const isTitle = (w: string) => w[0] === w[0].toUpperCase() && w[0] !== w[0].toLowerCase();
   703→  let bestRun: string[] = [];
   704→  let currentRun: string[] = [];
   705→  for (const w of words) {
   706→    if (isTitle(w)) {
   707→      currentRun.push(w.replace(/[^a-zA-Z0-9&'-]/g, ''));
   708→    } else {
   709→      if (currentRun.length > bestRun.length) bestRun = currentRun;
   710→      currentRun = [];
   711→    }
   712→  }
   713→  if (currentRun.length > bestRun.length) bestRun = currentRun;
   714→
   715→  if (bestRun.length >= 1) {
   716→    const candidate = bestRun.slice(0, 3).join(' ');
   717→    if (candidate.length >= 2 && candidate.length <= 40) return candidate;
   718→  }
   719→
   720→  return 'My Store';
   721→}
