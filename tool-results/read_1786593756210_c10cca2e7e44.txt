    91→    91→function LandingPage() {
    92→    92→  const [promptText, setPromptText] = useState('')
    93→    93→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    94→    94→  const progressTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    95→    95→  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
    96→    96→  const abortRef = useRef<AbortController | null>(null)
    97→    97→
    98→    98→  const {
    99→    99→    isGenerating,
   100→   100→    setIsGenerating,
   101→   101→    setStore,
   102→   102→    setStoreWithFallback,
   103→   103→  } = useStoreEditor()
   104→   104→
   105→   105→  // Local UI state (not in Zustand — this is view-level)
   106→   106→  const [generationStatus, setGenerationStatus] = useState('')
   107→   107→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   108→   108→  const [error, setError] = useState<string | null>(null)
   109→   109→
   110→   110→  const clearTimers = useCallback(() => {
   111→   111→    if (progressTimerRef.current) { clearTimeout(progressTimerRef.current); progressTimerRef.current = null }
   112→   112→    if (elapsedTimerRef.current) { clearInterval(elapsedTimerRef.current); elapsedTimerRef.current = null }
   113→   113→  }, [])
   114→   114→
   115→   115→  const handleCancel = useCallback(() => {
   116→   116→    abortRef.current?.abort()
   117→   117→    clearTimers()
   118→   118→    setIsGenerating(false)
   119→   119→    setGenerationStatus('')
   120→   120→    setElapsedSeconds(0)
   121→   121→    setError(null)
   122→   122→  }, [clearTimers, setIsGenerating])
   123→   123→
   124→   124→  const handleGenerate = useCallback(async () => {
   125→   125→    const trimmed = promptText.trim()
   126→   126→    if (!trimmed) {
   127→   127→      setError('Please describe your store to get started.')
   128→   128→      textareaRef.current?.focus()
   129→   129→      return
   130→   130→    }
   131→   131→
   132→   132→    // Reset state
   133→   133→    setError(null)
   134→   134→    setIsGenerating(true)
   135→   135→    setGenerationStatus('Starting generation...')
   136→   136→    setElapsedSeconds(0)
   137→   137→
   138→   138→    // Create abort controller for this request
   139→   139→    const controller = new AbortController()
   140→   140→    abortRef.current = controller
   141→   141→
   142→   142→    // Elapsed time counter
   143→   143→    elapsedTimerRef.current = setInterval(() => {
   144→   144→      setElapsedSeconds((prev) => prev + 1)
   145→   145→    }, 1000)
   146→   146→
   147→   147→    // Fallback progress cycling (used only if SSE progress events stop)
   148→   148→    let fallbackIdx = 0
   149→   149→    let receivedSSEProgress = false
   150→   150→    progressTimerRef.current = setInterval(() => {
   151→   151→      if (!receivedSSEProgress) {
   152→   152→        fallbackIdx = (fallbackIdx + 1) % progressMessages.length
   153→   153→        setGenerationStatus(progressMessages[fallbackIdx])
   154→   154→      }
   155→   155→    }, 3000)
   156→   156→
   157→   157→    try {
   158→   158→      console.log('[Storqly] Starting SSE store generation for prompt:', trimmed)
   159→   159→
   160→   160→      const res = await fetch('/api/store/generate', {
   161→   161→        method: 'POST',
   162→   162→        headers: { 'Content-Type': 'application/json' },
   163→   163→        body: JSON.stringify({ prompt: trimmed }),
   164→   164→        signal: controller.signal,
   165→   165→      })
   166→   166→
   167→   167→      if (!res.ok) {
   168→   168→        let errorMsg = `Server error (${res.status})`
   169→   169→        try {
   170→   170→          const errorData = await res.json()
   171→   171→          errorMsg = errorData.error || errorMsg
   172→   172→        } catch { /* ignore parse error */ }
   173→   173→        throw new Error(errorMsg)
   174→   174→      }
   175→   175→
   176→   176→      // ── Consume SSE stream ──
   177→   177→      const reader = res.body?.getReader()
   178→   178→      if (!reader) throw new Error('No response stream received.')
   179→   179→
   180→   180→      const decoder = new TextDecoder()
   181→   181→      let buffer = ''
   182→   182→      let resolved = false
   183→   183→
   184→   184→      while (true) {
   185→   185→        const { done, value } = await reader.read()
   186→   186→        if (done) break
   187→   187→
   188→   188→        buffer += decoder.decode(value, { stream: true })
   189→   189→
   190→   190→        // Parse SSE events from buffer
   191→   191→        // SSE format: "event: <name>\ndata: <json>\n\n" or ": comment\n\n"
   192→   192→        const lines = buffer.split('\n')
   193→   193→        // Keep incomplete last line in buffer
   194→   194→        buffer = lines.pop() || ''
   195→   195→
   196→   196→        let currentEvent = ''
   197→   197→        for (const line of lines) {
   198→   198→          if (line.startsWith(':')) {
   199→   199→            // SSE comment (heartbeat) — skip
   200→   200→            continue
   201→   201→          }
   202→   202→          if (line.startsWith('event: ')) {
   203→   203→            currentEvent = line.slice(7).trim()
   204→   204→          } else if (line.startsWith('data: ')) {
   205→   205→            const dataStr = line.slice(6)
   206→   206→
   207→   207→            // Handle SSE events
   208→   208→            if (currentEvent === 'progress') {
   209→   209→              receivedSSEProgress = true
   210→   210→              try {
   211→   211→                const data = JSON.parse(dataStr)
   212→   212→                setGenerationStatus(data.message || 'Processing...')
   213→   213→              } catch { /* ignore */ }
   214→   214→            } else if (currentEvent === 'result') {
   215→   215→              try {
   216→   216→                const data = JSON.parse(dataStr)
   217→   217→                if (!data.store) {
   218→   218→                  throw new Error('The AI response was missing store data.')
   219→   219→                }
   220→   220→                console.log('[Storqly] Store generated via SSE. isFallback:', data._isFallback, 'name:', data.store.name)
   221→   221→                resolved = true
   222→   222→                clearTimers()
   223→   223→
   224→   224→                if (data._isFallback) {
   225→   225→                  setStoreWithFallback(data.store, true, data._fallbackReason || 'AI generation failed')
   226→   226→                } else {
   227→   227→                  setStore(data.store)
   228→   228→                }
   229→   229→              } catch (parseErr) {
   230→   230→                const msg = parseErr instanceof Error ? parseErr.message : String(parseErr)
   231→   231→                console.error('[Storqly] Failed to parse result event:', msg)
   232→   232→                throw new Error(msg)
   233→   233→              }
   234→   234→            } else if (currentEvent === 'error') {
   235→   235→              try {
   236→   236→                const data = JSON.parse(dataStr)
   237→   237→                throw new Error(data.message || 'Generation failed')
   238→   238→              } catch (e) {
   239→   239→                if (e instanceof Error && e.message !== 'Generation failed') throw e
   240→   240→                throw new Error('Generation failed')
   241→   241→              }
   242→   242→            }
   243→   243→            // Reset event after processing its data
   244→   244→            currentEvent = ''
   245→   245→          }
   246→   246→        }
   247→   247→      }
   248→   248→
   249→   249→      if (!resolved) {
   250→   250→        throw new Error('Stream ended without a result. The server may have disconnected.')
   251→   251→      }
   252→   252→    } catch (err: unknown) {
   253→   253→      clearTimers()
   254→   254→
   255→   255→      let message = 'Something went wrong. Please try again.'
   256→   256→      if (err instanceof Error) {
   257→   257→        if (err.name === 'AbortError') {
   258→   258→          message = 'Generation was cancelled.'
   259→   259→        } else {
   260→   260→          console.error('[Storqly] Generation error:', err)
   261→   261→          message = err.message
   262→   262→        }
   263→   263→      }
   264→   264→
   265→   265→      setError(message)
   266→   266→      setIsGenerating(false)
   267→   267→      setGenerationStatus('')
   268→   268→      setElapsedSeconds(0)
   269→   269→      toast.error('Store generation failed', { description: message })
   270→   270→    }
   271→   271→  }, [promptText, setIsGenerating, setStore, setStoreWithFallback, clearTimers])
   272→   272→
   273→   273→  // Cleanup on unmount
   274→   274→  useEffect(() => {
   275→   275→    return () => {
   276→   276→      clearTimers()
   277→   277→      abortRef.current?.abort()
   278→   278→    }
   279→   279→  }, [clearTimers])
   280→   280→
   281→   281→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   282→   282→    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   283→   283→      e.preventDefault()
   284→   284→      handleGenerate()
   285→   285→    }
   286→   286→  }
   287→   287→
   288→   288→  const formatElapsed = (seconds: number) => {
   289→   289→    if (seconds < 60) return `${seconds}s`
   290→   290→    const m = Math.floor(seconds / 60)
   291→   291→    const s = seconds % 60
   292→   292→    return `${m}:${s.toString().padStart(2, '0')}`
   293→   293→  }

   300→   300→          <span className="text-2xl font-bold tracking-tight bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   301→   301→            Storqly

   308→   308→
   309→   309→      {/* ── Hero Section ── */}

   311→   311→        <div className="pointer-events-none absolute inset-0 overflow-hidden">
   312→   312→          <div className="absolute left-1/2 top-0 h-[600px] w-[800px] -translate-x-1/2 -translate-y-1/3 rounded-full bg-gradient-to-br from-[#a855f7]/15 via-[#ec4899]/10 to-transparent blur-3xl" />
   313→   313→        </div>
   314→   314→
   315→   315→        <div className="relative z-10 mx-auto w-full max-w-3xl text-center">
   316→   316→          <motion.h1
   317→   317→            custom={0}
   318→   318→            variants={fadeUp}
   319→   319→            initial="hidden"
   320→   320→            animate="visible"
   321→   321→            className="text-4xl font-extrabold leading-[1.1] tracking-tight sm:text-5xl md:text-6xl lg:text-7xl"
   322→   322→          >
   323→   323→            Build Your Store{' '}
   324→   324→            <span className="bg-gradient-to-r from-[#a855f7] via-[#ec4899] to-[#f43f5e] bg-clip-text text-transparent">
   325→   325→              in Seconds

   328→   328→
   329→   329→          <motion.p
   330→   330→            custom={1}
   331→   331→            variants={fadeUp}
   332→   332→            initial="hidden"
   333→   333→            animate="visible"
   334→   334→            className="mt-6 text-lg text-zinc-400 sm:text-xl"

   339→   339→          <motion.div

   345→   345→          >
   346→   346→            <div className="group relative rounded-2xl border border-white/[0.08] bg-white/[0.03] shadow-2xl shadow-black/40 transition-all duration-300 focus-within:border-[#a855f7]/40 focus-within:shadow-[0_0_40px_-8px_rgba(168,85,247,0.25)]">
   347→   347→              <Textarea

   359→   359→                  Press{' '}

   885→   885→function PublishedStoreViewer({ slug }: { slug: string }) {
   886→   886→  const [store, setStore] = useState<Store | null>(null)
   887→   887→  const [loading, setLoading] = useState(true)
   888→   888→  const [error, setError] = useState<string | null>(null)
   889→   889→
   890→   890→  useEffect(() => {
   891→   891→    let cancelled = false
   892→   892→    async function load() {
   893→   893→      try {
   894→   894→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
   895→   895→        if (!res.ok) {
   896→   896→          const data = await res.json().catch(() => ({}))
   897→   897→          throw new Error(data.error || `Store not found (${res.status})`)
   898→   898→        }
   899→   899→        const data = await res.json()
   900→   900→        if (!cancelled) setStore(data.store)
   901→   901→      } catch (err: unknown) {
   902→   902→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
   903→   903→      } finally {
   904→   904→        if (!cancelled) setLoading(false)
