   386→                finishWithError(msg)

   399→              }

   402→          }

   424→        }
   425→      }

   427→      setIsGenerating(false)
   428→      setGenerationStatus('')
   429→      setElapsedSeconds(0)
   430→      toast.error('Store generation failed', { description: message })
   431→    }

   943→function EditorToolbar({
   944→  onToggleLeft,
   945→  onToggleRight,
   946→  showLeft,
   947→  showRight,
   948→}: {
   949→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
   950→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
   951→  showLeft: boolean
   952→  showRight: boolean
   953→}) {
   954→  const store = useStoreEditor((s) => s.store)
   955→  const reset = useStoreEditor((s) => s.reset)
   956→  const isPublishing = useStoreEditor((s) => s.isPublishing)
   957→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
   958→  const isPublished = useStoreEditor((s) => s.isPublished)
   959→  const setIsPublished = useStoreEditor((s) => s.setIsPublished)
   960→  const [isSaving, setIsSaving] = useState(false)
   961→  const [publishedUrl, setPublishedUrl] = useState<string | null>(null)
   962→  const [copied, setCopied] = useState(false)
   963→  const [authOpen, setAuthOpen] = useState(false)
   964→
   965→  const handleBack = () => {
   966→    reset()
   967→  }
   968→
   969→  const handleSave = async () => {
   970→    if (!store || isSaving) return
   971→    setIsSaving(true)
   972→    try {
   973→      const res = await fetch('/api/store/save', {
   974→        method: 'POST',
   975→        headers: { 'Content-Type': 'application/json' },
   976→        body: JSON.stringify({ store }),
   977→      })
   978→      if (!res.ok) throw new Error('Save failed')
   979→      const data = await res.json()
   980→      toast.success(`Draft saved — ID: ${data.id}`, {
   981→        description: `Slug: ${data.slug}`,
   982→        duration: 5000,
   983→      })
   984→    } catch {
   985→      toast.error('Failed to save store')
   986→    } finally {
   987→      setIsSaving(false)
   988→    }
   989→  }
   990→
   991→  const handlePublish = async () => {
   992→    if (!store || isPublishing) return
   993→    setIsPublishing(true)
   994→    try {
   995→      const res = await fetch('/api/store/publish', {
   996→        method: 'POST',
   997→        headers: { 'Content-Type': 'application/json' },
   998→        body: JSON.stringify({ store }),
   999→      })
  1000→      if (!res.ok) throw new Error('Publish failed')
  1001→      const data = await res.json()
  1002→      const slug = data.slug
  1003→      setIsPublished(true)
  1004→      // Build a real viewable URL using the current origin
  1005→      const baseUrl = window.location.origin
  1006→      const viewUrl = `${baseUrl}/?store=${slug}`
  1007→      setPublishedUrl(viewUrl)
  1008→      toast.success('Store published successfully!')
  1009→    } catch {
  1010→      toast.error('Failed to publish store')
  1011→    } finally {
  1012→      setIsPublishing(false)
  1013→    }
  1014→  }
  1015→
  1016→  const handleCopyUrl = useCallback(async () => {
  1017→    if (!publishedUrl) return
  1018→    try {
  1019→      await navigator.clipboard.writeText(publishedUrl)
  1020→      setCopied(true)
  1021→      toast.success('URL copied to clipboard!')
  1022→      setTimeout(() => setCopied(false), 2000)
  1023→    } catch {
  1024→      // Fallback for older browsers
  1025→      const textarea = document.createElement('textarea')
  1026→      textarea.value = publishedUrl
  1027→      document.body.appendChild(textarea)
  1028→      textarea.select()
  1029→      document.execCommand('copy')
  1030→      document.body.removeChild(textarea)
  1031→      setCopied(true)
  1032→      toast.success('URL copied to clipboard!')
  1033→      setTimeout(() => setCopied(false), 2000)
  1034→    }
  1035→  }, [publishedUrl])
  1036→
  1037→  return (
  1038→    <>
  1039→      <div className="flex h-12 items-center justify-between border-b border-zinc-800 bg-zinc-950 px-3">
  1040→        <div className="flex items-center gap-1">
  1041→          <Button
  1042→            variant="ghost"
  1043→            size="icon"
  1044→            className="h-8 w-8 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"
  1045→            onClick={handleBack}
  1046→            aria-label="Back to home"
  1047→          >
  1048→            <ArrowLeft className="h-4 w-4" />
  1049→          </Button>
  1050→          <div className="h-5 w-px bg-zinc-800" />
  1051→          <Button
  1052→            variant={showLeft ? 'secondary' : 'ghost'}
  1053→            size="sm"
  1054→            className={`h-8 gap-1.5 ${showLeft ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
  1055→            onClick={() => onToggleLeft((v) => !v)}
  1056→          >
  1057→            <Layers className="h-3.5 w-3.5" />
  1058→            <span className="hidden sm:inline">Sections</span>
  1059→          </Button>
  1060→          <Button
  1061→            variant={showRight ? 'secondary' : 'ghost'}
  1062→            size="sm"
  1063→            className={`h-8 gap-1.5 ${showRight ? 'bg-zinc-800 text-zinc-200 hover:bg-zinc-700' : 'text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800'}`}
  1064→            onClick={() => onToggleRight((v) => !v)}
  1065→          >
  1066→            <MessageSquare className="h-3.5 w-3.5" />
  1067→            <span className="hidden sm:inline">Chat</span>
  1068→          </Button>
  1069→        </div>
  1070→
  1071→        <div className="flex items-center gap-1.5">
  1072→          {publishedUrl && (
  1073→            <div className="mr-2 hidden sm:flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-2.5 py-1">
  1074→              <Check className="h-3 w-3 text-emerald-400" />
  1075→              <span className="max-w-[180px] truncate text-xs font-medium text-emerald-300">
  1076→                {publishedUrl}
  1077→              </span>
  1078→              <button
  1079→                onClick={handleCopyUrl}
  1080→                className="ml-1 rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"
  1081→                aria-label="Copy URL"
  1082→              >
  1083→                {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
  1084→              </button>
  1085→              <a
  1086→                href={publishedUrl}
  1087→                target="_blank"
  1088→                rel="noopener noreferrer"
  1089→                className="rounded p-0.5 text-emerald-400 hover:text-emerald-200 hover:bg-emerald-500/20 transition-colors"

  1096→          <AuthButton onSignIn={() => setAuthOpen(true)} />

  1101→          <Button
  1102→            variant="ghost"
  1103→            size="sm"
  1104→            className="h-8 gap-1.5 text-zinc-400 hover:text-zinc-200 hover:bg-zinc-800"

  1208→function PublishedStoreViewer({ slug }: { slug: string }) {
  1209→  const [store, setStore] = useState<Store | null>(null)
  1210→  const [loading, setLoading] = useState(true)
  1211→  const [error, setError] = useState<string | null>(null)
  1212→
  1213→  useEffect(() => {
  1214→    let cancelled = false
  1215→    async function load() {
  1216→      try {
  1217→        const res = await fetch(`/api/store/lookup?slug=${encodeURIComponent(slug)}`)
  1218→        if (!res.ok) {
  1219→          const data = await res.json().catch(() => ({}))
  1220→          throw new Error(data.error || `Store not found (${res.status})`)
  1221→        }
  1222→        const data = await res.json()
  1223→        if (!cancelled) setStore(data.store)
  1224→      } catch (err: unknown) {
  1225→        if (!cancelled) setError(err instanceof Error ? err.message : 'Failed to load store')
  1226→      } finally {
  1227→        if (!cancelled) setLoading(false)
  1228→      }
  1229→    }
  1230→    load()
  1231→    return () => { cancelled = true }
  1232→  }, [slug])
  1233→
  1234→  if (loading) {
  1235→    return (
  1236→      <div className="flex min-h-screen items-center justify-center bg-white">
  1237→        <div className="text-center">
  1238→          <Loader2 className="mx-auto h-8 w-8 animate-spin text-[#a855f7]" />
  1239→          <p className="mt-3 text-sm text-gray-500">Loading store…</p>
  1240→        </div>
  1241→      </div>
  1242→    )
  1243→  }
  1244→
  1245→  if (error || !store) {
  1246→    return (
  1247→      <div className="flex min-h-screen flex-col items-center justify-center bg-white px-6">
  1248→        <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-50">
  1249→          <AlertCircle className="h-8 w-8 text-red-400" />
  1250→        </div>
  1251→        <h2 className="mb-2 text-xl font-semibold text-gray-900">Store Not Found</h2>
  1252→        <p className="mb-6 max-w-sm text-center text-sm text-gray-500">
  1253→          {error || 'This store could not be found or has not been published.'}
  1254→        </p>
  1255→        <Button
  1256→          variant="outline"
  1257→          onClick={() => window.location.href = '/'}
  1258→          className="gap-2"
  1259→        >
  1260→          <ArrowLeft className="h-4 w-4" />
  1261→          Back to Storqly
  1262→        </Button>
  1263→      </div>
  1264→    )
  1265→  }
  1266→
  1267→  return (
  1268→    <>
  1269→      <StoreRenderer store={store} />
  1270→      <Toaster />
  1271→      {/* Minimal footer for published stores */}
  1272→      <div className="fixed bottom-3 right-3 z-40">
  1273→        <a
  1274→          href="/"
  1275→          className="flex items-center gap-1.5 rounded-full border border-gray-200 bg-white/90 px-3 py-1.5 text-xs font-medium text-gray-500 shadow-sm backdrop-blur transition-colors hover:border-gray-300 hover:text-gray-700"
  1276→        >
  1277→          <Sparkles className="h-3 w-3" />
  1278→          Built with Storqly
  1279→        </a>
  1280→      </div>
  1281→    </>
  1282→  )
  1283→}
