   103→   103→// ─── Background Image Enrichment (lazy, non-blocking) ─────
   104→   104→// Triggered after store generation completes. Fetches real product
   105→   105→// images sequentially without blocking the user or exhausting rate limits.
   106→   106→
   107→   107→function triggerBackgroundImageEnrichment(store: Store) {
   108→   108→  if (!store.products || store.products.length === 0) return
   109→   109→
   110→   110→  // Only enrich products that still have AI-hallucinated Unsplash URLs
   111→   111→  // (Real enriched URLs from the image search service will have different patterns)
   112→   112→  const needsEnrichment = store.products.filter(p =>
   113→   113→    p.images.length > 0 && p.images[0].includes('unsplash.com/photo-')
   114→   114→  )
   115→   115→
   116→   116→  // Scan homepage sections for backgroundImage URLs that need enrichment
   117→   117→  const homepage = store.pages.find(p => p.isHomepage)
   118→   118→  const sectionBackgrounds: { sectionId: string; query: string; currentUrl: string }[] = []
   119→   119→  if (homepage) {
   120→   120→    for (const section of homepage.sections) {
   121→   121→      const bgUrl = section.style?.backgroundImage
   122→   122→      if (bgUrl && bgUrl.includes('unsplash.com/photo-')) {
   123→   123→        // Build a query from the section type and store name
   124→   124→        const query = `${store.name} ${section.type} lifestyle setting photo`
   125→   125→        sectionBackgrounds.push({ sectionId: section.id, query, currentUrl: bgUrl })
   126→   126→      }
   127→   127→    }
   128→   128→  }
   129→   129→
   130→   130→  // Skip if nothing needs enrichment
   131→   131→  if (needsEnrichment.length === 0 && sectionBackgrounds.length === 0) return
   132→   132→
   133→   133→  console.log(`[Storqly] Background image enrichment: ${needsEnrichment.length}/${store.products.length} products, ${sectionBackgrounds.length} section backgrounds`)
   134→   134→
   135→   135→  fetch('/api/store/enrich-images', {
   136→   136→    method: 'POST',
   137→   137→    headers: { 'Content-Type': 'application/json' },
   138→   138→    body: JSON.stringify({
   139→   139→      products: needsEnrichment.map(p => ({
   140→   140→        id: p.id,
   141→   141→        name: p.name,
   142→   142→        images: [...p.images],
   143→   143→        category: p.category,
   144→   144→        description: p.description,
   145→   145→      })),
   146→   146→      storeName: store.name,
   147→   147→      sectionBackgrounds: sectionBackgrounds.length > 0 ? sectionBackgrounds : undefined,
   148→   148→    }),
   149→   149→  })
   150→   150→    .then(res => res.json())
   151→   151→    .then(data => {
   152→   152→      // Update product images in the store via the Zustand store
   153→   153→      const hasProductEnrichment = data.enriched > 0
   154→   154→      const hasSectionEnrichment = data.sectionBackgrounds && data.sectionBackgrounds.length > 0
   155→   155→
   156→   156→      if (hasProductEnrichment || hasSectionEnrichment) {
   157→   157→        console.log(`[Storqly] Background enrichment complete: ${data.enriched} enriched, ${data.kept} kept, ${data.failed} failed in ${data.latencyMs}ms`)
   158→   158→        // Dynamic import to avoid circular dependency
   159→   159→        import('@/lib/store').then(({ useStoreEditor }) => {
   160→   160→          const currentStore = useStoreEditor.getState().store
   161→   161→          if (currentStore) {
   162→   162→            // Update product images
   163→   163→            let updatedProducts = currentStore.products
   164→   164→            if (hasProductEnrichment) {
   165→   165→              updatedProducts = currentStore.products.map(prod => {
   166→   166→                const enriched = data.products?.find((ep: { id: string; images: string[] }) => ep.id === prod.id)
   167→   167→                if (enriched && enriched.images[0] !== prod.images[0]) {
   168→   168→                  return { ...prod, images: enriched.images }
   169→   169→                }
   170→   170→                return prod
   171→   171→              })
   172→   172→            }
   173→   173→
   174→   174→            // Update section background images
   175→   175→            let updatedPages = currentStore.pages
   176→   176→            if (hasSectionEnrichment) {
   177→   177→              const bgMap = new Map(data.sectionBackgrounds.map((sb: { sectionId: string; url: string }) => [sb.sectionId, sb.url]))
   178→   178→              updatedPages = currentStore.pages.map(page => ({
   179→   179→                ...page,
   180→   180→                sections: page.sections.map(section => {
   181→   181→                  const newBg = bgMap.get(section.id)
   182→   182→                  if (newBg && section.style?.backgroundImage !== newBg) {
   183→   183→                    return { ...section, style: { ...section.style, backgroundImage: newBg } }
   184→   184→                  }
   185→   185→                  return section
   186→   186→                }),
   187→   187→              }))
   188→   188→            }
   189→   189→
   190→   190→            useStoreEditor.getState().setStore({ ...currentStore, products: updatedProducts, pages: updatedPages })
   191→   191→          }
   192→   192→        })
   193→   193→      }
   194→   194→    })
   195→   195→    .catch(err => {
   196→   196→      // Non-fatal — products still show AI placeholder images
   197→   197→      console.warn('[Storqly] Background enrichment failed (non-fatal):', err)
   198→   198→    })
   199→   199→}

  1155→  1155→function EditorToolbar({
  1156→  1156→  onToggleLeft,
  1157→  1157→  onToggleRight,
  1158→  1158→  showLeft,
  1159→  1159→  showRight,
  1160→  1160→}: {
  1161→  1161→  onToggleLeft: (v: boolean | ((prev: boolean) => boolean)) => void
  1162→  1162→  onToggleRight: (v: boolean | ((prev: boolean) => boolean)) => void
  1163→  1163→  showLeft: boolean
  1164→  1164→  showRight: boolean
  1165→  1165→}) {
  1166→  1166→  const store = useStoreEditor((s) => s.store)
  1167→  1167→  const reset = useStoreEditor((s) => s.reset)
  1168→  1168→  const isPublishing = useStoreEditor((s) => s.isPublishing)
  1169→  1169→  const setIsPublishing = useStoreEditor((s) => s.setIsPublishing)
  1170→  1170→  const isPublished = useStoreEditor((s) => s.isPublished)

  1174→  1174→  const [copied, setCopied] = useState(false)
  1175→  1175→  const [authOpen, setAuthOpen] = useState(false)
  1176→  1176→
  1177→  1177→  const handleBack = () => {
  1178→  1178→    reset()
  1179→  1179→  }
  1180→  1180→
  1181→  1181→  const handleSave = async () => {
  1182→  1182→    if (!store || isSaving) return
  1183→  1183→    setIsSaving(true)
  1184→  1184→    try {
  1185→  1185→      const res = await fetch('/api/store/save', {

  1189→  1189→      })
  1190→  1190→      if (!res.ok) throw new Error(`Save failed (${res.status})`)
  1191→  1191→      const data = await res.json()
  1192→  1192→      toast.success(`Draft saved — ID: ${data.id}`, {
  1193→  1193→        description: `Slug: ${data.slug}`,
  1194→  1194→        duration: 5000,
  1195→  1195→      })
  1196→  1196→    } catch (err: unknown) {
  1197→  1197→      const msg = err instanceof Error ? err.message : ''
  1198→  1198→      if (msg.includes('Failed to fetch') || msg.includes('NetworkError') || msg.includes('fetch')) {
  1199→  1199→        toast.error('Connection lost while saving', {
  1200→  1200→          description: 'Your work is still in the editor. Please check your connection and try again.',
  1201→  1201→          duration: 6000,
  1202→  1202→        })
  1203→  1203→      } else {
  1204→  1204→        toast.error('Failed to save store', {
  1205→  1205→          description: msg || 'An unexpected error occurred. Please try again.',
  1206→  1206→          duration: 5000,
  1207→  1207→        })
  1208→  1208→      }
  1209→  1209→    } finally {
  1210→  1210→      setIsSaving(false)
  1211→  1211→    }
  1212→  1212→  }
  1213→  1213→

  1215→  1215→    if (!store || isPublishing) return
  1216→  1216→    setIsPublishing(true)
  1217→  1217→    try {
  1218→  1218→      const res = await fetch('/api/store/publish', {
  1219→  1219→        method: 'POST',
  1220→  1220→        headers: { 'Content-Type': 'application/json' },
  1221→  1221→        body: JSON.stringify({ store }),
  1222→  1222→      })
  1223→  1223→      if (!res.ok) {
  1224→  1224→        // Try to get server error details
  1225→  1225→        let serverMsg = ''
  1226→  1226→        try {
  1227→  1227→          const errorData = await res.json()
  1228→  1228→          serverMsg = errorData.error || ''
  1229→  1229→        } catch { /* ignore */ }
  1230→  1230→
  1231→  1231→        if (res.status === 401) {
  1232→  1232→          throw new Error('AUTH_REQUIRED')
  1233→  1233→        } else if (res.status === 502 || res.status === 503 || res.status === 504) {
  1234→  1234→          throw new Error('GATEWAY_ERROR')
  1235→  1235→        } else {
  1236→  1236→          throw new Error(serverMsg || `Server error (${res.status})`)
  1237→  1237→        }
  1238→  1238→      }
  1239→  1239→      const data = await res.json()
  1240→  1240→      const slug = data.slug
  1241→  1241→      setIsPublished(true)
  1242→  1242→      // Build a real viewable URL using the current origin
  1243→  1243→      const baseUrl = window.location.origin
  1244→  1244→      const viewUrl = `${baseUrl}/?store=${slug}`
  1245→  1245→      setPublishedUrl(viewUrl)
  1246→  1246→      toast.success('Store published successfully!')
  1247→  1247→    } catch (err: unknown) {
  1248→  1248→      const msg = err instanceof Error ? err.message : ''
  1249→  1249→
  1250→  1250→      if (msg === 'AUTH_REQUIRED') {
  1251→  1251→        toast.error('Please sign in to publish', {
  1252→  1252→          description: 'You need to be signed in before publishing your store.',
  1253→  1253→          duration: 6000,
  1254→  1254→        })
  1255→  1255→        setAuthOpen(true)
  1256→  1256→      } else if (msg === 'GATEWAY_ERROR') {
  1257→  1257→        toast.error('Server temporarily unavailable', {
  1258→  1258→          description: 'The server seems to be restarting. Please wait a moment and try again.',
  1259→  1259→          duration: 6000,
  1260→  1260→        })
  1261→  1261→      } else if (msg.includes('Failed to fetch') || msg.includes('NetworkError') || msg.includes('fetch') || msg.includes('Load failed')) {
  1262→  1262→        toast.error('Connection lost while publishing', {
  1263→  1263→          description: 'Your store data is safe. Please check your connection and try again.',
  1264→  1264→          duration: 6000,
  1265→  1265→        })
  1266→  1266→      } else {
  1267→  1267→        toast.error('Failed to publish store', {
  1268→  1268→          description: msg || 'An unexpected error occurred. Please try again.',
  1269→  1269→          duration: 5000,
  1270→  1270→        })
  1271→  1271→      }
  1272→  1272→    } finally {
  1273→  1273→      setIsPublishing(false)
  1274→  1274→    }
  1275→  1275→  }
  1276→  1276→
  1277→  1277→  const handleCopyUrl = useCallback(async () => {
  1278→  1278→    if (!publishedUrl) return
  1279→  1279→    try {
  1280→  1280→      await navigator.clipboard.writeText(publishedUrl)
  1281→  1281→      setCopied(true)
  1282→  1282→      toast.success('URL copied to clipboard!')
  1283→  1283→      setTimeout(() => setCopied(false), 2000)
  1284→  1284→    } catch {
  1285→  1285→      // Fallback for older browsers
  1286→  1286→      const textarea = document.createElement('textarea')
  1287→  1287→      textarea.value = publishedUrl
  1288→  1288→      document.body.appendChild(textarea)
  1289→  1289→      textarea.select()
  1290→  1290→      document.execCommand('copy')
  1291→  1291→      document.body.removeChild(textarea)
  1292→  1292→      setCopied(true)
  1293→  1293→      toast.success('URL copied to clipboard!')
